```go
package main

import (
	缓冲输入输出包 "bufio"
	错误判断包 "errors"
	输入输出包 "io"
	操作系统包 "os"
	文件路径包 "path/filepath"
	字符串包 "strings"
	测试包 "testing"

	SQLite包 "github.com/xiaoyu98628/callum/internal/storage/sqlite"
)

func Test资源加锁排除服务无打开中数据库(测试器 *测试包.测试器) {
	数据库 := 文件路径包.合并(测试器.创建临时目录(), "private", "callum.db")
	输入, 写入器 := 输入输出包.管道()
	读取器, 输出 := 输入输出包.管道()
	defer 写入器.关闭()
	defer 读取器.关闭()
	完成 := make(chan error, 1)
	go func() { 完成 <- 持有资源加锁(测试器.上下文(), 数据库, 输入, 输出) }()
	扫描器 := 缓冲输入输出包.新建扫描器(读取器)
	if !扫描器.扫描() || !字符串包.包含(扫描器.文本(), `"resource_lock":true`) {
		测试器.格式化报告致命错误("资源锁未就绪: %s %v", 扫描器.文本(), <-完成)
	}
	if _, 错误 := 操作系统包.状态信息(数据库); !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		测试器.格式化报告致命错误("资源锁创建了业务数据库: %v", 错误)
	}
	if 所属对象, 错误 := SQLite包.获取所有权(数据库); 错误 == nil {
		所属对象.关闭()
		测试器.报告致命错误("服务绕过了资源安装锁")
	}
	if _, 错误 := 输入输出包.写入字符串(写入器, "{\"jsonrpc\":\"2.0\",\"id\":\"validation\",\"method\":\"runtime.validate\",\"params\":{\"name\":\"runtimes\"}}\n"); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if !扫描器.扫描() || !字符串包.包含(扫描器.文本(), `"valid":false`) {
		测试器.格式化报告致命错误("缺失资源未报告校验失败: %s", 扫描器.文本())
	}
	if _, 错误 := 输入输出包.写入字符串(写入器, `{"jsonrpc":"2.0","id":"pinned","method":"runtime.validate","params":{"name":"runtimes","manifest_sha256":"`+字符串包.重复("a", 64)+`"}}`+"\n"); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if !扫描器.扫描() || !字符串包.包含(扫描器.文本(), `"valid":false`) {
		测试器.格式化报告致命错误("带摘要的校验未报告资源缺失: %s", 扫描器.文本())
	}
	for _, 摘要 := range []string{`""`, `null`, `"wrong"`, `"` + 字符串包.重复("A", 64) + `"`} {
		if _, 错误 := 输入输出包.写入字符串(写入器, `{"jsonrpc":"2.0","id":"bad-digest","method":"runtime.validate","params":{"name":"runtimes","manifest_sha256":`+摘要+`}}`+"\n"); 错误 != nil {
			测试器.报告致命错误(错误)
		}
		if !扫描器.扫描() || !字符串包.包含(扫描器.文本(), `"code":-32602`) {
			测试器.格式化报告致命错误("未拒绝无效的清单摘要 %s: %s", 摘要, 扫描器.文本())
		}
	}
	if _, 错误 := 输入输出包.写入字符串(写入器, "{\"jsonrpc\":\"2.0\",\"id\":\"invalid\",\"method\":\"runtime.validate\",\"params\":{\"name\":\"../runtimes\"}}\n"); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if !扫描器.扫描() || !字符串包.包含(扫描器.文本(), `"code":-32602`) {
		测试器.格式化报告致命错误("资源校验请求越过数据目录: %s", 扫描器.文本())
	}
	if _, 错误 := 输入输出包.写入字符串(写入器, "{\"jsonrpc\":\"2.0\",\"id\":\"x\",\"method\":\"system.shutdown\",\"params\":{}}\n"); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if !扫描器.扫描() || !字符串包.包含(扫描器.文本(), `"shutdown":true`) {
		测试器.报告致命错误("资源锁关闭未响应")
	}
	if 错误 := <-完成; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	所属对象, 错误 := SQLite包.获取所有权(数据库)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 所属对象.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
}
```
