package 附件

import (
	上下文包 "context"
	错误判断包 "errors"
	输入输出包 "io"
	结构化日志包 "log/slog"
	操作系统包 "os"
	文件路径包 "path/filepath"
	字符串包 "strings"

func Test访问副本集当前会话并且清理后续(测试器 *测试包.测试器) {
	执行上下文 := 测试器.上下文()
	数据 := 测试器.创建临时目录()
	存储, 错误 := 打开(执行上下文, 数据, func(上下文包.上下文, string, string) (bool, error) { return true, nil }, 结构化日志包.新建(结构化日志包.新建文本处理器(输入输出包.丢弃, nil)))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 存储.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	会话, 运行 := UUID包.必须(UUID包.新建UUIDv7()).字符串(), UUID包.必须(UUID包.新建UUIDv7()).字符串()
	原始 := 文件路径包.合并(测试器.创建临时目录(), "report.csv")
	if 错误 := 操作系统包.写入文件(原始, []byte("name,total\nA,42\n"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	内容块集 := []模型包.内容块{{类型: "file", 路径: 原始, 名称: "report.csv"}}
	批次, 错误 := 存储.阶段(执行上下文, 会话, 运行, 内容块集)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	批次.结束(执行上下文, true)
	当前 := true
	解析 := func(_ 上下文包.上下文, 标识, 引用 string) (模型包.内容块, error) {
		if 当前 && 标识 == 会话 && 引用 == 内容块集[0].附件引用 {
			return 内容块集[0], nil
		}
		return 模型包.内容块{}, 模型包.附件错误
	}
	访问, 错误 := 打开访问(执行上下文, 数据, 存储, 解析)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 访问.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	文件集, 错误 := 访问.开始(执行上下文, 会话, 运行)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	内容块, 置顶, 错误 := 访问.元数据(执行上下文, 会话, 运行, 内容块集[0].附件引用)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 置顶.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	路径, 工作, 错误 := 访问.导出(执行上下文, 会话, 运行, 内容块, 置顶)
	if 错误 != nil || 工作 != 文件集.工作 || 文件路径包.基准(路径) != "report.csv" {
		测试器.格式化报告致命错误("副本导出失败: %s %s %v", 路径, 工作, 错误)
	}
	if 错误 := 操作系统包.移除(原始); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	内容, 错误 := 操作系统包.读取文件(路径)
	if 错误 != nil || !字符串包.包含(string(内容), "42") {
		测试器.格式化报告致命错误("副本不独立: %s %v", 内容, 错误)
	}
	其他 := UUID包.必须(UUID包.新建UUIDv7()).字符串()
	if _, 文件, 错误 := 访问.元数据(执行上下文, 其他, 运行, 内容块.附件引用); 错误 == nil {
		if 文件 != nil {
			_ = 文件.关闭()
		}
		测试器.报告致命错误("跨会话附件被允许")
	}
	当前 = false
	if _, _, 错误 := 访问.导出(执行上下文, 会话, 运行, 内容块, 置顶); 错误 == nil {
		测试器.报告致命错误("被替换消息仍可导出")
	}
	已生成, 错误 := 访问.保存已生成(执行上下文, 会话, 运行, "file", "result.txt", []byte("retained result"), true)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 访问.保存已生成(执行上下文, 其他, 运行, "file", "result.txt", nil, true); 错误 == nil {
		测试器.报告致命错误("跨会话生成物被允许")
	}
	if _, 错误 := 访问.保存已生成(执行上下文, 会话, 运行, "file", "../result.txt", nil, true); 错误 == nil {
		测试器.报告致命错误("非法交付名称被允许")
	}
	if 值, 成功 := 访问.文件集(会话, 运行); !成功 || len(值.交付物集) != 1 {
		测试器.格式化报告致命错误("交付记录丢失: %+v", 值)
	}
	if 错误 := 访问.结束(运行); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 操作系统包.状态信息(工作); !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		测试器.格式化报告致命错误("结束后未回收: %v", 错误)
	}
	if 错误 := 访问.结束(运行); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 操作系统包.状态信息(文件路径包.合并(数据, "attachments", 文件路径包.来源斜杠(内容块.附件引用))); 错误 != nil {
		测试器.格式化报告致命错误("原件被清理: %v", 错误)
	}
	if 数据, _, 错误 := 存储.读取(执行上下文, 已生成.附件引用); 错误 != nil || string(数据) != "retained result" {
		测试器.格式化报告致命错误("结束后交付物不可读: %q %v", 数据, 错误)
	}
}

func Test访问恢复仅已记录私有目录集(测试器 *测试包.测试器) {
	数据 := 测试器.创建临时目录()
	存储, 错误 := 打开(测试器.上下文(), 数据, func(上下文包.上下文, string, string) (bool, error) { return true, nil }, 结构化日志包.新建(结构化日志包.新建文本处理器(输入输出包.丢弃, nil)))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 存储.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	解析 := func(上下文包.上下文, string, string) (模型包.内容块, error) { return 模型包.内容块{}, 模型包.附件错误 }
	首个, 错误 := 打开访问(测试器.上下文(), 数据, 存储, 解析)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	会话, 运行 := UUID包.必须(UUID包.新建UUIDv7()).字符串(), UUID包.必须(UUID包.新建UUIDv7()).字符串()
	文件集, 错误 := 首个.开始(测试器.上下文(), 会话, 运行)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	// 同时覆盖正式标记和进程在原子改名之前退出的情况。
	待处理执行 := UUID包.必须(UUID包.新建UUIDv7()).字符串()
	待处理, 错误 := 首个.开始(测试器.上下文(), 会话, 待处理执行)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 操作系统包.重命名(首个.标记(待处理执行), 文件路径包.合并(首个.标记集, 待处理执行+".pending")); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	无效执行 := UUID包.必须(UUID包.新建UUIDv7()).字符串()
	const 已截取 = `"incomplete`
	if 错误 := 操作系统包.写入文件(首个.标记(无效执行), []byte(已截取), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	秒, 错误 := 打开访问(测试器.上下文(), 数据, 存储, 解析)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	for _, 工作 := range []string{文件集.工作, 待处理.工作} {
		if _, 错误 := 操作系统包.状态信息(工作); !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
			测试器.格式化报告致命错误("异常退出的工作目录未恢复: %v", 错误)
		}
	}
	已隔离, 错误 := 操作系统包.读取文件(文件路径包.合并(首个.标记集, 无效执行+".invalid"))
	if 错误 != nil || string(已隔离) != 已截取 {
		测试器.格式化报告致命错误("未保留不完整标记: %q %v", 已隔离, 错误)
	}
	if _, 错误 := 打开访问(测试器.上下文(), 数据, 存储, 解析); 错误 != nil {
		测试器.格式化报告致命错误("遗留的不完整标记阻止重新启动: %v", 错误)
	}
	if 错误 := 秒.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 首个.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
}
