package 附件

import (
	字节包 "bytes"
	上下文包 "context"
	SHA256包 "crypto/sha256"
	十六进制包 "encoding/hex"
	JSON包 "encoding/json"
	错误判断包 "errors"
	输入输出包 "io"
	操作系统包 "os"
	文件路径包 "path/filepath"
	运行时包 "runtime"
	字符串包 "strings"
	同步包 "sync"

	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

// 执行文件集 由服务端创建并绑定一次执行，不能由模型或客户端指定。
type 执行文件集 struct {
	会话标识 string
	输入集         string
	工作           string
	交付物集      []模型包.内容块
}

// 执行访问 的 门控 串行化本执行的文件操作与回收；就绪、关闭中 和公共快照由 附件访问器.互斥锁 保护。
// 配额和导出索引仅在持有 门控 时访问，不把长时间 标识/打开标志 放进全局锁。
type 执行访问 struct {
	执行文件集
	门控           chan struct{}
	就绪, 关闭中 bool
	目录      string
	字节          int64
	文件集          map[string]string
}

// 附件访问器 保管命令读取附件所需的工作副本。实例锁覆盖恢复，运行调度器结束后由 运行时 关闭。
type 附件访问器 struct {
	存储              *存储
	解析            func(上下文包.上下文, string, string) (模型包.内容块, error)
	标记集, 临时 string
	互斥锁                 同步包.互斥锁
	执行集               map[string]*执行访问
	已关闭             bool
}

// 打开访问 恢复上次异常退出的自有工作目录，不接触用户原图或文档。
func 打开访问(执行上下文 上下文包.上下文, 数据目录 string, 存储 *存储, 解析 func(上下文包.上下文, string, string) (模型包.内容块, error)) (*附件访问器, error) {
	if 存储 == nil || 解析 == nil {
		return nil, 错误判断包.新建("附件访问配置无效")
	}
	标记集 := 文件路径包.合并(数据目录, ".attachment-runs")
	if 错误 := 私有目录(标记集); 错误 != nil {
		return nil, 错误
	}
	临时, 错误 := 文件路径包.求值符号链接集(操作系统包.创建临时目录())
	if 错误 != nil {
		return nil, 错误
	}
	甲 := &附件访问器{存储: 存储, 解析: 解析, 标记集: 标记集, 临时: 临时, 执行集: make(map[string]*执行访问)}
	if 错误 := 甲.recover(执行上下文); 错误 != nil {
		return nil, 错误
	}
	return 甲, nil
}

func (当前附件访问器 *附件访问器) 标记(执行标识 string) string { return 文件路径包.合并(当前附件访问器.标记集, 执行标识+".json") }

// 文件集 仅向装配好的工具调度器返回当前执行的路径，不接受模型指定根目录。
func (当前附件访问器 *附件访问器) 文件集(会话标识, 执行标识 string) (执行文件集, bool) {
	当前附件访问器.互斥锁.加锁()
	defer 当前附件访问器.互斥锁.解锁()
	运行 := 当前附件访问器.执行集[执行标识]
	if 当前附件访问器.已关闭 || 运行 == nil || !运行.就绪 || 运行.关闭中 || 运行.会话标识 != 会话标识 {
		return 执行文件集{}, false
	}
	return 执行快照(运行), true
}

// 执行快照 必须在 附件访问器.互斥锁 内调用，只复制公开快照，不读写由 门控 保护的配额与索引。
func 执行快照(运行 *执行访问) 执行文件集 {
	return 执行文件集{会话标识: 运行.会话标识, 输入集: 运行.输入集, 工作: 运行.工作,
		交付物集: append([]模型包.内容块(nil), 运行.交付物集...)}
}

func (当前附件访问器 *附件访问器) 获取(执行上下文 上下文包.上下文, 会话标识, 执行标识 string) (*执行访问, error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return nil, 错误
	}
	当前附件访问器.互斥锁.加锁()
	运行 := 当前附件访问器.执行集[执行标识]
	允许 := !当前附件访问器.已关闭 && 运行 != nil && 运行.就绪 && !运行.关闭中 && 运行.会话标识 == 会话标识
	当前附件访问器.互斥锁.解锁()
	if !允许 {
		return nil, 模型包.附件错误
	}
	select {
	case 运行.门控 <- struct{}{}:
	case <-执行上下文.完成():
		return nil, 执行上下文.获取错误()
	}
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		运行.释放()
		return nil, 错误
	}
	// 等待本执行的其他操作时，结束 或 关闭 可能已撤销入口，取得 门控 后再次核验。
	当前附件访问器.互斥锁.加锁()
	允许 = !当前附件访问器.已关闭 && 当前附件访问器.执行集[执行标识] == 运行 && !运行.关闭中
	当前附件访问器.互斥锁.解锁()
	if !允许 {
		运行.释放()
		return nil, 模型包.附件错误
	}
	return 运行, nil
}

func (当前执行访问 *执行访问) 释放() { <-当前执行访问.门控 }

func (当前附件访问器 *附件访问器) recover(执行上下文 上下文包.上下文) (返回错误 error) {
	文件, 错误 := 操作系统包.打开(当前附件访问器.标记集)
	if 错误 != nil {
		return 错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 文件.关闭()) }()
	for {
		条目集, 错误 := 文件.读取目录(64)
		if 错误 != nil && 错误 != 输入输出包.文件结束 {
			return 错误
		}
		for _, 条目 := range 条目集 {
			if 错误 := 执行上下文.获取错误(); 错误 != nil {
				return 错误
			}
			名称 := 条目.名称()
			if 执行标识, 成功 := 字符串包.截断后缀(名称, ".invalid"); 成功 && 有效标识(执行标识) && 条目.类型().是否普通文件() {
				continue
			}
			执行标识, 已识别 := 字符串包.截断后缀(名称, ".json")
			if !已识别 {
				执行标识, 已识别 = 字符串包.截断后缀(名称, ".pending")
			}
			if !已识别 || !有效标识(执行标识) || !条目.类型().是否普通文件() {
				return 错误判断包.新建("附件工作目录恢复标记无效")
			}
			路径 := 文件路径包.合并(当前附件访问器.标记集, 名称)
			标记, 错误 := 操作系统包.打开文件(路径, 读取参数集, 0)
			if 错误 != nil {
				return 错误
			}
			信息, 状态信息错误 := 标记.状态信息()
			if 状态信息错误 != nil || !信息.模式().是否普通文件() || 信息.大小() > 4096 {
				return 错误判断包.合并(错误判断包.新建("附件恢复标记类型或大小无效"), 状态信息错误, 标记.关闭())
			}
			数据, 读取错误 := 输入输出包.读取全部(输入输出包.上限读取器(标记, 4097))
			if 错误 := 错误判断包.合并(读取错误, 标记.关闭()); 错误 != nil {
				return 错误
			}
			if len(数据) > 4096 {
				return 错误判断包.新建("附件工作目录恢复记录不可读")
			}
			if !JSON包.有效(数据) {
				// 旧版正式标记和新版待提交标记都可能只写了一部分。留下记录，
				// 不从不完整的数据推断需要删除的目录。
				if 错误 := 操作系统包.重命名(路径, 文件路径包.合并(当前附件访问器.标记集, 执行标识+".invalid")); 错误 != nil {
					return 错误
				}
				continue
			}
			var 目录 string
			if JSON包.反序列化(数据, &目录) != nil || 文件路径包.目录(目录) != 当前附件访问器.临时 || !字符串包.存在前缀(文件路径包.基准(目录), "callum-attachments-"+执行标识+"-") {
				return 错误判断包.新建("附件工作目录恢复路径无效")
			}
			if 错误 := 操作系统包.移除全部(目录); 错误 != nil {
				return 错误
			}
			if 错误 := 操作系统包.移除(路径); 错误 != nil {
				return 错误
			}
		}
		if 错误 == 输入输出包.文件结束 {
			return nil
		}
	}
}

// 开始 创建独立私有目录。副本直到对应执行的所有工具结束后才回收。
func (当前附件访问器 *附件访问器) 开始(执行上下文 上下文包.上下文, 会话标识, 执行标识 string, 已发布 ...模型包.内容块) (_ 执行文件集, 返回错误 error) {
	if !有效标识(会话标识) || !有效标识(执行标识) {
		return 执行文件集{}, 模型包.附件错误
	}
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 执行文件集{}, 错误
	}
	当前附件访问器.互斥锁.加锁()
	if 当前附件访问器.已关闭 || 当前附件访问器.执行集[执行标识] != nil || len(当前附件访问器.执行集) >= 64 {
		当前附件访问器.互斥锁.解锁()
		return 执行文件集{}, 模型包.附件错误
	}
	运行 := &执行访问{门控: make(chan struct{}, 1)}
	运行.门控 <- struct{}{}
	当前附件访问器.执行集[执行标识] = 运行
	当前附件访问器.互斥锁.解锁()
	// 预留项使并发 结束/关闭 能等待初始化，失败时移除本次预留，保留清理错误。
	defer func() {
		if 返回错误 != nil {
			当前附件访问器.互斥锁.加锁()
			if 当前附件访问器.执行集[执行标识] == 运行 {
				delete(当前附件访问器.执行集, 执行标识)
			}
			当前附件访问器.互斥锁.解锁()
		}
		运行.释放()
	}()
	目录, 错误 := 操作系统包.创建目录临时(当前附件访问器.临时, "callum-attachments-"+执行标识+"-")
	if 错误 != nil {
		return 执行文件集{}, 错误
	}
	标记已写入 := false
	defer func() {
		if 返回错误 != nil {
			清理错误 := 操作系统包.移除全部(目录)
			返回错误 = 错误判断包.合并(返回错误, 清理错误)
			// 回收失败时保留已写入的恢复标记，重启后仍能定位自有临时目录。
			if 标记已写入 && 清理错误 == nil {
				if 错误 := 操作系统包.移除(当前附件访问器.标记(执行标识)); 错误 != nil && !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
					返回错误 = 错误判断包.合并(返回错误, 错误)
				}
			}
		}
	}()
	值 := 执行文件集{会话标识: 会话标识, 输入集: 文件路径包.合并(目录, "inputs"), 工作: 文件路径包.合并(目录, "work"), 交付物集: append([]模型包.内容块(nil), 已发布...)}
	for _, 路径 := range []string{值.输入集, 值.工作} {
		if 错误 := 操作系统包.创建目录(路径, 0700); 错误 != nil {
			return 执行文件集{}, 错误
		}
	}
	if 错误 := 当前附件访问器.写入标记(执行标识, 目录); 错误 != nil {
		return 执行文件集{}, 错误
	}
	标记已写入 = true
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 执行文件集{}, 错误
	}
	当前附件访问器.互斥锁.加锁()
	defer 当前附件访问器.互斥锁.解锁()
	if 当前附件访问器.已关闭 || 运行.关闭中 {
		return 执行文件集{}, 模型包.附件错误
	}
	运行.执行文件集, 运行.目录, 运行.文件集, 运行.就绪 = 值, 目录, make(map[string]string), true
	return 执行快照(运行), nil
}

func (当前附件访问器 *附件访问器) 写入标记(执行标识, 目录 string) (返回错误 error) {
	路径 := 当前附件访问器.标记(执行标识)
	if _, 错误 := 操作系统包.读取链接自身状态(路径); !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		if 错误 == nil {
			return 操作系统包.存在错误
		}
		return 错误
	}
	数据, 错误 := JSON包.序列化(目录)
	if 错误 != nil {
		return 错误
	}
	待处理 := 文件路径包.合并(当前附件访问器.标记集, 执行标识+".pending")
	标记, 错误 := 操作系统包.打开文件(待处理, 操作系统包.创建文件标志|操作系统包.独占创建标志|操作系统包.仅写打开标志, 0600)
	if 错误 != nil {
		return 错误
	}
	defer func() {
		if 错误 := 操作系统包.移除(待处理); 错误 != nil && !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
			返回错误 = 错误判断包.合并(返回错误, 错误)
		}
	}()
	_, 写入错误 := 标记.写入(数据)
	if 错误 := 错误判断包.合并(写入错误, 标记.同步(), 标记.关闭()); 错误 != nil {
		return 错误
	}
	if 错误 := 操作系统包.重命名(待处理, 路径); 错误 != nil {
		return 错误
	}
	// 文件内容先同步，恢复标记再固定，异常退出时仅按本实例记录回收。
	if 运行时包.目标操作系统 != "windows" {
		父级, 错误 := 操作系统包.打开(当前附件访问器.标记集)
		if 错误 != nil {
			return 错误
		}
		if 错误 := 错误判断包.合并(父级.同步(), 父级.关闭()); 错误 != nil {
			return 错误
		}
	}
	return nil
}

// 元数据 只从当前消息分支解析引用，并固定原件身份；调用者负责关闭 文件。
func (当前附件访问器 *附件访问器) 元数据(执行上下文 上下文包.上下文, 会话标识, 执行标识, 引用 string) (模型包.内容块, *文件, error) {
	当前附件访问器.互斥锁.加锁()
	运行 := 当前附件访问器.执行集[执行标识]
	允许 := !当前附件访问器.已关闭 && 运行 != nil && 运行.就绪 && !运行.关闭中 && 运行.会话标识 == 会话标识
	当前附件访问器.互斥锁.解锁()
	if !允许 || !有效引用(引用, 会话标识) {
		return 模型包.内容块{}, nil, 模型包.附件错误
	}
	内容块, 错误 := 当前附件访问器.解析(执行上下文, 会话标识, 引用)
	if 错误 != nil || 内容块.类型 != "file" || 内容块.附件引用 != 引用 {
		return 模型包.内容块{}, nil, 错误判断包.合并(模型包.附件错误, 错误)
	}
	文件, 错误 := 当前附件访问器.存储.固定(执行上下文, 引用)
	return 内容块, 文件, 错误
}

// 导出 在审批完成后校验原件哈希并生成独立副本，不使用硬链接。
func (当前附件访问器 *附件访问器) 导出(执行上下文 上下文包.上下文, 会话标识, 执行标识 string, 内容块 模型包.内容块, 文件 *文件) (string, string, error) {
	运行, 错误 := 当前附件访问器.获取(执行上下文, 会话标识, 执行标识)
	if 错误 != nil {
		return "", "", 错误
	}
	defer 运行.释放()
	if _, 错误 := 当前附件访问器.解析(执行上下文, 会话标识, 内容块.附件引用); 错误 != nil {
		return "", "", 错误判断包.合并(模型包.附件错误, 错误)
	}
	数据, 错误 := 文件.读取(执行上下文)
	if 错误 != nil {
		return "", "", 错误
	}
	if 路径 := 运行.文件集[内容块.附件引用]; 路径 != "" {
		已缓存, 错误 := 操作系统包.打开文件(路径, 读取参数集, 0)
		if 错误 != nil {
			return "", "", 错误
		}
		信息, 状态信息错误 := 已缓存.状态信息()
		if 状态信息错误 != nil || !信息.模式().是否普通文件() || 信息.大小() != int64(len(数据)) {
			return "", "", 错误判断包.合并(模型包.附件错误, 状态信息错误, 已缓存.关闭())
		}
		当前, 读取错误 := 读取字节(执行上下文, 已缓存)
		错误 = 错误判断包.合并(读取错误, 已缓存.关闭())
		if 错误 != nil || !字节包.相等(当前, 数据) {
			return "", "", 模型包.附件错误
		}
		return 路径, 运行.工作, nil
	}
	if 运行.字节+int64(len(数据)) > 128<<20 || len(运行.文件集) >= 1024 {
		return "", "", 模型包.附件错误
	}
	名称 := 内容块.名称
	if 名称 == "" {
		名称 = 文件路径包.基准(内容块.附件引用)
	}
	if !有效名称(名称) {
		return "", "", 模型包.附件错误
	}
	哈希 := SHA256包.校验和256([]byte(内容块.附件引用))
	目录 := 文件路径包.合并(运行.输入集, 十六进制包.编码目标字符串(哈希[:]))
	if 错误 := 操作系统包.创建目录全部(目录, 0700); 错误 != nil {
		return "", "", 错误
	}
	路径 := 文件路径包.合并(目录, 名称)
	输出, 错误 := 操作系统包.打开文件(路径, 操作系统包.创建文件标志|操作系统包.独占创建标志|操作系统包.仅写打开标志, 0600)
	if 错误 != nil {
		return "", "", 错误
	}
	_, 写入错误 := 输出.写入(数据)
	if 错误 := 错误判断包.合并(写入错误, 输出.关闭()); 错误 != nil {
		return "", "", 错误判断包.合并(错误, 操作系统包.移除(路径))
	}
	运行.字节 += int64(len(数据))
	运行.文件集[内容块.附件引用] = 路径
	return 路径, 运行.工作, 执行上下文.获取错误()
}

// 结束 可重复调用，正常、取消与失败共用清理。失败保留恢复标记供重启处理。
func (当前附件访问器 *附件访问器) 结束(执行标识 string) error {
	当前附件访问器.互斥锁.加锁()
	运行 := 当前附件访问器.执行集[执行标识]
	if 运行 == nil {
		当前附件访问器.互斥锁.解锁()
		return nil
	}
	运行.关闭中 = true
	当前附件访问器.互斥锁.解锁()
	// 禁止新操作后只等待本执行，其他执行仍可查询与读写。
	运行.门控 <- struct{}{}
	defer 运行.释放()
	当前附件访问器.互斥锁.加锁()
	当前 := 当前附件访问器.执行集[执行标识] == 运行
	当前附件访问器.互斥锁.解锁()
	if !当前 {
		return nil
	}
	if 错误 := 操作系统包.移除全部(运行.目录); 错误 != nil {
		return 错误
	}
	if 错误 := 操作系统包.移除(当前附件访问器.标记(执行标识)); 错误 != nil && !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		return 错误
	}
	当前附件访问器.互斥锁.加锁()
	delete(当前附件访问器.执行集, 执行标识)
	当前附件访问器.互斥锁.解锁()
	return nil
}

func (当前附件访问器 *附件访问器) 关闭() error {
	当前附件访问器.互斥锁.加锁()
	当前附件访问器.已关闭 = true
	标识集 := make([]string, 0, len(当前附件访问器.执行集))
	for 标识 := range 当前附件访问器.执行集 {
		标识集 = append(标识集, 标识)
	}
	当前附件访问器.互斥锁.解锁()
	var 错误 error
	for _, 标识 := range 标识集 {
		错误 = 错误判断包.合并(错误, 当前附件访问器.结束(标识))
	}
	return 错误
}
