```go
// 包 附件 保管随会话提交的原始附件。
package 附件

import (
	上下文包 "context"
	SHA256包 "crypto/sha256"
	十六进制包 "encoding/hex"
	错误判断包 "errors"
	输入输出包 "io"
	结构化日志包 "log/slog"
	操作系统包 "os"
	文件路径包 "path/filepath"
	运行时包 "runtime"
	字符串包 "strings"
	时间包 "time"

	标识符包 "github.com/xiaoyu98628/callum/internal/id"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	图片输入包 "github.com/xiaoyu98628/callum/internal/model/imageinput"
)

type 存储 struct {
	根    *操作系统包.根
	数据目录 string
	存在  func(上下文包.上下文, string, string) (bool, error)
	日志记录器  *结构化日志包.日志记录器
}

// 打开 在运行时实例锁内打开私有目录并恢复未完成导入。调用方负责 关闭。
func 打开(执行上下文 上下文包.上下文, 数据目录 string, 存在 func(上下文包.上下文, string, string) (bool, error), 日志记录器 *结构化日志包.日志记录器) (_ *存储, 返回错误 error) {
	路径 := 文件路径包.合并(数据目录, "attachments")
	if 错误 := 私有目录(路径); 错误 != nil {
		return nil, 错误
	}
	根, 错误 := 操作系统包.打开根(路径)
	if 错误 != nil {
		return nil, 错误
	}
	defer func() {
		if 返回错误 != nil {
			返回错误 = 错误判断包.合并(返回错误, 根.关闭())
		}
	}()
	if 错误 := 私有目录(文件路径包.合并(路径, ".pending")); 错误 != nil {
		return nil, 错误
	}
	当前状态 := &存储{根: 根, 数据目录: 数据目录, 存在: 存在, 日志记录器: 日志记录器}
	if 错误 := 当前状态.recover(执行上下文); 错误 != nil {
		return nil, 错误
	}
	return 当前状态, nil
}

func 私有目录(路径 string) error {
	if 错误 := 操作系统包.创建目录(路径, 0700); 错误 != nil && !错误判断包.是否属于(错误, 操作系统包.存在错误) {
		return 错误
	}
	信息, 错误 := 操作系统包.读取链接自身状态(路径)
	if 错误 != nil {
		return 错误
	}
	if !信息.是否目录() || (运行时包.目标操作系统 != "windows" && 信息.模式().权限()&0077 != 0) {
		return 错误判断包.新建("附件目录不是私有普通目录")
	}
	return nil
}

func (当前存储 *存储) 关闭() error { return 当前存储.根.关闭() }

var 扩展集 = map[string]string{
	"image/png":  ".png",
	"image/jpeg": ".jpg",
	"image/gif":  ".gif",
	"image/webp": ".webp",
}

func 有效标识(值 string) bool { return 标识符包.有效版本7(值) }

func 有效引用(引用, 会话标识 string) bool {
	部分集 := 字符串包.拆分(引用, "/")
	if len(部分集) != 3 || !有效标识(部分集[0]) || !有效标识(部分集[1]) || (会话标识 != "" && 部分集[0] != 会话标识) {
		return false
	}
	扩展名 := 文件路径包.扩展名(部分集[2])
	if !有效扩展名(扩展名) {
		return false
	}
	哈希 := 字符串包.裁剪后缀(部分集[2], 扩展名)
	数据, 错误 := 十六进制包.解码字符串(哈希)
	return 错误 == nil && len(数据) == SHA256包.大小 && 十六进制包.编码目标字符串(数据) == 哈希
}

// 读取 校验引用及内容哈希，模型和预览共用这个读取边界。
func (当前存储 *存储) 读取(执行上下文 上下文包.上下文, 引用 string) (_ []byte, _ string, 返回错误 error) {
	文件, 错误 := 当前存储.固定(执行上下文, 引用)
	if 错误 != nil {
		return nil, "", 错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 文件.关闭()) }()
	数据, 错误 := 文件.读取(执行上下文)
	if 错误 != nil {
		return nil, "", 错误
	}
	return 数据, 内容类型(文件路径包.扩展名(引用), 数据), nil
}

type 批次 struct {
	存储                 *存储
	会话标识, 执行标识 string
}

func (当前批次 *批次) 目录() string { return 文件路径包.合并(当前批次.会话标识, 当前批次.执行标识) }
func (当前批次 *批次) 标记() string    { return 文件路径包.合并(".pending", 当前批次.会话标识+"_"+当前批次.执行标识) }

// 阶段 将原始附件落盘后返回副本引用。成功后调用方须根据 SQLite 受理结果收尾。
func (当前存储 *存储) 阶段(执行上下文 上下文包.上下文, 会话标识, 执行标识 string, 内容块集 []模型包.内容块) (_ *批次, 返回错误 error) {
	if !有效标识(会话标识) || !有效标识(执行标识) {
		return nil, 模型包.图片读取错误
	}
	var 批次 *批次
	defer func() {
		if 返回错误 != nil && 批次 != nil {
			返回错误 = 错误判断包.合并(返回错误, 批次.丢弃())
		}
	}()
	总计, 数量 := 0, 0
	for 索引, 内容块 := range 内容块集 {
		if 内容块.类型 != "image" && 内容块.类型 != "file" {
			continue
		}
		数量++
		if 数量 > 图片输入包.最大图片集 {
			return nil, 模型包.图片上限错误
		}
		var 数据 []byte
		var MIME string
		var 错误 error
		if 内容块.附件引用 != "" {
			if !有效引用(内容块.附件引用, 会话标识) {
				return nil, 模型包.图片读取错误
			}
			数据, _, 错误 = 当前存储.读取(执行上下文, 内容块.附件引用)
		} else {
			数据, MIME, 错误 = 当前存储.读取来源(执行上下文, 内容块.路径, 内容块.类型)
		}
		if 错误 != nil {
			return nil, 错误
		}
		总计 += len(数据)
		if 总计 > 图片输入包.最大来源批次字节 {
			return nil, 模型包.图片上限错误
		}
		if 内容块.附件引用 != "" {
			continue
		}
		if 批次 == nil {
			乙 := &批次{当前存储, 会话标识, 执行标识}
			if 错误 := 当前存储.写入(执行上下文, 乙.标记(), nil); 错误 != nil {
				return nil, 错误
			}
			批次 = 乙
			if 错误 := 当前存储.同步目录(".pending"); 错误 != nil {
				return nil, 错误
			}
			if 错误 := 当前存储.根.创建目录全部(批次.目录(), 0700); 错误 != nil {
				return nil, 错误
			}
		}
		名称 := 内容块.名称
		if 名称 == "" {
			名称 = 文件路径包.基准(内容块.路径)
		}
		扩展名 := 扩展集[MIME]
		if 内容块.类型 == "file" {
			扩展名 = 字符串包.目标小写(文件路径包.扩展名(名称))
			if !有效扩展名(扩展名) {
				扩展名 = ".bin"
			}
			MIME = 内容类型(扩展名, 数据)
		}
		哈希 := SHA256包.校验和256(数据)
		引用 := 文件路径包.目标斜杠(文件路径包.合并(批次.目录(), 十六进制包.编码目标字符串(哈希[:])+扩展名))
		// 相同内容在本批只存一次，消息块仍保持用户选择的顺序。
		if 错误 := 当前存储.写入(执行上下文, 文件路径包.来源斜杠(引用), 数据); 错误 != nil && !错误判断包.是否属于(错误, 操作系统包.存在错误) {
			return nil, 错误
		}
		内容块集[索引] = 模型包.内容块{类型: 内容块.类型, 附件引用: 引用, 名称: 名称, MIME: MIME, 大小: int64(len(数据))}
	}
	if 批次 != nil {
		// 先固定文件与目录项，再允许 SQLite 提交引用。
		for _, 目录 := range []string{批次.目录(), 会话标识, "."} {
			if 错误 := 当前存储.同步目录(目录); 错误 != nil {
				return nil, 错误
			}
		}
	}
	return 批次, 执行上下文.获取错误()
}

func (当前存储 *存储) 写入(执行上下文 上下文包.上下文, 路径 string, 数据 []byte) (返回错误 error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	文件, 错误 := 当前存储.根.打开文件(路径, 操作系统包.仅写打开标志|操作系统包.创建文件标志|操作系统包.独占创建标志, 0600)
	if 错误 != nil {
		return 错误
	}
	defer func() {
		返回错误 = 错误判断包.合并(返回错误, 文件.关闭())
		if 返回错误 != nil {
			返回错误 = 错误判断包.合并(返回错误, 当前存储.根.移除(路径))
		}
	}()
	for len(数据) > 0 {
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			return 错误
		}
		部分 := 数据[:min(len(数据), 256<<10)]
		if _, 错误 := 文件.写入(部分); 错误 != nil {
			return 错误
		}
		数据 = 数据[len(部分):]
	}
	return 文件.同步()
}

func (当前存储 *存储) 同步目录(路径 string) error {
	// Windows 文件句柄已同步，目录句柄不支持 同步。
	if 运行时包.目标操作系统 == "windows" {
		return nil
	}
	目录, 错误 := 当前存储.根.打开(路径)
	if 错误 != nil {
		return 错误
	}
	return 错误判断包.合并(目录.同步(), 目录.关闭())
}

func (当前批次 *批次) 丢弃() error {
	if 当前批次 == nil {
		return nil
	}
	if 错误 := 当前批次.存储.根.移除全部(当前批次.目录()); 错误 != nil {
		return 错误
	}
	// 保留会话目录。确认目录项已持久化后才移除恢复标记。
	if 错误 := 当前批次.存储.同步目录(当前批次.会话标识); 错误 != nil && !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		return 错误
	}
	return 当前批次.移除标记()
}

func (当前批次 *批次) 移除标记() error {
	错误 := 当前批次.存储.根.移除(当前批次.标记())
	if 错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		return nil
	}
	return 错误
}

// 结束 在受理成功后只移除标记。提交结果未知时核对记录再回收。
func (当前批次 *批次) 结束(执行上下文 上下文包.上下文, 已受理 bool) {
	if 当前批次 == nil {
		return
	}
	var 错误 error
	if 已受理 {
		错误 = 当前批次.移除标记()
	} else {
		检查, 取消函数 := 上下文包.创建超时上下文(上下文包.无取消(执行上下文), 2*时间包.秒)
		defer 取消函数()
		错误 = 当前批次.recover(检查)
	}
	if 错误 != nil {
		当前批次.存储.日志记录器.警告("附件导入待恢复", "module", "attachments", "run_id", 当前批次.执行标识)
	}
}

func (当前批次 *批次) recover(执行上下文 上下文包.上下文) error {
	存在, 错误 := 当前批次.存储.存在(执行上下文, 当前批次.会话标识, 当前批次.执行标识)
	if 错误 != nil {
		return 错误
	}
	if 存在 {
		return 当前批次.移除标记()
	}
	return 当前批次.丢弃()
}

func (当前存储 *存储) recover(执行上下文 上下文包.上下文) (返回错误 error) {
	目录, 错误 := 当前存储.根.打开(".pending")
	if 错误 != nil {
		return 错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 目录.关闭()) }()
	for {
		名称集, 读取错误 := 目录.读取目录项名称(128)
		for _, 名称 := range 名称集 {
			if 错误 := 执行上下文.获取错误(); 错误 != nil {
				return 错误
			}
			会话标识, 执行标识, 成功 := 字符串包.截断(名称, "_")
			if !成功 || !有效标识(会话标识) || !有效标识(执行标识) {
				return 错误判断包.新建("附件恢复标记无效")
			}
			乙 := &批次{当前存储, 会话标识, 执行标识}
			if 错误 := 乙.recover(执行上下文); 错误 != nil {
				return 错误
			}
		}
		if 读取错误 != nil {
			if 错误判断包.是否属于(读取错误, 输入输出包.文件结束) {
				return nil
			}
			return 读取错误
		}
	}
}

// 移除后续消息 回收已移除队列项私有的附件目录。
func (当前存储 *存储) 移除后续消息(会话标识, 标识 string) {
	if !有效标识(会话标识) || !有效标识(标识) {
		return
	}
	批次 := &批次{当前存储, 会话标识, 标识}
	if 错误 := 批次.丢弃(); 错误 != nil {
		当前存储.日志记录器.警告("清理后续消息附件失败", "module", "attachments", "follow_up_id", 标识)
	}
}
```
