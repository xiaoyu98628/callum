func 访问测试夹具(测试器 *测试包.测试器, 存在 func(上下文包.上下文, string, string) (bool, error)) *附件访问器 {
	测试器.标记测试辅助函数()
	目录 := 测试器.创建临时目录()
	存储, 错误 := 打开(测试器.上下文(), 目录, 存在, 结构化日志包.新建(结构化日志包.新建文本处理器(输入输出包.丢弃, nil)))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 存储.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	访问, 错误 := 打开访问(测试器.上下文(), 目录, 存储, func(上下文包.上下文, string, string) (模型包.内容块, error) {
		return 模型包.内容块{}, 模型包.附件错误
	})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 访问.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	return 访问
}

func Test访问隔离执行集并且等待之前清理(测试器 *测试包.测试器) {
	for _, 关闭全部 := range []bool{false, true} {
		测试器.执行(格式化包.格式化字符串("close=%t", 关闭全部), func(测试器 *测试包.测试器) {
			会话, 缓慢, 其他 := UUID包.必须(UUID包.新建UUIDv7()).字符串(), UUID包.必须(UUID包.新建UUIDv7()).字符串(), UUID包.必须(UUID包.新建UUIDv7()).字符串()
			已开始, 释放 := make(chan struct{}), make(chan struct{})
			解除阻塞 := 同步包.一次函数(func() { close(释放) })
			访问 := 访问测试夹具(测试器, func(_ 上下文包.上下文, _, 执行标识 string) (bool, error) {
				if 执行标识 == 缓慢 {
					close(已开始)
					<-释放
				}
				return true, nil
			})
			defer 解除阻塞()
			文件集, 错误 := 访问.开始(测试器.上下文(), 会话, 缓慢)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			if _, 错误 := 访问.开始(测试器.上下文(), 会话, 其他); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			已结束 := make(chan error, 1)
			go func() {
				_, 错误 := 访问.保存已生成(测试器.上下文(), 会话, 缓慢, "file", "slow.txt", []byte("slow"), true)
				已结束 <- 错误
			}()
			select {
			case <-已开始:
			case <-时间包.之后(3 * 时间包.秒):
				测试器.报告致命错误("附件操作未开始")
			}
			// 一个执行的归属查询被阻塞时，另一个执行仍能查询、生成和回收。
			独立 := make(chan error, 1)
			go func() {
				if _, 成功 := 访问.文件集(会话, 其他); !成功 {
					独立 <- 错误判断包.新建("独立执行不可查询")
					return
				}
				_, 错误 := 访问.保存已生成(测试器.上下文(), 会话, 其他, "file", "other.txt", []byte("other"), true)
				独立 <- 错误判断包.合并(错误, 访问.结束(其他))
			}()
			select {
			case 错误 := <-独立:
				if 错误 != nil {
					测试器.报告致命错误(错误)
				}
			case <-时间包.之后(3 * 时间包.秒):
				测试器.报告致命错误("另一个执行被全局附件锁阻塞")
			}
			// 等待同一执行的操作可以取消，不需要等当前文件操作结束。
			执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
			等待中 := make(chan error, 1)
			go func() {
				_, 错误 := 访问.保存已生成(执行上下文, 会话, 缓慢, "file", "waiting.txt", nil, true)
				等待中 <- 错误
			}()
			取消函数()
			select {
			case 错误 := <-等待中:
				if !错误判断包.是否属于(错误, 上下文包.已取消错误) {
					测试器.格式化报告致命错误("等待未保留取消: %v", 错误)
				}
			case <-时间包.之后(3 * 时间包.秒):
				测试器.报告致命错误("等待文件操作不能取消")
			}
			已清理 := make(chan error, 1)
			go func() {
				if 关闭全部 {
					已清理 <- 访问.关闭()
				} else {
					已清理 <- 访问.结束(缓慢)
				}
			}()
			select {
			case 错误 := <-已清理:
				测试器.格式化报告致命错误("操作尚未完成便回收: %v", 错误)
			case <-时间包.之后(30 * 时间包.毫秒):
			}
			if _, 错误 := 操作系统包.状态信息(文件集.工作); 错误 != nil {
				测试器.格式化报告致命错误("在途操作的目录已被回收: %v", 错误)
			}
			解除阻塞()
			if 错误 := <-已结束; 错误 != nil {
				测试器.报告致命错误(错误)
			}
			if 错误 := <-已清理; 错误 != nil {
				测试器.报告致命错误(错误)
			}
			if _, 错误 := 操作系统包.状态信息(文件集.工作); !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
				测试器.格式化报告致命错误("操作结束后目录未回收: %v", 错误)
			}
			if _, 错误 := 操作系统包.状态信息(访问.标记(缓慢)); !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
				测试器.格式化报告致命错误("恢复标记未回收: %v", 错误)
			}
			if 错误 := 访问.结束(缓慢); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			if !关闭全部 {
				if _, 错误 := 访问.开始(测试器.上下文(), 会话, 缓慢); 错误 != nil {
					测试器.格式化报告致命错误("同 ID 的后续租约不能重新建立: %v", 错误)
				}
			}
		})
	}
}

func Test并发发布中保留交付物配额(测试器 *测试包.测试器) {
	访问 := 访问测试夹具(测试器, func(上下文包.上下文, string, string) (bool, error) { return true, nil })
	会话, 运行 := UUID包.必须(UUID包.新建UUIDv7()).字符串(), UUID包.必须(UUID包.新建UUIDv7()).字符串()
	if _, 错误 := 访问.开始(测试器.上下文(), 会话, 运行); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	var 工作线程集 同步包.等待组
	var 成功 原子操作包.整数32
	for 索引 := range 65 {
		工作线程集.Go(func() {
			_, 错误 := 访问.保存已生成(测试器.上下文(), 会话, 运行, "file", 格式化包.格式化字符串("%d.txt", 索引), []byte(格式化包.转为字符串(索引)), true)
			if 错误 == nil {
				成功.添加(1)
			} else if !错误判断包.是否属于(错误, 模型包.附件错误) {
				测试器.格式化创建错误("发布失败: %v", 错误)
			}
		})
	}
	工作线程集.等待()
	文件集, 成功 := 访问.文件集(会话, 运行)
	if !成功 || len(文件集.交付物集) != 63 || 成功.加载() != 63 {
		测试器.格式化报告致命错误("并发发布突破或丢失配额: %d %d", len(文件集.交付物集), 成功.加载())
	}
	文件集.交付物集[0].名称 = "changed"
	再次, _ := 访问.文件集(会话, 运行)
	if 再次.交付物集[0].名称 == "changed" {
		测试器.报告致命错误("交付物快照仍借用内部切片")
	}
}

func Test访问重试集失败初始化并且清理(测试器 *测试包.测试器) {
	访问 := 访问测试夹具(测试器, func(上下文包.上下文, string, string) (bool, error) { return true, nil })
	会话, 运行 := UUID包.必须(UUID包.新建UUIDv7()).字符串(), UUID包.必须(UUID包.新建UUIDv7()).字符串()
	标记 := 访问.标记(运行)
	if 错误 := 操作系统包.写入文件(标记, []byte(`"existing"`), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 访问.开始(测试器.上下文(), 会话, 运行); 错误 == nil {
		测试器.报告致命错误("初始化覆盖了已有恢复标记")
	}
	if _, 成功 := 访问.文件集(会话, 运行); 成功 {
		测试器.报告致命错误("失败初始化暴露了工作目录")
	}
	if 错误 := 操作系统包.移除(标记); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	文件集, 错误 := 访问.开始(测试器.上下文(), 会话, 运行)
	if 错误 != nil {
		测试器.格式化报告致命错误("初始化失败后仍占用执行槽位: %v", 错误)
	}
	if 错误 := 操作系统包.移除(标记); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 操作系统包.创建目录(标记, 0700); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	子级 := 文件路径包.合并(标记, "block-removal")
	if 错误 := 操作系统包.写入文件(子级, nil, 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 访问.结束(运行); 错误 == nil {
		测试器.报告致命错误("清理失败被忽略")
	}
	if _, 成功 := 访问.文件集(会话, 运行); 成功 {
		测试器.报告致命错误("清理失败后仍接受新操作")
	}
	if 错误 := 操作系统包.移除(子级); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 访问.结束(运行); 错误 != nil {
		测试器.格式化报告致命错误("不能重试清理: %v", 错误)
	}
	if _, 错误 := 操作系统包.状态信息(文件集.工作); !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		测试器.格式化报告致命错误("工作目录未回收: %v", 错误)
	}
	if 错误 := 访问.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 访问.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
}
