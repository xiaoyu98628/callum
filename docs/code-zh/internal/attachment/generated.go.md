package 附件

import (
	上下文包 "context"
	SHA256包 "crypto/sha256"
	十六进制包 "encoding/hex"
	错误判断包 "errors"
	操作系统包 "os"
	文件路径包 "path/filepath"
	字符串包 "strings"

	模型包 "github.com/xiaoyu98628/callum/internal/model"
	图片输入包 "github.com/xiaoyu98628/callum/internal/model/imageinput"
)

// 有效已发布内容块 校验已持久化的交付物元数据及其会话和执行归属。
func 有效已发布内容块(内容块 模型包.内容块, 会话标识, 执行标识 string) bool {
	return 内容块.类型 == "file" && 内容块.路径 == "" && 内容块.文本 == "" &&
		内容块.名称 != "" && 有效名称(内容块.名称) && 内容块.大小 >= 0 && 内容块.大小 <= 图片输入包.最大来源字节 &&
		有效引用(内容块.附件引用, 会话标识) && 字符串包.拆分(内容块.附件引用, "/")[1] == 执行标识
}

// 保存已生成 保留可信工具已经读取的固定内容；只对已受理执行写入会话存储。
// 源路径不再读取，文件与校验结果始终来自同一份字节。
func (当前存储 *存储) 保存已生成(执行上下文 上下文包.上下文, 会话标识, 执行标识, 类别, 名称 string, 数据 []byte) (模型包.内容块, error) {
	if !有效标识(会话标识) || !有效标识(执行标识) || (类别 != "image" && 类别 != "file") || 名称 == "" || !有效名称(名称) || len(数据) > 图片输入包.最大来源字节 {
		return 模型包.内容块{}, 模型包.附件错误
	}
	存在, 错误 := 当前存储.存在(执行上下文, 会话标识, 执行标识)
	if 错误 != nil || !存在 {
		return 模型包.内容块{}, 错误判断包.合并(模型包.附件错误, 错误)
	}
	扩展名 := 字符串包.目标小写(文件路径包.扩展名(名称))
	if !有效扩展名(扩展名) {
		扩展名 = ".bin"
	}
	MIME := 内容类型(扩展名, 数据)
	if 类别 == "image" {
		if _, 错误 := 图片输入包.准备(执行上下文, 数据, MIME); 错误 != nil {
			return 模型包.内容块{}, 错误
		}
		if 扩展集[MIME] == "" {
			return 模型包.内容块{}, 模型包.图片格式错误
		}
		扩展名 = 扩展集[MIME]
	}
	目录 := 文件路径包.合并(会话标识, 执行标识)
	if 错误 := 当前存储.根.创建目录全部(目录, 0700); 错误 != nil {
		return 模型包.内容块{}, 错误
	}
	哈希 := SHA256包.校验和256(数据)
	引用 := 文件路径包.目标斜杠(文件路径包.合并(目录, 十六进制包.编码目标字符串(哈希[:])+扩展名))
	if 错误 := 当前存储.写入(执行上下文, 文件路径包.来源斜杠(引用), 数据); 错误 != nil {
		if !错误判断包.是否属于(错误, 操作系统包.存在错误) {
			return 模型包.内容块{}, 错误
		}
		if _, _, 错误 := 当前存储.读取(执行上下文, 引用); 错误 != nil {
			return 模型包.内容块{}, 错误
		}
	}
	for _, 名称 := range []string{目录, 会话标识, "."} {
		if 错误 := 当前存储.同步目录(名称); 错误 != nil {
			return 模型包.内容块{}, 错误
		}
	}
	return 模型包.内容块{类型: 类别, 附件引用: 引用, 名称: 名称, MIME: MIME, 大小: int64(len(数据))}, nil
}

// 保存已生成 将本轮工具输出保留为视觉上下文或最终交付物，受本轮资源配额限制。
func (当前附件访问器 *附件访问器) 保存已生成(执行上下文 上下文包.上下文, 会话标识, 执行标识, 类别, 名称 string, 数据 []byte, 发布 bool) (模型包.内容块, error) {
	运行, 错误 := 当前附件访问器.获取(执行上下文, 会话标识, 执行标识)
	if 错误 != nil {
		return 模型包.内容块{}, 错误
	}
	defer 运行.释放()
	if 运行.字节+int64(len(数据)) > 128<<20 || len(运行.文件集) >= 1024 || (发布 && len(运行.交付物集) >= 63) {
		return 模型包.内容块{}, 模型包.附件错误
	}
	内容块, 错误 := 当前附件访问器.存储.保存已生成(执行上下文, 会话标识, 执行标识, 类别, 名称, 数据)
	if 错误 != nil {
		return 内容块, 错误
	}
	键 := "generated:" + 内容块.附件引用
	if _, 存在 := 运行.文件集[键]; !存在 {
		运行.文件集[键] = 内容块.附件引用
		运行.字节 += int64(len(数据))
	}
	if 发布 {
		for _, 前一个 := range 运行.交付物集 {
			if 前一个.附件引用 == 内容块.附件引用 {
				return 内容块, nil
			}
		}
		当前附件访问器.互斥锁.加锁()
		运行.交付物集 = append(运行.交付物集, 内容块)
		当前附件访问器.互斥锁.解锁()
	}
	return 内容块, nil
}
