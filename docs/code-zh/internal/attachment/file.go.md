package 附件

// 有效名称 允许未指定名称。显示名称使用单个文件名，不能包含控制字符或目录。
func 有效名称(名称 string) bool {
	return UTF8包.有效字符串(名称) && len(名称) <= 1024 &&
		!字符串包.包含任一字符(名称, "/\\") &&
		!字符串包.包含函数(名称, Unicode包.是控制) && 名称 != "." && 名称 != ".."
}

func 有效扩展名(扩展名 string) bool {
	if len(扩展名) < 2 || len(扩展名) > 17 || 扩展名[0] != '.' {
		return false
	}
	for _, 丙 := range 扩展名[1:] {
		if !(丙 >= 'a' && 丙 <= 'z' || 丙 >= '0' && 丙 <= '9') {
			return false
		}
	}
	return true
}

func 内容类型(扩展名 string, 数据 []byte) string {
	if 预期 := MIME包.类型依据扩展名(扩展名); 预期 != "" {
		return 字符串包.拆分数量(预期, ";", 2)[0]
	}
	return 字符串包.拆分数量(HTTP包.检测内容类型(数据), ";", 2)[0]
}

// 文件 固定一次读取的文件身份及内容哈希，持有者负责关闭。
type 文件 struct {
	文件 *操作系统包.文件
	预期 操作系统包.文件信息
	引用 string
}

func (当前存储 *存储) 固定(执行上下文 上下文包.上下文, 引用 string) (*文件, error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return nil, 错误
	}
	if !有效引用(引用, "") {
		return nil, 模型包.附件错误
	}
	路径 := 文件路径包.来源斜杠(引用)
	信息, 错误 := 当前存储.根.读取链接自身状态(路径)
	if 错误 != nil || !信息.模式().是否普通文件() || 信息.大小() > 图片输入包.最大来源字节 {
		return nil, 模型包.附件错误
	}
	文件, 错误 := 当前存储.根.打开文件(路径, 读取参数集, 0)
	if 错误 != nil {
		return nil, 错误判断包.合并(模型包.附件错误, 错误)
	}
	已打开, 错误 := 文件.状态信息()
	if 错误 != nil || !操作系统包.相同文件(信息, 已打开) || !已打开.模式().是否普通文件() {
		return nil, 错误判断包.合并(模型包.附件错误, 错误, 文件.关闭())
	}
	return &文件{文件: 文件, 预期: 已打开, 引用: 引用}, nil
}

func (当前文件 *文件) 读取(执行上下文 上下文包.上下文) ([]byte, error) {
	信息, 错误 := 当前文件.文件.状态信息()
	if 错误 != nil || !操作系统包.相同文件(信息, 当前文件.预期) || 信息.大小() > 图片输入包.最大来源字节 {
		return nil, 模型包.附件错误
	}
	数据, 错误 := 读取字节(执行上下文, 输入输出包.新建区域读取器(当前文件.文件, 0, 信息.大小()+1))
	if 错误 != nil {
		return nil, 错误
	}
	哈希 := SHA256包.校验和256(数据)
	if 十六进制包.编码目标字符串(哈希[:]) != 字符串包.裁剪后缀(文件路径包.基准(当前文件.引用), 文件路径包.扩展名(当前文件.引用)) {
		return nil, 模型包.附件错误
	}
	return 数据, nil
}

func (当前文件 *文件) 关闭() error { return 当前文件.文件.关闭() }

type 上下文读取器 struct {
	执行上下文 上下文包.上下文
	读取器   输入输出包.读取器
}

func (当前上下文读取器 上下文读取器) 读取(字节数据 []byte) (int, error) {
	if 错误 := 当前上下文读取器.执行上下文.获取错误(); 错误 != nil {
		return 0, 错误
	}
	return 当前上下文读取器.读取器.读取(字节数据)
}

func 读取字节(执行上下文 上下文包.上下文, 读取器 输入输出包.读取器) ([]byte, error) {
	数据, 错误 := 输入输出包.读取全部(输入输出包.上限读取器(上下文读取器{执行上下文, 读取器}, 图片输入包.最大来源字节+1))
	if 错误 != nil {
		return nil, 错误
	}
	if len(数据) > 图片输入包.最大来源字节 {
		return nil, 模型包.附件错误
	}
	return 数据, 执行上下文.获取错误()
}

func (当前存储 *存储) 读取来源(执行上下文 上下文包.上下文, 路径, 类别 string) (_ []byte, _ string, 返回错误 error) {
	if !文件路径包.是否绝对路径(路径) || 文件路径包.清理(路径) != 路径 {
		return nil, "", 模型包.附件错误
	}
	规范化, 错误 := 文件路径包.求值符号链接集(路径)
	if 错误 != nil {
		return nil, "", 模型包.附件错误
	}
	相对, 相对路径错误 := 文件路径包.相对路径(字符串包.目标小写(当前存储.数据目录), 字符串包.目标小写(规范化))
	if 相对路径错误 == nil && 文件路径包.是否本地路径(相对) {
		部分集 := 字符串包.拆分(相对, string(文件路径包.分隔符))
		if len(部分集) != 3 || 部分集[0] != ".attachment-staging" || !字符串包.存在前缀(部分集[1], "upload-") {
			return nil, "", 模型包.附件错误
		}
	}
	文件, 错误 := 操作系统包.打开文件(规范化, 读取参数集, 0)
	if 错误 != nil {
		return nil, "", 模型包.附件错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 文件.关闭()) }()
	信息, 错误 := 文件.状态信息()
	if 错误 != nil || !信息.模式().是否普通文件() || 信息.大小() > 图片输入包.最大来源字节 {
		return nil, "", 模型包.附件错误
	}
	for _, 后缀 := range []string{"", "-wal", "-shm", "-journal", ".lock"} {
		数据库信息, 错误 := 操作系统包.状态信息(文件路径包.合并(当前存储.数据目录, "callum.db") + 后缀)
		if 错误 != nil && !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
			return nil, "", 错误
		}
		if 错误 == nil && 操作系统包.相同文件(信息, 数据库信息) {
			return nil, "", 模型包.附件错误
		}
	}
	数据, 错误 := 读取字节(执行上下文, 文件)
	if 错误 != nil {
		return nil, "", 错误
	}
	已检测, _, _ := 字符串包.截断(HTTP包.检测内容类型(数据), ";")
	if 类别 == "image" && 扩展集[已检测] == "" {
		return nil, "", 模型包.图片格式错误
	}
	if 类别 == "image" && len(数据) == 0 {
		return nil, "", 模型包.图片格式错误
	}
	if 类别 == "file" && 字节包.存在前缀(数据, []byte("SQLite format 3\x00")) {
		// SQLite 身份校验之外，也拒绝导入数据库副本。
		return nil, "", 模型包.附件错误
	}
	return 数据, 已检测, nil
}
