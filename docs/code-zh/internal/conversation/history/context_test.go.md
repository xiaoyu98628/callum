package 历史测试

import (
	错误判断包 "errors"
	反射包 "reflect"
	字符串包 "strings"
	测试包 "testing"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	历史包 "github.com/xiaoyu98628/callum/internal/conversation/history"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

func Test上下文恢复事实集之后对应用户消息(测试器 *测试包.测试器) {
	消息集 := []会话包.消息{
		{执行标识: "failed", 角色: "user", 内容: []模型包.内容块{{类型: "text", 文本: "执行操作"}}},
		{执行标识: "failed", 角色: "assistant", 状态: 智能体包.失败, 内容: []模型包.内容块{{类型: "text", 文本: "未完成回答"}}},
		{执行标识: "completed", 角色: "user", 内容: []模型包.内容块{{类型: "text", 文本: "查询事实"}}},
		{执行标识: "completed", 角色: "assistant", 状态: 智能体包.已完成, 内容: []模型包.内容块{{类型: "text", 文本: "最终回答"}}},
		{执行标识: "next", 角色: "user", 内容: []模型包.内容块{{类型: "text", 文本: "继续"}}},
	}
	事实集 := []模型包.消息{{角色: 模型包.角色助手, 工具调用集: []模型包.工具调用{{标识: "call", 名称: "operation"}}}, {角色: 模型包.角色工具, 工具调用标识: "call", 内容: `{"operation":{"version":1,"outcome":"applied"}}`}}
	输出, 错误 := 历史包.上下文(消息集, map[string][]模型包.消息{"failed": 事实集, "unrelated": {{角色: 模型包.角色助手, 内容: "其他会话"}}})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if len(输出) != 6 || !反射包.深度相等(输出[1:3], 事实集) || 输出[4].内容块集[0].文本 != "最终回答" || 输出[5].内容块集[0].文本 != "继续" {
		测试器.格式化报告致命错误("上下文错序或丢失: %+v", 输出)
	}
	for _, 消息 := range 输出 {
		for _, 内容块 := range 消息.内容块集 {
			if 内容块.文本 == "未完成回答" {
				测试器.报告致命错误("恢复了失败回答")
			}
		}
	}
	// 成功执行的完整轨迹替代回答消息，避免重复回答。
	输出, 错误 = 历史包.上下文(消息集[2:4], map[string][]模型包.消息{"completed": {{角色: 模型包.角色助手, 内容: "最终回答"}}})
	if 错误 != nil || len(输出) != 2 || 输出[1].内容 != "最终回答" {
		测试器.格式化报告致命错误("成功回答重复: %+v %v", 输出, 错误)
	}
}

func Test技能上下文保留任务标题并且独立引用(测试器 *测试包.测试器) {
	内容 := []模型包.内容块{{类型: "skill", 名称: "Review", 路径: "/skills/review/SKILL.md"}, {类型: "text", 文本: "核对正文"}}
	之前 := append([]模型包.内容块(nil), 内容...)
	if 实际值 := 历史包.标题(内容); 实际值 != "核对正文" {
		测试器.格式化报告致命错误("标题包含技能引用: %q", 实际值)
	}
	内容块集 := 历史包.模型内容(内容)
	if len(内容块集) != 2 || 内容块集[0].文本 != "$Review 核对正文" || !字符串包.包含(内容块集[1].文本, 内容[0].路径) {
		测试器.格式化报告致命错误("模型缺少引用或正文: %+v", 内容块集)
	}
	if !反射包.深度相等(内容, 之前) {
		测试器.报告致命错误("组装模型上下文修改了持久化内容")
	}
}

func Test已恢复事实集仍然遵循上下文上限集(测试器 *测试包.测试器) {
	for _, 事实集 := range [][]模型包.消息{
		make([]模型包.消息, 256),
		{{角色: 模型包.角色工具, 内容: 字符串包.重复("x", 历史包.最大字节)}},
	} {
		_, 错误 := 历史包.上下文([]会话包.消息{{角色: "user", 执行标识: "run"}}, map[string][]模型包.消息{"run": 事实集})
		if !错误判断包.是否属于(错误, 模型包.输入上限错误) {
			测试器.格式化报告致命错误("恢复轨迹绕过输入限制: %v", 错误)
		}
	}
}
