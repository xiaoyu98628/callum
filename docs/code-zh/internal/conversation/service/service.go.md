package 服务

import (

	文件路径包 "path/filepath"
	字符串包 "strings"
	UTF8包 "unicode/utf8"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	执行器包 "github.com/xiaoyu98628/callum/internal/agent/executor"

// 服务 负责会话业务与发送受理；异步执行由装配层注入的执行协调器完成。
type 服务 struct {
	仓储    会话包.仓储
	解析    智能体包.解析器
	管理器   *执行器包.管理器
	引擎    执行协调器
	发出事件  func(智能体包.事件) error
	日志记录器 *结构化日志包.日志记录器
	附件集   *附件包.存储
	技能    *技能包.存储
}

func 新建(仓储 会话包.仓储, 解析 智能体包.解析器, 管理器 *执行器包.管理器, 引擎 执行协调器, 附件集 *附件包.存储, 技能存储 *技能包.存储, 发出事件 func(智能体包.事件) error, 日志记录器 *结构化日志包.日志记录器) (*服务, error) {
	if 仓储 == nil || 解析 == nil || 管理器 == nil || 引擎 == nil || 附件集 == nil || 技能存储 == nil || 发出事件 == nil || 日志记录器 == nil {
		return nil, 会话包.无效错误
	}
	return &服务{仓储: 仓储, 解析: 解析, 管理器: 管理器, 引擎: 引擎, 附件集: 附件集, 技能包: 技能存储, 发出事件: 发出事件, 日志记录器: 日志记录器}, nil
}

func 有效标识(标识 string) bool { return 标识符包.有效版本7(标识) }

func 新建标识() (string, error) {
	标识, 错误 := UUID包.新建UUIDv7()
	if 错误 != nil {
		return "", 错误
	}
	return 标识.字符串(), nil
}

func (当前服务 *服务) 获取(执行上下文 上下文包.上下文, 标识 string) (会话包.会话, error) {
	if !有效标识(标识) {
		return 会话包.会话{}, 会话包.无效错误
	}
	return 当前服务.仓储.获取(执行上下文, 标识)
}

func (当前服务 *服务) 列表(执行上下文 上下文包.上下文, 当前筛选 会话包.筛选) ([]会话包.会话, error) {
	if 当前筛选.上限 == 0 {
		当前筛选.上限 = 100
	}
	if 当前筛选.上限 < 1 || 当前筛选.上限 > 200 || 当前筛选.偏移 < 0 || (当前筛选.工作区标识 != nil && !有效标识(*当前筛选.工作区标识)) || (当前筛选.未绑定 && 当前筛选.工作区标识 != nil) {
		return nil, 会话包.无效错误
	}
	return 当前服务.仓储.列表(执行上下文, 当前筛选)
}

// 历史 按执行边界返回有界历史，独立压缩即使没有对应消息也可继续翻页。
func (当前服务 *服务) 历史(执行上下文 上下文包.上下文, 标识, 之前 string, 上限 int) (会话包.历史页, error) {
	if 上限 == 0 {
		上限 = 25
	}
	if !有效标识(标识) || (之前 != "" && !有效标识(之前)) || 上限 < 1 || 上限 > 25 {
		return 会话包.历史页{}, 会话包.无效错误
	}
	return 当前服务.仓储.历史(执行上下文, 标识, 之前, 上限)
}

func (当前服务 *服务) 更新(执行上下文 上下文包.上下文, 标识 string, 输入 会话包.更新) (会话包.会话, error) {
	if !有效标识(标识) || (输入.工作区标识 != nil && !有效标识(*输入.工作区标识)) || (输入.解除关联工作区 && 输入.工作区标识 != nil) {
		return 会话包.会话{}, 会话包.无效错误
	}
	变更 := 会话包.变更{标题: 输入.标题, 工作区标识: 输入.工作区标识, 解除关联工作区: 输入.解除关联工作区, 置顶: 输入.置顶, 已归档: 输入.已归档, 已删除: 输入.已删除}
	if 变更.标题 != nil {
		值 := 字符串包.裁剪空间(*变更.标题)
		if 值 == "" || len(值) > 1024 || !UTF8包.有效字符串(值) {
			return 会话包.会话{}, 会话包.无效错误
		}
		变更.标题 = &值
	}
	if 输入.模型标识 != "" {
		if !有效标识(输入.模型标识) {
			return 会话包.会话{}, 会话包.无效错误
		}
		已解析, 错误 := 当前服务.解析(执行上下文, 输入.模型标识)
		if 错误 != nil {
			return 会话包.会话{}, 错误
		}
		变更.模型 = &已解析.快照
	}
	值, 错误 := 当前服务.仓储.更新(执行上下文, 标识, 变更)
	if 错误 == nil {
		当前服务.发布(智能体包.事件{类型: "conversation.updated", 会话标识: 值.标识, 数据: 值})
	}
	return 值, 错误
}

func 校验(输入 会话包.发送) error {
	if 输入.会话标识 != "" && !有效标识(输入.会话标识) {
		return 会话包.无效错误
	}
	if 输入.模型标识 != "" && !有效标识(输入.模型标识) {
		return 会话包.无效错误
	}
	if 输入.工作区标识 != nil && (!有效标识(*输入.工作区标识) || 输入.会话标识 != "") {
		return 会话包.无效错误
	}
	if 输入.消息标识 != "" && (!有效标识(输入.消息标识) || 输入.会话标识 == "") {
		return 会话包.无效错误
	}
	if 输入.预期执行标识 != "" && (输入.消息标识 == "" || !有效标识(输入.预期执行标识)) {
		return 会话包.无效错误
	}
	if len(输入.已保留附件集) > 0 && (输入.消息标识 == "" || 输入.预期执行标识 == "") {
		return 会话包.无效错误
	}
	if len(输入.内容)+len(输入.已保留附件集) > 64 {
		return 会话包.无效错误
	}
	有意义 := len(输入.已保留附件集) != 0
	存在技能 := false
	大小 := 0
	for _, 内容块 := range 输入.内容 {
		大小 += len(内容块.文本) + len(内容块.路径) + len(内容块.附件引用) + len(内容块.名称) + len(内容块.MIME)
		switch 内容块.类型 {
		case "skill":
			if 存在技能 || 内容块.名称 == "" || len(内容块.名称) > 128 || !UTF8包.有效字符串(内容块.名称) || !文件路径包.是否绝对路径(内容块.路径) || 文件路径包.清理(内容块.路径) != 内容块.路径 || 内容块.文本 != "" || 内容块.附件引用 != "" || 内容块.MIME != "" || 内容块.大小 != 0 {
				return 会话包.无效错误
			}
			存在技能 = true
		case "text":
			if 内容块.路径 != "" || 内容块.附件引用 != "" || 内容块.名称 != "" || 内容块.MIME != "" || 内容块.大小 != 0 || !UTF8包.有效字符串(内容块.文本) {
				return 会话包.无效错误
			}
			有意义 = 有意义 || 字符串包.裁剪空间(内容块.文本) != ""
		case "image", "file":
			if 内容块.文本 != "" || (内容块.附件引用 != "" && (内容块.路径 != "" || 输入.消息标识 == "")) || (内容块.附件引用 == "" && (!文件路径包.是否绝对路径(内容块.路径) || 文件路径包.清理(内容块.路径) != 内容块.路径)) {
				return 会话包.无效错误
			}
			if !附件包.有效名称(内容块.名称) || len(内容块.MIME) > 255 || 内容块.大小 < 0 {
				return 会话包.无效错误
			}
			有意义 = true
		default:
			return 会话包.无效错误
		}
	}
	if !有意义 || 大小 > 1<<20 {
		return 会话包.无效错误
	}
	return nil
}

// 发送 在事务提交后调用 已受理；回调成功后才启动异步调用和发布执行事件。
// 执行上下文 只用于受理；执行使用独立运行上下文，输入内容在受理前复制。
func (当前服务 *服务) 发送(执行上下文 上下文包.上下文, 输入 会话包.发送, 已受理 func(会话包.执行) error) error {
	return 当前服务.内部发送(执行上下文, 输入, nil, 已受理)
}

// 继续 重新受理原执行，只为恢复后的模型请求分配新的步骤标识。
func (当前服务 *服务) 继续(执行上下文 上下文包.上下文, 输入 会话包.继续, 已受理 func(会话包.执行) error) error {
	if 已受理 == nil || !有效标识(输入.会话标识) || !有效标识(输入.执行标识) || (输入.模型标识 != "" && !有效标识(输入.模型标识)) {
		return 会话包.无效错误
	}
	步骤标识, 错误 := 新建标识()
	if 错误 != nil {
		return 错误
	}
	执行租约, 错误 := 当前服务.管理器.预留智能体上下文(执行上下文, 输入.执行标识, 输入.会话标识)
	if 错误 != nil {
		return 错误
	}
	defer 执行租约.中止()
	查找, 取消函数 := 受理上下文(执行上下文, 执行租约)
	defer 取消函数()
	if 输入.模型标识 == "" {
		会话, 错误 := 当前服务.仓储.获取(查找, 输入.会话标识)
		if 错误 != nil {
			return 错误
		}
		输入.模型标识 = 会话.模型.标识
	}
	已解析, 错误 := 当前服务.解析模型(查找, 输入.模型标识)
	if 错误 != nil {
		return 错误
	}
	本次执行, 错误 := 当前服务.仓储.开始(查找, 会话包.开始{继续: true, 继续图片输入: 已解析.图片输入,
		会话标识: 输入.会话标识, 执行标识: 输入.执行标识, 步骤标识: 步骤标识, 模型: 已解析.快照, 协议: 已解析.协议})
	if 错误 != nil {
		return 错误
	}
	return 当前服务.启动(执行租约, 本次执行, 已受理, 当前服务.引擎.准备执行(执行租约, 本次执行, 已解析))
}

func (当前服务 *服务) 内部发送(执行上下文 上下文包.上下文, 输入 会话包.发送, 后续消息 *会话包.后续消息开始, 已受理 func(会话包.执行) error) error {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	if 已受理 == nil {
		return 会话包.无效错误
	}
	if 后续消息 == nil {
		if 错误 := 校验(输入); 错误 != nil {
			return 错误
		}
	}
	输入.内容 = append([]模型包.内容块(nil), 输入.内容...)
	if 错误 := 当前服务.校验技能(执行上下文, 输入.内容); 错误 != nil {
		return 错误
	}
	创建 := 输入.会话标识 == ""
	if 创建 {
		if 输入.模型标识 == "" {
			return 会话包.无效错误
		}
		标识, 错误 := 新建标识()
		if 错误 != nil {
			return 错误
		}
		输入.会话标识 = 标识
	} else if 输入.模型标识 == "" {
		会话, 错误 := 当前服务.仓储.获取(执行上下文, 输入.会话标识)
		if 错误 != nil {
			return 错误
		}
		输入.模型标识 = 会话.模型.标识
	}
	标识集 := make([]string, 4)
	for 索引 := range 标识集 {
		标识, 错误 := 新建标识()
		if 错误 != nil {
			return 错误
		}
		标识集[索引] = 标识
	}
	执行租约, 错误 := 当前服务.管理器.预留智能体上下文(执行上下文, 标识集[0], 输入.会话标识)
	if 错误 != nil {
		return 错误
	}
	defer 执行租约.中止()
	查找, 取消函数 := 受理上下文(执行上下文, 执行租约)
	defer 取消函数()
	已解析, 错误 := 当前服务.解析模型(查找, 输入.模型标识)
	if 错误 != nil {
		return 错误
	}
	if len(输入.已保留附件集) > 0 {
		输入.内容, 错误 = 当前服务.已保留附件集(查找, 输入)
		if 错误 != nil {
			return 错误
		}
	}
	批次, 错误 := 当前服务.阶段附件集(查找, 输入.会话标识, 标识集[0], 输入.内容, 已解析)
	if 错误 != nil {
		return 错误
	}
	流 := true
	if 输入.流 != nil {
		流 = *输入.流
	}
	本次执行, 错误 := 当前服务.仓储.开始(查找, 会话包.开始{后续消息: 后续消息, 创建: 创建, 会话标识: 输入.会话标识, 工作区标识: 输入.工作区标识, 执行标识: 标识集[0], 用户消息标识: 标识集[1], 助手消息标识: 标识集[2], 步骤标识: 标识集[3], 编辑重发消息标识: 输入.消息标识, 编辑重发执行标识: 输入.预期执行标识, 模型: 已解析.快照, 协议: 已解析.协议, 内容: 输入.内容, 流: 流})
	批次.结束(查找, 错误 == nil)
	if 错误 != nil {
		return 错误
	}
	return 当前服务.启动(执行租约, 本次执行, 已受理, 当前服务.引擎.准备执行(执行租约, 本次执行, 已解析))
}

// 受理上下文 随请求或租约取消受理工作，已受理执行仍使用自己的上下文。
func 受理上下文(执行上下文 上下文包.上下文, 执行租约 *执行器包.执行租约) (上下文包.上下文, 上下文包.取消函数) {
	查找, 取消函数 := 上下文包.创建可取消上下文(执行上下文)
	停止 := 上下文包.之后函数(执行租约.上下文(), 取消函数)
	return 查找, func() {
		停止()
		取消函数()
	}
}

func (当前服务 *服务) 解析模型(执行上下文 上下文包.上下文, 标识 string) (智能体包.已解析, error) {
	已解析, 错误 := 当前服务.解析(执行上下文, 标识)
	if 错误 != nil {
		return 智能体包.已解析{}, 错误
	}
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 智能体包.已解析{}, 错误
	}
	if 已解析.客户端 == nil || 已解析.快照.标识 != 标识 || 字符串包.裁剪空间(已解析.快照.远程模型标识) == "" {
		return 智能体包.已解析{}, 会话包.无效错误
	}
	return 已解析, nil
}

func (当前服务 *服务) 启动(执行租约 *执行器包.执行租约, 本次执行 会话包.执行, 已受理 func(会话包.执行) error, 执行 func(上下文包.上下文)) error {
	已开始 := false
	defer func() {
		if !已开始 {
			当前服务.引擎.放弃(执行租约, 本次执行, 上下文包.已取消错误)
		}
	}()
	已受理值 := 本次执行
	if 本次执行.执行.设置快照 != nil {
		// 受理回调借用独立快照，不能通过指针修改即将执行的权限设置。
		快照 := 本次执行.执行.设置快照.复制()
		已受理值.执行.设置快照 = &快照
	}
	if 本次执行.上下文 != nil {
		值 := 本次执行.上下文.复制()
		已受理值.上下文 = &值
	}
	if 错误 := 已受理(已受理值); 错误 != nil {
		return 错误
	}
	执行租约.启动(执行)
	已开始 = true
	return nil
}

// 发布 在业务提交后尽力通知，订阅故障不改变已经提交的会话操作结果。
func (当前服务 *服务) 发布(事件 智能体包.事件) {
	if 错误 := 当前服务.发出事件(事件); 错误 != nil {
		当前服务.日志记录器.警告("发布会话变更事件失败", "module", "conversation", "conversation_id", 事件.会话标识, "event", 事件.类型)
	}
}

func (当前服务 *服务) 后续后续消息已更新(会话标识, 执行标识 string) {
	当前服务.发布(智能体包.事件{类型: "conversation.follow_ups_updated", 会话标识: 会话标识, 执行标识: 执行标识})
}
