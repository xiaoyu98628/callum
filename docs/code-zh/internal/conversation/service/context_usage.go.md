package 服务

import (
	上下文包 "context"
	UTF8包 "unicode/utf8"

	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	图片输入包 "github.com/xiaoyu98628/callum/internal/model/imageinput"
)

// 上下文用量 返回当前模型下的只读上下文预览；草稿不会保存或发送到模型。
func (当前服务 *服务) 上下文用量(执行上下文 上下文包.上下文, 输入 会话包.上下文用量请求) (会话包.上下文用量, error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 会话包.上下文用量{}, 错误
	}
	if (输入.会话标识 != "" && !有效标识(输入.会话标识)) || (输入.模型标识 != "" && !有效标识(输入.模型标识)) || (输入.工作区标识 != nil && (!有效标识(*输入.工作区标识) || 输入.会话标识 != "")) || 输入.草稿图片集 < 0 || 输入.草稿图片集 > 图片输入包.最大图片集 || len(输入.草稿) > 1<<20 || !UTF8包.有效字符串(输入.草稿) {
		return 会话包.上下文用量{}, 会话包.无效错误
	}
	if 输入.会话标识 != "" {
		if 值, 成功, 错误 := 当前服务.引擎.实时上下文用量(输入); 成功 || 错误 != nil {
			return 值, 错误
		}
	}
	状态, 错误 := 当前服务.仓储.上下文状态(执行上下文, 输入.会话标识)
	if 错误 != nil {
		return 会话包.上下文用量{}, 错误
	}
	if 状态.设置.压缩 == nil || !状态.设置.压缩.有效() {
		return 会话包.上下文用量{}, 会话包.无效错误
	}
	if 输入.模型标识 == "" {
		输入.模型标识 = 状态.会话.模型.标识
	}
	if !有效标识(输入.模型标识) {
		return 会话包.上下文用量{}, 会话包.无效错误
	}
	if 状态.活动执行 != nil && 状态.活动执行.模型.标识 != 输入.模型标识 {
		return 会话包.上下文用量{}, 会话包.繁忙错误
	}
	已解析, 错误 := 当前服务.解析(执行上下文, 输入.模型标识)
	if 错误 != nil {
		return 会话包.上下文用量{}, 错误
	}
	if 已解析.快照.标识 != 输入.模型标识 || 已解析.客户端 == nil {
		return 会话包.上下文用量{}, 会话包.无效错误
	}
	return 当前服务.引擎.预览上下文(执行上下文, 输入, 状态, 已解析)
}
