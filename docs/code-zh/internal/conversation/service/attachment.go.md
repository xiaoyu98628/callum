package 服务

import (
	上下文包 "context"
	错误判断包 "errors"
	文件路径包 "path/filepath"
	切片包 "slices"
	字符串包 "strings"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	附件包 "github.com/xiaoyu98628/callum/internal/attachment"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	历史包 "github.com/xiaoyu98628/callum/internal/conversation/history"
	推理调用包 "github.com/xiaoyu98628/callum/internal/inference"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	图片输入包 "github.com/xiaoyu98628/callum/internal/model/imageinput"
)

// 阶段附件集 在写入事务前准备托管引用和校验模型输入，事务结果由调用方交给 批次.结束。
func (当前服务 *服务) 阶段附件集(执行上下文 上下文包.上下文, 会话标识, 所属对象标识 string, 内容 []模型包.内容块, 已解析 智能体包.已解析) (*附件包.批次, error) {
	存在附件集 := false
	for _, 内容块 := range 内容 {
		if 内容块.类型 == "image" && !已解析.图片输入 {
			return nil, 推理调用包.不支持错误
		}
		存在附件集 = 存在附件集 || 内容块.类型 == "image" || 内容块.类型 == "file"
	}
	if !存在附件集 {
		return nil, nil
	}
	批次, 错误 := 当前服务.附件集.阶段(执行上下文, 会话标识, 所属对象标识, 内容)
	if 错误 != nil {
		return nil, 错误
	}
	if 校验器, 成功 := 已解析.客户端.(模型包.输入校验器); 成功 {
		请求 := 模型包.请求{模型: 已解析.快照.远程模型标识, 允许图片输入: 已解析.图片输入,
			消息集: []模型包.消息{{角色: 模型包.角色用户, 内容块集: 历史包.模型内容(内容)}}}
		if 错误 := 校验器.校验输入(执行上下文, 请求); 错误 != nil {
			// 尚未进入受理事务，可以直接回收拒绝的附件。
			return nil, 错误判断包.合并(错误, 批次.丢弃())
		}
	}
	return 批次, nil
}

// 读取附件分片 只读当前分支中消息明确持有的副本，不接受客户端指定任意路径。
// 每次核对完整哈希，再返回不超过 1 MiB 的片段，保持 标准输入输出 响应有界。
func (当前服务 *服务) 读取附件分片(执行上下文 上下文包.上下文, 会话标识, 消息标识, 预期执行标识 string, 索引 int, 偏移 int64) (会话包.附件分片, error) {
	var 结果 会话包.附件分片
	if !有效标识(会话标识) || !有效标识(消息标识) || (预期执行标识 != "" && !有效标识(预期执行标识)) || 索引 < 0 || 索引 >= 64 || 偏移 < 0 {
		return 结果, 会话包.无效错误
	}
	消息, 错误 := 当前服务.仓储.消息(执行上下文, 会话标识, 消息标识)
	if 错误 != nil {
		return 结果, 错误
	}
	if 预期执行标识 != "" && 预期执行标识 != 消息.执行标识 {
		return 结果, 会话包.附件变化通知错误
	}
	if 索引 >= len(消息.内容) {
		return 结果, 会话包.未找到错误
	}
	内容块 := 消息.内容[索引]
	if (内容块.类型 != "file" && 内容块.类型 != "image") || 内容块.附件引用 == "" {
		return 结果, 会话包.未找到错误
	}
	数据, MIME, 错误 := 当前服务.附件集.读取(执行上下文, 内容块.附件引用)
	if 错误 != nil {
		return 结果, 错误
	}
	if 偏移 > int64(len(数据)) || (偏移 == int64(len(数据)) && 偏移 != 0) {
		return 结果, 会话包.无效错误
	}
	结束 := min(偏移+(1<<20), int64(len(数据)))
	分片 := make([]byte, 结束-偏移)
	copy(分片, 数据[偏移:结束])
	名称 := 内容块.名称
	if 名称 == "" {
		名称 = 文件路径包.基准(文件路径包.来源斜杠(内容块.附件引用))
	}
	// 读取 已核对文件内容与引用中的摘要，跨分片再用它固定实际内容。
	摘要 := 字符串包.裁剪后缀(文件路径包.基准(文件路径包.来源斜杠(内容块.附件引用)), 文件路径包.扩展名(内容块.附件引用))
	结果 = 会话包.附件分片{执行标识: 消息.执行标识, SHA256: 摘要, 名称: 名称, MIME: MIME, 大小: int64(len(数据)), 偏移: 偏移, 数据: 分片}
	if 结束 < int64(len(数据)) {
		结果.下一个偏移 = new(结束)
	}
	return 结果, nil
}

// 预览附件 返回有界预览。历史图从固定消息索引读取，草稿使用本地路径。
func (当前服务 *服务) 预览附件(执行上下文 上下文包.上下文, 会话标识, 消息标识 string, 索引 int, 路径 string) ([]byte, string, error) {
	var 内容块 模型包.内容块
	if 路径 != "" {
		if 会话标识 != "" || 消息标识 != "" || 索引 != 0 || !文件路径包.是否绝对路径(路径) || 文件路径包.清理(路径) != 路径 {
			return nil, "", 会话包.无效错误
		}
		内容块.路径 = 路径
	} else {
		if !有效标识(会话标识) || !有效标识(消息标识) || 索引 < 0 || 索引 >= 64 {
			return nil, "", 会话包.无效错误
		}
		消息, 错误 := 当前服务.仓储.消息(执行上下文, 会话标识, 消息标识)
		if 错误判断包.是否属于(错误, 会话包.未找到错误) {
			草稿, 查找错误 := 当前服务.仓储.后续消息(执行上下文, 会话标识, 消息标识)
			if 查找错误 == nil && 草稿.状态 == 会话包.后续消息草稿 {
				消息, 错误 = 会话包.消息{角色: string(模型包.角色用户), 内容: 草稿.内容}, nil
			} else if 查找错误 != nil && !错误判断包.是否属于(查找错误, 会话包.未找到错误) {
				错误 = 查找错误
			}
		}
		if 错误 != nil {
			return nil, "", 错误
		}
		if 消息.角色 != string(模型包.角色用户) || 索引 >= len(消息.内容) || 消息.内容[索引].类型 != "image" {
			return nil, "", 会话包.未找到错误
		}
		内容块 = 消息.内容[索引]
	}
	var 数据 []byte
	var MIME string
	var 错误 error
	if 内容块.附件引用 != "" {
		数据, MIME, 错误 = 当前服务.附件集.读取(执行上下文, 内容块.附件引用)
	} else {
		数据, MIME, 错误 = 图片输入包.读取(执行上下文, 内容块.路径)
	}
	if 错误 != nil {
		return nil, "", 错误
	}
	数据, 错误 = 图片输入包.准备(执行上下文, 数据, MIME)
	return 数据, MIME, 错误
}

// 已保留附件集 从正在编辑的消息取回引用，客户端只能选择该消息的块索引。
func (当前服务 *服务) 已保留附件集(执行上下文 上下文包.上下文, 输入 会话包.发送) ([]模型包.内容块, error) {
	消息, 错误 := 当前服务.仓储.消息(执行上下文, 输入.会话标识, 输入.消息标识)
	if 错误 != nil {
		return nil, 错误
	}
	if 消息.角色 != string(模型包.角色用户) || 消息.执行标识 != 输入.预期执行标识 {
		return nil, 会话包.冲突错误
	}
	已见 := make(map[int]bool, len(输入.已保留附件集))
	已保留 := make([]模型包.内容块, 0, len(输入.已保留附件集))
	for _, 索引 := range 输入.已保留附件集 {
		if 索引 < 0 || 索引 >= len(消息.内容) || (消息.内容[索引].类型 != "image" && 消息.内容[索引].类型 != "file") || 已见[索引] {
			return nil, 会话包.无效错误
		}
		已见[索引] = true
		已保留 = append(已保留, 消息.内容[索引])
	}
	首个图片 := 切片包.索引函数(输入.内容, func(内容块 模型包.内容块) bool { return 内容块.类型 == "image" || 内容块.类型 == "file" })
	if 首个图片 < 0 {
		首个图片 = len(输入.内容)
	}
	输入.内容 = 切片包.插入(输入.内容, 首个图片, 已保留...)
	输入.已保留附件集 = nil
	if 错误 := 校验(输入); 错误 != nil {
		return nil, 错误
	}
	return 输入.内容, nil
}
