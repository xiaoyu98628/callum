package 服务

import (
	上下文包 "context"

	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
)

// 压缩 受理独立的历史压缩，复用会话互斥、全局容量和连接所属执行管理。
func (当前服务 *服务) 压缩(执行上下文 上下文包.上下文, 输入 会话包.压缩, 已受理 func(会话包.执行) error) error {
	if !有效标识(输入.会话标识) || (输入.模型标识 != "" && !有效标识(输入.模型标识)) || 已受理 == nil {
		return 会话包.无效错误
	}
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	if 输入.模型标识 == "" {
		值, 错误 := 当前服务.仓储.获取(执行上下文, 输入.会话标识)
		if 错误 != nil {
			return 错误
		}
		输入.模型标识 = 值.模型.标识
	}
	执行标识, 错误 := 新建标识()
	if 错误 != nil {
		return 错误
	}
	步骤标识, 错误 := 新建标识()
	if 错误 != nil {
		return 错误
	}
	执行租约, 错误 := 当前服务.管理器.预留智能体上下文(执行上下文, 执行标识, 输入.会话标识)
	if 错误 != nil {
		return 错误
	}
	defer 执行租约.中止()
	查找, 取消函数 := 受理上下文(执行上下文, 执行租约)
	defer 取消函数()
	已解析, 错误 := 当前服务.解析模型(查找, 输入.模型标识)
	if 错误 != nil {
		return 错误
	}
	本次执行, 错误 := 当前服务.仓储.开始压缩(查找, 会话包.开始{会话标识: 输入.会话标识, 执行标识: 执行标识, 步骤标识: 步骤标识, 模型: 已解析.快照, 协议: 已解析.协议})
	if 错误 != nil {
		return 错误
	}
	return 当前服务.启动(执行租约, 本次执行, 已受理, func(执行上下文 上下文包.上下文) {
		当前服务.引擎.执行压缩(执行上下文, 执行租约, 本次执行, 已解析)
	})
}

func (当前服务 *服务) 上下文版本集(执行上下文 上下文包.上下文, 标识 string, 上限, 偏移 int) ([]会话包.上下文版本, error) {
	if 上限 == 0 {
		上限 = 50
	}
	if !有效标识(标识) || 上限 < 1 || 上限 > 200 || 偏移 < 0 {
		return nil, 会话包.无效错误
	}
	return 当前服务.仓储.上下文版本集(执行上下文, 标识, 上限, 偏移)
}

func (当前服务 *服务) 上下文版本(执行上下文 上下文包.上下文, 会话标识, 标识 string) (会话包.上下文版本, error) {
	if !有效标识(会话标识) || !有效标识(标识) {
		return 会话包.上下文版本{}, 会话包.无效错误
	}
	return 当前服务.仓储.上下文版本(执行上下文, 会话标识, 标识)
}
