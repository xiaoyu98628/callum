package 服务

import (
	上下文包 "context"
	错误判断包 "errors"
	字符串包 "strings"

	执行器包 "github.com/xiaoyu98628/callum/internal/agent/executor"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	推理调用包 "github.com/xiaoyu98628/callum/internal/inference"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

func (当前服务 *服务) 后续消息集(执行上下文 上下文包.上下文, 标识 string) ([]会话包.后续消息, error) {
	if !有效标识(标识) {
		return nil, 会话包.无效错误
	}
	return 当前服务.仓储.后续消息集(执行上下文, 标识)
}

func (当前服务 *服务) 添加后续消息(执行上下文 上下文包.上下文, 输入 会话包.后续消息输入) (输出 会话包.后续消息, 错误 error) {
	if !有效标识(输入.会话标识) || !有效标识(输入.预期执行标识) || (输入.模式 != "queue" && 输入.模式 != "steer") {
		return 输出, 会话包.无效错误
	}
	var 执行租约 *执行器包.执行租约
	if 输入.模式 == "steer" {
		执行租约 = 当前服务.引擎.引导执行租约(输入.预期执行标识)
		if 执行租约 == nil {
			return 输出, 会话包.后续消息变化通知错误
		}
	}
	内容 := append([]模型包.内容块(nil), 输入.内容...)
	if 错误 := 校验(会话包.发送{会话标识: 输入.会话标识, 模型标识: 输入.模型标识, 内容: 内容}); 错误 != nil {
		return 输出, 错误
	}
	if 错误 := 当前服务.校验技能(执行上下文, 内容); 错误 != nil {
		return 输出, 错误
	}
	会话, 错误 := 当前服务.仓储.获取(执行上下文, 输入.会话标识)
	if 错误 != nil {
		return 输出, 错误
	}
	if 输入.模型标识 == "" {
		输入.模型标识 = 会话.模型.标识
	}
	if 输入.模式 == "steer" {
		运行, 错误 := 当前服务.引擎.获取(执行上下文, 输入.预期执行标识)
		if 错误 != nil || 运行.会话标识 != 输入.会话标识 {
			return 输出, 会话包.后续消息变化通知错误
		}
		输入.模型标识 = 运行.模型.标识
	}
	已解析, 错误 := 当前服务.解析模型(执行上下文, 输入.模型标识)
	if 错误 != nil {
		return 输出, 错误
	}
	标识, 错误 := 新建标识()
	if 错误 != nil {
		return 输出, 错误
	}
	批次, 错误 := 当前服务.阶段附件集(执行上下文, 输入.会话标识, 标识, 内容, 已解析)
	if 错误 != nil {
		return 输出, 错误
	}
	defer func() { 批次.结束(执行上下文, 错误 == nil) }()
	状态 := 会话包.后续消息已排队
	if 输入.模式 == "steer" {
		状态 = 会话包.后续消息引导
	}
	写入 := func() (会话包.后续消息, error) {
		输出, 错误 = 当前服务.仓储.创建后续消息(执行上下文, 会话包.后续消息{标识: 标识, 会话标识: 输入.会话标识,
			预期执行标识: 输入.预期执行标识, 模型标识: 输入.模型标识, 状态: 状态, 内容: 内容})
		return 输出, 错误
	}
	if 输入.模式 == "steer" {
		错误 = 当前服务.引擎.接受引导(执行租约, 写入)
	} else {
		_, 错误 = 写入()
		if 错误 == nil {
			当前服务.后续后续消息已更新(输入.会话标识, 输入.预期执行标识)
		}
	}
	return 输出, 错误
}

func (当前服务 *服务) 变更后续消息(执行上下文 上下文包.上下文, 输入 会话包.后续消息变更) (输出 会话包.后续消息, 错误 error) {
	引导 := 输入.动作 == "steer" || 输入.动作 == "steer_draft"
	队列 := 输入.动作 == "queue_draft"
	编辑 := 输入.动作 == "edit" || 输入.动作 == "steer_draft" || 队列
	if !有效标识(输入.会话标识) || !有效标识(输入.标识) || 输入.修订号 < 1 ||
		((引导 || 队列) && 输入.预期执行标识 == "") ||
		(输入.预期执行标识 != "" && (!(引导 || 队列 || 输入.动作 == "restore_draft") || !有效标识(输入.预期执行标识))) ||
		编辑 != (输入.文本 != nil) || (输入.模型标识 != "" && (!队列 || !有效标识(输入.模型标识))) {
		return 输出, 会话包.无效错误
	}
	var 执行租约 *执行器包.执行租约
	if 引导 {
		执行租约 = 当前服务.引擎.引导执行租约(输入.预期执行标识)
		if 执行租约 == nil {
			return 输出, 会话包.后续消息变化通知错误
		}
	}
	项, 错误 := 当前服务.仓储.后续消息(执行上下文, 输入.会话标识, 输入.标识)
	if 错误 != nil {
		return 输出, 错误
	}
	var 内容 []模型包.内容块
	if 编辑 {
		内容, 错误 = 已编辑后续消息内容(项, *输入.文本)
		if 错误 != nil {
			return 输出, 错误
		}
	}
	if 引导 || 队列 {
		模型标识 := 项.模型标识
		if 引导 {
			运行, 错误 := 当前服务.引擎.获取(执行上下文, 输入.预期执行标识)
			if 错误 != nil || 运行.会话标识 != 输入.会话标识 {
				return 输出, 会话包.后续消息变化通知错误
			}
			模型标识 = 运行.模型.标识
		} else if 输入.模型标识 != "" {
			模型标识 = 输入.模型标识
		}
		已解析, 错误 := 当前服务.解析模型(执行上下文, 模型标识)
		if 错误 != nil {
			return 输出, 错误
		}
		for _, 内容块 := range 项.内容 {
			if 内容块.类型 == "image" && !已解析.图片输入 {
				return 输出, 推理调用包.不支持错误
			}
		}
	}
	写入 := func() (会话包.后续消息, error) {
		输出, 错误 = 当前服务.仓储.变更后续消息(执行上下文, 输入, 内容)
		return 输出, 错误
	}
	if 引导 {
		错误 = 当前服务.引擎.接受引导(执行租约, 写入)
	} else {
		_, 错误 = 写入()
		if 错误 == nil {
			当前服务.后续后续消息已更新(输入.会话标识, 输入.预期执行标识)
		}
	}
	if 错误 != nil {
		return 输出, 错误
	}
	if 输入.动作 == "remove" {
		当前服务.附件集.移除后续消息(输入.会话标识, 输入.标识)
	}
	return 输出, nil
}

// 编辑保留服务端已经保管的附件与技能，客户端只能提交正文。
func 已编辑后续消息内容(项 会话包.后续消息, 文本 string) ([]模型包.内容块, error) {
	内容 := make([]模型包.内容块, 0, len(项.内容)+1)
	if 文本 = 字符串包.裁剪空间(文本); 文本 != "" {
		内容 = append(内容, 模型包.内容块{类型: "text", 文本: 文本})
	}
	for _, 内容块 := range 项.内容 {
		if 内容块.类型 != "text" {
			内容 = append(内容, 内容块)
		}
	}
	return 内容, 校验(会话包.发送{会话标识: 项.会话标识, 消息标识: 项.标识, 内容: 内容})
}

func (当前服务 *服务) 开始后续消息(执行上下文 上下文包.上下文, 输入 会话包.后续消息开始, 已受理 func(会话包.执行) error) error {
	if 已受理 == nil || !有效标识(输入.会话标识) || !有效标识(输入.标识) || 输入.修订号 < 1 ||
		(输入.之后执行标识 != "" && (!有效标识(输入.之后执行标识) || 输入.文本 != nil)) ||
		(输入.模型标识 != "" && (输入.文本 == nil || !有效标识(输入.模型标识))) {
		return 会话包.无效错误
	}
	项, 错误 := 当前服务.仓储.后续消息(执行上下文, 输入.会话标识, 输入.标识)
	if 错误 != nil {
		return 错误
	}
	状态 := 会话包.后续消息已排队
	if 输入.文本 != nil {
		状态 = 会话包.后续消息草稿
	}
	if 项.状态 != 状态 || 项.修订号 != 输入.修订号 {
		return 会话包.后续消息变化通知错误
	}
	内容, 模型标识 := 项.内容, 项.模型标识
	if 输入.文本 != nil {
		内容, 错误 = 已编辑后续消息内容(项, *输入.文本)
		if 错误 != nil {
			return 错误
		}
		if 输入.模型标识 != "" {
			模型标识 = 输入.模型标识
		}
	}
	return 当前服务.内部发送(执行上下文, 会话包.发送{会话标识: 输入.会话标识, 模型标识: 模型标识, 内容: 内容}, &输入, func(本次执行 会话包.执行) error {
		当前服务.后续后续消息已更新(本次执行.会话.标识, 本次执行.执行.标识)
		return 已受理(本次执行)
	})
}

// 是后续消息附件所属对象 保留已入库队列及其后续执行引用。
func 是后续消息附件所属对象(项 会话包.后续消息, 错误 error) (bool, error) {
	if 错误判断包.是否属于(错误, 会话包.未找到错误) {
		return false, nil
	}
	return 项.状态 != 会话包.后续消息已移除, 错误
}
