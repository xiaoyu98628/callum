package 压缩测试

import (
	错误判断包 "errors"
	字符串包 "strings"
	测试包 "testing"

	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	压缩包 "github.com/xiaoyu98628/callum/internal/conversation/compaction"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	词元包 "github.com/xiaoyu98628/callum/internal/model/tokens"
)

func Test计划保留完成工具配对集并且当前轮次(测试器 *测试包.测试器) {
	行集 := []会话包.消息{
		{标识: "u1", 执行标识: "r1", 序号: 1, 角色: "user", 内容: []模型包.内容块{{类型: "text", 文本: "旧问题"}}},
		{标识: "a1", 执行标识: "r1", 序号: 2, 角色: "assistant", 状态: "completed"},
		{标识: "u2", 执行标识: "r2", 序号: 3, 角色: "user", 内容: []模型包.内容块{{类型: "text", 文本: "保留的问题"}}},
		{标识: "a2", 执行标识: "r2", 序号: 4, 角色: "assistant", 状态: "completed"},
		{标识: "u3", 执行标识: "r3", 序号: 5, 角色: "user", 内容: []模型包.内容块{{类型: "text", 文本: "当前问题"}}},
	}
	轨迹 := []模型包.消息{{角色: 模型包.角色助手, 工具调用集: []模型包.工具调用{{标识: "c", 名称: "read_file", 参数: []byte(`{}`)}}}, {角色: 模型包.角色工具, 工具调用标识: "c", 内容: "事实"}}
	计划, 错误 := 压缩包.构建(行集, map[string][]模型包.消息{"r1": 轨迹}, 2)
	if 错误 != nil || 计划.通过 != 2 || len(计划.来源集) != 2 || len(计划.消息集) != 3 {
		测试器.格式化报告致命错误("边界错误: %+v %v", 计划, 错误)
	}
	if len(计划.消息集[1].工具调用集) != 1 || 计划.消息集[2].工具调用标识 != "c" {
		测试器.报告致命错误("工具配对丢失")
	}
	哈希 := 计划.来源集[0].指纹
	轨迹[1].内容 = "改变的事实"
	变化通知, 错误 := 压缩包.构建(行集, map[string][]模型包.消息{"r1": 轨迹}, 2)
	if 错误 != nil || 变化通知.来源集[0].指纹 == 哈希 {
		测试器.报告致命错误("轨迹变化未反映到指纹")
	}
}

func Test计划批次集范围内词元预算无拆分轮次集(测试器 *测试包.测试器) {
	行集 := []会话包.消息{
		{标识: "u1", 执行标识: "r1", 序号: 1, 角色: "user", 内容: []模型包.内容块{{类型: "text", 文本: 字符串包.重复("old ", 100)}}},
		{标识: "a1", 执行标识: "r1", 序号: 2, 角色: "assistant", 状态: "completed"},
		{标识: "u2", 执行标识: "r2", 序号: 3, 角色: "user", 内容: []模型包.内容块{{类型: "text", 文本: 字符串包.重复("large ", 500)}}},
		{标识: "a2", 执行标识: "r2", 序号: 4, 角色: "assistant", 状态: "completed"},
		{标识: "u3", 执行标识: "r3", 序号: 5, 角色: "user", 内容: []模型包.内容块{{类型: "text", 文本: "current"}}},
	}
	适配 := func(消息集 []模型包.消息) bool { return 词元包.估算(消息集, nil, 8192) <= 400 }
	计划, 错误 := 压缩包.构建范围内(行集, nil, 1, 适配)
	if 错误 != nil || 计划.通过 != 2 || len(计划.来源集) != 2 {
		测试器.格式化报告致命错误("未按完整旧轮次缩小摘要源批次: %+v %v", 计划, 错误)
	}
	if _, 错误 := 压缩包.构建范围内(行集, nil, 1, func([]模型包.消息) bool { return false }); !错误判断包.是否属于(错误, 模型包.词元上限错误) {
		测试器.格式化报告致命错误("不可容纳完整源轮次时未明确失败: %v", 错误)
	}
}

func Test摘要需要有界纯文本文本(测试器 *测试包.测试器) {
	const 摘要 = "## 当前进展\n- 结果未知，不能重复执行\n\n参考值为 {\"status\":\"unknown\"}。"
	for _, 输入 := range []string{摘要, "\r\n  " + 摘要 + "  \r\n", 字符串包.重复("x", 压缩包.最大摘要字节)} {
		if 实际值, 错误 := 压缩包.校验(输入); 错误 != nil || 实际值 != 字符串包.裁剪空间(输入) {
			测试器.格式化报告致命错误("未按原样保留文本摘要: %q, %v", 实际值[:min(len(实际值), 100)], 错误)
		}
	}
	const 旧摘要 = `{"goals":["任务"],"constraints":[],"decisions":[],"facts":[],"open_questions":[]}`
	for _, 无效 := range []string{"", " \r\n\t", "摘要\x00", "摘要\xff", 字符串包.重复("x", 压缩包.最大摘要字节+1), 旧摘要, "```json\n" + 旧摘要 + "\n```", `["任务"]`} {
		if _, 错误 := 压缩包.校验(无效); !错误判断包.是否属于(错误, 模型包.无效响应错误) {
			测试器.格式化报告致命错误("未拒绝无效文本摘要: %q, %v", 无效[:min(len(无效), 100)], 错误)
		}
	}
}
