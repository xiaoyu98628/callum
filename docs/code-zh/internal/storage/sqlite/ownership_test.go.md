package SQLite

import (
	文件路径包 "path/filepath"
	测试包 "testing"
)

func Test所有权排除秒运行时并且释放(测试器 *测试包.测试器) {
	路径 := 文件路径包.合并(测试器.创建临时目录(), "data", "callum.db")
	数据库, 错误 := 打开(测试器.上下文(), 路径)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 数据库.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	首个, 错误 := 获取所有权(路径)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 首个.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	if 其他, 错误 := 获取所有权(路径); 错误 == nil {
		其他.关闭()
		测试器.报告致命错误("第二个服务实例取得运行权")
	}
	if 错误 := 首个.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 首个.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	秒, 错误 := 获取所有权(路径)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 秒.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
}
