package 工作区测试

import (
	错误判断包 "errors"
	文件路径包 "path/filepath"
	切片包 "slices"
	测试包 "testing"

	UUID包 "github.com/google/uuid"

	SQLite包 "github.com/xiaoyu98628/callum/internal/storage/sqlite"
	表结构包 "github.com/xiaoyu98628/callum/internal/storage/sqlite/schema"
	存储包 "github.com/xiaoyu98628/callum/internal/storage/sqlite/workspace"
	工作区包 "github.com/xiaoyu98628/callum/internal/workspace"
)

func Test工作区文件夹集回滚(测试器 *测试包.测试器) {
	执行上下文 := 测试器.上下文()
	数据库, 错误 := SQLite包.打开(执行上下文, 文件路径包.合并(测试器.创建临时目录(), "data", "callum.db"))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 数据库.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	仓储, 错误 := 存储包.新建仓储(数据库.数据库)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	根, 其他 := 测试器.创建临时目录(), 测试器.创建临时目录()
	输入 := 工作区包.工作区{标识: UUID包.必须(UUID包.新建UUIDv7()).字符串(), 名称: "项目", 根路径: 根, 其他路径集: []string{其他, 其他}}
	if _, 错误 := 仓储.创建(执行上下文, 输入); 错误 == nil {
		测试器.报告致命错误("项目目录唯一约束未生效")
	}
	if _, 错误 := 仓储.获取(执行上下文, 输入.标识); !错误判断包.是否属于(错误, 工作区包.未找到错误) {
		测试器.格式化报告致命错误("目录写入失败后项目仍然存在: %v", 错误)
	}
	输入.其他路径集 = []string{其他}
	if _, 错误 := 仓储.创建(执行上下文, 输入); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 仓储.更新(执行上下文, 输入.标识, 工作区包.变更{已删除: new(true)}); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	输入.名称 = "恢复名称"
	输入.其他路径集 = []string{其他, 其他}
	if _, 错误 := 仓储.创建(执行上下文, 输入); 错误 == nil {
		测试器.报告致命错误("恢复时接受了重复目录")
	}
	值, 错误 := 仓储.获取(执行上下文, 输入.标识)
	if 错误 != nil || 值.删除时间 == nil || 值.名称 != "项目" || !切片包.相等(值.其他路径集, []string{其他}) {
		测试器.格式化报告致命错误("恢复失败后状态发生变化: %+v %v", 值, 错误)
	}
}

func Test工作区主要目录事务(测试器 *测试包.测试器) {
	执行上下文 := 测试器.上下文()
	数据库, 错误 := SQLite包.打开(执行上下文, 文件路径包.合并(测试器.创建临时目录(), "data", "callum.db"))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 数据库.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	仓储, 错误 := 存储包.新建仓储(数据库.数据库)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	根, 其他 := 测试器.创建临时目录(), 测试器.创建临时目录()
	项目, 错误 := 仓储.创建(执行上下文, 工作区包.工作区{标识: UUID包.必须(UUID包.新建UUIDv7()).字符串(), 名称: "项目", 根路径: 根, 其他路径集: []string{其他}})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	会话 := 表结构包.会话{标识: UUID包.必须(UUID包.新建UUIDv7()).字符串(), 工作区标识: &项目.标识, 标题: "保留的会话"}
	if 错误 := 数据库.数据库.带有上下文(执行上下文).创建(&会话).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	运行 := 表结构包.执行{标识: UUID包.必须(UUID包.新建UUIDv7()).字符串(), 会话标识: 会话.标识, 状态: "running"}
	if 错误 := 数据库.数据库.带有上下文(执行上下文).创建(&运行).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	for _, 状态 := range []string{"running", "awaiting_approval", "awaiting_user_input"} {
		if 错误 := 数据库.数据库.带有上下文(执行上下文).模型(&运行).更新("status", 状态).错误; 错误 != nil {
			测试器.报告致命错误(错误)
		}
		if _, 错误 := 仓储.更新(执行上下文, 项目.标识, 工作区包.变更{名称: new("不得保存"), 根路径: &其他}); !错误判断包.是否属于(错误, 工作区包.繁忙错误) {
			测试器.格式化报告致命错误("%s 时没有阻止主目录切换: %v", 状态, 错误)
		}
		if _, 错误 := 仓储.更新(执行上下文, 项目.标识, 工作区包.变更{根路径: &根, 其他路径集: new([]string{}), 预期文件夹集: new([]string{根, 其他})}); !错误判断包.是否属于(错误, 工作区包.繁忙错误) {
			测试器.格式化报告致命错误("%s 时允许增删源目录: %v", 状态, 错误)
		}
	}
	// 未改变主目录的保存仍可在执行期间改名。
	if _, 错误 := 仓储.更新(执行上下文, 项目.标识, 工作区包.变更{名称: new("已改名"), 根路径: &根}); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 数据库.数据库.带有上下文(执行上下文).模型(&运行).更新集(map[string]any{"status": "cancelled", "finished_at": 1}).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	值, 错误 := 仓储.更新(执行上下文, 项目.标识, 工作区包.变更{根路径: &其他})
	if 错误 != nil || 值.标识 != 项目.标识 || 值.名称 != "已改名" || 值.根路径 != 其他 || !切片包.相等(值.其他路径集, []string{根}) {
		测试器.格式化报告致命错误("切换主目录未保留项目及完整目录集合: %+v %v", 值, 错误)
	}
	// 在附加目录更新后强制主表写入失败，验证两表及名称同时回滚。
	if 错误 := 数据库.数据库.带有上下文(执行上下文).进程执行(`CREATE TRIGGER reject_root_change BEFORE UPDATE OF root_path ON workspaces BEGIN SELECT RAISE(ABORT, 'controlled failure'); END`).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 仓储.更新(执行上下文, 项目.标识, 工作区包.变更{名称: new("回滚名称"), 根路径: &根}); 错误 == nil {
		测试器.报告致命错误("主表写入失败未返回错误")
	}
	值, 错误 = 仓储.获取(执行上下文, 项目.标识)
	if 错误 != nil || 值.名称 != "已改名" || 值.根路径 != 其他 || !切片包.相等(值.其他路径集, []string{根}) {
		测试器.格式化报告致命错误("失败后主目录、附加目录或名称发生变化: %+v %v", 值, 错误)
	}
}
