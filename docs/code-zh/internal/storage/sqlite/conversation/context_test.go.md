func 上下文测试夹具(测试器 *测试包.测试器) (*SQLite包.数据库, *存储包.仓储, *智能体仓储包.仓储, 会话包.执行) {
	测试器.标记测试辅助函数()
	数据库, 错误 := SQLite包.打开(测试器.上下文(), 文件路径包.合并(测试器.创建临时目录(), "data", "callum.db"))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 数据库.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	仓储, 错误 := 存储包.新建仓储(数据库.数据库)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	执行集, 错误 := 智能体仓储包.新建仓储(数据库.数据库)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	输入 := 输入(测试器)
	输入.内容[0].文本 = 字符串包.重复("原始用户约束", 200)
	var 最后 会话包.执行
	for range 4 {
		最后, 错误 = 仓储.开始(测试器.上下文(), 输入)
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		if 错误 := 执行集.保存消息(测试器.上下文(), 最后.助手.标识, []模型包.内容块{{类型: "text", 文本: 字符串包.重复("保留原始回答", 100)}}, 智能体包.已完成); 错误 != nil {
			测试器.报告致命错误(错误)
		}
		最后.执行.状态, 最后.执行.更新时间 = 智能体包.已完成, 时间包.当前时间().Unix毫秒()
		最后.执行.结束时间 = new(最后.执行.更新时间)
		if 错误 := 执行集.结束(测试器.上下文(), 最后.执行); 错误 != nil {
			测试器.报告致命错误(错误)
		}
		输入.创建, 输入.执行标识, 输入.用户消息标识, 输入.助手消息标识, 输入.步骤标识 = false, 标识(测试器), 标识(测试器), 标识(测试器), 标识(测试器)
	}
	return 数据库, 仓储, 执行集, 最后
}

func 构建中上下文(测试器 *测试包.测试器, 仓储 *存储包.仓储, 执行集 *智能体仓储包.仓储, 前一个 会话包.执行, 保留 int) (会话包.执行, 会话包.上下文版本) {
	测试器.标记测试辅助函数()
	本次执行, 错误 := 仓储.开始压缩(测试器.上下文(), 会话包.开始{会话标识: 前一个.会话.标识, 执行标识: 标识(测试器), 步骤标识: 标识(测试器), 模型: 前一个.执行.模型})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	当前时间 := 时间包.当前时间().Unix毫秒()
	本次执行.步骤.状态, 本次执行.步骤.开始时间 = 智能体包.运行中, &当前时间
	if 错误 := 执行集.保存步骤(测试器.上下文(), 本次执行.步骤); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	下限 := int64(0)
	if 本次执行.上下文 != nil {
		下限 = 本次执行.上下文.通过序号
	}
	轨迹集, 错误 := 执行集.上下文执行轨迹(测试器.上下文(), 本次执行.会话.标识, 下限)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	计划, 错误 := 压缩包.构建(本次执行.历史, 轨迹集, 保留)
	if 错误 != nil || len(计划.来源集) == 0 {
		测试器.格式化报告致命错误("无法准备压缩: %v", 错误)
	}
	值 := 会话包.上下文版本{标识: 标识(测试器), 会话标识: 本次执行.会话.标识, 已生成依据步骤标识: 本次执行.步骤.标识, 通过序号: 计划.通过, 来源集: 计划.来源集}
	if 本次执行.上下文 != nil {
		值.基准版本标识 = &本次执行.上下文.标识
	}
	值, 错误 = 仓储.创建上下文(测试器.上下文(), 值)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	值.摘要 = "目标和约束\n- 已确认的事实"
	值.统计 = JSON包.原始JSON消息(`{"before_messages":6,"after_messages":1,"before_bytes":1000,"after_bytes":200}`)
	return 本次执行, 值
}

func 结束上下文执行(测试器 *测试包.测试器, 执行集 *智能体仓储包.仓储, 本次执行 会话包.执行) {
	测试器.标记测试辅助函数()
	本次执行.执行.状态, 本次执行.执行.更新时间 = 智能体包.已完成, 时间包.当前时间().Unix毫秒()
	本次执行.执行.结束时间 = new(本次执行.执行.更新时间)
	if 错误 := 执行集.结束(测试器.上下文(), 本次执行.执行); 错误 != nil {
		测试器.报告致命错误(错误)
	}
}

func Test上下文激活检查集来源并且回滚返回作为一个事务(测试器 *测试包.测试器) {
	数据库, 仓储, 执行集, 最后 := 上下文测试夹具(测试器)
	首个, 版本 := 构建中上下文(测试器, 仓储, 执行集, 最后, 2)
	if 错误 := 仓储.激活上下文(测试器.上下文(), 版本); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	结束上下文执行(测试器, 执行集, 首个)
	下一个, 替换 := 构建中上下文(测试器, 仓储, 执行集, 最后, 1)
	if 错误 := 数据库.数据库.进程执行(`CREATE TRIGGER fail_context_activation BEFORE UPDATE OF status ON conversation_context_versions WHEN NEW.status='active' BEGIN SELECT RAISE(ABORT,'controlled activation failure'); END`).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 仓储.激活上下文(测试器.上下文(), 替换); 错误 == nil {
		测试器.报告致命错误("激活故障未返回错误")
	}
	活动, 错误 := 仓储.活动上下文(测试器.上下文(), 最后.会话.标识)
	if 错误 != nil || 活动 == nil || 活动.标识 != 版本.标识 {
		测试器.格式化报告致命错误("激活失败破坏旧版本: %+v %v", 活动, 错误)
	}
	if 错误 := 数据库.数据库.进程执行("DROP TRIGGER fail_context_activation").错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	变化通知 := []模型包.内容块{{类型: "text", 文本: "来源已变化"}}
	原始, 错误 := JSON包.序列化(变化通知)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 数据库.数据库.模型(&表结构包.消息{}).条件("id=?", 替换.来源集[0].消息标识).更新("content", string(原始)).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 仓储.激活上下文(测试器.上下文(), 替换); !错误判断包.是否属于(错误, 会话包.冲突错误) {
		测试器.格式化报告致命错误("激活未验证来源: %v", 错误)
	}
	if 错误 := 执行集.恢复(测试器.上下文()); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	已保存, 错误 := 仓储.上下文版本(测试器.上下文(), 最后.会话.标识, 替换.标识)
	if 错误 != nil || 已保存.状态 != "failed" {
		测试器.报告致命错误("恢复留下 building 版本")
	}
	运行, 错误 := 执行集.获取(测试器.上下文(), 下一个.执行.标识)
	if 错误 != nil || 运行.状态 != 智能体包.已中断 {
		测试器.报告致命错误("恢复未中断压缩执行")
	}
	活动, 错误 = 仓储.活动上下文(测试器.上下文(), 最后.会话.标识)
	if 错误 != nil || 活动.标识 != 版本.标识 {
		测试器.报告致命错误("恢复改写有效版本")
	}
	if 错误 := 数据库.数据库.模型(&表结构包.上下文版本{}).条件("id=?", 版本.标识).更新("summary", `{"goals":[]}`).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 仓储.活动上下文(测试器.上下文(), 最后.会话.标识); !错误判断包.是否属于(错误, 模型包.无效响应错误) {
		测试器.报告致命错误("损坏的有效摘要被继续使用")
	}
}

func Test编辑重发清空已覆盖摘要并且其模型结果(测试器 *测试包.测试器) {
	_, 仓储, 执行集, 最后 := 上下文测试夹具(测试器)
	本次执行, 错误 := 仓储.开始压缩(测试器.上下文(), 会话包.开始{会话标识: 最后.会话.标识, 执行标识: 标识(测试器), 步骤标识: 标识(测试器), 模型: 最后.执行.模型})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	本次执行.步骤.状态 = 智能体包.运行中
	if 错误 := 执行集.保存步骤(测试器.上下文(), 本次执行.步骤); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	// 仓储也保护覆盖到最后问题的历史版本；服务当前会保留最近轮次，通常无需失效该前缀。
	var 来源集 []会话包.上下文来源
	轨迹集, 错误 := 执行集.上下文执行轨迹(测试器.上下文(), 最后.会话.标识, 0)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	for _, 行 := range 本次执行.历史 {
		var 轨迹 []模型包.消息
		if 行.角色 == "user" {
			轨迹 = 轨迹集[行.执行标识]
		}
		哈希, 错误 := 压缩包.指纹(行, 轨迹)
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		来源集 = append(来源集, 会话包.上下文来源{消息标识: 行.标识, 执行标识: 行.执行标识, 序号: 行.序号, 指纹: 哈希})
	}
	版本, 错误 := 仓储.创建上下文(测试器.上下文(), 会话包.上下文版本{标识: 标识(测试器), 会话标识: 本次执行.会话.标识, 已生成依据步骤标识: 本次执行.步骤.标识, 通过序号: 来源集[len(来源集)-1].序号, 来源集: 来源集})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	版本.摘要 = "旧问题的文本摘要"
	版本.统计 = JSON包.原始JSON消息(`{"before_bytes":1000,"after_bytes":200}`)
	当前时间 := 时间包.当前时间().Unix毫秒()
	子级, 错误 := 执行集.创建步骤(测试器.上下文(), 智能体包.步骤{标识: 标识(测试器), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 父级步骤标识: &本次执行.步骤.标识, 类型: "model_call", 名称: "生成历史摘要", 状态: 智能体包.已完成, 输入: JSON包.原始JSON消息(`{"purpose":"context_summary"}`), 输出: JSON包.原始JSON消息(`{"text":"旧问题的摘要副本"}`), 用量: JSON包.原始JSON消息(`{"total_tokens":7}`), 创建时间: 当前时间, 更新时间: 当前时间, 结束时间: &当前时间})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 仓储.激活上下文(测试器.上下文(), 版本); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	结束上下文执行(测试器, 执行集, 本次执行)
	输入 := 输入(测试器)
	输入.创建, 输入.会话标识, 输入.编辑重发消息标识 = false, 最后.会话.标识, 最后.用户.标识
	已受理, 错误 := 仓储.开始(测试器.上下文(), 输入)
	if 错误 != nil || 已受理.上下文 != nil {
		测试器.格式化报告致命错误("重发复用了失效摘要: %v", 错误)
	}
	已保存, 错误 := 仓储.上下文版本(测试器.上下文(), 最后.会话.标识, 版本.标识)
	if 错误 != nil || 已保存.状态 != "invalid" || 已保存.摘要 != "" || 已保存.已失效位于 == nil {
		测试器.报告致命错误("未清理受影响的摘要")
	}
	步骤, 错误 := 执行集.获取步骤(测试器.上下文(), 子级.标识)
	if 错误 != nil || len(步骤.输出) != 0 || len(步骤.用量) == 0 {
		测试器.报告致命错误("摘要模型副本未清理或实际用量丢失")
	}
}
