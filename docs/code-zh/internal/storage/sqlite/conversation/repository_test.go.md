package 会话测试

func 标识(测试器 *测试包.测试器) string {
	测试器.标记测试辅助函数()
	值, 错误 := UUID包.新建UUIDv7()
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	return 值.字符串()
}
func 输入(测试器 *测试包.测试器) 会话包.开始 {
	测试器.标记测试辅助函数()
	return 会话包.开始{创建: true, 会话标识: 标识(测试器), 用户消息标识: 标识(测试器), 助手消息标识: 标识(测试器), 执行标识: 标识(测试器), 步骤标识: 标识(测试器), 协议: "chatcompletions", 模型: 模型包.快照{标识: 标识(测试器), 显示名称: "模型", 远程模型标识: "remote", 供应商: 模型包.供应商快照{标识: 标识(测试器), 名称: "厂商"}}, 内容: []模型包.内容块{{类型: "text", 文本: "原问题"}}, 流: true}
}

func Test继续图片检查使用活动上下文并且已生效引导(测试器 *测试包.测试器) {
	for _, 测试用例 := range []struct {
		名称 string
		引导 bool
	}{
		{名称: "历史图片由有效摘要覆盖"},
		{名称: "原执行的已应用引导带图", 引导: true},
	} {
		测试器.执行(测试用例.名称, func(测试器 *测试包.测试器) {
			数据库, 仓储, 执行集, 最后 := 上下文测试夹具(测试器)
			执行上下文 := 测试器.上下文()
			图片 := []模型包.内容块{{类型: "image", 附件引用: "stored.png"}}
			if !测试用例.引导 {
				原始, 错误 := JSON包.序列化(图片)
				if 错误 != nil {
					测试器.报告致命错误(错误)
				}
				if 错误 := 数据库.数据库.模型(&表结构包.消息{}).条件("conversation_id=? AND sequence=1", 最后.会话.标识).
					更新("content", string(原始)).错误; 错误 != nil {
					测试器.报告致命错误(错误)
				}
			}
			输入 := 输入(测试器)
			输入.创建, 输入.会话标识 = false, 最后.会话.标识
			目标, 错误 := 仓储.开始(执行上下文, 输入)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			if 测试用例.引导 {
				项, 错误 := 仓储.创建后续消息(执行上下文, 会话包.后续消息{标识: 标识(测试器), 会话标识: 输入.会话标识,
					预期执行标识: 输入.执行标识, 模型标识: 输入.模型.标识, 状态: 会话包.后续消息引导, 内容: 图片})
				if 错误 != nil {
					测试器.报告致命错误(错误)
				}
				if _, 错误 := 仓储.应用引导(执行上下文, 输入.执行标识, 目标.步骤.标识, []string{项.标识}); 错误 != nil {
					测试器.报告致命错误(错误)
				}
			}
			当前时间 := 时间包.当前时间().Unix毫秒()
			目标.执行.状态, 目标.执行.结束时间, 目标.执行.更新时间 = 智能体包.已取消, &当前时间, 当前时间
			if 错误 := 执行集.结束(执行上下文, 目标.执行); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			继续 := 会话包.开始{继续: true, 会话标识: 输入.会话标识, 执行标识: 输入.执行标识, 步骤标识: 标识(测试器), 模型: 输入.模型, 协议: 输入.协议}
			if _, 错误 := 仓储.开始(执行上下文, 继续); !错误判断包.是否属于(错误, 模型包.不支持输出错误) {
				测试器.格式化报告致命错误("该图片未参与继续任务校验: %v", 错误)
			}
			if 测试用例.引导 {
				继续.继续图片输入 = true
			} else {
				压缩, 版本 := 构建中上下文(测试器, 仓储, 执行集, 最后, 2)
				if 错误 := 仓储.激活上下文(执行上下文, 版本); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				结束上下文执行(测试器, 执行集, 压缩)
			}
			if 本次执行, 错误 := 仓储.开始(执行上下文, 继续); 错误 != nil || 本次执行.执行.标识 != 输入.执行标识 {
				测试器.格式化报告致命错误("任务无法继续: %+v %v", 本次执行.执行, 错误)
			}
		})
	}
}

func Test会话约束集并且编辑重发回滚(测试器 *测试包.测试器) {
	数据库, 错误 := SQLite包.打开(测试器.上下文(), 文件路径包.合并(测试器.创建临时目录(), "data", "callum.db"))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer func() {
		if 错误 := 数据库.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	}()
	仓储, 错误 := 存储包.新建仓储(数据库.数据库)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	执行集, 错误 := 智能体仓储包.新建仓储(数据库.数据库)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	首个输入, 秒输入 := 输入(测试器), 输入(测试器)
	首个, 错误 := 仓储.开始(测试器.上下文(), 首个输入)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	秒, 错误 := 仓储.开始(测试器.上下文(), 秒输入)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	已结束位于 := 时间包.当前时间().Unix毫秒()
	外键 := 表结构包.执行{标识: 标识(测试器), 会话标识: 秒.会话.标识, 模型: 秒.执行.模型, 状态: 智能体包.已完成, 创建时间: 已结束位于, 更新时间: 已结束位于, 结束时间: &已结束位于}
	if 错误 := 数据库.数据库.省略("Conversation", "Replaces").创建(&外键).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 数据库.数据库.模型(&表结构包.消息{}).条件("id=?", 首个.助手.标识).更新("run_id", 外键.标识).错误; 错误 == nil {
		测试器.报告致命错误("允许消息引用其他会话执行")
	}
	if 错误 := 数据库.数据库.模型(&表结构包.消息{}).条件("id=?", 首个.助手.标识).更新("sequence", 首个.用户.序号).错误; 错误 == nil {
		测试器.报告致命错误("允许同会话消息序号重复")
	}
	重复 := 表结构包.执行{标识: 标识(测试器), 会话标识: 首个.会话.标识, 模型: 首个.执行.模型, 状态: 智能体包.运行中, 创建时间: 1, 更新时间: 1}
	if 错误 := 数据库.数据库.省略("Conversation", "Replaces").创建(&重复).错误; 错误 == nil {
		测试器.报告致命错误("允许同会话两个活动执行")
	}
	if 错误 := 数据库.数据库.模型(&表结构包.步骤{}).条件("id=?", 首个.步骤.标识).更新("usage", `"not-an-object"`).错误; 错误 == nil {
		测试器.报告致命错误("usage 未限制为 JSON 对象")
	}
	子级 := 表结构包.步骤{标识: 标识(测试器), 会话标识: 首个.会话.标识, 执行标识: 首个.执行.标识, 父级步骤标识: &秒.步骤.标识, 序号: 2, 类型: "model_call", 名称: "错误父级", 状态: 智能体包.待处理, 输入: JSON包.原始JSON消息(`{}`), 创建时间: 1, 更新时间: 1}
	if 错误 := 数据库.数据库.省略("Run", "Parent").创建(&子级).错误; 错误 == nil {
		测试器.报告致命错误("允许父步骤跨执行")
	}
	当前时间 := 时间包.当前时间().Unix毫秒()
	步骤 := 首个.步骤
	步骤.状态, 步骤.结束时间, 步骤.更新时间 = 智能体包.已完成, &当前时间, 当前时间
	步骤.输出 = JSON包.原始JSON消息(`{"response":{"text":"旧回答"}}`)
	步骤.用量 = JSON包.原始JSON消息(`{"total_tokens":2}`)
	if 错误 := 执行集.保存步骤(测试器.上下文(), 步骤); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 执行集.保存消息(测试器.上下文(), 首个.助手.标识, []模型包.内容块{{类型: "text", 文本: "旧回答"}}, 智能体包.已完成); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	已结束 := 首个.执行
	已结束.状态, 已结束.结束时间, 已结束.更新时间 = 智能体包.已完成, &当前时间, 当前时间
	if 错误 := 执行集.结束(测试器.上下文(), 已结束); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	编辑重发 := 输入(测试器)
	编辑重发.创建 = false
	编辑重发.会话标识 = 首个.会话.标识
	编辑重发.编辑重发消息标识 = 首个.用户.标识
	编辑重发.内容 = []模型包.内容块{{类型: "text", 文本: "修改后"}}
	编辑重发.步骤标识 = 首个.步骤.标识
	if _, 错误 := 仓储.开始(测试器.上下文(), 编辑重发); 错误 == nil {
		测试器.报告致命错误("步骤主键冲突没有使重发失败")
	}
	行集, 错误 := 仓储.消息集(测试器.上下文(), 首个.会话.标识, 0, 10)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if len(行集) != 2 || 行集[0].执行标识 != 首个.执行.标识 || 行集[0].内容[0].文本 != "原问题" || 行集[1].标识 != 首个.助手.标识 || 行集[1].内容[0].文本 != "旧回答" {
		测试器.格式化报告致命错误("失败重发没有完整回滚: %+v", 行集)
	}
	已保留, 错误 := 执行集.获取步骤(测试器.上下文(), 首个.步骤.标识)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if string(已保留.输出) != string(步骤.输出) {
		测试器.格式化报告致命错误("回滚没有恢复旧模型结果: %s", 已保留.输出)
	}
	if _, 错误 := 执行集.获取(测试器.上下文(), 编辑重发.执行标识); !错误判断包.是否属于(错误, 智能体包.未找到错误) {
		测试器.格式化报告致命错误("遗留部分新执行: %v", 错误)
	}
}
