func Test撤回后续消息并且自动发送使用该相同认领(测试器 *测试包.测试器) {
	for _, 撤回首个 := range []bool{true, false} {
		名称 := "send wins"
		if 撤回首个 {
			名称 = "withdraw wins"
		}
		测试器.执行(名称, func(测试器 *测试包.测试器) {
			执行上下文 := 测试器.上下文()
			数据库, 错误 := SQLite包.打开(执行上下文, 文件路径包.合并(测试器.创建临时目录(), "data", "callum.db"))
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			测试器.注册清理函数(func() {
				if 错误 := 数据库.关闭(); 错误 != nil {
					测试器.错误文本(错误)
				}
			})
			仓储, _ := 存储包.新建仓储(数据库.数据库)
			执行集, _ := 智能体仓储包.新建仓储(数据库.数据库)
			本次执行, 错误 := 仓储.开始(执行上下文, 输入(测试器))
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			项, 错误 := 仓储.创建后续消息(执行上下文, 会话包.后续消息{
				标识: 标识(测试器), 会话标识: 本次执行.会话.标识, 预期执行标识: 本次执行.执行.标识,
				模型标识: 本次执行.执行.模型.标识, 状态: 会话包.后续消息已排队,
				内容: []模型包.内容块{{类型: "text", 文本: "原队列正文"}},
			})
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			本次执行.执行.状态, 本次执行.执行.结束时间 = 智能体包.已完成, new(时间包.当前时间().Unix毫秒())
			if 错误 := 执行集.结束(执行上下文, 本次执行.执行); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			撤回 := 会话包.后续消息变更{会话标识: 项.会话标识, 标识: 项.标识, 修订号: 项.修订号, 动作: "withdraw"}
			下一个 := 输入(测试器)
			下一个.创建, 下一个.会话标识, 下一个.模型 = false, 项.会话标识, 本次执行.执行.模型
			下一个.后续消息 = &会话包.后续消息开始{会话标识: 项.会话标识, 标识: 项.标识, 修订号: 项.修订号, 之后执行标识: 本次执行.执行.标识}
			下一个.内容 = 项.内容
			if !撤回首个 {
				if _, 错误 := 仓储.开始(执行上下文, 下一个); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				if _, 错误 := 仓储.变更后续消息(执行上下文, 撤回, nil); !错误判断包.是否属于(错误, 会话包.后续消息已发送错误) {
					测试器.格式化报告致命错误("已发消息错误地进入草稿: %v", 错误)
				}
				return
			}
			草稿, 错误 := 仓储.变更后续消息(执行上下文, 撤回, nil)
			if 错误 != nil || 草稿.状态 != 会话包.后续消息草稿 || 草稿.内容[0].文本 != "原队列正文" {
				测试器.格式化报告致命错误("取出的草稿不完整: %+v, %v", 草稿, 错误)
			}
			if _, 错误 := 仓储.开始(执行上下文, 下一个); !错误判断包.是否属于(错误, 会话包.后续消息变化通知错误) {
				测试器.格式化报告致命错误("已取出的消息被自动发送: %v", 错误)
			}
			下一个.后续消息.修订号, 下一个.后续消息.之后执行标识, 下一个.后续消息.文本 = 草稿.修订号, "", new("手动提交的编辑")
			下一个.内容 = []模型包.内容块{{类型: "text", 文本: *下一个.后续消息.文本}}
			已受理, 错误 := 仓储.开始(执行上下文, 下一个)
			if 错误 != nil || 已受理.用户.内容[0].文本 != "手动提交的编辑" {
				测试器.格式化报告致命错误("草稿无法手动发送: %+v, %v", 已受理.用户, 错误)
			}
		})
	}
}

func Test引导历史保留回执位置跨新建步骤集(测试器 *测试包.测试器) {
	执行上下文 := 测试器.上下文()
	数据库, 错误 := SQLite包.打开(执行上下文, 文件路径包.合并(测试器.创建临时目录(), "data", "callum.db"))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 数据库.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	仓储, _ := 存储包.新建仓储(数据库.数据库)
	执行集, _ := 智能体仓储包.新建仓储(数据库.数据库)
	本次执行, 错误 := 仓储.开始(执行上下文, 输入(测试器))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	添加 := func(状态 string) 会话包.后续消息 {
		测试器.标记测试辅助函数()
		后续消息, 错误 := 仓储.创建后续消息(执行上下文, 会话包.后续消息{
			标识: 标识(测试器), 会话标识: 本次执行.会话.标识, 预期执行标识: 本次执行.执行.标识,
			模型标识: 本次执行.执行.模型.标识, 状态: 状态, 内容: []模型包.内容块{{类型: "text", 文本: "接着核对目录"}},
		})
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		return 后续消息
	}
	检查 := func(消息 *会话包.消息, 之后 int64) {
		测试器.标记测试辅助函数()
		if 消息 == nil || 消息.引导之后步骤 == nil || *消息.引导之后步骤 != 之后 {
			测试器.格式化报告致命错误("引导应在步骤 %d 之后，实际为 %+v", 之后, 消息)
		}
	}
	历史 := func() map[string]会话包.消息 {
		测试器.标记测试辅助函数()
		页, 错误 := 仓储.历史(执行上下文, 本次执行.会话.标识, "", 1)
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		结果 := make(map[string]会话包.消息, len(页.消息集))
		for _, 消息 := range 页.消息集 {
			结果[消息.标识] = 消息
		}
		return 结果
	}
	添加步骤 := func(类别, 名称, 数据 string) 智能体包.步骤 {
		测试器.标记测试辅助函数()
		当前时间 := 时间包.当前时间().Unix毫秒()
		步骤, 错误 := 执行集.创建步骤(执行上下文, 智能体包.步骤{标识: 标识(测试器), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识,
			类型: 类别, 名称: 名称, 状态: 智能体包.待处理, 输入: JSON包.原始JSON消息(数据), 创建时间: 当前时间, 更新时间: 当前时间})
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		return 步骤
	}
	保存 := func(步骤 智能体包.步骤, 状态 string) {
		测试器.标记测试辅助函数()
		当前时间 := 时间包.当前时间().Unix毫秒()
		步骤.状态, 步骤.更新时间 = 状态, 当前时间
		if 状态 != 智能体包.运行中 {
			步骤.结束时间 = &当前时间
		}
		if 错误 := 执行集.保存步骤(执行上下文, 步骤); 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	提前 := 添加(会话包.后续消息引导)
	检查(提前.回执, 0)
	保存(本次执行.步骤, 智能体包.运行中)
	命令 := 添加步骤("tool_call", "run_command", `{}`)
	保存(命令, 智能体包.运行中)
	已排队 := 添加(会话包.后续消息已排队)
	引导, 错误 := 仓储.变更后续消息(执行上下文, 会话包.后续消息变更{
		会话标识: 本次执行.会话.标识, 标识: 已排队.标识, 修订号: 已排队.修订号,
		预期执行标识: 本次执行.执行.标识, 动作: "steer",
	}, nil)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	检查(引导.回执, 命令.序号)
	保存(本次执行.步骤, 智能体包.已完成) // 隔离中的模型步骤仍持有受理前的快照。
	已跳过 := 添加步骤("tool_call", "read_file", `{}`)
	保存(已跳过, 智能体包.失败)
	待处理 := 历史()
	检查(new(待处理[提前.标识]), 0)
	检查(new(待处理[引导.标识]), 命令.序号)

	请求 := 添加步骤("model_call", "继续生成回答", `{"steering_ids":["`+提前.标识+`","`+引导.标识+`"]}`)
	保存(请求, 智能体包.已完成)
	已更新, 错误 := 仓储.应用引导(执行上下文, 本次执行.执行.标识, 请求.标识, []string{提前.标识, 引导.标识})
	if 错误 != nil || len(已更新) != 2 {
		测试器.格式化报告致命错误("应用引导失败: %v, %v", 已更新, 错误)
	}
	检查(&已更新[0], 0)
	检查(&已更新[1], 命令.序号)
	已结束 := 历史()
	检查(new(已结束[提前.标识]), 0)
	检查(new(已结束[引导.标识]), 命令.序号)
}

func Test历史页集保留旧执行集并且独立压缩集(测试器 *测试包.测试器) {
	数据库, 错误 := SQLite包.打开(测试器.上下文(), 文件路径包.合并(测试器.创建临时目录(), "data", "callum.db"))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 数据库.关闭()
	仓储, _ := 存储包.新建仓储(数据库.数据库)
	执行集, _ := 智能体仓储包.新建仓储(数据库.数据库)
	var 会话标识 string
	var 旧执行 string
	for 索引 := range 205 {
		输入 := 输入(测试器)
		if 索引 > 0 {
			输入.创建 = false
			输入.会话标识 = 会话标识
		}
		本次执行, 错误 := 仓储.开始(测试器.上下文(), 输入)
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		会话标识 = 本次执行.会话.标识
		if 索引 == 0 {
			旧执行 = 本次执行.执行.标识
		}
		本次执行.执行.状态 = 智能体包.已完成
		本次执行.执行.结束时间 = new(时间包.当前时间().Unix毫秒())
		if 错误 := 执行集.结束(测试器.上下文(), 本次执行.执行); 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	// 密集的独立压缩不应隐藏消息，也不能被挂在更早的消息页末尾。
	for range 30 {
		运行 := 表结构包.执行{标识: 标识(测试器), 会话标识: 会话标识, 状态: 智能体包.已完成, 结束时间: new(int64(1))}
		if 错误 := 数据库.数据库.创建(&运行).错误; 错误 != nil {
			测试器.报告致命错误(错误)
		}
		步骤 := 表结构包.步骤{标识: 标识(测试器), 执行标识: 运行.标识, 会话标识: 会话标识, 序号: 1, 类型: "context_compression", 名称: "压缩历史上下文", 状态: 智能体包.已完成, 输入: []byte(`{"purpose":"context_compaction","trigger":{"type":"manual"}}`), 结束时间: new(int64(1))}
		if 错误 := 数据库.数据库.创建(&步骤).错误; 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	已见 := map[string]bool{}
	游标 := ""
	总计消息集 := 0
	for {
		页, 错误 := 仓储.历史(测试器.上下文(), 会话标识, 游标, 25)
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		if 游标 == "" && len(页.消息集) != 0 {
			测试器.报告致命错误("独立压缩分页混入了更早消息")
		}
		标识集 := map[string]bool{}
		for _, 运行 := range 页.执行集 {
			if 已见[运行.标识] || (游标 != "" && 运行.标识 >= 游标) {
				测试器.报告致命错误("执行游标未推进")
			}
			已见[运行.标识], 标识集[运行.标识] = true, true
		}
		for _, 消息 := range 页.消息集 {
			if !标识集[消息.执行标识] {
				测试器.报告致命错误("消息缺失执行")
			}
		}
		总计消息集 += len(页.消息集)
		if !页.还有更多 {
			break
		}
		游标 = 页.之前执行标识
	}
	if len(已见) != 235 || 总计消息集 != 410 || !已见[旧执行] {
		测试器.报告致命错误("长历史读取不完整", len(已见), 总计消息集)
	}
}
