func 新建标识(测试器 *测试包.测试器) string {

	值, 错误 := UUID包.新建UUIDv7()

func 测试夹具(测试器 *测试包.测试器) (*SQLite包.数据库, *存储包.仓储, 会话包.执行, 智能体包.步骤) {

	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 数据库.关闭(); 错误 != nil {

		}

	值, 错误 := 仓储.创建审批(测试器.上下文(), 智能体包.审批{标识: 新建标识(测试器), 执行标识: 步骤.执行标识, 步骤标识: 步骤.标识, 请求: 请求, 请求哈希: 十六进制包.编码目标字符串(校验和[:]), 状态: 智能体包.审批待处理, 创建时间: 1, 更新时间: 1})

func Test审批绑定并且原子操作认领(测试器 *测试包.测试器) {
	数据库, 仓储, 执行, 步骤 := 测试夹具(测试器)
	值 := 新建审批(测试器, 仓储, 步骤)
	行 := 表结构包.审批{标识: 新建标识(测试器), 执行标识: 执行.执行.标识, 步骤标识: 执行.步骤.标识, 请求: 值.请求, 请求哈希: 值.请求哈希, 状态: 智能体包.审批待处理, 创建时间: 1, 更新时间: 1}
	// 同一审批步骤的唯一性由数据库保证。
	行.步骤标识 = 步骤.标识
	if 错误 := 数据库.数据库.省略("Run", "Step").创建(&行).错误; 错误 == nil {
		测试器.报告致命错误("允许一个步骤多条审批")
	}
	// 已有步骤无法被另一个执行引用。
	外键 := 表结构包.执行{标识: 新建标识(测试器), 会话标识: 执行.会话.标识, 模型: 执行.执行.模型, 状态: 智能体包.已完成, 创建时间: 1, 更新时间: 1, 结束时间: new(int64(1))}
	if 错误 := 数据库.数据库.省略("Conversation", "Replaces").创建(&外键).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	行.执行标识, 行.步骤标识 = 外键.标识, 执行.步骤.标识
	if 错误 := 数据库.数据库.省略("Run", "Step").创建(&行).错误; 错误 == nil {
		测试器.报告致命错误("审批允许跨执行关联步骤")
	}
	if _, 错误 := 仓储.决定审批(测试器.上下文(), 值.标识, 智能体包.审批已批准, new("user"), nil); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 仓储.开始工具(测试器.上下文(), 步骤.执行标识, 步骤.标识, 值.标识, "changed"); !错误判断包.是否属于(错误, 智能体包.冲突错误) {
		测试器.格式化报告致命错误("未核对批准参数: %v", 错误)
	}
	var 等待组 同步包.等待组
	结果集 := make(chan error, 2)
	for range 2 {
		等待组.Go(func() {
			_, 错误 := 仓储.开始工具(测试器.上下文(), 步骤.执行标识, 步骤.标识, 值.标识, 值.请求哈希)
			结果集 <- 错误
		})
	}
	等待组.等待()
	close(结果集)
	成功集 := 0
	for 错误 := range 结果集 {
		if 错误 == nil {
			成功集++
		}
	}
	if 成功集 != 1 {
		测试器.格式化报告致命错误("工具被认领 %d 次", 成功集)
	}
}

func Test审批取消决定竞态并且恢复(测试器 *测试包.测试器) {
	for _, 动作 := range []string{"finish", "recover"} {
		测试器.执行(动作, func(测试器 *测试包.测试器) {
			_, 仓储, 执行, 步骤 := 测试夹具(测试器)
			值 := 新建审批(测试器, 仓储, 步骤)
			var 等待组 同步包.等待组
			错误集 := make(chan error, 2)
			等待组.Go(func() {
				_, 错误 := 仓储.决定审批(测试器.上下文(), 值.标识, 智能体包.审批已批准, new("user"), nil)
				if 错误判断包.是否属于(错误, 智能体包.冲突错误) {
					错误 = nil
				}
				错误集 <- 错误
			})
			等待组.Go(func() {
				if 动作 == "recover" {
					错误集 <- 仓储.恢复(测试器.上下文())
				} else {
					当前时间 := 时间包.当前时间().Unix毫秒()
					运行 := 执行.执行
					运行.状态, 运行.更新时间, 运行.结束时间 = 智能体包.已取消, 当前时间, &当前时间
					错误集 <- 仓储.结束(测试器.上下文(), 运行)
				}
			})
			等待组.等待()
			close(错误集)
			for 错误 := range 错误集 {
				if 错误 != nil {
					测试器.报告致命错误(错误)
				}
			}
			已保存, 错误 := 仓储.获取审批(测试器.上下文(), 值.标识)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			if 已保存.状态 == 智能体包.审批待处理 || 已保存.已决定位于 == nil {
				测试器.格式化报告致命错误("终止后仍有待办: %+v", 已保存)
			}
			if _, 错误 := 仓储.开始工具(测试器.上下文(), 步骤.执行标识, 步骤.标识, 值.标识, 值.请求哈希); 错误 == nil {
				测试器.报告致命错误("终止后启动了工具")
			}
			运行, 错误 := 仓储.获取(测试器.上下文(), 执行.执行.标识)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			预期 := 智能体包.已取消
			if 动作 == "recover" {
				预期 = 智能体包.已中断
			}
			if 运行.状态 != 预期 {
				测试器.报告致命错误(运行.状态)
			}
		})
	}
}
