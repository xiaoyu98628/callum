package 智能体

func 记录(输入 业务域包.步骤) 表结构包.步骤 {
	return 表结构包.步骤{标识: 输入.标识, 会话标识: 输入.会话标识, 执行标识: 输入.执行标识, 父级步骤标识: 输入.父级步骤标识, 序号: 输入.序号, 类型: 输入.类型, 名称: 输入.名称, 定义: 输入.定义, 状态: 输入.状态, 输入: 输入.输入, 输出: 输入.输出, 用量: 输入.用量, 模型: 输入.模型, 供应商请求标识: 输入.供应商请求标识, 工具调用标识: 输入.工具调用标识, 错误类别: 输入.错误类别, 错误消息: 输入.错误消息, 创建时间: 输入.创建时间, 开始时间: 输入.开始时间, 更新时间: 输入.更新时间, 结束时间: 输入.结束时间}

// 执行行 包含普通执行行与由步骤推导的恢复时间，不参与数据库结构检查。
type 执行行 struct {
	表结构包.执行 `gorm:"embedded"`
	已继续位于   *int64
}

// 执行 从 SQLite 查询投影转换执行状态，供历史与独立执行查询共享。
func 执行(行 执行行) 业务域包.执行 {
	return 业务域包.执行{标识: 行.标识, 会话标识: 行.会话标识, 替换执行标识: 行.替换执行标识, 模型: 行.模型, 流: 行.流, 设置快照: 行.设置快照, 状态: 行.状态, 错误类别: 行.错误类别, 错误消息: 行.错误消息, 创建时间: 行.创建时间, 已继续位于: 行.已继续位于, 更新时间: 行.更新时间, 结束时间: 行.结束时间}
}

// 带有继续时间 投影最新恢复步骤的受理时间，保留执行原创建时间用于排序。
func 带有继续时间(数据库 *GORM包.数据库) *GORM包.数据库 {
	最新 := 继续步骤集(数据库.会话连接(&GORM包.会话连接{新建数据库: true})).选择("created_at").
		条件("run_id=conversation_runs.id").顺序("sequence DESC").上限(1)
	return 数据库.模型(&表结构包.执行{}).选择("conversation_runs.*, (?) AS resumed_at", 最新)
}
