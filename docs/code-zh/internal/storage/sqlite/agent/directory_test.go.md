package 智能体测试

import (
	SHA256包 "crypto/sha256"
	十六进制包 "encoding/hex"
	JSON包 "encoding/json"
	错误判断包 "errors"
	文件路径包 "path/filepath"
	测试包 "testing"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

func Test目录撤销之前原子操作认领(测试器 *测试包.测试器) {
	_, 当前读取器, 本次执行, 步骤 := 测试夹具(测试器)
	根 := 测试器.创建临时目录()
	范围 := 工具包.目录范围{根路径: 根, 附件访问器: "read"}
	请求, 错误 := JSON包.序列化(map[string]any{"tool": 工具包.定义{影响集: []工具包.影响{工具包.影响读取}}, "target": 工具包.目标{路径: 根, 类别: "directory"}, "directory_scope": 范围})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	步骤.输入, 错误 = JSON包.序列化(map[string]any{"approval_policy": "on-request", "authorization": map[string]any{"action": "ask", "reason": "outside_workspace"}})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 当前读取器.保存步骤(测试器.上下文(), 步骤); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	校验和 := SHA256包.校验和256(请求)
	甲, 错误 := 当前读取器.创建审批(测试器.上下文(), 智能体包.审批{标识: 新建标识(测试器), 执行标识: 步骤.执行标识, 步骤标识: 步骤.标识, 请求: 请求, 请求哈希: 十六进制包.编码目标字符串(校验和[:]), 创建时间: 1, 更新时间: 1})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	甲, 错误 = 当前读取器.决定审批(测试器.上下文(), 甲.标识, 智能体包.审批已批准, new("user"), nil, 智能体包.审批会话目录)
	if 错误 != nil || 甲.目录授权 == nil {
		测试器.格式化报告致命错误("未创建授权: %+v %v", 甲, 错误)
	}
	if _, 错误 := 当前读取器.决定审批(测试器.上下文(), 甲.标识, 智能体包.审批已批准, new("user"), nil, 智能体包.审批会话目录); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 当前读取器.开始工具(测试器.上下文(), 步骤.执行标识, 步骤.标识, 甲.标识, 甲.请求哈希); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	继承 := 步骤
	继承.标识 = 新建标识(测试器)
	继承.状态 = 智能体包.待处理
	继承.序号 = 0
	继承.输入, 错误 = JSON包.序列化(map[string]any{"approval_policy": "on-request", "authorization": map[string]any{"action": "allow", "reason": "conversation_directory_grant"}, "directory_grant_ids": []string{甲.目录授权.标识}, "request": map[string]any{"tool": 工具包.定义{影响集: []工具包.影响{工具包.影响读取}}, "target": 工具包.目标{路径: 文件路径包.合并(根, "child.txt"), 类别: "file"}}})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	继承, 错误 = 当前读取器.创建步骤(测试器.上下文(), 继承)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 当前读取器.撤销目录授权(测试器.上下文(), 本次执行.会话.标识, 甲.目录授权.标识); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 当前读取器.开始工具(测试器.上下文(), 步骤.执行标识, 继承.标识, "", "unused"); !错误判断包.是否属于(错误, 工具包.已拒绝错误) {
		测试器.格式化报告致命错误("撤销后仍可认领工具: %v", 错误)
	}
	if _, 错误 := 当前读取器.决定审批(测试器.上下文(), 甲.标识, 智能体包.审批已批准, new("user"), nil, 智能体包.审批会话目录); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	授权集, 错误 := 当前读取器.列表目录授权集(测试器.上下文(), 本次执行.会话.标识, 100, 0)
	if 错误 != nil || len(授权集) != 1 || 授权集[0].撤销时间 == nil {
		测试器.格式化报告致命错误("幂等提交重新激活了撤销授权: %+v %v", 授权集, 错误)
	}
}
