package 智能体测试

import (

	时间包 "time"

func 上下文输出(测试器 *测试包.测试器, 消息 模型包.消息, 字段集 map[string]any) JSON包.原始JSON消息 {
	测试器.标记测试辅助函数()
	if 字段集 == nil {
		字段集 = map[string]any{}
	}
	字段集["context_message"] = 消息
	原始, 错误 := JSON包.序列化(字段集)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	return 原始
}

func 操作测试夹具(测试器 *测试包.测试器, 带版本 bool) (*SQLite包.数据库, *存储包.仓储, 会话包.执行, 智能体包.步骤) {
	测试器.标记测试辅助函数()
	数据库, 当前读取器, 本次执行, 步骤 := 测试夹具(测试器)
	步骤.工具调用标识 = new("call-a")
	if 错误 := 数据库.数据库.模型(&表结构包.步骤{}).条件("id=?", 步骤.标识).更新("tool_call_id", *步骤.工具调用标识).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	步骤.定义 = JSON包.原始JSON消息(`{"effects":["write"]}`)
	if 带版本 {
		步骤.输入 = JSON包.原始JSON消息(`{"operation_version":1,"request_hash":"fixed-operation"}`)
	}
	if 错误 := 当前读取器.保存步骤(测试器.上下文(), 步骤); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	父级 := 本次执行.步骤
	父级.状态 = 智能体包.已完成
	父级.结束时间 = new(int64(2))
	父级.输出 = 上下文输出(测试器, 模型包.消息{角色: 模型包.角色助手, 内容: "尚未完成的说明", 工具调用集: []模型包.工具调用{{标识: "call-a", 名称: "operation", 参数: JSON包.原始JSON消息(`{}`)}, {标识: "call-b", 名称: "operation", 参数: JSON包.原始JSON消息(`{}`)}}}, nil)
	if 错误 := 当前读取器.保存步骤(测试器.上下文(), 父级); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	return 数据库, 当前读取器, 本次执行, 步骤
}

func 保存操作(测试器 *测试包.测试器, 当前仓储 *存储包.仓储, 步骤 智能体包.步骤, 操作结果 工具包.操作结果, 带版本 bool) {
	测试器.标记测试辅助函数()
	字段集 := map[string]any{"result": map[string]any{"written": true}}
	if 带版本 {
		字段集["operation"] = 工具包.操作结果记录{版本: 1, 操作结果: 操作结果, 请求哈希: "fixed-operation"}
	}
	正文, 错误 := JSON包.序列化(字段集)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	步骤.输出 = 上下文输出(测试器, 模型包.消息{角色: 模型包.角色工具, 工具调用标识: *步骤.工具调用标识, 内容: string(正文)}, 字段集)
	步骤.状态 = 智能体包.已完成
	步骤.开始时间, 步骤.结束时间 = new(int64(2)), new(int64(3))
	if 错误 := 当前仓储.保存步骤(测试器.上下文(), 步骤); 错误 != nil {
		测试器.报告致命错误(错误)
	}
}

func 结束操作执行(测试器 *测试包.测试器, 当前仓储 *存储包.仓储, 本次执行 会话包.执行, 状态 string) {
	测试器.标记测试辅助函数()
	本次执行.执行.状态 = 状态
	本次执行.执行.更新时间 = 时间包.当前时间().Unix毫秒()
	本次执行.执行.结束时间 = new(本次执行.执行.更新时间)
	if 错误 := 当前仓储.结束(测试器.上下文(), 本次执行.执行); 错误 != nil {
		测试器.报告致命错误(错误)
	}
}

func Test失败执行集恢复已配对影响集无失败回答(测试器 *测试包.测试器) {
	for _, 状态 := range []string{智能体包.失败, 智能体包.已取消, 智能体包.已中断} {
		for _, 操作结果 := range []工具包.操作结果{工具包.操作结果已生效, 工具包.操作结果部分, 工具包.操作结果未知} {
			测试器.执行(状态+"_"+string(操作结果), func(测试器 *测试包.测试器) {
				_, 当前读取器, 本次执行, 步骤 := 操作测试夹具(测试器, true)
				保存操作(测试器, 当前读取器, 步骤, 操作结果, true)
				结束操作执行(测试器, 当前读取器, 本次执行, 状态)
				执行轨迹, 错误 := 当前读取器.执行轨迹(测试器.上下文(), 本次执行.会话.标识)
				if 错误 != nil {
					测试器.报告致命错误(错误)
				}
				消息集 := 执行轨迹[本次执行.执行.标识]
				if len(消息集) != 3 || 消息集[0].内容 != "" || len(消息集[0].工具调用集) != 2 || 消息集[1].工具调用标识 != "call-a" || 消息集[2].工具调用标识 != "call-b" {
					测试器.格式化报告致命错误("事实或配对丢失: %+v", 消息集)
				}
				var 实际, 不已开始 struct {
					操作 工具包.操作结果记录 `json:"operation"`
					结果 struct {
						已写入 bool `json:"written"`
					} `json:"result"`
				}
				if JSON包.反序列化([]byte(消息集[1].内容), &实际) != nil || 实际.操作.操作结果 != 操作结果 || !实际.结果.已写入 {
					测试器.格式化报告致命错误("实际输出丢失: %s", 消息集[1].内容)
				}
				if JSON包.反序列化([]byte(消息集[2].内容), &不已开始) != nil || 不已开始.操作.操作结果 != 工具包.操作结果不已生效 {
					测试器.格式化报告致命错误("未执行调用被错误恢复: %s", 消息集[2].内容)
				}
			})
		}
	}
}

func Test恢复结束仅新建未完成操作集(测试器 *测试包.测试器) {
	for _, 动作 := range []string{"recover", "finish"} {
		for _, 测试 := range []struct {
			名称       string
			已开始, 带版本 bool
			检查点      工具包.操作结果
			预期       工具包.操作结果
		}{
			{"pending", false, true, "", 工具包.操作结果不已生效},
			{"started", true, true, "", 工具包.操作结果未知},
			{"checkpoint", true, true, 工具包.操作结果未知, 工具包.操作结果未知},
			{"known_applied", true, true, 工具包.操作结果已生效, 工具包.操作结果已生效},
			{"legacy", true, false, "", ""},
		} {
			测试器.执行(动作+"_"+测试.名称, func(测试器 *测试包.测试器) {
				_, 当前读取器, 本次执行, 步骤 := 操作测试夹具(测试器, 测试.带版本)
				if 测试.已开始 {
					var 错误 error
					步骤, 错误 = 当前读取器.开始工具(测试器.上下文(), 本次执行.执行.标识, 步骤.标识, "", "fixed-operation")
					if 错误 != nil {
						测试器.报告致命错误(错误)
					}
				}
				if 测试.检查点 != "" {
					步骤.输出 = 上下文输出(测试器, 模型包.消息{角色: 模型包.角色工具, 工具调用标识: "call-a", 内容: `{}`}, map[string]any{"operation": 工具包.操作结果记录{版本: 1, 操作结果: 测试.检查点, 请求哈希: "fixed-operation"}, "result": map[string]bool{"written": true}})
					if 错误 := 当前读取器.保存步骤(测试器.上下文(), 步骤); 错误 != nil {
						测试器.报告致命错误(错误)
					}
				}
				if 动作 == "recover" {
					if 错误 := 当前读取器.恢复(测试器.上下文()); 错误 != nil {
						测试器.报告致命错误(错误)
					}
				} else {
					结束操作执行(测试器, 当前读取器, 本次执行, 智能体包.已取消)
				}
				已保存, 错误 := 当前读取器.获取步骤(测试器.上下文(), 步骤.标识)
				if 错误 != nil {
					测试器.报告致命错误(错误)
				}
				if !测试.带版本 {
					if len(已保存.输出) != 0 {
						测试器.格式化报告致命错误("替旧记录推断了事实: %s", 已保存.输出)
					}
				} else {
					var 值 struct {
						操作 工具包.操作结果记录     `json:"operation"`
						结果 JSON包.原始JSON消息 `json:"result"`
					}
					if JSON包.反序列化(已保存.输出, &值) != nil || 值.操作.操作结果 != 测试.预期 || !值.操作.已恢复 {
						测试器.格式化报告致命错误("恢复事实错误: %s", 已保存.输出)
					}
					if 测试.检查点 != "" && !字符串包.包含(string(值.结果), `"written":true`) {
						测试器.报告致命错误("恢复覆盖了实际输出")
					}
				}
				执行轨迹, 错误 := 当前读取器.执行轨迹(测试器.上下文(), 本次执行.会话.标识)
				if 错误 != nil {
					测试器.报告致命错误(错误)
				}
				if (len(执行轨迹[本次执行.执行.标识]) > 0) != 测试.预期.可能拥有影响集() {
					测试器.格式化报告致命错误("恢复上下文范围错误: %+v", 执行轨迹)
				}
				// 重复恢复不修改已结束记录。
				if 错误 := 当前读取器.恢复(测试器.上下文()); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				再次, 错误 := 当前读取器.获取步骤(测试器.上下文(), 步骤.标识)
				if 错误 != nil || string(再次.输出) != string(已保存.输出) {
					测试器.格式化报告致命错误("重复恢复改变事实: %v", 错误)
				}
			})
		}
	}
}

func Test旧版失败并且已重发审计不进入上下文(测试器 *测试包.测试器) {
	for _, 旧版 := range []bool{true, false} {
		测试器.执行(map[bool]string{true: "legacy_failure", false: "resent_audit"}[旧版], func(测试器 *测试包.测试器) {
			数据库, 当前读取器, 本次执行, 步骤 := 操作测试夹具(测试器, !旧版)
			保存操作(测试器, 当前读取器, 步骤, 工具包.操作结果已生效, !旧版)
			结束操作执行(测试器, 当前读取器, 本次执行, 智能体包.失败)
			if !旧版 {
				会话集, 错误 := 会话仓储包.新建仓储(数据库.数据库)
				if 错误 != nil {
					测试器.报告致命错误(错误)
				}
				_, 错误 = 会话集.开始(测试器.上下文(), 会话包.开始{会话标识: 本次执行.会话.标识, 执行标识: 新建标识(测试器), 用户消息标识: 新建标识(测试器), 助手消息标识: 新建标识(测试器), 步骤标识: 新建标识(测试器), 编辑重发消息标识: 本次执行.用户.标识, 模型: 本次执行.执行.模型, 协议: "chatcompletions", 内容: []模型包.内容块{{类型: "text", 文本: "更正问题"}}})
				if 错误 != nil {
					测试器.报告致命错误(错误)
				}
			}
			执行轨迹, 错误 := 当前读取器.执行轨迹(测试器.上下文(), 本次执行.会话.标识)
			if 错误 != nil || len(执行轨迹) != 0 {
				测试器.格式化报告致命错误("旧失败或重发分支进入上下文: %+v %v", 执行轨迹, 错误)
			}
			// 审计步骤与真实操作事实仍保存在数据库中。
			已保存, 错误 := 当前读取器.获取步骤(测试器.上下文(), 步骤.标识)
			if 错误 != nil || !字符串包.包含(string(已保存.输出), `"written":true`) {
				测试器.格式化报告致命错误("审计被删除: %s %v", 已保存.输出, 错误)
			}
		})
	}
}
