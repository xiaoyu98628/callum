	会话集, 错误 := 会话仓储包.新建仓储(数据库.数据库)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}

	if 创建 {
		标识 = 新建标识(测试器)
	}

func 事实步骤(测试器 *测试包.测试器, 本次执行 会话包.执行, 哈希 string, 操作结果 工具包.操作结果, 失败 bool) 智能体包.步骤 {
	测试器.标记测试辅助函数()
	字段集 := map[string]any{"operation": 工具包.操作结果记录{版本: 工具包.操作版本, 请求哈希: 哈希, 操作结果: 操作结果}}

		字段集["error"] = map[string]string{"kind": "tool_failed"}
	}
	输入, 错误 := JSON包.序列化(map[string]any{"operation_version": 工具包.操作版本, "request_hash": 哈希})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	输出, 错误 := JSON包.序列化(字段集)

		测试器.报告致命错误(错误)
	}

func Test操作事实集使用最新结果并且当前分支(测试器 *测试包.测试器) {
	数据库, 当前读取器, 本次执行, _ := 测试夹具(测试器)
	追加事实(测试器, 当前读取器, 本次执行, "cleared", 工具包.操作结果未知, false)
	追加事实(测试器, 当前读取器, 本次执行, "cleared", 工具包.操作结果已生效, false)
	部分 := 追加事实(测试器, 当前读取器, 本次执行, "partial", 工具包.操作结果部分, false)
	追加事实(测试器, 当前读取器, 本次执行, "partial", 工具包.操作结果不已生效, false)
	失败 := 追加事实(测试器, 当前读取器, 本次执行, "failed", 工具包.操作结果已生效, true)
	追加事实(测试器, 当前读取器, 本次执行, "", 工具包.操作结果未知, false)
	结束操作执行(测试器, 当前读取器, 本次执行, 智能体包.已完成)
	下一个 := 开始查询执行(测试器, 数据库, 本次执行, false, "")
	最新 := 追加事实(测试器, 当前读取器, 下一个, "cleared", 工具包.操作结果未知, false)
	外键 := 开始查询执行(测试器, 数据库, 本次执行, true, "")
	追加事实(测试器, 当前读取器, 外键, "cleared", 工具包.操作结果已生效, false)
	追加事实(测试器, 当前读取器, 外键, "other_conversation", 工具包.操作结果未知, false)
	事实集, 错误 := 当前读取器.操作事实集(测试器.上下文(), 本次执行.会话.标识)
	if 错误 != nil || len(事实集) != 3 {
		测试器.格式化报告致命错误("操作去重或会话隔离错误: %+v %v", 事实集, 错误)
	}
	预期 := map[string]智能体包.步骤{"cleared": 最新, "partial": 部分, "failed": 失败}
	for 索引, 事实 := range 事实集 {
		步骤, 成功 := 预期[事实.请求哈希]
		if !成功 || 事实.步骤标识 != 步骤.标识 || 事实.执行标识 != 步骤.执行标识 || (索引 > 0 && 事实集[索引-1].步骤标识 >= 事实.步骤标识) {
			测试器.格式化报告致命错误("没有按当前分支的最新结果返回: %+v", 事实集)
		}
	}
	结束操作执行(测试器, 当前读取器, 下一个, 智能体包.已完成)
	开始查询执行(测试器, 数据库, 下一个, false, 下一个.用户.标识)
	事实集, 错误 = 当前读取器.操作事实集(测试器.上下文(), 本次执行.会话.标识)
	if 错误 != nil || len(事实集) != 2 {
		测试器.格式化报告致命错误("重发后仍恢复旧分支事实: %+v %v", 事实集, 错误)
	}
	for _, 事实 := range 事实集 {
		if 事实.请求哈希 == "cleared" {
			测试器.格式化报告致命错误("旧成功之前的未知结果重新进入上下文: %+v", 事实集)
		}
	}
}
