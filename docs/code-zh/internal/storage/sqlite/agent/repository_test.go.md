func Test消息保存保留取消并且后续写入(测试器 *测试包.测试器) {
	数据库, 仓储, 执行, _ := 测试夹具(测试器)
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	defer 取消函数()
	const 回调 = "controlled_cancel_message_save"
	if 错误 := 数据库.数据库.回调().更新().之前("gorm:update").注册(回调, func(事务 *GORM包.数据库) {
		if 事务.语句.上下文 != 执行上下文 {
			return
		}
		取消函数()
		// 将隐式事务固定在已回滚状态，复现 数据库/SQL 的取消回滚竞态。
		// 这与调度器先完成自动回滚的结果一致，不依赖随机等待。
		if 事务, 成功 := 事务.语句.连接池.(*SQL包.事务); 成功 {
			if 错误 := 事务.回滚(); 错误 != nil && !错误判断包.是否属于(错误, SQL包.事务完成错误) {
				测试器.错误文本(错误)
			}
		}
		事务.添加错误(执行上下文.获取错误())
	}); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 数据库.数据库.回调().更新().移除(回调); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	var 之前, 之后 表结构包.消息
	if 错误 := 数据库.数据库.条件("id=?", 执行.助手.标识).首个(&之前).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	错误 := 仓储.保存消息(执行上下文, 执行.助手.标识, []模型包.内容块{{类型: "text", 文本: "must not be written"}}, 智能体包.运行中)
	if !错误判断包.是否属于(错误, 上下文包.已取消错误) {
		测试器.格式化报告致命错误("取消错误类型被回滚覆盖: %v", 错误)
	}
	if 错误 := 数据库.数据库.条件("id=?", 执行.助手.标识).首个(&之后).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if !切片包.相等(之前.内容, 之后.内容) || 之前.状态 != 之后.状态 {
		测试器.报告致命错误("取消的写入修改了原消息")
	}
	if 错误 := 仓储.保存消息(测试器.上下文(), 执行.助手.标识, []模型包.内容块{{类型: "text", 文本: "subsequent write"}}, 智能体包.运行中); 错误 != nil {
		测试器.格式化报告致命错误("取消污染了后续写入: %v", 错误)
	}
}

func Test消息保存保留真实数据库错误判断(测试器 *测试包.测试器) {
	数据库, 仓储, 执行, _ := 测试夹具(测试器)
	if 错误 := 数据库.数据库.进程执行(`CREATE TRIGGER reject_message_save BEFORE UPDATE ON conversation_messages BEGIN SELECT RAISE(ABORT,'controlled message write failure'); END`).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	错误 := 仓储.保存消息(测试器.上下文(), 执行.助手.标识, []模型包.内容块{{类型: "text", 文本: "rejected"}}, 智能体包.运行中)
	if 错误 == nil || 错误判断包.是否属于(错误, 上下文包.已取消错误) || !字符串包.包含(错误.错误文本(), "controlled message write failure") {
		测试器.格式化报告致命错误("真实数据库错误被丢失: %v", 错误)
	}
	if 错误 := 数据库.数据库.进程执行("DROP TRIGGER reject_message_save").错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 仓储.保存消息(测试器.上下文(), 执行.助手.标识, []模型包.内容块{{类型: "text", 文本: "after failed write"}}, 智能体包.运行中); 错误 != nil {
		测试器.报告致命错误(错误)
	}
}

func Test消息保存保留调用方事务(测试器 *测试包.测试器) {
	数据库, _, 执行, _ := 测试夹具(测试器)
	var 之前, 之后 表结构包.消息
	if 错误 := 数据库.数据库.条件("id=?", 执行.助手.标识).首个(&之前).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	回滚 := 错误判断包.新建("controlled caller rollback")
	错误 := 数据库.数据库.带有上下文(测试器.上下文()).事务(func(事务 *GORM包.数据库) error {
		仓储, 错误 := 存储包.新建仓储(事务)
		if 错误 != nil {
			return 错误
		}
		if 错误 := 仓储.保存消息(测试器.上下文(), 执行.助手.标识, []模型包.内容块{{类型: "text", 文本: "inside caller transaction"}}, 智能体包.运行中); 错误 != nil {
			return 错误
		}
		return 回滚
	})
	if !错误判断包.是否属于(错误, 回滚) {
		测试器.格式化报告致命错误("调用方事务错误丢失: %v", 错误)
	}
	if 错误 := 数据库.数据库.条件("id=?", 执行.助手.标识).首个(&之后).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if !切片包.相等(之前.内容, 之后.内容) || 之前.状态 != 之后.状态 {
		测试器.报告致命错误("消息写入绕过了调用方事务")
	}
}

func Test步骤查询集读取元数据并且UTF8Windows(测试器 *测试包.测试器) {
	数据库, 仓储, 执行, 工具步骤 := 测试夹具(测试器)
	执行上下文 := 测试器.上下文()
	内容 := "a🙂中\x00尾"
	输出 := `{"reasoning":"a🙂中\u0000尾","context_message":{"role":"assistant","content":"a🙂中\u0000尾"},"provider_extra":"` +
		字符串包.重复("源", 1<<18) + `"}`
	for _, 标识 := range []string{执行.步骤.标识, 工具步骤.标识} {
		// 模拟旧记录，JSON 仓储解码只去掉顶层前后的空白。
		if 错误 := 数据库.数据库.模型(&表结构包.步骤{}).条件("id=?", 标识).
			更新集(map[string]any{"output": "\t" + 输出 + "\r\n", "usage": `{"total_tokens":5}`, "definition": `{"tools":[]}`}).错误; 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	for _, 之后 := range []int64{0, 1} {
		摘要集, 错误 := 仓储.步骤摘要集(执行上下文, 执行.执行.标识, 之后, 1)
		if 错误 != nil || len(摘要集) != 1 || 摘要集[0].序号 != 之后+1 {
			测试器.格式化报告致命错误("步骤游标错误: %+v %v", 摘要集, 错误)
		}
		摘要 := 摘要集[0]
		完整, 错误 := 仓储.获取步骤(执行上下文, 摘要.标识)
		if 错误 != nil || !反射包.深度相等(摘要, 智能体包.生成摘要步骤(完整)) || 摘要.输出字节 != len(输出) {
			测试器.格式化报告致命错误("列表元数据与旧查询不一致: %+v %v", 摘要, 错误)
		}
		单个, 错误 := 仓储.获取步骤摘要(执行上下文, 摘要.标识)
		if 错误 != nil || !反射包.深度相等(摘要, 单个) {
			测试器.格式化报告致命错误("单个步骤的元数据与列表不一致: %+v %v", 单个, 错误)
		}
	}
	步骤标识 := 执行.步骤.标识
	for _, 测试用例 := range []struct {
		字段, 预期 string
		偏移     int
		下一个    int
		总计     int
		存在     bool
	}{
		{字段: "output", 预期: 输出[:4], 下一个: 4, 总计: len(输出), 存在: true},
		{字段: "reasoning", 预期: "a", 下一个: 1, 总计: len(内容), 存在: true},
		{字段: "reasoning", 预期: "🙂", 偏移: 1, 下一个: 5, 总计: len(内容), 存在: true},
		{字段: "reasoning", 预期: "中\x00", 偏移: 5, 下一个: 9, 总计: len(内容), 存在: true},
		{字段: "reasoning", 偏移: len(内容), 下一个: len(内容), 总计: len(内容), 存在: true},
		{字段: "text", 预期: "a", 下一个: 1, 总计: len(内容), 存在: true},
		{字段: "text", 预期: "🙂", 偏移: 1, 下一个: 5, 总计: len(内容), 存在: true},
		{字段: "text", 预期: "中\x00", 偏移: 5, 下一个: 9, 总计: len(内容), 存在: true},
		{字段: "usage", 预期: `{"to`, 下一个: 4, 总计: len(`{"total_tokens":5}`), 存在: true},
		{字段: "definition", 预期: `{"to`, 下一个: 4, 总计: len(`{"tools":[]}`), 存在: true},
	} {
		预期 := 智能体包.步骤数据{存在: 测试用例.存在, 片段: 测试用例.预期, 总计字节: 测试用例.总计, 下一个偏移: 测试用例.下一个}
		if 实际值, 错误 := 仓储.步骤数据(执行上下文, 步骤标识, 测试用例.字段, 测试用例.偏移, 4); 错误 != nil || 实际值 != 预期 {
			测试器.格式化报告致命错误("%s at %d: %+v %v, want %+v", 测试用例.字段, 测试用例.偏移, 实际值, 错误, 预期)
		}
	}
	for _, 字段 := range []string{"reasoning", "text"} {
		for _, 偏移 := range []int{2, len(内容) + 1, 数学包.最大整数} {
			if _, 错误 := 仓储.步骤数据(执行上下文, 步骤标识, 字段, 偏移, 4); !错误判断包.是否属于(错误, 智能体包.无效错误) {
				测试器.格式化报告致命错误("%s 接受无效的字节偏移 %d: %v", 字段, 偏移, 错误)
			}
		}
		if 实际值, 错误 := 仓储.步骤数据(执行上下文, 工具步骤.标识, 字段, 0, 4); 错误 != nil || 实际值 != (智能体包.步骤数据{}) {
			测试器.格式化报告致命错误("非模型步骤暴露了 %s: %+v %v", 字段, 实际值, 错误)
		}
	}
	for _, 测试用例 := range []struct {
		字段 string
		原始 any
		存在 bool
		错误 error
	}{
		{字段: "reasoning", 原始: nil},
		{字段: "reasoning", 原始: `{"text":"旧记录"}`},
		{字段: "reasoning", 原始: `{"reasoning":null}`},
		{字段: "reasoning", 原始: `{"reasoning":""}`, 存在: true},
		{字段: "reasoning", 原始: `{"reasoning":23}`, 错误: 智能体包.步骤推理错误},
		{字段: "reasoning", 原始: `[]`, 错误: 智能体包.步骤推理错误},
		{字段: "text", 原始: nil},
		{字段: "text", 原始: `{"text":"旧记录"}`},
		{字段: "text", 原始: `{"context_message":{"role":"assistant","content":null}}`},
		{字段: "text", 原始: `{"context_message":{"role":"assistant","content":""}}`, 存在: true},
		{字段: "text", 原始: `{"context_message":{"role":"assistant","content":23}}`},
		{字段: "text", 原始: `{"context_message":{"role":"tool","content":"工具结果"}}`},
	} {
		if 错误 := 数据库.数据库.模型(&表结构包.步骤{}).条件("id=?", 步骤标识).更新("output", 测试用例.原始).错误; 错误 != nil {
			测试器.报告致命错误(错误)
		}
		实际值, 错误 := 仓储.步骤数据(执行上下文, 步骤标识, 测试用例.字段, 0, 4)
		if !错误判断包.是否属于(错误, 测试用例.错误) || 实际值 != (智能体包.步骤数据{存在: 测试用例.存在}) {
			测试器.格式化报告致命错误("%s 字段 %v: %+v %v", 测试用例.字段, 测试用例.原始, 实际值, 错误)
		}
	}
}
