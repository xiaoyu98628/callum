```go
package 智能体测试

import (
	上下文包 "context"
	JSON包 "encoding/json"
	错误判断包 "errors"
	输入输出包 "io"
	结构化日志包 "log/slog"
	测试包 "testing"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	附件包 "github.com/xiaoyu98628/callum/internal/attachment"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	会话仓储包 "github.com/xiaoyu98628/callum/internal/storage/sqlite/conversation"
	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

func Test恢复已发布交付物集输入当前分支(测试器 *测试包.测试器) {
	数据库, 执行集, 本次执行, _ := 测试夹具(测试器)
	执行上下文 := 测试器.上下文()
	会话集, 错误 := 会话仓储包.新建仓储(数据库.数据库)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	存储, 错误 := 附件包.打开(执行上下文, 测试器.创建临时目录(), func(上下文包.上下文, string, string) (bool, error) {
		return true, nil
	}, 结构化日志包.新建(结构化日志包.新建文本处理器(输入输出包.丢弃, nil)))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 存储.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	访问, 错误 := 附件包.打开访问(执行上下文, 测试器.创建临时目录(), 存储, 会话集.附件)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 访问.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	if _, 错误 := 访问.开始(执行上下文, 本次执行.会话.标识, 本次执行.执行.标识); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	const 原始 = "a durable report"
	报告, 错误 := 访问.保存已生成(执行上下文, 本次执行.会话.标识, 本次执行.执行.标识, "file", "report.txt", []byte(原始), true)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	失败, 错误 := 访问.保存已生成(执行上下文, 本次执行.会话.标识, 本次执行.执行.标识, "file", "failed.txt", []byte("another file"), true)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	定义 := JSON包.原始JSON消息(`{"source":"builtin","name":"publish_artifact"}`)
	for _, 发布 := range []struct {
		内容块 模型包.内容块
		状态  string
	}{{报告, 智能体包.已完成}, {报告, 智能体包.已完成}, {失败, 智能体包.失败}} {
		步骤, 错误 := 执行集.创建步骤(执行上下文, 智能体包.步骤{
			标识: 新建标识(测试器), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识,
			父级步骤标识: &本次执行.步骤.标识, 工具调用标识: new("call"), 类型: "tool_call", 名称: "publish_artifact",
			状态: 智能体包.待处理, 定义: 定义, 输入: JSON包.原始JSON消息(`{}`), 创建时间: 1, 更新时间: 1,
		})
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		步骤.状态, 步骤.结束时间 = 发布.状态, new(int64(2))
		步骤.输出, 错误 = JSON包.序列化(map[string]any{
			"operation": 工具包.操作结果记录{版本: 工具包.操作版本, 操作结果: 工具包.操作结果无},
			"result":    map[string]any{"attachment": 发布.内容块, "published": true},
		})
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		if 错误 := 执行集.保存步骤(执行上下文, 步骤); 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	if 错误 := 执行集.保存消息(执行上下文, 本次执行.助手.标识, []模型包.内容块{{类型: "text", 文本: "partial answer"}}, 智能体包.运行中); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	// 模拟只持久化了工具结果的一次中断，以及服务再次启动。
	for range 2 {
		if 错误 := 执行集.恢复(执行上下文); 错误 != nil {
			测试器.报告致命错误(错误)
		}
		消息, 错误 := 会话集.消息(执行上下文, 本次执行.会话.标识, 本次执行.助手.标识)
		if 错误 != nil || 消息.状态 != 智能体包.已中断 || len(消息.内容) != 2 ||
			消息.内容[0].文本 != "partial answer" || 消息.内容[1] != 报告 {
			测试器.格式化报告致命错误("交付引用未恢复或重复恢复: %+v %v", 消息, 错误)
		}
	}
	内容块, 错误 := 会话集.附件(执行上下文, 本次执行.会话.标识, 报告.附件引用)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	数据, _, 错误 := 存储.读取(执行上下文, 内容块.附件引用)
	if 错误 != nil || string(数据) != 原始 {
		测试器.格式化报告致命错误("恢复后的交付物不可读: %q %v", 数据, 错误)
	}
	if _, 错误 := 会话集.附件(执行上下文, 本次执行.会话.标识, 失败.附件引用); !错误判断包.是否属于(错误, 会话包.未找到错误) {
		测试器.格式化报告致命错误("恢复了失败步骤的交付物: %v", 错误)
	}
	_, 错误 = 会话集.开始(执行上下文, 会话包.开始{
		会话标识: 本次执行.会话.标识, 执行标识: 新建标识(测试器), 步骤标识: 新建标识(测试器), 助手消息标识: 新建标识(测试器),
		编辑重发消息标识: 本次执行.用户.标识, 编辑重发执行标识: 本次执行.执行.标识, 模型: 本次执行.执行.模型, 协议: "chatcompletions",
		内容: []模型包.内容块{{类型: "text", 文本: "resend"}},
	})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 执行集.恢复(执行上下文); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 会话集.附件(执行上下文, 本次执行.会话.标识, 报告.附件引用); !错误判断包.是否属于(错误, 会话包.未找到错误) {
		测试器.格式化报告致命错误("恢复了已替换分支的交付物: %v", 错误)
	}
}
```
