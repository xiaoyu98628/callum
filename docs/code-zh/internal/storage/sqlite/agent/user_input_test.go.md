package 智能体测试

import (
	JSON包 "encoding/json"
	错误判断包 "errors"
	同步包 "sync"
	测试包 "testing"
	时间包 "time"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
)

func Test用户输入取消竞态并且恢复(测试器 *测试包.测试器) {
	for _, 恢复 := range []bool{false, true} {
		_, 仓储, 本次执行, _ := 测试夹具(测试器)
		当前时间 := 时间包.当前时间().Unix毫秒()
		输入, 错误 := 仓储.创建用户输入(测试器.上下文(), 智能体包.步骤{标识: 新建标识(测试器), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 父级步骤标识: new(本次执行.步骤.标识), 工具调用标识: new("call-question"), 类型: "user_input", 名称: "问题", 状态: 智能体包.等待中用户输入, 输入: JSON包.原始JSON消息(`{"question":"请选择","options":["选项"]}`), 创建时间: 当前时间, 更新时间: 当前时间})
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		if 恢复 {
			if 错误 := 仓储.恢复(测试器.上下文()); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			步骤, 错误 := 仓储.获取步骤(测试器.上下文(), 输入.标识)
			if 错误 != nil || 步骤.状态 != 智能体包.已中断 {
				测试器.报告致命错误(步骤, 错误)
			}
			if _, _, 错误 := 仓储.回答用户输入(测试器.上下文(), 输入.标识, 输入.执行标识, 智能体包.输入回答{文本: "迟到回复"}); !错误判断包.是否属于(错误, 智能体包.冲突错误) {
				测试器.报告致命错误("中断后接受了回复", 错误)
			}
			continue
		}
		var 等待组 同步包.等待组
		结果集 := make(chan error, 2)
		等待组.Go(func() {
			_, _, 错误 := 仓储.回答用户输入(测试器.上下文(), 输入.标识, 输入.执行标识, 智能体包.输入回答{选项索引: new(0)})
			结果集 <- 错误
		})
		等待组.Go(func() { _, 错误 := 仓储.取消用户输入(测试器.上下文(), 输入.标识); 结果集 <- 错误 })
		等待组.等待()
		close(结果集)
		for 错误 := range 结果集 {
			if 错误 != nil && !错误判断包.是否属于(错误, 智能体包.冲突错误) {
				测试器.报告致命错误(错误)
			}
		}
		步骤, 错误 := 仓储.获取步骤(测试器.上下文(), 输入.标识)
		if 错误 != nil || 步骤.状态 != 智能体包.已完成 && 步骤.状态 != 智能体包.已取消 {
			测试器.报告致命错误(步骤, 错误)
		}
		值, 错误 := 智能体包.输入来源步骤(步骤)
		if 错误 != nil || 值.状态 == "pending" {
			测试器.报告致命错误(值, 错误)
		}
		if 步骤.状态 == 智能体包.已取消 && 值.回答 != nil {
			测试器.报告致命错误("取消生成了虚假回复")
		}
	}
}
