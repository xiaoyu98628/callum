func 已存储检查点(测试器 *测试包.测试器, 当前仓储 *存储包.仓储, 本次执行 会话包.执行, 通过 int64, 事实, 状态 string) (智能体包.步骤, 智能体包.上下文检查点) {
	测试器.标记测试辅助函数()
	标识 := 新建标识(测试器)
	值 := 智能体包.上下文检查点{版本: 智能体包.上下文检查点版本, 标识: 标识, 会话标识: 本次执行.会话.标识, 来源执行标识: 本次执行.执行.标识, 通过: 通过, 来源集: []智能体包.批次来源{{步骤标识: 本次执行.步骤.标识, 通过: 通过, 指纹: 字符串包.重复("0", 64)}}, 摘要: 事实, 统计: JSON包.原始JSON消息(`{}`)}
	输出, 错误 := JSON包.序列化(map[string]any{"checkpoint": 值})

	}
	步骤, 错误 := 当前仓储.创建步骤(测试器.上下文(), 智能体包.步骤{标识: 标识, 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 类型: "context_compression", 名称: "已保存检查点", 状态: 状态, 输入: JSON包.原始JSON消息(`{}`), 输出: 输出, 结束时间: new(int64(3))})
	if 错误 != nil {
		测试器.报告致命错误(错误)

}

func Test检查点查询选择最新每个可见执行(测试器 *测试包.测试器) {
	数据库, 当前读取器, 本次执行, 子级 := 操作测试夹具(测试器, true)
	保存操作(测试器, 当前读取器, 子级, 工具包.操作结果已生效, true)
	已存储检查点(测试器, 当前读取器, 本次执行, 子级.序号, "最新摘要", 智能体包.已完成)
	旧, _ := 已存储检查点(测试器, 当前读取器, 本次执行, 本次执行.步骤.序号, "旧摘要", 智能体包.已完成)
	// 旧版本已被新范围覆盖；查询应在数据库中选定最新版本后才解码。
	if 错误 := 数据库.数据库.模型(&表结构包.步骤{}).条件("id=?", 旧.标识).更新("output", `{"checkpoint":{"version":0}}`).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	已存储检查点(测试器, 当前读取器, 本次执行, 子级.序号+100, "失败摘要", 智能体包.失败)
	其他类型, _ := 已存储检查点(测试器, 当前读取器, 本次执行, 子级.序号+100, "其他步骤类型", 智能体包.已完成)
	if 错误 := 数据库.数据库.模型(&表结构包.步骤{}).条件("id=?", 其他类型.标识).更新("type", "model_call").错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	结束操作执行(测试器, 当前读取器, 本次执行, 智能体包.失败)
	下一个 := 开始查询执行(测试器, 数据库, 本次执行, false, "")
	下一个.步骤.状态, 下一个.步骤.结束时间 = 智能体包.已完成, new(int64(3))
	下一个.步骤.输出 = 上下文输出(测试器, 模型包.消息{角色: 模型包.角色助手, 内容: "第二执行回答"}, nil)
	if 错误 := 当前读取器.保存步骤(测试器.上下文(), 下一个.步骤); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	已存储检查点(测试器, 当前读取器, 下一个, 下一个.步骤.序号, "第二执行摘要", 智能体包.已完成)
	结束操作执行(测试器, 当前读取器, 下一个, 智能体包.已完成)
	外键 := 开始查询执行(测试器, 数据库, 本次执行, true, "")
	外键步骤, 遮蔽 := 已存储检查点(测试器, 当前读取器, 外键, 子级.序号+200, "其他会话", 智能体包.已完成)
	遮蔽.来源执行标识 = 本次执行.执行.标识
	输出, 错误 := JSON包.序列化(map[string]any{"checkpoint": 遮蔽})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 数据库.数据库.模型(&表结构包.步骤{}).条件("id=?", 外键步骤.标识).更新("output", string(输出)).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	轨迹集, 错误 := 当前读取器.上下文执行轨迹(测试器.上下文(), 本次执行.会话.标识, 0)
	if 错误 != nil || len(轨迹集) != 2 || len(轨迹集[本次执行.执行.标识]) != 1 || len(轨迹集[下一个.执行.标识]) != 1 {
		测试器.格式化报告致命错误("未按来源执行恢复检查点: %+v %v", 轨迹集, 错误)
	}
	if !字符串包.包含(轨迹集[本次执行.执行.标识][0].内容, "最新摘要") || !字符串包.包含(轨迹集[下一个.执行.标识][0].内容, "第二执行摘要") {
		测试器.格式化报告致命错误("选择了旧版本或其他会话检查点: %+v", 轨迹集)
	}
	轨迹集, 错误 = 当前读取器.上下文执行轨迹(测试器.上下文(), 本次执行.会话.标识, 0, 下一个.用户.序号)
	if 错误 != nil || len(轨迹集) != 1 || len(轨迹集[本次执行.执行.标识]) != 1 {
		测试器.格式化报告致命错误("检查点查询越过历史上界: %+v %v", 轨迹集, 错误)
	}
}

func Test检查点激活拒绝变化通知来源并且保留前一个(测试器 *测试包.测试器) {
	数据库, 当前读取器, 本次执行, 子级 := 操作测试夹具(测试器, true)
	保存操作(测试器, 当前读取器, 子级, 工具包.操作结果已生效, true)
	// 调用-乙 没有完成，活动批次不能被摘要覆盖。
	if 批次, 错误 := 当前读取器.上下文批次(测试器.上下文(), 本次执行.会话.标识, 本次执行.执行.标识, 0); 错误 != nil || 批次 != nil {
		测试器.格式化报告致命错误("未完成批次可被压缩: %+v %v", 批次, 错误)
	}
	秒, 错误 := 当前读取器.创建步骤(测试器.上下文(), 智能体包.步骤{标识: 新建标识(测试器), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 父级步骤标识: &本次执行.步骤.标识, 工具调用标识: new("call-b"), 类型: "tool_call", 名称: "operation", 状态: 智能体包.已完成, 输入: JSON包.原始JSON消息(`{}`), 输出: 上下文输出(测试器, 模型包.消息{角色: 模型包.角色工具, 工具调用标识: "call-b", 内容: `{}`}, nil), 结束时间: new(int64(3))})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	批次, 错误 := 当前读取器.上下文批次(测试器.上下文(), 本次执行.会话.标识, 本次执行.执行.标识, 0)
	if 错误 != nil || 批次 == nil || len(批次.消息集) != 3 {
		测试器.格式化报告致命错误("完整批次读取失败: %+v %v", 批次, 错误)
	}
	当前时间 := 时间包.当前时间().Unix毫秒()
	父级, 错误 := 当前读取器.创建步骤(测试器.上下文(), 智能体包.步骤{标识: 新建标识(测试器), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 类型: "context_compression", 名称: "压缩过程", 状态: 智能体包.运行中, 输入: JSON包.原始JSON消息(`{}`), 开始时间: &当前时间, 创建时间: 当前时间, 更新时间: 当前时间})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	值 := 智能体包.上下文检查点{版本: 智能体包.上下文检查点版本, 标识: 父级.标识, 会话标识: 本次执行.会话.标识, 来源执行标识: 本次执行.执行.标识, 通过: 批次.通过, 来源集: []智能体包.批次来源{批次.批次来源}, 摘要: "操作已发生", 统计: JSON包.原始JSON消息(`{}`)}
	变化通知 := 上下文输出(测试器, 模型包.消息{角色: 模型包.角色工具, 工具调用标识: "call-b", 内容: `{"changed":true}`}, nil)
	if 错误 := 数据库.数据库.模型(&表结构包.步骤{}).条件("id=?", 秒.标识).更新("output", string(变化通知)).错误; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 当前读取器.保存检查点(测试器.上下文(), 值); !错误判断包.是否属于(错误, 智能体包.冲突错误) {
		测试器.格式化报告致命错误("来源变化仍激活摘要: %v", 错误)
	}
	if 当前, 错误 := 当前读取器.最新检查点(测试器.上下文(), 本次执行.会话.标识, 本次执行.执行.标识); 错误 != nil || 当前 != nil {
		测试器.报告致命错误("失败激活留下半成品")
	}
	批次, 错误 = 当前读取器.上下文批次(测试器.上下文(), 本次执行.会话.标识, 本次执行.执行.标识, 0)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	值.来源集 = []智能体包.批次来源{批次.批次来源}
	if 错误 := 当前读取器.保存检查点(测试器.上下文(), 值); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 当前读取器.保存检查点(测试器.上下文(), 值); !错误判断包.是否属于(错误, 智能体包.冲突错误) {
		测试器.格式化报告致命错误("旧父检查点仍能覆盖新状态: %v", 错误)
	}
	已取消, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	取消函数()
	if 错误 := 当前读取器.保存检查点(已取消, 值); 错误 == nil {
		测试器.报告致命错误("取消后仍保存")
	}
	当前, 错误 := 当前读取器.最新检查点(测试器.上下文(), 本次执行.会话.标识, 本次执行.执行.标识)
	if 错误 != nil || 当前 == nil || 当前.标识 != 值.标识 {
		测试器.报告致命错误("取消或冲突损坏了此前有效摘要")
	}
}
