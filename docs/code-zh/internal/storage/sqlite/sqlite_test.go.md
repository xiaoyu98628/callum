func Test数据库URI(测试器 *测试包.测试器) {
	for _, 测试用例 := range []struct {
		路径 string
		预期 string
	}{
		{"C:/Callum profile/测试 #1%/callum.db", "file:///C:/Callum%20profile/%E6%B5%8B%E8%AF%95%20%231%25/callum.db"},
		{"/var/tmp/Callum profile/callum.db", "file:///var/tmp/Callum%20profile/callum.db"},
		{"//server/share/callum.db", "file:////server/share/callum.db"},
	} {
		测试器.执行(测试用例.路径, func(测试器 *测试包.测试器) {
			实际值, _, _ := 字符串包.截断(数据库URI(测试用例.路径), "?")
			if 实际值 != 测试用例.预期 {
				测试器.格式化报告致命错误("数据库 URI = %q，期望 %q", 实际值, 测试用例.预期)
			}
		})
	}
}

func Test打开遵循截止时间期间已加锁(测试器 *测试包.测试器) {
	路径 := 文件路径包.合并(测试器.创建临时目录(), "data", "callum.db")
	数据库, 错误 := 打开(测试器.上下文(), 路径)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer func() {
		if 错误 := 数据库.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	}()
	连接, 错误 := 数据库.池.连接(测试器.上下文())
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer func() {
		if 错误 := 连接.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	}()
	if _, 错误 := 连接.进程执行上下文(测试器.上下文(), "BEGIN EXCLUSIVE"); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer func() {
		if _, 错误 := 连接.进程执行上下文(测试器.上下文(), "ROLLBACK"); 错误 != nil {
			测试器.错误文本(错误)
		}
	}()
	执行上下文, 取消函数 := 上下文包.创建超时上下文(测试器.上下文(), 50*时间包.毫秒)
	defer 取消函数()
	其他, 错误 := 打开(执行上下文, 路径)
	if 其他 != nil {
		if 错误 := 其他.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	}
	if !错误判断包.是否属于(错误, 上下文包.超出截止时间错误) {
		测试器.格式化报告致命错误("初始化没有保留调用方的超时错误: %v", 错误)
	}
}

func Test已有数据不是已重建(测试器 *测试包.测试器) {
	目录 := 文件路径包.合并(测试器.创建临时目录(), "data")
	if 错误 := 操作系统包.创建目录(目录, 0700); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	路径 := 文件路径包.合并(目录, "callum.db")
	原始 := []byte("existing data that is not a SQLite database")
	if 错误 := 操作系统包.写入文件(路径, 原始, 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 打开(测试器.上下文(), 路径); 错误 == nil {
		测试器.报告致命错误("接受了无效数据库")
	}
	实际值, 错误 := 操作系统包.读取文件(路径)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if !字节包.相等(实际值, 原始) {
		测试器.报告致命错误("初始化失败后原文件被改写")
	}
}

func Test已有权限不是变化通知(测试器 *测试包.测试器) {
	if 运行时包.目标操作系统 == "windows" {
		测试器.跳过("Windows 访问控制使用 ACL")
	}
	目录 := 文件路径包.合并(测试器.创建临时目录(), "data")
	if 错误 := 操作系统包.创建目录(目录, 0700); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	路径 := 文件路径包.合并(目录, "callum.db")
	if 错误 := 操作系统包.写入文件(路径, nil, 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 操作系统包.修改权限(路径, 0644); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 打开(测试器.上下文(), 路径); 错误 == nil {
		测试器.报告致命错误("接受了权限过宽的文件")
	}
	信息, 错误 := 操作系统包.状态信息(路径)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 信息.模式().权限() != 0644 {
		测试器.报告致命错误("擅自改变了已有文件权限")
	}
}

	if 错误 := 操作系统包.创建目录(目录, 0700); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	实际 := 文件路径包.合并(目录, "actual.db")
	if 错误 := 操作系统包.写入文件(实际, nil, 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	链接 := 文件路径包.合并(目录, "linked.db")
	if 错误 := 操作系统包.符号链接(实际, 链接); 错误 != nil {

	}

	if 错误 := 操作系统包.符号链接(目录, 目录链接); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 打开(测试器.上下文(), 文件路径包.合并(目录链接, "callum.db")); 错误 == nil {
		测试器.报告致命错误("接受了数据目录末级符号链接")
	}

func Test打开并且权限(测试器 *测试包.测试器) {
	路径 := 文件路径包.合并(测试器.创建临时目录(), "data", "callum.db")
	数据库, 错误 := 打开(测试器.上下文(), 路径)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 数据库.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 数据库.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 运行时包.目标操作系统 != "windows" {
		for 当前对象, 模式 := range map[string]操作系统包.文件模式{文件路径包.目录(路径): 0700, 路径: 0600} {
			信息, 错误 := 操作系统包.状态信息(当前对象)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			if 信息.模式().权限() != 模式 {
				测试器.格式化报告致命错误("%s 权限为 %o", 当前对象, 信息.模式().权限())
			}
		}
	}
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	取消函数()
	if _, 错误 := 打开(执行上下文, 路径); !错误判断包.是否属于(错误, 上下文包.已取消错误) {
		测试器.格式化报告致命错误("初始化取消: %v", 错误)
	}
	if _, 错误 := 打开(测试器.上下文(), "relative.db"); 错误 == nil {
		测试器.报告致命错误("接受了相对路径")
	}
}
