```go
package SQLite

import (
	错误判断包 "errors"
	格式化包 "fmt"
	操作系统包 "os"
	文件路径包 "path/filepath"
	运行时包 "runtime"
)

func 准备文件(路径 string) error {
	目录 := 文件路径包.目录(路径)
	if 错误 := 操作系统包.创建目录全部(目录, 0700); 错误 != nil {
		return 格式化包.格式化创建错误("创建数据目录: %w", 错误)
	}
	if 错误 := 检查私有(目录, true); 错误 != nil {
		return 错误
	}
	文件, 错误 := 操作系统包.打开文件(路径, 操作系统包.创建文件标志|操作系统包.独占创建标志|操作系统包.读写打开标志, 0600)
	if 错误 == nil {
		return 文件.关闭()
	}
	if !错误判断包.是否属于(错误, 操作系统包.存在错误) {
		return 格式化包.格式化创建错误("创建数据库文件: %w", 错误)
	}
	return 检查私有(路径, false)
}

func 检查私有(路径 string, 目录 bool) error {
	信息, 错误 := 操作系统包.读取链接自身状态(路径)
	if 错误 != nil {
		return 格式化包.格式化创建错误("检查数据路径: %w", 错误)
	}
	if 目录 && !信息.是否目录() || !目录 && !信息.模式().是否普通文件() {
		return 错误判断包.新建("数据路径类型无效，不接受符号链接")
	}
	// Windows 需要由部署环境配置 ACL，不能通过 Unix 权限位推断。
	if 运行时包.目标操作系统 != "windows" && 信息.模式().权限()&0077 != 0 {
		return 错误判断包.新建("已有数据路径权限过宽，需要用户调整访问权限")
	}
	return nil
}
```
