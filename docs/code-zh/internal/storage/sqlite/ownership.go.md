```go
package SQLite

import (
	错误判断包 "errors"
	操作系统包 "os"
	同步包 "sync"

	文件锁包 "github.com/xiaoyu98628/callum/internal/filelock"
)

// 所有权 持有服务实例的文件锁；关闭 在运行任务结束后释放，可重复调用。
type 所有权 struct {
	文件 *操作系统包.文件
	一次 同步包.一次
	错误  error
}

// 获取所有权 防止另一个服务实例在同一数据目录执行恢复或写入运行状态。
// SQLite 普通连接仍可用于配置查询；应用模块 必须在 恢复 前取得并持有此锁。
func 获取所有权(数据库路径 string) (*所有权, error) {
	路径 := 数据库路径 + ".lock"
	if 错误 := 准备文件(路径); 错误 != nil {
		return nil, 错误
	}
	文件, 错误 := 操作系统包.打开文件(路径, 操作系统包.读写打开标志, 0600)
	if 错误 != nil {
		return nil, 错误
	}
	if 错误 := 文件锁包.尝试(文件); 错误 != nil {
		if 错误判断包.是否属于(错误, 文件锁包.已加锁错误) {
			错误 = 错误判断包.合并(错误判断包.新建("该数据目录已有运行中的 Callum 服务"), 错误)
		}
		return nil, 错误判断包.合并(错误, 文件.关闭())
	}
	return &所有权{文件: 文件}, nil
}

func (当前所有权 *所有权) 关闭() error { 当前所有权.一次.执行(func() { 当前所有权.错误 = 当前所有权.文件.关闭() }); return 当前所有权.错误 }
```
