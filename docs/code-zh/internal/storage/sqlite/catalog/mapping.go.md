```go
package 目录管理

import (
	供应商包 "github.com/xiaoyu98628/callum/internal/model/provider"
	表结构包 "github.com/xiaoyu98628/callum/internal/storage/sqlite/schema"
)

// 供应商摘要 是厂商公开查询的数据库投影，既不读取明文凭据，也不让 ORM 映射业务类型。
type 供应商摘要 struct {
	标识       string
	名称       string
	协议       string
	基础URL    string
	已配置API密钥 bool
	创建时间     int64
	更新时间     int64
}

func (当前供应商摘要 供应商摘要) 目标供应商() 供应商包.供应商 {
	return 供应商包.供应商{标识: 当前供应商摘要.标识, 名称: 当前供应商摘要.名称, 协议: 当前供应商摘要.协议, 基础URL: 当前供应商摘要.基础URL,
		已配置API密钥: 当前供应商摘要.已配置API密钥, 创建时间: 当前供应商摘要.创建时间, 更新时间: 当前供应商摘要.更新时间}
}

// 目标记录 将配置转为数据库记录，未标注的内容类型统一保存为空数组。
func 目标记录(输入 供应商包.模型输入) 表结构包.模型 {
	类型集 := 输入.模型类型
	if 类型集 == nil {
		类型集 = []string{}
	}
	return 表结构包.模型{
		供应商标识: 输入.供应商标识, 显示名称: 输入.显示名称, 远程模型标识: 输入.远程模型标识,
		已启用: new(输入.已启用), 上下文窗口: 输入.上下文窗口, 最大输出词元: 输入.最大输出词元,
		能力集: 输入.能力集, 模型类型: 类型集, 推理等级集: 输入.推理等级集,
	}
}

// 来源记录 还原数据库映射并检查必需的启用值，完整配置校验由业务服务负责。
func 来源记录(行 表结构包.模型) (供应商包.模型, error) {
	if 行.已启用 == nil {
		return 供应商包.模型{}, 供应商包.无效配置错误
	}
	输入 := 供应商包.模型输入{
		供应商标识: 行.供应商标识, 显示名称: 行.显示名称, 远程模型标识: 行.远程模型标识,
		已启用: *行.已启用, 上下文窗口: 行.上下文窗口, 最大输出词元: 行.最大输出词元,
		能力集: 行.能力集, 模型类型: 行.模型类型, 推理等级集: 行.推理等级集,
	}
	return 供应商包.模型{模型输入: 输入, 标识: 行.标识, 创建时间: 行.创建时间, 更新时间: 行.更新时间}, nil
}
```
