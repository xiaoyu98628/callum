func Test目录管理页集保留稳定游标并且字节边界(测试器 *测试包.测试器) {
	数据库, 错误 := SQLite包.打开(测试器.上下文(), 文件路径包.合并(测试器.创建临时目录(), "data", "callum.db"))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 数据库.关闭()
	服务, 错误 := 新建服务(数据库.数据库)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	进程标识, 错误 := 服务.创建供应商(测试器.上下文(), 供应商包.供应商输入{名称: "p", 协议: "chatcompletions", 基础URL: "https://example.invalid"})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	for 索引 := range 501 {
		_, 错误 := 服务.创建模型(测试器.上下文(), 供应商包.模型输入{供应商标识: 进程标识, 显示名称: "模型", 远程模型标识: 格式化包.转为字符串(索引), 已启用: 索引%2 == 0})
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	for _, 仅 := range []bool{false, true} {
		之后, 数量 := "", 0
		for {
			页, 错误 := 服务.列表模型集(测试器.上下文(), 仅, 供应商包.列表选项集{之后标识: 之后, 上限: 200})
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			for _, 行 := range 页.项集 {
				if 行.标识 <= 之后 || (仅 && !行.已启用) {
					测试器.报告致命错误("游标或筛选错误")
				}
				之后 = 行.标识
				数量++
			}
			if !页.还有更多 {
				if 页.下一个之后标识 != "" {
					测试器.报告致命错误("末页有游标")
				}
				break
			}
			if 页.下一个之后标识 != 之后 || len(页.项集) == 0 {
				测试器.报告致命错误("分页没有推进")
			}
		}
		预期 := 501
		if 仅 {
			预期 = 251
		}
		if 数量 != 预期 {
			测试器.格式化报告致命错误("读取 %d，期望 %d", 数量, 预期)
		}
	}
	for _, 选项集 := range []供应商包.列表选项集{{上限: -1}, {上限: 201}, {之后标识: "invalid"}} {
		if _, 错误 := 服务.列表模型集(测试器.上下文(), false, 选项集); !错误判断包.是否属于(错误, 供应商包.无效配置错误) {
			测试器.格式化报告致命错误("接受错误分页: %v", 错误)
		}
	}
	首个, 错误 := 服务.列表模型集(测试器.上下文(), false, 供应商包.列表选项集{上限: 1})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	游标 := 首个.下一个之后标识
	if 错误 := 服务.删除模型(测试器.上下文(), 游标); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	下一个, 错误 := 服务.列表模型集(测试器.上下文(), false, 供应商包.列表选项集{之后标识: 游标, 上限: 1})
	if 错误 != nil || len(下一个.项集) != 1 || 下一个.项集[0].标识 <= 游标 {
		测试器.格式化报告致命错误("删除游标后分页漂移: %+v %v", 下一个, 错误)
	}
	// 单条配置合法，但合计超过存储页预算时仍可按游标读完，不默默截断。
	for 索引 := range 12 {
		if _, 错误 := 服务.创建供应商(测试器.上下文(), 供应商包.供应商输入{名称: 字符串包.重复("中", 70000), 协议: "chatcompletions", 基础URL: 格式化包.格式化字符串("https://example.invalid/%d", 索引)}); 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	页, 错误 := 服务.列表供应商集(测试器.上下文(), 供应商包.列表选项集{上限: 200})
	if 错误 != nil || !页.还有更多 || len(页.项集) >= 13 {
		测试器.格式化报告致命错误("字节预算未生效: %d %v", len(页.项集), 错误)
	}
	总计 := len(页.项集)
	for 页.还有更多 {
		页, 错误 = 服务.列表供应商集(测试器.上下文(), 供应商包.列表选项集{之后标识: 页.下一个之后标识, 上限: 200})
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		总计 += len(页.项集)
	}
	if 总计 != 13 {
		测试器.格式化报告致命错误("字节分片丢失记录: %d", 总计)
	}
}
