package 目录管理测试

import (
	上下文包 "context"
	SQL包 "database/sql"
	错误判断包 "errors"
	文件路径包 "path/filepath"
	测试包 "testing"
	时间包 "time"

	供应商包 "github.com/xiaoyu98628/callum/internal/model/provider"
	SQLite包 "github.com/xiaoyu98628/callum/internal/storage/sqlite"
)

func Test等待中用于连接能够超时(测试器 *测试包.测试器) {
	数据库, 错误 := SQLite包.打开(测试器.上下文(), 文件路径包.合并(测试器.创建临时目录(), "data", "callum.db"))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 数据库.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	当前状态, 错误 := 新建服务(数据库.数据库)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	池, 错误 := 数据库.数据库.数据库()
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	连接, 错误 := 池.连接(测试器.上下文())
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer func() {
		if 错误 := 连接.关闭(); 错误 != nil && !错误判断包.是否属于(错误, SQL包.连接完成错误) {
			测试器.错误文本(错误)
		}
	}()
	执行上下文, 取消函数 := 上下文包.创建超时上下文(测试器.上下文(), 50*时间包.毫秒)
	defer 取消函数()
	if _, 错误 := 当前状态.列表供应商集(执行上下文); !错误判断包.是否属于(错误, 上下文包.超出截止时间错误) {
		测试器.格式化报告致命错误("等待连接时没有传递超时: %v", 错误)
	}
	if 错误 := 连接.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 当前状态.列表供应商集(测试器.上下文()); 错误 != nil {
		测试器.格式化报告致命错误("超时后连接池无法继续使用: %v", 错误)
	}
}

func TestGORM零值集(测试器 *测试包.测试器) {
	数据库, 错误 := SQLite包.打开(测试器.上下文(), 文件路径包.合并(测试器.创建临时目录(), "data", "callum.db"))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer func() {
		if 错误 := 数据库.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	}()
	当前状态, 错误 := 新建服务(数据库.数据库)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	进程标识, 错误 := 当前状态.创建供应商(测试器.上下文(), 供应商包.供应商输入{名称: "厂商", 协议: "chatcompletions", 基础URL: "https://example.invalid"})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	输入 := 供应商包.模型输入{供应商标识: 进程标识, 显示名称: "模型", 远程模型标识: "remote-model", 已启用: false}
	标识, 错误 := 当前状态.创建模型(测试器.上下文(), 输入)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	映射, 错误 := 当前状态.获取模型(测试器.上下文(), 标识)
	if 错误 != nil || 映射.已启用 {
		测试器.格式化报告致命错误("显式 false 被默认值覆盖: %+v %v", 映射, 错误)
	}
	输入.已启用 = true
	输入.上下文窗口 = new(int64(1000))
	输入.最大输出词元 = new(int64(100))
	输入.能力集 = map[string]bool{"reasoning": true}
	输入.模型类型 = []string{"text", "image"}
	输入.推理等级集 = []供应商包.推理级别{{显示名称: "标准", 远程值: "medium"}}
	if 错误 := 当前状态.更新模型(测试器.上下文(), 标识, 输入); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	输入.已启用 = false
	输入.上下文窗口 = nil
	输入.最大输出词元 = nil
	输入.能力集 = nil
	输入.模型类型 = nil
	输入.推理等级集 = nil
	if 错误 := 当前状态.更新模型(测试器.上下文(), 标识, 输入); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	映射, 错误 = 当前状态.获取模型(测试器.上下文(), 标识)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 映射.已启用 || 映射.上下文窗口 != nil || 映射.最大输出词元 != nil || 映射.能力集 != nil || len(映射.模型类型) != 0 || 映射.模型类型 == nil || 映射.推理等级集 != nil {
		测试器.格式化报告致命错误("零值或 NULL 更新失败: %+v", 映射)
	}
	if 错误 := 当前状态.更新模型(测试器.上下文(), "01900000-0000-7000-8000-000000000001", 输入); !错误判断包.是否属于(错误, 供应商包.未找到错误) {
		测试器.格式化报告致命错误("无效更新: %v", 错误)
	}
	全部, 错误 := 当前状态.列表模型集(测试器.上下文(), false)
	if 错误 != nil || len(全部.项集) != 1 {
		测试器.格式化报告致命错误("更新错误地新增了记录: %+v %v", 全部, 错误)
	}
}
