```go
package 配置模块

import (
	错误判断包 "errors"
	命令行参数包 "flag"
	格式化包 "fmt"
	输入输出包 "io"
	文件路径包 "path/filepath"
	字符串包 "strings"
	时间包 "time"
)

const (
	// 默认最大并发执行集 是活动执行和实际模型请求的共同数量上限。
	默认最大并发执行集 = 4
	// 默认调用超时 是单次模型请求的超时（不含审批等待）。
	默认调用超时 = 5 * 时间包.分钟
)

type 工具参数集 struct {
	工具   *时间包.时长
	运行时集 *string
}

func 参数集(当前写入器 输入输出包.写入器) (*命令行参数包.命令行参数设置, *string, *int, *时间包.时长, 工具参数集) {
	文件 := 命令行参数包.新建命令行参数设置("callum", 命令行参数包.继续开启错误)
	文件.设置输出(当前写入器)
	目录 := 文件.字符串("data-dir", "", "数据目录；未指定时按开发或发布构建选择")
	上限 := 文件.整数("max-concurrent-runs", 默认最大并发执行集, "同时活动执行及实际模型请求的上限（1 至 64）")
	超时 := 文件.时长("call-timeout", 默认调用超时, "每次模型调用的超时，例如 5m")
	额外 := 工具参数集{
		工具:   文件.时长("tool-timeout", 30*时间包.秒, "单次工具执行超时"),
		运行时集: 文件.字符串("runtime-dir", "", "可选的内置 Python/Node/Git 平台资源目录，必须是绝对路径"),
	}
	return 文件, 目录, 上限, 超时, 额外
}

// 加载参数 解析启动参数，再计算路径；不会创建数据库或修改文件。
func 加载参数(参数 []string) (配置, error) {
	文件, 目录, 上限, 超时, 额外 := 参数集(输入输出包.丢弃)
	if 错误 := 文件.解析(参数); 错误 != nil {
		return 配置{}, 错误
	}
	if 文件.剩余参数数量() != 0 {
		return 配置{}, 错误判断包.新建("不接受位置参数，请通过 stdin 发送协议请求")
	}
	if *上限 < 1 || *上限 > 64 || *超时 <= 0 {
		return 配置{}, 错误判断包.新建("并发上限必须为 1 至 64，调用超时必须大于零")
	}
	启动配置, 错误 := 加载(*目录)
	if 错误 != nil {
		return 配置{}, 错误
	}
	if *额外.工具 <= 0 {
		return 配置{}, 错误判断包.新建("工具超时必须大于零")
	}
	启动配置.最大并发执行集, 启动配置.调用超时 = *上限, *超时
	启动配置.工具超时 = *额外.工具
	if *额外.运行时集 != "" {
		if !文件路径包.是否绝对路径(*额外.运行时集) || 字符串包.包含字符(*额外.运行时集, 0) {
			return 配置{}, 错误判断包.新建("运行时目录必须是绝对路径")
		}
		启动配置.运行时目录 = 文件路径包.清理(*额外.运行时集)
	}
	return 启动配置, nil
}

// 输出使用说明 向指定输出写入帮助，程序入口应传入 标准错误输出。
func 输出使用说明(当前写入器 输入输出包.写入器) {
	文件, _, _, _, _ := 参数集(当前写入器)
	格式化包.输出文本并换行(当前写入器, "用法：callum [启动参数]，启动后每行输入一条 JSON 请求。")
	文件.打印默认值()
}
```
