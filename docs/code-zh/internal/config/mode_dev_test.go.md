//go:build !release

package 配置测试

import (
	错误判断包 "errors"
	操作系统包 "os"
	文件路径包 "path/filepath"
	测试包 "testing"

	配置模块包 "github.com/xiaoyu98628/callum/internal/config"
)

func Test开发默认数据目录(测试器 *测试包.测试器) {
	根 := 测试器.创建临时目录()
	if 错误 := 操作系统包.写入文件(文件路径包.合并(根, "go.mod"), []byte("module \"github.com/xiaoyu98628/callum\" // Callum\n"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	嵌套 := 文件路径包.合并(根, "internal", "config")
	if 错误 := 操作系统包.创建目录全部(嵌套, 0700); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.切换目录(嵌套)
	// 开发默认路径无需访问用户家目录。
	测试器.设置环境变量("HOME", "")
	测试器.设置环境变量("USERPROFILE", "")
	启动配置, 错误 := 配置模块包.加载("")
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 启动配置.数据目录 != 文件路径包.合并(根, ".callum", "dev") || 启动配置.数据库路径 != 文件路径包.合并(根, ".callum", "dev", "callum.db") {
		测试器.格式化报告致命错误("开发默认路径错误: %+v", 启动配置)
	}
	if _, 错误 := 操作系统包.状态信息(文件路径包.合并(根, ".callum")); !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		测试器.格式化报告致命错误("路径计算不应创建目录: %v", 错误)
	}
}

func Test开发使用Callum模块根(测试器 *测试包.测试器) {
	根 := 测试器.创建临时目录()
	if 错误 := 操作系统包.写入文件(文件路径包.合并(根, "go.mod"), []byte("module github.com/xiaoyu98628/callum\n"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	其他 := 文件路径包.合并(根, "nested")
	if 错误 := 操作系统包.创建目录(其他, 0700); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 操作系统包.写入文件(文件路径包.合并(其他, "go.mod"), []byte("module example.com/other\n"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.切换目录(其他)
	启动配置, 错误 := 配置模块包.加载("")
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 启动配置.数据目录 != 文件路径包.合并(根, ".callum", "dev") {
		测试器.格式化报告致命错误("错误地使用了其他模块的目录: %+v", 启动配置)
	}
}

func Test开发缺失根不回退返回目标家目录(测试器 *测试包.测试器) {
	外部 := 测试器.创建临时目录()
	if 错误 := 操作系统包.写入文件(文件路径包.合并(外部, "go.mod"), []byte("module example.com/other\n"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.切换目录(外部)
	家目录 := 测试器.创建临时目录()
	测试器.设置环境变量("HOME", 家目录)
	测试器.设置环境变量("USERPROFILE", 家目录)
	if _, 错误 := 配置模块包.加载(""); 错误 == nil {
		测试器.报告致命错误("未找到 Callum 根目录时应返回错误")
	}
	if _, 错误 := 操作系统包.状态信息(文件路径包.合并(家目录, ".callum")); !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		测试器.格式化报告致命错误("开发路径失败后不应创建家目录数据: %v", 错误)
	}
}
