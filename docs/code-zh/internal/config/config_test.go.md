package 配置模块

import (
	文件路径包 "path/filepath"
	测试包 "testing"
)

func Test显式数据目录不读取家目录(测试器 *测试包.测试器) {
	测试器.设置环境变量("HOME", "")
	测试器.设置环境变量("USERPROFILE", "")
	测试器.切换目录(测试器.创建临时目录())
	目录 := 测试器.创建临时目录()
	启动配置, 错误 := 加载(目录)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 启动配置.数据目录 != 目录 || 启动配置.数据库路径 != 文件路径包.合并(目录, "callum.db") {
		测试器.格式化报告致命错误("路径错误: %+v", 启动配置)
	}
}

func Test显式相对数据目录(测试器 *测试包.测试器) {
	根 := 测试器.创建临时目录()
	测试器.切换目录(根)
	启动配置, 错误 := 加载(文件路径包.合并(".callum", "custom"))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 启动配置.数据目录 != 文件路径包.合并(根, ".callum", "custom") || 启动配置.数据库路径 != 文件路径包.合并(根, ".callum", "custom", "callum.db") {
		测试器.格式化报告致命错误("显式相对路径错误: %+v", 启动配置)
	}
}
