//go:build release

package 配置测试

import (
	错误判断包 "errors"
	操作系统包 "os"
	文件路径包 "path/filepath"
	测试包 "testing"

	配置模块包 "github.com/xiaoyu98628/callum/internal/config"
)

func Test释放默认数据目录(测试器 *测试包.测试器) {
	家目录 := 测试器.创建临时目录()
	测试器.设置环境变量("HOME", 家目录)
	测试器.设置环境变量("USERPROFILE", 家目录)
	测试器.切换目录(测试器.创建临时目录())
	启动配置, 错误 := 配置模块包.加载("")
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 启动配置.数据目录 != 文件路径包.合并(家目录, ".callum") || 启动配置.数据库路径 != 文件路径包.合并(家目录, ".callum", "callum.db") {
		测试器.格式化报告致命错误("发布默认路径错误: %+v", 启动配置)
	}
	if _, 错误 := 操作系统包.状态信息(启动配置.数据目录); !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		测试器.格式化报告致命错误("路径解析不应创建数据目录: %v", 错误)
	}
}

func Test释放缺失家目录返回错误(测试器 *测试包.测试器) {
	测试器.设置环境变量("HOME", "")
	测试器.设置环境变量("USERPROFILE", "")
	测试器.切换目录(测试器.创建临时目录())
	if _, 错误 := 配置模块包.加载(""); 错误 == nil {
		测试器.报告致命错误("家目录缺失时应返回错误")
	}
}

func Test释放拒绝相对家目录(测试器 *测试包.测试器) {
	测试器.设置环境变量("HOME", "relative-home")
	测试器.设置环境变量("USERPROFILE", "relative-home")
	测试器.切换目录(测试器.创建临时目录())
	if _, 错误 := 配置模块包.加载(""); 错误 == nil {
		测试器.报告致命错误("家目录为相对路径时应返回错误")
	}
}
