```go
package 运行时资源

import (
	字节包 "bytes"
	上下文包 "context"
	SHA256包 "crypto/sha256"
	JSON包 "encoding/json"
	错误判断包 "errors"
	格式化包 "fmt"
	输入输出包 "io"
	操作系统包 "os"
	文件路径包 "path/filepath"
	运行时包 "runtime"
	切片包 "slices"
	字符串包 "strings"
)

// 资源包 是启动时校验的只读资源描述；不持有文件或后台任务。
// 同账户的外部修改不构成隔离边界；命令执行器另固定句柄并复核入口。
type 资源包 struct {
	目录 string
	清单  资源清单
	身份  操作系统包.文件信息
	Python    string
	节点      string
	Git       string
}

// 加载 校验平台、所有文件内容和链接，返回可供装配层借用的固定描述。
func 加载(执行上下文 上下文包.上下文, 目录 string) (*资源包, error) {
	return 加载资源实现(执行上下文, 目录, nil)
}

// 加载带有清单SHA256 将完整校验绑定到安装器提供的原始清单摘要。
func 加载带有清单SHA256(执行上下文 上下文包.上下文, 目录 string, 预期 [SHA256包.大小]byte) (*资源包, error) {
	return 加载资源实现(执行上下文, 目录, &预期)
}

func 加载资源实现(执行上下文 上下文包.上下文, 目录 string, 预期 *[SHA256包.大小]byte) (_ *资源包, 返回错误 error) {
	if !文件路径包.是否绝对路径(目录) {
		return nil, 错误判断包.新建("运行时目录必须是绝对路径")
	}
	实际, 错误 := 文件路径包.求值符号链接集(目录)
	if 错误 != nil {
		return nil, 格式化包.格式化创建错误("定位内置运行时: %w", 错误)
	}
	根, 错误 := 操作系统包.打开根(实际)
	if 错误 != nil {
		return nil, 错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 根.关闭()) }()
	if 预期 != nil {
		信息, 错误 := 根.读取链接自身状态("manifest.json")
		if 错误 != nil {
			return nil, 格式化包.格式化创建错误("读取内置运行时清单: %w", 错误)
		}
		if !信息.模式().是否普通文件() {
			return nil, 错误判断包.新建("待校验的运行时清单必须是普通文件")
		}
	}
	文件, 错误 := 根.打开("manifest.json")
	if 错误 != nil {
		return nil, 格式化包.格式化创建错误("读取内置运行时清单: %w", 错误)
	}
	var 读取器 输入输出包.读取器 = 文件
	哈希 := SHA256包.新建()
	if 预期 != nil {
		读取器 = 输入输出包.分流读取器(文件, 哈希)
	}
	清单, 解码错误 := 解码(读取器)
	if 错误 := 错误判断包.合并(解码错误, 文件.关闭()); 错误 != nil {
		return nil, 错误
	}
	if 预期 != nil && !字节包.相等(哈希.校验和(nil), 预期[:]) {
		return nil, 错误判断包.新建("运行时清单与预期摘要不符")
	}
	if 清单.平台 != 运行时包.目标操作系统 || 清单.架构 != 运行时包.目标架构 {
		return nil, 错误判断包.新建("内置运行时与当前平台或架构不匹配")
	}
	if 错误 := 校验布局(实际, 清单); 错误 != nil {
		return nil, 错误
	}
	文件集, 错误 := 资源库存(执行上下文, 实际)
	if 错误 != nil {
		return nil, 错误
	}
	if !切片包.相等(文件集, 清单.文件集) {
		return nil, 错误判断包.新建("内置运行时资源缺失、损坏或存在清单外文件，请重新准备资源")
	}
	信息, 错误 := 操作系统包.状态信息(实际)
	if 错误 != nil {
		return nil, 错误
	}
	return &资源包{目录: 实际, 清单: 清单, 身份: 信息, Python: 文件路径包.合并(实际, 文件路径包.来源斜杠(清单.Python.路径)), 节点: 文件路径包.合并(实际, 文件路径包.来源斜杠(清单.节点.路径)), Git: 文件路径包.合并(实际, 文件路径包.来源斜杠(清单.Git.路径))}, nil
}

// 目录 返回已解析符号链接的实际资源根目录。
func (当前资源包 *资源包) 获取目录() string { return 当前资源包.目录 }

// 存在工作区依赖集 报告通用依赖资源是否已通过启动校验，不参与技能安装或发现。
func (当前资源包 *资源包) 存在工作区依赖集() bool {
	return 当前资源包 != nil && 当前资源包.清单.工作区 != nil
}

// 工作区路径集 返回与技能约定兼容的绝对路径快照，不借出可变清单。
func (当前资源包 *资源包) 工作区路径集() map[string]string {
	if !当前资源包.存在工作区依赖集() {
		return nil
	}
	当前读取器 := 当前资源包.清单.工作区
	绝对 := func(路径 string) string { return 文件路径包.合并(当前资源包.目录, 文件路径包.来源斜杠(路径)) }
	return map[string]string{"bundle_version": 当前读取器.版本, "python": 当前资源包.Python, "node": 当前资源包.节点, "node_modules": 绝对(当前读取器.节点模块集), "bin_directory": 绝对(当前读取器.二进制目录)}
}

// 工作区环境 固定原技能脚本需要的环境变量；只用于命令子进程。
func (当前资源包 *资源包) 工作区环境() []string {
	路径集 := 当前资源包.工作区路径集()
	if 路径集 == nil {
		return nil
	}
	return []string{"RUNTIME_PYTHON=" + 路径集["python"], "RUNTIME_NODE=" + 路径集["node"], "RUNTIME_NODE_MODULES=" + 路径集["node_modules"], "RUNTIME_BIN_DIR=" + 路径集["bin_directory"]}
}

// 二进制目录集 返回仅供命令子进程使用的搜索目录，不修改系统或服务进程 路径。
func (当前资源包 *资源包) 二进制目录集() []string {
	目录集 := []string{文件路径包.目录(当前资源包.Python), 文件路径包.目录(当前资源包.节点), 文件路径包.目录(当前资源包.Git), 文件路径包.合并(当前资源包.目录, 文件路径包.来源斜杠(当前资源包.清单.Git.进程执行路径))}
	if 当前资源包.存在工作区依赖集() {
		目录集 = append([]string{当前资源包.工作区路径集()["bin_directory"]}, 目录集...)
	}
	if 当前资源包.清单.平台 == "windows" {
		// Windows 的 动态链接库、HTTPS/GCM 和 MSYS Shell/ssh 分布在两个附属 二进制 目录。
		前缀 := 文件路径包.目录(文件路径包.目录(文件路径包.合并(当前资源包.目录, 文件路径包.来源斜杠(当前资源包.清单.Git.进程执行路径))))
		目录集 = append(目录集, 文件路径包.合并(前缀, "bin"), 文件路径包.合并(当前资源包.目录, "git", "usr", "bin"))
	}
	return 目录集
}

// 解析 仅映射裸命令名；显式路径继续使用现有路径校验与审批规则。
func (当前资源包 *资源包) 解析(名称 string) (string, bool) {
	if 当前资源包 == nil {
		return "", false
	}
	if 运行时包.目标操作系统 == "windows" {
		名称 = 字符串包.裁剪后缀(字符串包.目标小写(名称), ".exe")
	}
	switch 名称 {
	case "python", "python3":
		return 当前资源包.Python, true
	case "node":
		return 当前资源包.节点, true
	case "git":
		return 当前资源包.Git, true
	default:
		return "", false
	}
}

// 检查 在每次准备操作时复核目录、已声明的解释器入口及 Git 配置；完整资源校验在启动时完成。
func (当前资源包 *资源包) 检查(执行上下文 上下文包.上下文) (返回错误 error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	实际, 错误 := 文件路径包.求值符号链接集(当前资源包.目录)
	if 错误 != nil || 实际 != 当前资源包.目录 {
		return 错误判断包.新建("内置运行时目录已变化")
	}
	信息, 错误 := 操作系统包.状态信息(实际)
	if 错误 != nil || !操作系统包.相同文件(信息, 当前资源包.身份) {
		return 错误判断包.新建("内置运行时目录身份已变化")
	}
	根, 错误 := 操作系统包.打开根(实际)
	if 错误 != nil {
		return 错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 根.关闭()) }()
	可执行文件集 := []可执行文件{当前资源包.清单.Python, 当前资源包.清单.节点, 当前资源包.清单.Git.可执行文件, {路径: 当前资源包.清单.Git.配置路径}}
	for _, 可执行文件 := range 可执行文件集 {
		名称 := 可执行文件.路径
		for {
			索引 := 切片包.索引函数(当前资源包.清单.文件集, func(文件 文件) bool { return 文件.路径 == 名称 })
			if 索引 < 0 {
				return 错误判断包.新建("内置命令不在清单中")
			}
			文件 := 当前资源包.清单.文件集[索引]
			if 文件.链接 != "" {
				// 资源库存 已验证链接链；此处解析实际目标并按清单校验内容。
				已解析, 错误 := 文件路径包.求值符号链接集(文件路径包.合并(实际, 文件路径包.来源斜杠(名称)))
				if 错误 != nil {
					return 错误
				}
				相对路径, 错误 := 文件路径包.相对路径(实际, 已解析)
				if 错误 != nil || !文件路径包.是否本地路径(相对路径) || 文件路径包.目标斜杠(相对路径) == 名称 {
					return 错误判断包.新建("内置命令链接无效")
				}
				名称 = 文件路径包.目标斜杠(相对路径)
				continue
			}
			哈希, 大小, 错误 := 哈希文件(执行上下文, 根, 名称)
			if 错误 != nil {
				return 格式化包.格式化创建错误("内置命令完整性检查失败: %w", 错误)
			}
			if 哈希 != 文件.SHA256 || 大小 != 文件.大小 {
				return 错误判断包.新建("内置命令完整性检查失败")
			}
			break
		}
	}
	return nil
}

// 写入清单 只供离线资源准备和测试使用；运行期不写入资源目录。
func 写入清单(目录 string, 清单 资源清单) error {
	数据, 错误 := JSON包.序列化缩进(清单, "", "  ")
	if 错误 != nil || len(数据) > 最大清单字节 {
		return 错误判断包.新建("无法生成运行时清单")
	}
	return 操作系统包.写入文件(文件路径包.合并(目录, "manifest.json"), append(数据, '\n'), 0444)
}

// 校验布局 检查清单入口及 Git 配套目录，供准备阶段和运行期启动共用。
func 校验布局(目录 string, 清单 资源清单) error {
	if 当前读取器 := 清单.工作区; 当前读取器 != nil {
		if 当前读取器.版本 == "" || !字符串包.存在前缀(当前读取器.节点模块集, "node/") || 当前读取器.二进制目录 != "native" {
			return 错误判断包.新建("工作区依赖资源版本无效")
		}
		for _, 项 := range []struct {
			路径      string
			目录 bool
		}{{当前读取器.节点模块集, true}, {当前读取器.二进制目录, true}} {
			if !本地路径(项.路径) {
				return 错误判断包.新建("工作区依赖资源路径无效")
			}
			信息, 错误 := 操作系统包.状态信息(文件路径包.合并(目录, 文件路径包.来源斜杠(项.路径)))
			if 错误 != nil || 信息.是否目录() != 项.目录 || (!项.目录 && (!信息.模式().是否普通文件() || (清单.平台 != "windows" && 信息.模式().权限()&0111 == 0))) {
				return 错误判断包.新建("工作区依赖资源入口缺失")
			}
		}
	}
	已见包集 := map[string]bool{}
	if len(清单.Python包集) > 32 {
		return 错误判断包.新建("Python 依赖数量超过限制")
	}
	for _, 依赖 := range 清单.Python包集 {
		if 依赖.名称 == "" || 依赖.版本 == "" || 已见包集[依赖.名称] || !本地路径(依赖.路径) || !字符串包.存在前缀(依赖.路径, "python/") {
			return 错误判断包.新建("Python 依赖清单无效")
		}
		已见包集[依赖.名称] = true
		信息, 错误 := 操作系统包.状态信息(文件路径包.合并(目录, 文件路径包.来源斜杠(依赖.路径)))
		if 错误 != nil || !信息.模式().是否普通文件() {
			return 错误判断包.新建("Python 依赖入口缺失")
		}
	}
	for 组件, 可执行文件 := range map[string]可执行文件{"python": 清单.Python, "node": 清单.节点, "git": 清单.Git.可执行文件} {
		if 可执行文件.版本 == "" || !本地路径(可执行文件.路径) || !字符串包.存在前缀(可执行文件.路径, 组件+"/") {
			return 错误判断包.新建("内置命令入口无效")
		}
		信息, 错误 := 操作系统包.状态信息(文件路径包.合并(目录, 文件路径包.来源斜杠(可执行文件.路径)))
		if 错误 != nil || !信息.模式().是否普通文件() || (清单.平台 != "windows" && 信息.模式().权限()&0111 == 0) {
			return 错误判断包.新建("内置命令不存在或不可执行")
		}
	}
	for _, 资源 := range []struct {
		名称        string
		是目录 bool
	}{{清单.Git.进程执行路径, true}, {清单.Git.模板路径, true}, {清单.Git.配置路径, false}} {
		名称, 是目录 := 资源.名称, 资源.是目录
		if !本地路径(名称) || !字符串包.存在前缀(名称, "git/") {
			return 错误判断包.新建("内置 Git 辅助资源路径无效")
		}
		信息, 错误 := 操作系统包.状态信息(文件路径包.合并(目录, 文件路径包.来源斜杠(名称)))
		if 错误 != nil || 信息.是否目录() != 是目录 || (!是目录 && !信息.模式().是否普通文件()) {
			return 错误判断包.新建("内置 Git 辅助资源缺失或类型无效")
		}
	}
	return nil
}

// Git环境 仅供命令子进程使用；不写入用户配置，不启用 GUI 或终端交互。
func (当前资源包 *资源包) Git环境() []string {
	绝对 := func(名称 string) string { return 文件路径包.合并(当前资源包.目录, 文件路径包.来源斜杠(名称)) }
	return []string{"GIT_EXEC_PATH=" + 绝对(当前资源包.清单.Git.进程执行路径), "GIT_TEMPLATE_DIR=" + 绝对(当前资源包.清单.Git.模板路径), "GIT_CONFIG_SYSTEM=" + 绝对(当前资源包.清单.Git.配置路径), "GIT_OPTIONAL_LOCKS=0", "GIT_TERMINAL_PROMPT=0", "GCM_INTERACTIVE=0", "GIT_ASKPASS=", "SSH_ASKPASS=", "GIT_PAGER=cat"}
}
```
