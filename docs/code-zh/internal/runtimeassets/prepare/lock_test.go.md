package 准备模块

import (
	错误判断包 "errors"
	操作系统包 "os"
	文件路径包 "path/filepath"
	字符串包 "strings"
	测试包 "testing"

	文件锁包 "github.com/xiaoyu98628/callum/internal/filelock"
)

func Test准备加锁身份并且释放(测试器 *测试包.测试器) {
	目录 := 测试器.创建临时目录()
	首个, 错误 := 获取准备加锁(目录, "target")
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 首个.关闭()
	if 秒, 错误 := 获取准备加锁(目录, "target"); !错误判断包.是否属于(错误, 文件锁包.已加锁错误) {
		if 秒 != nil {
			秒.关闭()
		}
		测试器.格式化报告致命错误("并发准备未互斥: %v", 错误)
	}
	if 错误 := 首个.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	秒, 错误 := 获取准备加锁(目录, "target")
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 秒.关闭()
	信息, 错误 := 操作系统包.读取链接自身状态(文件路径包.合并(目录, ".prepare-target.lock"))
	if 错误 != nil || !信息.模式().是否普通文件() {
		测试器.格式化报告致命错误("锁文件未保留: %v", 错误)
	}
}

func Test准备保留旧版目录加锁(测试器 *测试包.测试器) {
	目录 := 测试器.创建临时目录()
	路径 := 文件路径包.合并(目录, ".prepare-target.lock")
	if 错误 := 操作系统包.创建目录(路径, 0700); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 文件, 错误 := 获取准备加锁(目录, "target"); 错误 == nil || !字符串包.包含(错误.错误文本(), "旧格式") {
		if 文件 != nil {
			文件.关闭()
		}
		测试器.格式化报告致命错误("旧锁被接受: %v", 错误)
	}
	if 信息, 错误 := 操作系统包.状态信息(路径); 错误 != nil || !信息.是否目录() {
		测试器.格式化报告致命错误("旧锁被移除: %v", 错误)
	}
}
