```go
package 准备模块

import (
	TAR包 "archive/tar"
	ZIP包 "archive/zip"
	GZIP包 "compress/gzip"
	上下文包 "context"
	错误判断包 "errors"
	输入输出包 "io"
	文件系统包 "io/fs"
	操作系统包 "os"
	路径包 "path"
	文件路径包 "path/filepath"
	字符串包 "strings"
)

func 安全路径(名称 string) bool {
	return 名称 != "." && 文件系统包.有效路径(名称) && !字符串包.包含任一字符(名称, `\:`) && !字符串包.包含字符(名称, 0)
}

type 解包器 struct {
	执行上下文 上下文包.上下文
	根     *操作系统包.根
	字节    int64
	数量    int
	已见    map[string]bool
	链接集   map[string]string
}

type 上下文读取器 struct {
	执行上下文 上下文包.上下文
	读取器   输入输出包.读取器
}

func (当前上下文读取器 上下文读取器) 读取(缓冲区 []byte) (int, error) {
	if 错误 := 当前上下文读取器.执行上下文.获取错误(); 错误 != nil {
		return 0, 错误
	}
	return 当前上下文读取器.读取器.读取(缓冲区)
}

func (当前解包器 *解包器) 条目(名称 string, 大小 int64, 模式 文件系统包.文件模式, 读取器 输入输出包.读取器, 链接 string) error {
	if 错误 := 当前解包器.执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	if !安全路径(名称) || 大小 < 0 || 当前解包器.数量 >= 30000 || 当前解包器.字节+大小 > 2<<30 {
		return 错误判断包.新建("归档路径或资源边界无效")
	}
	当前解包器.数量++
	键 := 字符串包.目标小写(名称)
	if 当前解包器.已见[键] && !模式.是否目录() {
		return 错误判断包.新建("归档包含重复路径")
	}
	当前解包器.已见[键] = true
	当前解包器.字节 += 大小
	if 模式.是否目录() {
		return 当前解包器.根.创建目录全部(文件路径包.来源斜杠(名称), 0755)
	}
	if 错误 := 当前解包器.根.创建目录全部(文件路径包.来源斜杠(路径包.目录(名称)), 0755); 错误 != nil {
		return 错误
	}
	if 模式&操作系统包.模式符号链接 != 0 {
		if 链接 == "" || 字符串包.包含任一字符(链接, `\:`) || 路径包.是否绝对路径(链接) || !安全路径(路径包.合并(路径包.目录(名称), 链接)) {
			return 错误判断包.新建("归档链接越出组件目录")
		}
		当前解包器.链接集[名称] = 链接
		return nil
	}
	if !模式.是否普通文件() {
		return 错误判断包.新建("归档包含不受支持的文件类型")
	}
	文件, 错误 := 当前解包器.根.打开文件(文件路径包.来源斜杠(名称), 操作系统包.仅写打开标志|操作系统包.创建文件标志|操作系统包.独占创建标志, 0644|模式.权限()&0111)
	if 错误 != nil {
		return 错误
	}
	_, 复制错误 := 输入输出包.复制数量(文件, 上下文读取器{执行上下文: 当前解包器.执行上下文, 读取器: 读取器}, 大小)
	return 错误判断包.合并(复制错误, 文件.关闭())
}

// 延后创建链接，归档中的后续文件不能借助先前链接写到其他位置。
func 提取(执行上下文 上下文包.上下文, 归档, 目录, 前缀 string) (返回错误 error) {
	if 错误 := 操作系统包.创建目录(目录, 0755); 错误 != nil {
		return 错误
	}
	return 提取进入(执行上下文, 归档, 目录, 前缀)
}

// 提取进入 只供同一私有准备目录组合 滚轮；独占创建标志 阻止覆盖已有解释器和其他包文件。
func 提取进入(执行上下文 上下文包.上下文, 归档, 目录, 前缀 string) (返回错误 error) {
	根, 错误 := 操作系统包.打开根(目录)
	if 错误 != nil {
		return 错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 根.关闭()) }()
	地址 := 解包器{执行上下文: 执行上下文, 根: 根, 已见: map[string]bool{}, 链接集: map[string]string{}}
	名称输入根 := func(名称 string) (string, error) {
		名称 = 字符串包.裁剪后缀(名称, "/")
		if 前缀 == "." {
			if 名称 == "." {
				return "", nil
			}
			// Dugite 归档直接从 ./二进制 等路径开始；只移除一个显式 ./，其余仍严格校验。
			名称 = 字符串包.裁剪前缀(名称, "./")
			if !安全路径(名称) {
				return "", 错误判断包.新建("归档根路径无效")
			}
			return 名称, nil
		}
		if 名称 == 前缀 {
			return "", nil
		}
		if !字符串包.存在前缀(名称, 前缀+"/") {
			return "", 错误判断包.新建("归档根目录与固定清单不符")
		}
		return 字符串包.裁剪前缀(名称, 前缀+"/"), nil
	}
	if 字符串包.存在后缀(归档, ".zip") || 字符串包.存在后缀(归档, ".whl") {
		读取器, 错误 := ZIP包.打开读取器(归档)
		if 错误 != nil {
			return 错误
		}
		defer func() { 返回错误 = 错误判断包.合并(返回错误, 读取器.关闭()) }()
		if len(读取器.文件) > 30000 {
			return 错误判断包.新建("ZIP 条目数量超过限制")
		}
		for _, 项 := range 读取器.文件 {
			名称, 错误 := 名称输入根(项.名称)
			if 错误 != nil {
				return 错误
			}
			if 名称 == "" {
				continue
			}
			if 项.未压缩大小64 > 2<<30 || 项.模式()&操作系统包.模式符号链接 != 0 {
				return 错误判断包.新建("ZIP 项目类型或大小无效")
			}
			流, 错误 := 项.打开()
			if 错误 != nil {
				return 错误
			}
			条目错误 := 地址.条目(名称, int64(项.未压缩大小64), 项.模式(), 流, "")
			if 错误 := 错误判断包.合并(条目错误, 流.关闭()); 错误 != nil {
				return 错误
			}
		}
	} else if 字符串包.存在后缀(归档, ".tar.gz") {
		文件, 错误 := 操作系统包.打开(归档)
		if 错误 != nil {
			return 错误
		}
		defer func() { 返回错误 = 错误判断包.合并(返回错误, 文件.关闭()) }()
		已压缩, 错误 := GZIP包.新建读取器(上下文读取器{执行上下文: 执行上下文, 读取器: 文件})
		if 错误 != nil {
			return 错误
		}
		defer func() { 返回错误 = 错误判断包.合并(返回错误, 已压缩.关闭()) }()
		读取器 := TAR包.新建读取器(已压缩)
		数量 := 0
		for {
			项, 错误 := 读取器.下一个()
			if 错误 == 输入输出包.文件结束 {
				break
			}
			if 错误 != nil {
				return 错误
			}
			数量++
			if 数量 > 30000 {
				return 错误判断包.新建("TAR 条目数量超过限制")
			}
			名称, 错误 := 名称输入根(项.名称)
			if 错误 != nil {
				return 错误
			}
			if 名称 == "" {
				continue
			}
			if 项.类型标记 != TAR包.类型注册表 && 项.类型标记 != TAR包.类型目录 && 项.类型标记 != TAR包.类型符号链接 {
				return 错误判断包.新建("TAR 包含特殊文件或硬链接")
			}
			if 错误 := 地址.条目(名称, 项.大小, 项.文件信息().模式(), 读取器, 项.链接名); 错误 != nil {
				return 错误
			}
		}
		if _, 错误 := 输入输出包.复制(输入输出包.丢弃, 输入输出包.上限读取器(已压缩, 1<<20)); 错误 != nil {
			return 错误
		}
	} else {
		return 错误判断包.新建("仅支持 tar.gz 和 zip 运行时归档")
	}
	for 名称, 链接 := range 地址.链接集 {
		if 错误 := 根.符号链接(链接, 文件路径包.来源斜杠(名称)); 错误 != nil {
			return 错误
		}
	}
	return nil
}
```
