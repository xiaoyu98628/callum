// 来源 固定归档的来源和校验值，归档 是本地缓存文件名。
type 来源 struct {
	版本     string `json:"version"`
	URL    string `json:"url"`
	SHA256 string `json:"sha256"`
	归档     string `json:"archive"`
	前缀     string `json:"prefix"`
	条目     string `json:"entry"`
}

// Git来源 固定 Git 归档中相对于组件目录的配套路径。
type Git来源 struct {
	来源
	进程执行路径 string `json:"exec_path"`
	模板路径   string `json:"template_path"`
	配置路径   string `json:"config_path"`
}

func (当前Git来源 Git来源) 清单() 运行时资源包.Git运行时 {
	return 运行时资源包.Git运行时{版本: 当前Git来源.版本, 路径: "git/" + 当前Git来源.条目, 进程执行路径: "git/" + 当前Git来源.进程执行路径, 模板路径: "git/" + 当前Git来源.模板路径, 配置路径: "git/" + 当前Git来源.配置路径}
}

// 目标 将客户端的平台资源目录映射为 Go 平台和三个完整发行包。
type 目标 struct {
	目录       string `json:"directory"`
	平台       string `json:"platform"`
	架构       string `json:"arch"`
	Python   来源     `json:"python"`
	节点       来源     `json:"node"`
	Git      Git来源  `json:"git"`
	Python包集 []包来源  `json:"python_packages,omitempty"`
	工作区      *工作区来源 `json:"workspace,omitempty"`
}

// 包来源 固定 PyPI 滚轮 的来源、校验值和模块入口；准备时只解压，不运行安装脚本。
type 包来源 struct {
	名称 string `json:"name"`
	来源
}

type 来源集 struct {
	格式版本 int  `json:"format_version"`
	目标集  []目标 `json:"targets"`
}

// 读取来源集 接收固定本地归档及显式 HTTPS 来源，不执行归档内容或解析 最新。
// URL 为空的用户提供资源只允许使用已有本地归档。
func 读取来源集(路径 string) (_ []目标, 返回错误 error) {
	文件, 错误 := 操作系统包.打开(路径)
	if 错误 != nil {
		return nil, 错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 文件.关闭()) }()
	解码器 := JSON包.新建解码器(输入输出包.上限读取器(文件, 64<<10))
	解码器.禁止未知字段集()
	var 值 来源集
	if 错误 := 解码器.解码(&值); 错误 != nil {
		return nil, 错误
	}
	if 解码器.解码(new(any)) != 输入输出包.文件结束 || 值.格式版本 != 运行时资源包.格式版本 || len(值.目标集) == 0 || len(值.目标集) > 16 {
		return nil, 错误判断包.新建("资源来源清单无效")
	}
	已见 := map[string]bool{}
	for _, 目标 := range 值.目标集 {
		if !简单名称(目标.目录) || 已见[目标.目录] || !切片包.包含([]string{"darwin", "windows", "linux"}, 目标.平台) || !切片包.包含([]string{"arm64", "amd64"}, 目标.架构) {
			return nil, 错误判断包.新建("资源目标平台无效或重复")
		}
		for _, 名称 := range []string{目标.Git.进程执行路径, 目标.Git.模板路径, 目标.Git.配置路径} {
			if !安全路径(名称) {
				return nil, 错误判断包.新建("Git 配套路径无效")
			}
		}
		已见[目标.目录] = true
		if 错误 := 校验包集(目标); 错误 != nil {
			return nil, 错误
		}
		if 错误 := 校验工作区(目标.工作区); 错误 != nil {
			return nil, 错误
		}
		for _, 来源 := range []来源{目标.Python, 目标.节点, 目标.Git.来源} {
			已解析, 错误 := URL包.解析(来源.URL)
			_, 哈希错误 := 十六进制包.解码字符串(来源.SHA256)
			if 错误 != nil || (来源.URL != "" && (已解析.方案 != "https" || 已解析.主机 == "" || 已解析.用户 != nil || 已解析.片段 != "" || 已解析.原始查询 != "")) || 哈希错误 != nil || len(来源.SHA256) != 64 || 来源.版本 == "" || !简单名称(来源.归档) || (!简单名称(来源.前缀) && 来源.前缀 != ".") || !安全路径(来源.条目) {
				return nil, 错误判断包.新建("运行时归档来源或入口无效")
			}
		}
	}
	return 值.目标集, nil
}

func 简单名称(名称 string) bool {
	return 名称 != "." && 名称 != ".." && 名称 != "" && !字符串包.包含任一字符(名称, `/\:`) && !字符串包.包含字符(名称, 0)
}

// 安装 在隔离暂存目录准备一个平台，并在全部校验成功后以 重命名 激活。
// 已存在的目录只做一致性校验，拒绝覆盖损坏或不同版本资源；离线 禁止下载缺失归档。
func 安装(执行上下文 上下文包.上下文, 目标 目标, 输出根, 缓存根 string, 离线 bool) (返回错误 error) {
	if !文件路径包.是否绝对路径(输出根) || !文件路径包.是否绝对路径(缓存根) || !简单名称(目标.目录) {
		return 错误判断包.新建("资源输出与缓存目录必须是绝对路径")
	}
	if 错误 := 操作系统包.创建目录全部(输出根, 0755); 错误 != nil {
		return 错误
	}
	加锁, 错误 := 获取准备加锁(输出根, 目标.目录)
	if 错误 != nil {
		return 错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 加锁.关闭()) }()
	归档集 := make(map[string]string, 3)
	包归档集 := make([]string, 0, len(目标.Python包集))
	var 工作区归档集 []string
	if 错误 := 校验工作区(目标.工作区); 错误 != nil {
		return 错误
	}
	if 目标.工作区 != nil {
		for _, 来源 := range 目标.工作区.归档集 {
			路径 := 文件路径包.合并(缓存根, 来源.归档)
			if 错误 := 核验归档(执行上下文, 路径, 来源.SHA256); 错误 != nil {
				return 格式化包.格式化创建错误("离线 工作区依赖归档 %s 不可用: %w", 来源.归档, 错误)
			}
			工作区归档集 = append(工作区归档集, 路径)
		}
	}
	for _, 依赖 := range 目标.Python包集 {
		归档, 错误 := 已缓存归档(执行上下文, 依赖.来源, 缓存根, 离线)
		if 错误 != nil {
			return 错误
		}
		包归档集 = append(包归档集, 归档)
	}
	for _, 项 := range []struct {
		名称 string
		来源 来源
	}{{"python", 目标.Python}, {"node", 目标.节点}, {"git", 目标.Git.来源}} {
		归档, 错误 := 已缓存归档(执行上下文, 项.来源, 缓存根, 离线)
		if 错误 != nil {
			return 错误
		}
		归档集[项.名称] = 归档
	}
	目标 := 文件路径包.合并(输出根, 目标.目录)
	if _, 错误 := 操作系统包.读取链接自身状态(目标); 错误 == nil {
		文件, 错误 := 操作系统包.打开(文件路径包.合并(目标, "manifest.json"))
		if 错误 != nil {
			return 错误判断包.新建("已有运行时目录缺少清单，拒绝覆盖")
		}
		清单, 解码错误 := 运行时资源包.解码(文件)
		if 错误 := 错误判断包.合并(解码错误, 文件.关闭()); 错误 != nil {
			return 错误
		}
		if !切片包.相等(清单.Python包集, 包清单(目标)) {
			return 错误判断包.新建("已有运行时的 Python 依赖与清单不一致，请在独立输出目录重新准备")
		}
		if !相同工作区(清单.工作区, 目标.工作区) {
			return 错误判断包.新建("已有运行时的完整 工作区依赖资源不一致，请在独立目录重新准备")
		}
		文件集, 错误 := 运行时资源包.资源库存(执行上下文, 目标)
		if 错误 != nil || 清单.平台 != 目标.平台 || 清单.架构 != 目标.架构 || 清单.Python != (运行时资源包.可执行文件{版本: 目标.Python.版本, 路径: "python/" + 目标.Python.条目}) || 清单.节点 != (运行时资源包.可执行文件{版本: 目标.节点.版本, 路径: "node/" + 目标.节点.条目}) || 清单.Git != 目标.Git.清单() || !切片包.相等(文件集, 清单.文件集) {
			return 错误判断包.新建("已有运行时资源与固定版本不一致，拒绝覆盖")
		}
		return 运行时资源包.校验布局(目标, 清单)
	} else if !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		return 错误
	}
	阶段, 错误 := 操作系统包.创建目录临时(输出根, ".prepare-"+目标.目录+"-")
	if 错误 != nil {
		return 错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 移除阶段(阶段)) }()
	for _, 项 := range []struct {
		名称 string
		来源 来源
	}{{"python", 目标.Python}, {"node", 目标.节点}, {"git", 目标.Git.来源}} {
		组件 := 文件路径包.合并(阶段, 项.名称)
		if 错误 := 提取(执行上下文, 归档集[项.名称], 组件, 项.来源.前缀); 错误 != nil {
			return 格式化包.格式化创建错误("解压 %s: %w", 项.名称, 错误)
		}
	}
	if 目标.平台 == "windows" {
		// Windows 不依赖符号链接权限；提供与 Unix 一致的 Python3 命令名称。
		原始 := 文件路径包.合并(阶段, "python", 文件路径包.来源斜杠(目标.Python.条目))
		别名 := 文件路径包.合并(文件路径包.目录(原始), "python3.exe")
		if _, 错误 := 操作系统包.状态信息(别名); 错误判断包.是否属于(错误, 操作系统包.不存在错误) {
			数据, 错误 := 操作系统包.读取文件(原始)
			if 错误 != nil {
				return 错误
			}
			if 错误 := 操作系统包.写入文件(别名, 数据, 0755); 错误 != nil {
				return 错误
			}
		}
	}
	if 错误 := 安装包集(执行上下文, 目标, 阶段, 包归档集); 错误 != nil {
		return 错误
	}
	if len(工作区归档集) > 0 {
		for _, 归档 := range 工作区归档集 {
			if 错误 := 提取进入(执行上下文, 归档, 阶段, "."); 错误 != nil {
				return 格式化包.格式化创建错误("解压工作区依赖: %w", 错误)
			}
		}
	}
	文件集, 错误 := 运行时资源包.资源库存(执行上下文, 阶段)
	if 错误 != nil {
		return 错误
	}
	清单 := 运行时资源包.资源清单{格式版本: 运行时资源包.格式版本, 平台: 目标.平台, 架构: 目标.架构, Python: 运行时资源包.可执行文件{版本: 目标.Python.版本, 路径: "python/" + 目标.Python.条目}, 节点: 运行时资源包.可执行文件{版本: 目标.节点.版本, 路径: "node/" + 目标.节点.条目}, Git: 目标.Git.清单(), Python包集: 包清单(目标), 文件集: 文件集}
	if 目标.工作区 != nil {
		清单.工作区 = new(目标.工作区.运行时)
	}
	if 错误 := 运行时资源包.校验布局(阶段, 清单); 错误 != nil {
		return 错误
	}
	if 错误 := 运行时资源包.写入清单(阶段, 清单); 错误 != nil {
		return 错误
	}
	if 错误 := 文件路径包.遍历目录(阶段, func(路径 string, 条目 文件系统包.目录条目, 错误 error) error {
		if 错误 != nil {
			return 错误
		}
		if 条目.类型()&操作系统包.模式符号链接 != 0 {
			return nil
		}
		信息, 错误 := 条目.获取信息()
		if 错误 != nil {
			return 错误
		}
		模式 := 文件系统包.文件模式(0444)
		if 信息.是否目录() || 信息.模式().权限()&0111 != 0 {
			模式 = 0555
		}
		return 操作系统包.修改权限(路径, 模式)
	}); 错误 != nil {
		return 错误
	}
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	return 操作系统包.重命名(阶段, 目标)
}

// 保留锁文件的身份。旧版目录锁可能由仍在运行的旧进程持有，不能推测为陈旧后删除。
func 获取准备加锁(输出根, 目标 string) (*操作系统包.文件, error) {
	路径 := 文件路径包.合并(输出根, ".prepare-"+目标+".lock")
	if 信息, 错误 := 操作系统包.读取链接自身状态(路径); 错误 == nil {
		if 信息.是否目录() {
			return nil, 错误判断包.新建("检测到旧格式资源准备目录锁，请确认旧准备进程已退出后处理该锁目录")
		}
		if !信息.模式().是否普通文件() {
			return nil, 错误判断包.新建("资源准备锁必须是普通文件")
		}
	} else if !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		return nil, 错误
	}
	文件, 错误 := 操作系统包.打开文件(路径, 操作系统包.创建文件标志|操作系统包.读写打开标志, 0600)
	if 错误 != nil {
		return nil, 错误
	}
	已打开, 状态信息错误 := 文件.状态信息()
	当前, 路径错误 := 操作系统包.读取链接自身状态(路径)
	if 状态信息错误 != nil || 路径错误 != nil || !当前.模式().是否普通文件() || !操作系统包.相同文件(已打开, 当前) {
		return nil, 错误判断包.合并(错误判断包.新建("资源准备锁身份已变化"), 状态信息错误, 路径错误, 文件.关闭())
	}
	if 错误 := 文件锁包.尝试(文件); 错误 != nil {
		return nil, 错误判断包.合并(格式化包.格式化创建错误("无法取得资源准备锁: %w", 错误), 文件.关闭())
	}
	return 文件, nil
}

func 移除阶段(目录 string) error {
	if _, 错误 := 操作系统包.状态信息(目录); 错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		return nil
	}
	if 错误 := 文件路径包.遍历目录(目录, func(路径 string, 条目 文件系统包.目录条目, 错误 error) error {
		if 错误 != nil {
			return 错误
		}
		if 条目.是否目录() {
			return 操作系统包.修改权限(路径, 0700)
		}
		return nil
	}); 错误 != nil {
		return 错误
	}
	return 操作系统包.移除全部(目录)
}

func 已缓存归档(执行上下文 上下文包.上下文, 来源 来源, 缓存 string, 离线 bool) (_ string, 返回错误 error) {
	if !离线 {
		if 错误 := 操作系统包.创建目录全部(缓存, 0700); 错误 != nil {
			return "", 错误
		}
	}
	路径 := 文件路径包.合并(缓存, 来源.归档)
	if _, 错误 := 操作系统包.状态信息(路径); 错误 == nil {
		return 路径, 核验归档(执行上下文, 路径, 来源.SHA256)
	} else if !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		return "", 错误
	}
	if 离线 {
		return "", 格式化包.格式化创建错误("缺少本地运行时归档 %s，离线模式不允许下载", 来源.归档)
	}
	if 来源.URL == "" {
		return "", 格式化包.格式化创建错误("归档 %s 仅支持已核验的本地资源", 来源.归档)
	}
	请求, 错误 := HTTP包.新建请求带有上下文(执行上下文, HTTP包.方法获取, 来源.URL, nil)
	if 错误 != nil {
		return "", 错误
	}
	传输模块 := &HTTP包.传输{代理: HTTP包.代理来源环境, 强制尝试HTTP2: true, TLS握手超时: 10 * 时间包.秒, 响应头部超时: 30 * 时间包.秒, 空闲连接超时: 30 * 时间包.秒}
	defer 传输模块.关闭空闲连接()
	客户端 := &HTTP包.客户端{传输: 传输模块, 超时: 10 * 时间包.分钟, 检查重定向: func(请求 *HTTP包.请求, 通过 []*HTTP包.请求) error {
		if 请求.URL.方案 != "https" || len(通过) >= 10 {
			return 错误判断包.新建("运行时下载重定向无效")
		}
		return nil
	}}
	响应, 错误 := 客户端.执行(请求)
	if 错误 != nil {
		return "", 错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 响应.正文.关闭()) }()
	if 响应.状态代码 != HTTP包.状态成功 {
		return "", 格式化包.格式化创建错误("运行时下载返回 HTTP %d", 响应.状态代码)
	}
	文件, 错误 := 操作系统包.创建临时(缓存, ".download-")
	if 错误 != nil {
		return "", 错误
	}
	defer func() {
		if 错误 := 操作系统包.移除(文件.名称()); 错误 != nil && !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
			返回错误 = 错误判断包.合并(返回错误, 错误)
		}
	}()
	数量, 复制错误 := 输入输出包.复制(文件, 输入输出包.上限读取器(响应.正文, (512<<20)+1))
	if 错误 := 错误判断包.合并(复制错误, 文件.关闭()); 错误 != nil {
		return "", 错误
	}
	if 数量 > 512<<20 {
		return "", 错误判断包.新建("运行时归档超过大小限制")
	}
	if 错误 := 核验归档(执行上下文, 文件.名称(), 来源.SHA256); 错误 != nil {
		return "", 错误
	}
	if 错误 := 操作系统包.重命名(文件.名称(), 路径); 错误 != nil {
		return "", 错误
	}
	return 路径, nil
}

func 核验归档(执行上下文 上下文包.上下文, 路径, 预期 string) (返回错误 error) {
	文件, 错误 := 操作系统包.打开(路径)
	if 错误 != nil {
		return 错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 文件.关闭()) }()
	哈希 := SHA256包.新建()
	缓冲区 := make([]byte, 128<<10)
	var 大小 int64
	for {
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			return 错误
		}
		数量, 错误 := 文件.读取(缓冲区)
		大小 += int64(数量)
		if 大小 > 512<<20 {
			return 错误判断包.新建("运行时归档超过大小限制")
		}
		_, _ = 哈希.写入(缓冲区[:数量])
		if 错误 == 输入输出包.文件结束 {
			break
		}
		if 错误 != nil {
			return 错误
		}
	}
	if 十六进制包.编码目标字符串(哈希.校验和(nil)) != 预期 {
		return 错误判断包.新建("运行时归档 SHA-256 不匹配")
	}
	return nil
}
