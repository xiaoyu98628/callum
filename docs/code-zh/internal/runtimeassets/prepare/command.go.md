```go
package 准备模块

import (
	上下文包 "context"
	错误判断包 "errors"
	命令行参数包 "flag"
	格式化包 "fmt"
	输入输出包 "io"
	结构化日志包 "log/slog"
	操作系统包 "os"
	文件路径包 "path/filepath"
)

// 运行命令 执行独立的资源准备命令；离线仅 调用方不能通过参数启用下载。
// 此入口不创建应用数据目录、数据库或 标准输入输出 服务。
func 运行命令(执行上下文 上下文包.上下文, 参数 []string, 日志记录器 *结构化日志包.日志记录器, 离线仅 bool) error {
	参数集 := 命令行参数包.新建命令行参数设置("prepare-runtimes", 命令行参数包.继续开启错误)
	参数集.设置输出(输入输出包.丢弃)
	来源 := 参数集.字符串("sources", "", "固定版本来源清单")
	输出 := 参数集.字符串("output", "", "客户端 runtimes 资源根目录")
	缓存 := 参数集.字符串("cache", "", "本地归档目录")
	目标 := 参数集.字符串("target", "", "可选的平台目录名；省略时准备全部目标")
	下载 := 参数集.布尔("download", false, "维护资源时允许下载缺失归档；默认严格离线")
	if 错误 := 参数集.解析(参数); 错误 != nil {
		if 错误判断包.是否属于(错误, 命令行参数包.请求帮助错误) {
			参数集.设置输出(操作系统包.标准错误输出)
			参数集.用量()
		}
		return 错误
	}
	if *来源 == "" || *输出 == "" || *缓存 == "" || 参数集.剩余参数数量() != 0 {
		return 格式化包.格式化创建错误("必须指定 --sources、--output 和 --cache")
	}
	目标集, 错误 := 读取来源集(*来源)
	if 错误 != nil {
		return 错误
	}
	输出路径, 错误 := 文件路径包.绝对值(*输出)
	if 错误 != nil {
		return 错误
	}
	缓存路径, 错误 := 文件路径包.绝对值(*缓存)
	if 错误 != nil {
		return 错误
	}
	if *下载 && 离线仅 {
		return 错误判断包.新建("此资源准备入口仅支持离线归档")
	}
	已匹配 := false
	for _, 条目 := range 目标集 {
		if *目标 != "" && 条目.目录 != *目标 {
			continue
		}
		已匹配 = true
		if 错误 := 安装(执行上下文, 条目, 输出路径, 缓存路径, !*下载); 错误 != nil {
			return 格式化包.格式化创建错误("准备 %s: %w", 条目.目录, 错误)
		}
		日志记录器.信息("运行时资源已校验", "platform", 条目.目录)
	}
	if !已匹配 {
		return 格式化包.格式化创建错误("来源清单没有指定目标")
	}
	return nil
}
```
