func 测试夹具(测试器 *测试包.测试器) string {
	测试器.标记测试辅助函数()
	目录 := 文件路径包.合并(测试器.创建临时目录(), "runtime with spaces")
	for _, 名称 := range []string{"python/bin/python3", "node/bin/node", "python/lib/stdlib.py", "git/bin/git", "git/etc/gitconfig", "git/libexec/git-core/helper", "git/share/templates/template"} {
		完整 := 文件路径包.合并(目录, 文件路径包.来源斜杠(名称))
		if 错误 := 操作系统包.创建目录全部(文件路径包.目录(完整), 0755); 错误 != nil {
			测试器.报告致命错误(错误)
		}
		if 错误 := 操作系统包.写入文件(完整, []byte("fixture-content"), 0755); 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	文件集, 错误 := 运行时资源包.资源库存(测试器.上下文(), 目录)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 运行时资源包.写入清单(目录, 运行时资源包.资源清单{格式版本: 运行时资源包.格式版本, 平台: 运行时包.目标操作系统, 架构: 运行时包.目标架构, Python: 运行时资源包.可执行文件{版本: "3.14.10", 路径: "python/bin/python3"}, 节点: 运行时资源包.可执行文件{版本: "26.13.0", 路径: "node/bin/node"}, Git: 运行时资源包.Git运行时{可执行文件: 运行时资源包.可执行文件{版本: "fixed", 路径: "git/bin/git"}, 进程执行路径: "git/libexec/git-core", 模板路径: "git/share/templates", 配置路径: "git/etc/gitconfig"}, 文件集: 文件集}); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	return 目录
}

func Test资源包拒绝缺失变化通知并且附加资源集(测试器 *测试包.测试器) {
	for _, 变更 := range []string{"library", "executable", "missing", "additional", "git", "git-config", "git-helper"} {
		测试器.执行(变更, func(测试器 *测试包.测试器) {
			目录 := 测试夹具(测试器)
			资源包, 错误 := 运行时资源包.加载(测试器.上下文(), 目录)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			if 错误 := 资源包.检查(测试器.上下文()); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			for _, 名称 := range []string{"python", "python3", "node", "git"} {
				条目, 成功 := 资源包.解析(名称)
				if !成功 || !文件路径包.是否绝对路径(条目) {
					测试器.格式化报告致命错误("入口没有固定为绝对路径: %q", 条目)
				}
			}
			if _, 成功 := 资源包.解析("./python"); 成功 {
				测试器.报告致命错误("显式路径被裸命令映射覆盖")
			}
			名称 := "python/lib/stdlib.py"
			if 变更 == "executable" {
				名称 = "python/bin/python3"
			} else if 变更 == "additional" {
				名称 = "python/lib/sitecustomize.py"
			}
			switch 变更 {
			case "git":
				名称 = "git/bin/git"
			case "git-config":
				名称 = "git/etc/gitconfig"
			case "git-helper":
				名称 = "git/libexec/git-core/helper"
			}
			路径 := 文件路径包.合并(目录, 文件路径包.来源斜杠(名称))
			if 变更 == "missing" {
				错误 = 操作系统包.移除(路径)
			} else {
				错误 = 操作系统包.写入文件(路径, []byte("changed-content"), 0755)
			}
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			if _, 错误 := 运行时资源包.加载(测试器.上下文(), 目录); 错误 == nil {
				测试器.报告致命错误("资源变更后仍接受清单")
			}
			if (变更 == "executable" || 变更 == "git" || 变更 == "git-config") && 资源包.检查(测试器.上下文()) == nil {
				测试器.报告致命错误("调用前未发现解释器被改写")
			}
		})
	}
}

func Test资源包上下文并且转义链接(测试器 *测试包.测试器) {
	目录 := 测试夹具(测试器)
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	取消函数()
	if _, 错误 := 运行时资源包.加载(执行上下文, 目录); !错误判断包.是否属于(错误, 上下文包.已取消错误) {
		测试器.格式化报告致命错误("忽略初始化取消: %v", 错误)
	}
	if 运行时包.目标操作系统 == "windows" {
		测试器.跳过("Windows 符号链接权限由原生验收覆盖")
	}
	外部 := 文件路径包.合并(测试器.创建临时目录(), "outside")
	if 错误 := 操作系统包.写入文件(外部, []byte("outside"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 操作系统包.符号链接(外部, 文件路径包.合并(目录, "python", "escape")); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 运行时资源包.资源库存(测试器.上下文(), 目录); 错误 == nil {
		测试器.报告致命错误("接受了资源外链接")
	}
}

func Test资源包清单SHA256绑定原始清单并且检查集文件集(测试器 *测试包.测试器) {
	目录 := 测试夹具(测试器)
	路径 := 文件路径包.合并(目录, "manifest.json")
	数据, 错误 := 操作系统包.读取文件(路径)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	预期 := SHA256包.校验和256(数据)
	if _, 错误 := 运行时资源包.加载带有清单SHA256(测试器.上下文(), 目录, 预期); 错误 != nil {
		测试器.格式化报告致命错误("拒绝符合预期清单的完整资源: %v", 错误)
	}
	if 错误 := 操作系统包.修改权限(路径, 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	变化通知 := append(数据, '\n')
	if 错误 := 操作系统包.写入文件(路径, 变化通知, 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 运行时资源包.加载(测试器.上下文(), 目录); 错误 != nil {
		测试器.格式化报告致命错误("旧校验调用应继续接受等价 JSON: %v", 错误)
	}
	if _, 错误 := 运行时资源包.加载带有清单SHA256(测试器.上下文(), 目录, 预期); 错误 == nil {
		测试器.报告致命错误("接受了原始字节与源清单不同的清单")
	}
	文件 := 文件路径包.合并(目录, "python", "lib", "stdlib.py")
	if 错误 := 操作系统包.写入文件(文件, []byte("changed-content"), 0755); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 运行时资源包.加载带有清单SHA256(测试器.上下文(), 目录, SHA256包.校验和256(变化通知)); 错误 == nil {
		测试器.报告致命错误("清单摘要相同后跳过了普通文件内容校验")
	}
}

func Test资源包拒绝不同架构(测试器 *测试包.测试器) {
	目录 := 测试夹具(测试器)
	路径 := 文件路径包.合并(目录, "manifest.json")
	文件, 错误 := 操作系统包.打开(路径)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	清单, 解码错误 := 运行时资源包.解码(文件)
	if 错误 := 错误判断包.合并(解码错误, 文件.关闭()); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	清单.架构 = "arm64"
	if 运行时包.目标架构 == "arm64" {
		清单.架构 = "amd64"
	}
	if 错误 := 操作系统包.修改权限(路径, 0644); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 运行时资源包.写入清单(目录, 清单); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 运行时资源包.加载(测试器.上下文(), 目录); 错误 == nil {
		测试器.报告致命错误("接受了不同架构的资源")
	}
}
