```go
// 包 服务 校验已注册的应用设置，不依赖通信协议或全局内存缓存。
package 服务

import (
	上下文包 "context"
	JSON包 "encoding/json"
	错误判断包 "errors"
	数学包 "math"

	设置包 "github.com/xiaoyu98628/callum/internal/settings"
)

// 服务 借用设置仓储，数据库生命周期由装配层负责。
type 服务 struct{ 仓储 设置包.仓储 }

func 新建(仓储 设置包.仓储) (*服务, error) {
	if 仓储 == nil {
		return nil, 设置包.无效错误
	}
	return &服务{仓储: 仓储}, nil
}

func (当前服务 *服务) 获取(执行上下文 上下文包.上下文, 键 string) (设置包.记录, error) {
	if 键 != 设置包.执行键 && 键 != 设置包.压缩键 && 键 != 设置包.常规键 {
		return 设置包.记录{}, 设置包.无效错误
	}
	值, 错误 := 当前服务.仓储.获取(执行上下文, 键)
	if 错误 != nil {
		return 值, 错误
	}
	switch 键 {
	case 设置包.执行键:
		快照, 错误 := 设置包.执行快照(值)
		if 错误 != nil {
			return 设置包.记录{}, 错误
		}
		// 将旧配置的明确兼容默认值返回给客户端，不改写历史记录和 修订号。
		值.值, 错误 = JSON包.序列化(快照.执行)
		if 错误 != nil {
			return 设置包.记录{}, 错误判断包.合并(设置包.存储错误, 错误)
		}
	case 设置包.压缩键:
		if _, 错误 := 设置包.解码压缩快照(值); 错误 != nil {
			return 设置包.记录{}, 错误
		}
	case 设置包.常规键:
		if _, 错误 := 设置包.解码常规记录(值); 错误 != nil {
			return 设置包.记录{}, 错误
		}
	}
	return 值, nil
}

// 更新 完整替换一个已注册配置组；调用方提交其读取到的 修订号，冲突后重新读取。
// 值在写入前完成校验，返回成功表示事务已经提交，后续受理执行即可读取新值。
func (当前服务 *服务) 更新(执行上下文 上下文包.上下文, 键 string, 修订号 int64, 原始 JSON包.原始JSON消息) (设置包.记录, error) {
	if (键 != 设置包.执行键 && 键 != 设置包.压缩键 && 键 != 设置包.常规键) || 修订号 < 1 || 修订号 == 数学包.最大整数64 {
		return 设置包.记录{}, 设置包.无效错误
	}
	var 值 any
	var 错误 error
	switch 键 {
	case 设置包.执行键:
		var 执行 设置包.执行
		执行, 错误 = 设置包.解码执行(原始)
		if 错误 == nil {
			var 字段集 map[string]JSON包.原始JSON消息
			if 错误 = JSON包.反序列化(原始, &字段集); 错误 == nil && 字段集["sandbox_mode"] == nil {
				// 旧客户端完整替换原有两字段时，保留用户已选的沙箱模式。
				var 当前 设置包.记录
				当前, 错误 = 当前服务.获取(执行上下文, 键)
				if 错误 == nil && 当前.修订号 != 修订号 {
					return 设置包.记录{}, 设置包.冲突错误
				}
				if 错误 == nil {
					var 快照 设置包.快照
					快照, 错误 = 设置包.执行快照(当前)
					执行.沙箱模式 = 快照.沙箱模式
				}
			}
		}
		if 错误 == nil && !执行.有效() {
			错误 = 设置包.无效错误
		}
		值 = 执行
	case 设置包.压缩键:
		值, 错误 = 设置包.解码压缩(原始)
	case 设置包.常规键:
		值, 错误 = 设置包.解码常规(原始)
	}
	if 错误 != nil {
		return 设置包.记录{}, 错误
	}
	规范化, 错误 := JSON包.序列化(值)
	if 错误 != nil {
		return 设置包.记录{}, 错误判断包.合并(设置包.无效错误, 错误)
	}
	return 当前服务.仓储.更新(执行上下文, 键, 修订号, 规范化)
}
```
