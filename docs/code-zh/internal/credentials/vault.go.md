```go
// 包 凭据 保护持久化凭据，明文仅在实际模型调用装配时解密。
package 凭据

import (
	上下文包 "context"
	AES包 "crypto/aes"
	密码算法包 "crypto/cipher"
	随机包 "crypto/rand"
	SHA256包 "crypto/sha256"
	密码辅助包 "crypto/subtle"
	基准64包 "encoding/base64"
	十六进制包 "encoding/hex"
	错误判断包 "errors"
	字符串包 "strings"
	同步包 "sync"
)

var (
	未找到错误 = 错误判断包.新建("系统凭据不存在")
	不可用错误 = 错误判断包.新建("系统凭据库不可用或已锁定")
	无效错误  = 错误判断包.新建("加密凭据无效或无法解密")
)

// 最大字节 与配置请求字节边界一致，不把系统主密钥条目的大小限制施加给 API 键。
const 最大字节 = 1 << 20

// 系统凭据库 只保存每个数据目录的随机主密钥，不能将其与数据库一起明文落盘。
// 实现负责调用范围、取消、输出边界和系统资源清理，不得记录密钥。
type 系统凭据库 interface {
	加载(上下文包.上下文, string) ([]byte, error)
	保存(上下文包.上下文, string, []byte) error
}

// 密码算法 由装配层提供给配置仓储；准备 仅在有旧凭据迁移时于事务外调用。
type 密码算法 interface {
	准备(上下文包.上下文) error
	加密封装(上下文包.上下文, string, string) (string, error)
	打开(上下文包.上下文, string, string) (string, error)
}

// 凭据保险库 缓存一个数据目录的主密钥；持有者在所有调用结束后 关闭。
type 凭据保险库 struct {
	互斥锁   同步包.互斥锁
	系统凭据库 系统凭据库
	名称    string
	键     []byte
	已关闭   bool
}

func 新建(系统凭据库 系统凭据库, 身份 string) (*凭据保险库, error) {
	if 系统凭据库 == nil || 身份 == "" {
		return nil, 无效错误
	}
	哈希 := SHA256包.校验和256([]byte(身份))
	return &凭据保险库{系统凭据库: 系统凭据库, 名称: 十六进制包.编码目标字符串(哈希[:])}, nil
}

func (当前凭据保险库 *凭据保险库) 主密钥(执行上下文 上下文包.上下文, 创建 bool) ([]byte, error) {
	当前凭据保险库.互斥锁.加锁()
	defer 当前凭据保险库.互斥锁.解锁()
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return nil, 错误
	}
	if 当前凭据保险库.已关闭 {
		return nil, 不可用错误
	}
	if 当前凭据保险库.键 == nil {
		键, 错误 := 当前凭据保险库.系统凭据库.加载(执行上下文, 当前凭据保险库.名称)
		if 错误判断包.是否属于(错误, 未找到错误) && 创建 {
			键 = make([]byte, 32)
			if _, 错误 := 随机包.读取(键); 错误 != nil {
				return nil, 错误
			}
			if 错误 := 当前凭据保险库.系统凭据库.保存(执行上下文, 当前凭据保险库.名称, 键); 错误 != nil {
				clear(键)
				return nil, 错误判断包.合并(不可用错误, 错误, 执行上下文.获取错误())
			}
			// 系统调用成功不替代写入核对；失败不能让数据库使用一个未安全保存的密钥。
			已存储, 读取错误 := 当前凭据保险库.系统凭据库.加载(执行上下文, 当前凭据保险库.名称)
			if 读取错误 != nil || 密码辅助包.常量时间比较(键, 已存储) != 1 {
				clear(键)
				clear(已存储)
				return nil, 错误判断包.合并(不可用错误, 读取错误, 执行上下文.获取错误())
			}
			clear(已存储)
		} else if 错误 != nil {
			clear(键)
			return nil, 错误判断包.合并(不可用错误, 错误, 执行上下文.获取错误())
		}
		if len(键) != 32 {
			clear(键)
			return nil, 无效错误
		}
		当前凭据保险库.键 = 键
	}
	return append([]byte(nil), 当前凭据保险库.键...), nil
}

func (当前凭据保险库 *凭据保险库) 准备(执行上下文 上下文包.上下文) error {
	键, 错误 := 当前凭据保险库.主密钥(执行上下文, true)
	clear(键)
	return 错误
}

func (当前凭据保险库 *凭据保险库) 认证加密(执行上下文 上下文包.上下文, 创建 bool) (密码算法包.认证加密, error) {
	键, 错误 := 当前凭据保险库.主密钥(执行上下文, 创建)
	if 错误 != nil {
		return nil, 错误
	}
	defer clear(键)
	内容块, 错误 := AES包.新建密码算法(键)
	if 错误 != nil {
		return nil, 无效错误
	}
	return 密码算法包.新建GCM带有随机随机数(内容块)
}

func (当前凭据保险库 *凭据保险库) 加密封装(执行上下文 上下文包.上下文, 标识, 文本 string) (string, error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return "", 错误
	}
	if 文本 == "" {
		return "", nil
	}
	if len(文本) > 最大字节 {
		return "", 无效错误
	}
	认证加密, 错误 := 当前凭据保险库.认证加密(执行上下文, true)
	if 错误 != nil {
		return "", 错误
	}
	return "v1:" + 当前凭据保险库.名称 + ":" + 基准64包.原始标准编码.编码目标字符串(认证加密.加密封装(nil, nil, []byte(文本), []byte("callum/provider/v1/"+当前凭据保险库.名称+"/"+标识))), nil
}

func (当前凭据保险库 *凭据保险库) 打开(执行上下文 上下文包.上下文, 标识, 已编码 string) (string, error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return "", 错误
	}
	if 已编码 == "" {
		return "", nil
	}
	if len(已编码) > 2*最大字节 {
		return "", 无效错误
	}
	部分集 := 字符串包.拆分(已编码, ":")
	if len(部分集) != 3 || 部分集[0] != "v1" || len(部分集[1]) != 64 {
		return "", 无效错误
	}
	if _, 错误 := 十六进制包.解码字符串(部分集[1]); 错误 != nil {
		return "", 无效错误
	}
	原始, 错误 := 基准64包.原始标准编码.解码字符串(部分集[2])
	if 错误 != nil || len(原始) > 最大字节+28 {
		return "", 无效错误
	}
	var 认证加密 密码算法包.认证加密
	if 部分集[1] == 当前凭据保险库.名称 {
		认证加密, 错误 = 当前凭据保险库.认证加密(执行上下文, false)
	} else {
		// 密文携带不含秘密的主密钥标识，同一系统用户移动数据目录后仍能读取。
		当前凭据保险库.互斥锁.加锁()
		if 当前凭据保险库.已关闭 {
			当前凭据保险库.互斥锁.解锁()
			return "", 不可用错误
		}
		键, 键错误 := 当前凭据保险库.系统凭据库.加载(执行上下文, 部分集[1])
		当前凭据保险库.互斥锁.解锁()
		if 键错误 != nil {
			clear(键)
			return "", 错误判断包.合并(不可用错误, 键错误, 执行上下文.获取错误())
		}
		defer clear(键)
		if len(键) != 32 {
			return "", 无效错误
		}
		内容块, 键错误 := AES包.新建密码算法(键)
		if 键错误 != nil {
			return "", 无效错误
		}
		认证加密, 错误 = 密码算法包.新建GCM带有随机随机数(内容块)
	}
	if 错误 != nil {
		return "", 错误
	}
	文本, 错误 := 认证加密.打开(nil, nil, 原始, []byte("callum/provider/v1/"+部分集[1]+"/"+标识))
	if 错误 != nil {
		return "", 无效错误
	}
	defer clear(文本)
	return string(文本), nil
}

func (当前凭据保险库 *凭据保险库) 关闭() error {
	当前凭据保险库.互斥锁.加锁()
	defer 当前凭据保险库.互斥锁.解锁()
	clear(当前凭据保险库.键)
	当前凭据保险库.键, 当前凭据保险库.已关闭 = nil, true
	return nil
}
```
