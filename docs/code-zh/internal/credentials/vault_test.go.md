package 凭据

import (
	上下文包 "context"
	基准64包 "encoding/base64"
	错误判断包 "errors"
	字符串包 "strings"
	测试包 "testing"
)

type 内存系统凭据库 struct {
	键    []byte
	失败   bool
	保存次数 int
}

func (当前内存系统凭据库 *内存系统凭据库) 加载(上下文包.上下文, string) ([]byte, error) {
	if 当前内存系统凭据库.失败 {
		return nil, 不可用错误
	}
	if 当前内存系统凭据库.键 == nil {
		return nil, 未找到错误
	}
	return append([]byte(nil), 当前内存系统凭据库.键...), nil
}
func (当前内存系统凭据库 *内存系统凭据库) 保存(_ 上下文包.上下文, _ string, 键 []byte) error {
	当前内存系统凭据库.保存次数++
	当前内存系统凭据库.键 = append([]byte(nil), 键...)
	return nil
}

func Test凭据保险库认证持久化并且闭包(测试器 *测试包.测试器) {
	存储 := &内存系统凭据库{}
	值, 错误 := 新建(存储, "temporary database")
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() { 值.关闭() })
	首个, 错误 := 值.加密封装(测试器.上下文(), "provider-a", "private-value")
	if 错误 != nil || 字符串包.包含(首个, "private-value") {
		测试器.报告致命错误("凭据未加密", 错误)
	}
	秒, 错误 := 值.加密封装(测试器.上下文(), "provider-a", "private-value")
	if 错误 != nil || 首个 == 秒 || 存储.保存次数 != 1 {
		测试器.报告致命错误("nonce 或主密钥生命周期错误", 错误)
	}
	已重新打开, _ := 新建(存储, "temporary database")
	defer 已重新打开.关闭()
	if 值, 错误 := 已重新打开.打开(测试器.上下文(), "provider-a", 首个); 错误 != nil || 值 != "private-value" {
		测试器.报告致命错误("无法恢复凭据", 错误)
	}
	部分集 := 字符串包.拆分(首个, ":")
	变化通知, _ := 基准64包.原始标准编码.解码字符串(部分集[2])
	变化通知[0] ^= 1
	被篡改 := 部分集[0] + ":" + 部分集[1] + ":" + 基准64包.原始标准编码.编码目标字符串(变化通知)
	for _, 测试用例 := range []struct{ 标识, 值 string }{{"provider-b", 首个}, {"provider-a", 被篡改}, {"provider-a", "%%%"}} {
		if _, 错误 := 值.打开(测试器.上下文(), 测试用例.标识, 测试用例.值); !错误判断包.是否属于(错误, 无效错误) {
			测试器.报告致命错误("接受被替换的密文", 错误)
		}
	}
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	取消函数()
	if _, 错误 := 值.加密封装(执行上下文, "id", "secret"); !错误判断包.是否属于(错误, 上下文包.已取消错误) {
		测试器.报告致命错误(错误)
	}
	if 错误 := 值.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 值.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 值.打开(测试器.上下文(), "provider-a", 首个); !错误判断包.是否属于(错误, 不可用错误) {
		测试器.报告致命错误("关闭后仍使用主密钥", 错误)
	}
}

func Test缺失或者已加锁键从不创建替换期间解密(测试器 *测试包.测试器) {
	for _, 失败 := range []bool{false, true} {
		存储 := &内存系统凭据库{失败: 失败}
		值, _ := 新建(存储, "temporary database")
		if _, 错误 := 值.打开(测试器.上下文(), "provider", "v1:"+值.名称+":"+字符串包.重复("A", 64)); !错误判断包.是否属于(错误, 不可用错误) || 存储.保存次数 != 0 {
			测试器.报告致命错误("解密时创建了替代主密钥", 错误)
		}
		值.关闭()
	}
}

func Test移动中数据库保留原始键引用(测试器 *测试包.测试器) {
	存储 := &内存系统凭据库{}
	原始, _ := 新建(存储, "original database path")
	值, 错误 := 原始.加密封装(测试器.上下文(), "provider", "private-value")
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	原始.关闭()
	已移动, _ := 新建(存储, "moved database path")
	文本, 错误 := 已移动.打开(测试器.上下文(), "provider", 值)
	if 错误 != nil || 文本 != "private-value" || 存储.保存次数 != 1 {
		测试器.报告致命错误("移动数据目录后丢失原凭据", 错误)
	}
	已移动.关闭()
	if _, 错误 := 已移动.打开(测试器.上下文(), "provider", 值); !错误判断包.是否属于(错误, 不可用错误) {
		测试器.报告致命错误("关闭后仍加载旧目录主密钥", 错误)
	}
}
