//go:build darwin

package 系统

import (
	上下文包 "context"
	随机包 "crypto/rand"
	SHA256包 "crypto/sha256"
	十六进制包 "encoding/hex"
	操作系统包 "os"
	进程执行包 "os/exec"
	切片包 "slices"
	测试包 "testing"
	时间包 "time"
)

func Test隔离系统钥匙串轮次往返(测试器 *测试包.测试器) {
	if 操作系统包.获取环境变量("CALLUM_LIVE_KEYRING_TEST") != "1" {
		测试器.跳过("系统凭据库实测需显式启用；默认测试使用隔离替身")
	}
	摘要 := SHA256包.校验和256([]byte("callum-credential-test-" + 随机包.文本()))
	名称 := 十六进制包.编码目标字符串(摘要[:])
	键 := make([]byte, 32)
	if _, 错误 := 随机包.读取(键); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer clear(键)
	执行上下文, 取消函数 := 上下文包.创建超时上下文(测试器.上下文(), 5*时间包.秒)
	defer 取消函数()
	// 只删除本次随机命名的测试条目，不枚举或读取其他用户凭据。
	测试器.注册清理函数(func() {
		清理, 停止 := 上下文包.创建超时上下文(上下文包.无取消(测试器.上下文()), 5*时间包.秒)
		defer 停止()
		if 错误 := 进程执行包.命令上下文(清理, "/usr/bin/security", "delete-generic-password", "-s", 服务, "-a", 名称).执行(); 错误 != nil {
			测试器.错误文本("清理测试钥匙串条目失败", 错误)
		}
	})
	存储 := 新建()
	if 错误 := 存储.保存(执行上下文, 名称, 键); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	实际值, 错误 := 存储.加载(执行上下文, 名称)
	defer clear(实际值)
	if 错误 != nil || !切片包.相等(实际值, 键) {
		测试器.报告致命错误("系统凭据读写不一致", 错误)
	}
}
