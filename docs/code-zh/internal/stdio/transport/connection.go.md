```go
package 传输模块

import (
	字节包 "bytes"
	上下文包 "context"
	JSON包 "encoding/json"
	错误判断包 "errors"
	格式化包 "fmt"
	输入输出包 "io"
	操作系统包 "os"
	同步包 "sync"
	时间包 "time"

	协议包 "github.com/xiaoyu98628/callum/internal/stdio/protocol"
)

const (
	// 最大请求字节 是单条协议请求的字节上限，不含换行。
	最大请求字节 = 1 << 20
	// 最大响应字节 是单条协议响应的字节上限，不含换行。
	最大响应字节    = 8 << 20
	输出队列大小     = 64
	默认写入超时 = 5 * 时间包.秒
)

var 缓慢消费者错误 = 错误判断包.新建("协议输出阻塞超过时限")

// 连接 持有标准流、取消信号和有界输出队列，统一串行输出协议消息。
// 创建者必须停止请求处理及事件生产者后再调用 关闭。
type 连接 struct {
	父级       上下文包.上下文
	执行上下文          上下文包.上下文
	取消函数       上下文包.取消原因函数
	输入           输入输出包.读取关闭器
	输出          输入输出包.写入关闭器
	队列        chan []byte
	完成         chan struct{}
	写入超时 时间包.时长
	关闭一次    同步包.一次
	关闭错误     error
	观察完成    chan struct{}
	停止观察    func() bool
	结束一次   同步包.一次
	结束错误    error
}

// 新建 接管输入输出并启动写入循环，关闭 必须能够中断底层 读取/写入。
// 外部取消或通信失败会关闭管道，创建失败时输入输出仍由调用方持有。
func 新建(执行上下文 上下文包.上下文, 输入 输入输出包.读取关闭器, 输出 输入输出包.写入关闭器) (*连接, error) {
	if 输入 == nil || 输出 == nil {
		return nil, 错误判断包.新建("协议输入输出不能为空")
	}
	父级 := 执行上下文
	执行上下文, 取消函数 := 上下文包.带有取消原因(执行上下文)
	丙 := &连接{父级: 父级, 执行上下文: 执行上下文, 取消函数: 取消函数, 输入: 输入, 输出: 输出, 队列: make(chan []byte, 输出队列大小), 完成: make(chan struct{}), 写入超时: 默认写入超时, 观察完成: make(chan struct{})}
	丙.停止观察 = 上下文包.之后函数(执行上下文, func() { 丙.关闭流集(); close(丙.观察完成) })
	go 丙.写入循环()
	return 丙, nil
}

// 上下文 控制本连接的读取、受理和事件转发，不控制共享运行时。
func (当前连接 *连接) 上下文() 上下文包.上下文 { return 当前连接.执行上下文 }

// 失败 以指定原因中断本连接，并关闭标准流解除读写阻塞。
func (当前连接 *连接) 失败(错误 error) { 当前连接.取消函数(错误) }

// 关闭 在连接正常时排空输出，关闭管道并保留通信与清理错误，可重复调用。
// 调用前必须停止 提供服务 与所有事件生产者，防止关闭队列后继续发送事件。
func (当前连接 *连接) 关闭() error {
	当前连接.结束一次.执行(func() {
		close(当前连接.队列)
		<-当前连接.完成
		if !当前连接.停止观察() {
			<-当前连接.观察完成
		}
		当前连接.关闭流集()
		原因 := 上下文包.原因(当前连接.执行上下文)
		if 当前连接.父级.获取错误() != nil && 错误判断包.是否属于(原因, 上下文包.原因(当前连接.父级)) {
			原因 = nil
		}
		当前连接.结束错误 = 错误判断包.合并(原因, 当前连接.关闭错误)
		当前连接.取消函数(上下文包.已取消错误)
	})
	return 当前连接.结束错误
}

// 规范化错误 避免重复报告由连接取消引起的管道错误，连接原始错误由 关闭 保留。
func (当前连接 *连接) 规范化错误(错误 error) error {
	if 当前连接.执行上下文.获取错误() != nil && (错误判断包.是否属于(错误, 上下文包.原因(当前连接.执行上下文)) || 错误判断包.是否属于(错误, 输入输出包.已关闭管道错误) || 错误判断包.是否属于(错误, 操作系统包.已关闭错误)) {
		return nil
	}
	return 错误
}
func (当前连接 *连接) 关闭流集() {
	当前连接.关闭一次.执行(func() { 当前连接.关闭错误 = 错误判断包.合并(当前连接.输入.关闭(), 当前连接.输出.关闭()) })
}

// 发送 编码并排队发送消息；超限、超时或编码失败会取消连接。
func (当前连接 *连接) 发送(值 any) error {
	数据, 错误 := JSON包.序列化(值)
	if 错误 != nil {
		当前连接.取消函数(错误判断包.新建("协议响应编码失败"))
		return 错误
	}
	if len(数据) > 最大响应字节 {
		错误 := 错误判断包.新建("协议响应超过大小限制")
		当前连接.取消函数(错误)
		return 错误
	}
	数据 = append(数据, '\n')
	计时器 := 时间包.新建计时器(当前连接.写入超时)
	defer 计时器.停止()
	select {
	case <-当前连接.执行上下文.完成():
		return 上下文包.原因(当前连接.执行上下文)
	case 当前连接.队列 <- 数据:
		return nil
	case <-计时器.字符:
		当前连接.取消函数(缓慢消费者错误)
		return 缓慢消费者错误
	}
}

func (当前连接 *连接) 写入循环() {
	defer close(当前连接.完成)
	for 数据 := range 当前连接.队列 {
		if 当前连接.执行上下文.获取错误() != nil {
			return
		}
		// 即使队列尚未填满，也限制单条消息的写入等待；超时关闭管道以解除阻塞。
		计时器 := 时间包.之后函数(当前连接.写入超时, func() { 当前连接.取消函数(缓慢消费者错误) })
		_, 错误 := 输入输出包.复制(当前连接.输出, 字节包.新建读取器(数据))
		计时器.停止()
		if 错误 != nil {
			当前连接.取消函数(格式化包.格式化创建错误("写入协议输出失败: %w", 错误))
			return
		}
	}
}

// 输入 返回供服务读取的输入；资源关闭由 连接 统一负责，只允许一个读取方。
func (当前连接 *连接) 获取输入() 输入输出包.读取器 { return 当前连接.输入 }

// 回复 按请求 标识 返回响应，合法通知不产生响应。
func (当前连接 *连接) 回复(请求 协议包.请求, 结果 any, 错误 *协议包.错误) error {
	if 请求.标识 == nil {
		return nil
	}
	return 当前连接.发送(协议包.响应{JSONRPC: "2.0", 标识: 请求.标识, 结果: 结果, 错误: 错误})
}

// 通知 发送服务端事件，多个调用者共享同一串行输出队列。
func (当前连接 *连接) 通知(方法 string, 参数 any) error {
	return 当前连接.发送(协议包.通知{JSONRPC: "2.0", 方法: 方法, 参数: 参数})
}
```
