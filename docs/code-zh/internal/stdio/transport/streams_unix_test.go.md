//go:build darwin || linux

package 传输模块

import (
	错误判断包 "errors"
	操作系统包 "os"
	字符串包 "strings"
	系统调用包 "syscall"
	测试包 "testing"
	时间包 "time"
)

func 文件参数集(测试器 *测试包.测试器, 文件 *操作系统包.文件) uintptr {
	测试器.标记测试辅助函数()
	原始, 错误 := 文件.系统调用连接()
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	var 参数集 uintptr
	var 操作错误 error
	if 错误 := 原始.控制(func(文件描述符 uintptr) {
		var 系统错误码 系统调用包.系统错误码
		参数集, _, 系统错误码 = 系统调用包.系统调用(系统调用包.系统文件控制, 文件描述符, 系统调用包.获取文件状态标志, 0)
		if 系统错误码 != 0 {
			操作错误 = 系统错误码
		}
	}); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 操作错误 != nil {
		测试器.报告致命错误(操作错误)
	}
	return 参数集
}

func Test继承标准输入输出可被已中断并且恢复参数集(测试器 *测试包.测试器) {
	for _, 写入中 := range []bool{false, true} {
		测试器.执行(map[bool]string{false: "读取", true: "写入"}[写入中], func(测试器 *测试包.测试器) {
			var 文件描述符集 [2]int
			if 错误 := 系统调用包.管道(文件描述符集[:]); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			读取器 := 操作系统包.新建文件(uintptr(文件描述符集[0]), "inherited-reader")
			写入器 := 操作系统包.新建文件(uintptr(文件描述符集[1]), "inherited-writer")
			测试器.注册清理函数(func() {
				if 错误 := 读取器.关闭(); 错误 != nil {
					测试器.错误文本(错误)
				}
				if 错误 := 写入器.关闭(); 错误 != nil {
					测试器.错误文本(错误)
				}
			})
			原始 := 读取器
			if 写入中 {
				原始 = 写入器
			}
			之前 := 文件参数集(测试器, 原始)
			已包装, 错误 := 新建可轮询文件(原始)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			测试器.注册清理函数(func() {
				if 错误 := 已包装.关闭(); 错误 != nil {
					测试器.错误文本(错误)
				}
			})
			if 错误 := 已包装.设置截止时间(时间包.当前时间().添加(时间包.小时)); 错误 != nil {
				测试器.格式化报告致命错误("继承描述符未注册到运行时轮询器: %v", 错误)
			}
			已开始 := make(chan struct{})
			完成 := make(chan error, 1)
			go func() {
				close(已开始)
				if 写入中 {
					_, 错误 := 已包装.写入([]byte(字符串包.重复("x", 8<<20)))
					完成 <- 错误
				} else {
					_, 错误 := 已包装.读取(make([]byte, 1))
					完成 <- 错误
				}
			}()
			<-已开始
			if 错误 := 已包装.关闭(); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			select {
			case 错误 := <-完成:
				if !错误判断包.是否属于(错误, 操作系统包.已关闭错误) {
					测试器.格式化报告致命错误("关闭未中断 I/O: %v", 错误)
				}
			case <-时间包.之后(时间包.秒):
				测试器.报告致命错误("关闭后 I/O 仍然阻塞")
			}
			if 实际值 := 文件参数集(测试器, 原始); 实际值 != 之前 {
				测试器.格式化报告致命错误("未恢复原始文件状态: before=%d after=%d", 之前, 实际值)
			}
			if _, 错误 := 原始.状态信息(); 错误 != nil {
				测试器.格式化报告致命错误("借用的原始描述符被关闭: %v", 错误)
			}
		})
	}
}

func Test共享标准输入输出参数集恢复之后两者流集关闭(测试器 *测试包.测试器) {
	var 文件描述符集 [2]int
	if 错误 := 系统调用包.管道(文件描述符集[:]); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	读取器 := 操作系统包.新建文件(uintptr(文件描述符集[0]), "shared-reader")
	写入器 := 操作系统包.新建文件(uintptr(文件描述符集[1]), "writer")
	测试器.注册清理函数(func() {
		if 错误 := 读取器.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
		if 错误 := 写入器.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	之前 := 文件参数集(测试器, 读取器)
	输入, 错误 := 新建可轮询文件(读取器)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 输入.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	输出, 错误 := 新建可轮询文件(读取器)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 输出.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	恢复一起(输入, 输出)
	if 错误 := 输入.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 文件参数集(测试器, 读取器)&系统调用包.非阻塞标志 == 0 {
		测试器.报告致命错误("另一个副本仍持有文件时过早恢复了阻塞模式")
	}
	if 错误 := 输出.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 文件参数集(测试器, 读取器) != 之前 {
		测试器.报告致命错误("两个副本关闭后未恢复原始标志")
	}
}
