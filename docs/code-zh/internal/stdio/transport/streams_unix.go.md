//go:build darwin || linux

package 传输模块

import (
	错误判断包 "errors"
	输入输出包 "io"
	操作系统包 "os"
	同步包 "sync"
	原子操作包 "sync/atomic"
	系统调用包 "syscall"
)

// 可轮询文件 持有标准流的副本，借用原始描述符以恢复共享的非阻塞标志。
// 原始标准流在副本关闭之前必须保持打开，main 在整个运行期间持有它们。
type 可轮询文件 struct {
	*操作系统包.文件
	原始        *操作系统包.文件
	恢复阻塞 bool
	一次            同步包.一次
	关闭错误        error
	之后关闭      func() error
}

func 新建可轮询文件(原始 *操作系统包.文件) (*可轮询文件, error) {
	原始, 错误 := 原始.系统调用连接()
	if 错误 != nil {
		return nil, 错误
	}
	重复 := -1
	var 操作错误 error
	var 恢复 bool
	控制错误 := 原始.控制(func(文件描述符 uintptr) {
		参数集, _, 系统错误码 := 系统调用包.系统调用(系统调用包.系统文件控制, 文件描述符, 系统调用包.获取文件状态标志, 0)
		if 系统错误码 != 0 {
			操作错误 = 系统错误码
			return
		}
		恢复 = 参数集&系统调用包.非阻塞标志 == 0
		重复, 操作错误 = 系统调用包.复制文件描述符(int(文件描述符))
		if 操作错误 != nil {
			return
		}
		系统调用包.关闭开启进程执行(重复)
		操作错误 = 系统调用包.设置非阻塞(重复, true)
	})
	if 错误 := 错误判断包.合并(控制错误, 操作错误); 错误 != nil {
		if 重复 >= 0 {
			错误 = 错误判断包.合并(错误, 系统调用包.关闭(重复))
		}
		return nil, 错误
	}
	// 新建文件 只有接收到非阻塞描述符才会将继承的终端或管道注册到运行时轮询器。
	// 此后 关闭 可以解除等待，避免 Ctrl+字符 或写入超时被阻塞系统调用拖住。
	return &可轮询文件{文件: 操作系统包.新建文件(uintptr(重复), 原始.名称()), 原始: 原始, 恢复阻塞: 恢复}, nil
}

func (当前可轮询文件 *可轮询文件) 关闭() error {
	当前可轮询文件.一次.执行(func() {
		// 先关闭副本、等待其读写退出，再恢复标志，防止正在读取的副本重新变成阻塞模式。
		当前可轮询文件.关闭错误 = 当前可轮询文件.文件.关闭()
		if 当前可轮询文件.之后关闭 != nil {
			当前可轮询文件.关闭错误 = 错误判断包.合并(当前可轮询文件.关闭错误, 当前可轮询文件.之后关闭())
		} else {
			当前可轮询文件.关闭错误 = 错误判断包.合并(当前可轮询文件.关闭错误, 当前可轮询文件.恢复())
		}
	})
	return 当前可轮询文件.关闭错误
}

func (当前可轮询文件 *可轮询文件) 恢复() error {
	if !当前可轮询文件.恢复阻塞 {
		return nil
	}
	原始, 错误 := 当前可轮询文件.原始.系统调用连接()
	if 错误 != nil {
		return 错误
	}
	var 恢复错误 error
	控制错误 := 原始.控制(func(文件描述符 uintptr) { 恢复错误 = 系统调用包.设置非阻塞(int(文件描述符), false) })
	return 错误判断包.合并(控制错误, 恢复错误)
}

func 恢复一起(输入, 输出 *可轮询文件) {
	var 剩余 原子操作包.整数32
	剩余.存储(2)
	// 终端的 标准输入/标准输出 可能共享同一文件状态，必须等两个副本都停止读写后再恢复。
	之后关闭 := func() error {
		if 剩余.添加(-1) == 0 {
			return 错误判断包.合并(输入.恢复(), 输出.恢复())
		}
		return nil
	}
	输入.之后关闭, 输出.之后关闭 = 之后关闭, 之后关闭
}

// 打开标准输入输出 创建可取消的标准流副本，接收方负责关闭；原始标准流保持打开。
func 打开标准输入输出() (输入输出包.读取关闭器, 输入输出包.写入关闭器, error) {
	输入, 错误 := 新建可轮询文件(操作系统包.标准输入)
	if 错误 != nil {
		return nil, nil, 错误
	}
	输出, 错误 := 新建可轮询文件(操作系统包.标准输出)
	if 错误 != nil {
		return nil, nil, 错误判断包.合并(错误, 输入.关闭())
	}
	恢复一起(输入, 输出)
	return 输入, 输出, nil
}
