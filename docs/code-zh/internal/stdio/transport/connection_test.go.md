package 传输模块

import (
	错误判断包 "errors"
	输入输出包 "io"
	字符串包 "strings"
	测试包 "testing"
	时间包 "time"

	协议包 "github.com/xiaoyu98628/callum/internal/stdio/protocol"
)

func Test已阻止输出取消并且关闭连接(测试器 *测试包.测试器) {
	读取器, 写入器 := 输入输出包.管道()
	defer func() {
		if 错误 := 读取器.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	}()
	连接, 错误 := 新建(测试器.上下文(), 输入输出包.空操作关闭器(字符串包.新建读取器("")), 写入器)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 连接.关闭(); !错误判断包.是否属于(错误, 缓慢消费者错误) {
			测试器.格式化创建错误("未保留阻塞错误: %v", 错误)
		}
	})
	连接.写入超时 = 30 * 时间包.毫秒
	if 错误 := 连接.发送(协议包.通知{JSONRPC: "2.0", 方法: "system.ready"}); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 连接.关闭(); !错误判断包.是否属于(错误, 缓慢消费者错误) {
		测试器.格式化报告致命错误("未报告阻塞: %v", 错误)
	}
}
