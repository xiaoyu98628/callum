import (
	上下文包 "context"
	错误判断包 "errors"

	供应商包 "github.com/xiaoyu98628/callum/internal/model/provider"
	协议包 "github.com/xiaoyu98628/callum/internal/stdio/protocol"
)

// 公开错误 只输出受控描述，禁止将存储、网络和供应商错误链直接发给客户端。
func 公开错误(错误 error) *协议包.错误 {
	当前条目 := &协议包.错误{代码: -32603, 消息: "内部处理失败"}
	类别 := "internal"
	switch {
	case 错误判断包.是否属于(错误, 上下文包.超出截止时间错误):
		当前条目.代码, 当前条目.消息, 类别 = 1008, "操作超时", "timeout"
	case 错误判断包.是否属于(错误, 上下文包.已取消错误):
		当前条目.代码, 当前条目.消息, 类别 = 1007, "操作已取消", "cancelled"
	case 错误判断包.是否属于(错误, 供应商包.无效配置错误):
		return 协议包.无效参数()
	case 错误判断包.是否属于(错误, 供应商包.未找到错误):
		当前条目.代码, 当前条目.消息, 类别 = 1001, "配置不存在", "not_found"
	case 错误判断包.是否属于(错误, 供应商包.冲突错误):
		当前条目.代码, 当前条目.消息, 类别 = 1002, "同一厂商下的远端模型标识已存在", "conflict"
	case 错误判断包.是否属于(错误, 供应商包.被引用错误):
		当前条目.代码, 当前条目.消息, 类别 = 1003, "关联配置不存在或仍被引用", "referenced"
	case 错误判断包.是否属于(错误, 供应商包.已停用错误):
		当前条目.代码, 当前条目.消息, 类别 = 1004, "模型已禁用", "disabled"
	case 错误判断包.是否属于(错误, 供应商包.凭据错误):
		当前条目.代码, 当前条目.消息, 类别 = 1012, "系统凭据库不可用或凭据无法解密，请解锁凭据库后重试。", "credentials_unavailable"
	}
	当前条目.数据 = map[string]any{"kind": 类别}
	return 当前条目
}
