```go
package 会话

import (
	错误判断包 "errors"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	业务域包 "github.com/xiaoyu98628/callum/internal/conversation"
	执行包 "github.com/xiaoyu98628/callum/internal/stdio/execution"
	协议包 "github.com/xiaoyu98628/callum/internal/stdio/protocol"
)

func 公开错误(错误 error) *协议包.错误 {
	if 错误 == nil {
		return nil
	}
	switch {
	case 错误判断包.是否属于(错误, 业务域包.无效错误), 错误判断包.是否属于(错误, 智能体包.无效错误):
		return 协议包.无效参数()
	case 错误判断包.是否属于(错误, 业务域包.未找到错误), 错误判断包.是否属于(错误, 智能体包.未找到错误):
		return &协议包.错误{代码: 1001, 消息: "会话、执行或消息不存在"}
	case 错误判断包.是否属于(错误, 业务域包.繁忙错误):
		return &协议包.错误{代码: 1006, 消息: "该会话正在执行"}
	case 错误判断包.是否属于(错误, 业务域包.非活动错误):
		return &协议包.错误{代码: 1003, 消息: "会话或工作区已归档或删除"}
	case 错误判断包.是否属于(错误, 业务域包.附件变化通知错误):
		return &协议包.错误{代码: 1003, 消息: 业务域包.附件变化通知错误.错误文本()}
	case 错误判断包.是否属于(错误, 业务域包.无法继续错误):
		return &协议包.错误{代码: 1003, 消息: 业务域包.无法继续错误.错误文本()}
	case 错误判断包.是否属于(错误, 业务域包.后续消息变化通知错误), 错误判断包.是否属于(错误, 业务域包.后续消息上限错误), 错误判断包.是否属于(错误, 业务域包.后续消息已发送错误), 错误判断包.是否属于(错误, 业务域包.后续消息草稿错误):
		return &协议包.错误{代码: 1003, 消息: 错误.错误文本()}
	case 错误判断包.是否属于(错误, 业务域包.冲突错误):
		return &协议包.错误{代码: 1003, 消息: "只能编辑最后一条用户消息并重发"}
	default:
		已转换 := 执行包.错误文本(错误)
		if 已转换.代码 == 1009 {
			return &协议包.错误{代码: -32603, 消息: "会话操作失败"}
		}
		return 已转换
	}
}
```
