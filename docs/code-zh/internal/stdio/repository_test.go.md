```go
package 标准输入输出

import (
	上下文包 "context"
	切片包 "slices"

	供应商包 "github.com/xiaoyu98628/callum/internal/model/provider"
)

// 内存仓储 为协议测试提供存储替身，SQLite 约束与持久化由存储集成测试验证。
// 协议请求串行访问此替身，测试中的模型解析不访问其状态。
type 内存仓储 struct {
	供应商集 map[string]供应商包.供应商输入
	模型集  map[string]供应商包.模型
}

func (当前内存仓储 *内存仓储) 创建供应商(_ 上下文包.上下文, 标识 string, 输入 供应商包.供应商输入) error {
	当前内存仓储.供应商集[标识] = 输入
	return nil
}
func (当前内存仓储 *内存仓储) 更新供应商(_ 上下文包.上下文, 标识 string, 输入 供应商包.供应商输入) error {
	旧, 成功 := 当前内存仓储.供应商集[标识]
	if !成功 {
		return 供应商包.未找到错误
	}
	if 输入.API密钥 == nil {
		输入.API密钥 = 旧.API密钥
	}
	当前内存仓储.供应商集[标识] = 输入
	return nil
}
func (当前内存仓储 *内存仓储) 获取供应商(_ 上下文包.上下文, 标识 string) (供应商包.供应商, error) {
	输入, 成功 := 当前内存仓储.供应商集[标识]
	if !成功 {
		return 供应商包.供应商{}, 供应商包.未找到错误
	}
	return 供应商包.供应商{标识: 标识, 名称: 输入.名称, 协议: 输入.协议, 基础URL: 输入.基础URL, 已配置API密钥: 输入.API密钥 != nil && *输入.API密钥 != ""}, nil
}
func (当前内存仓储 *内存仓储) 列表供应商集(执行上下文 上下文包.上下文, 选项集 ...供应商包.列表选项集) (供应商包.页[供应商包.供应商], error) {
	输出 := make([]供应商包.供应商, 0, len(当前内存仓储.供应商集))
	for 标识 := range 当前内存仓储.供应商集 {
		当前对象, 错误 := 当前内存仓储.获取供应商(执行上下文, 标识)
		if 错误 != nil {
			return 供应商包.页[供应商包.供应商]{}, 错误
		}
		输出 = append(输出, 当前对象)
	}
	切片包.排序函数(输出, func(当前供应商, 当前供应商 供应商包.供应商) int {
		if 当前供应商.标识 < 当前供应商.标识 {
			return -1
		}
		if 当前供应商.标识 > 当前供应商.标识 {
			return 1
		}
		return 0
	})
	return 内存页(输出, 选项集, func(当前供应商 供应商包.供应商) string { return 当前供应商.标识 })
}
func (当前内存仓储 *内存仓储) 删除供应商(_ 上下文包.上下文, 标识 string) error {
	if _, 成功 := 当前内存仓储.供应商集[标识]; !成功 {
		return 供应商包.未找到错误
	}
	for _, 映射 := range 当前内存仓储.模型集 {
		if 映射.供应商标识 == 标识 {
			return 供应商包.被引用错误
		}
	}
	delete(当前内存仓储.供应商集, 标识)
	return nil
}
func (当前内存仓储 *内存仓储) 创建模型(_ 上下文包.上下文, 标识 string, 输入 供应商包.模型输入) error {
	if _, 成功 := 当前内存仓储.供应商集[输入.供应商标识]; !成功 {
		return 供应商包.被引用错误
	}
	for _, 映射 := range 当前内存仓储.模型集 {
		if 映射.供应商标识 == 输入.供应商标识 && 映射.远程模型标识 == 输入.远程模型标识 {
			return 供应商包.冲突错误
		}
	}
	if 输入.模型类型 == nil {
		输入.模型类型 = []string{}
	}
	当前内存仓储.模型集[标识] = 供应商包.模型{标识: 标识, 模型输入: 输入}
	return nil
}
func (当前内存仓储 *内存仓储) 更新模型(_ 上下文包.上下文, 标识 string, 输入 供应商包.模型输入) error {
	if _, 成功 := 当前内存仓储.模型集[标识]; !成功 {
		return 供应商包.未找到错误
	}
	if 输入.模型类型 == nil {
		输入.模型类型 = []string{}
	}
	当前内存仓储.模型集[标识] = 供应商包.模型{标识: 标识, 模型输入: 输入}
	return nil
}
func (当前内存仓储 *内存仓储) 获取模型(_ 上下文包.上下文, 标识 string) (供应商包.模型, error) {
	映射, 成功 := 当前内存仓储.模型集[标识]
	if !成功 {
		return 映射, 供应商包.未找到错误
	}
	return 映射, nil
}
func (当前内存仓储 *内存仓储) 列表模型集(_ 上下文包.上下文, 已启用仅 bool, 选项集 ...供应商包.列表选项集) (供应商包.页[供应商包.模型], error) {
	输出 := make([]供应商包.模型, 0, len(当前内存仓储.模型集))
	for _, 映射 := range 当前内存仓储.模型集 {
		if !已启用仅 || 映射.已启用 {
			输出 = append(输出, 映射)
		}
	}
	切片包.排序函数(输出, func(当前模型, 当前模型 供应商包.模型) int {
		if 当前模型.标识 < 当前模型.标识 {
			return -1
		}
		if 当前模型.标识 > 当前模型.标识 {
			return 1
		}
		return 0
	})
	return 内存页(输出, 选项集, func(当前模型 供应商包.模型) string { return 当前模型.标识 })
}
func (当前内存仓储 *内存仓储) 设置模型已启用(_ 上下文包.上下文, 标识 string, 已启用 bool) error {
	映射, 成功 := 当前内存仓储.模型集[标识]
	if !成功 {
		return 供应商包.未找到错误
	}
	映射.已启用 = 已启用
	当前内存仓储.模型集[标识] = 映射
	return nil
}
func (当前内存仓储 *内存仓储) 删除模型(_ 上下文包.上下文, 标识 string) error {
	if _, 成功 := 当前内存仓储.模型集[标识]; !成功 {
		return 供应商包.未找到错误
	}
	delete(当前内存仓储.模型集, 标识)
	return nil
}
func (当前内存仓储 *内存仓储) 加载运行时(_ 上下文包.上下文, 标识 string) (供应商包.运行时配置, error) {
	映射, 成功 := 当前内存仓储.模型集[标识]
	if !成功 {
		return 供应商包.运行时配置{}, 供应商包.未找到错误
	}
	当前对象 := 当前内存仓储.供应商集[映射.供应商标识]
	键 := ""
	if 当前对象.API密钥 != nil {
		键 = *当前对象.API密钥
	}
	return 供应商包.运行时配置{模型: 映射, 协议: 当前对象.协议, 基础URL: 当前对象.基础URL, API密钥: 键}, nil
}

func 内存页[元素类型 any](项集 []元素类型, 选项集 []供应商包.列表选项集, 项标识 func(元素类型) string) (供应商包.页[元素类型], error) {
	值, 错误 := 供应商包.规范化列表选项集(选项集...)
	if 错误 != nil {
		return 供应商包.页[元素类型]{}, 错误
	}
	项集 = 切片包.删除函数(项集, func(当前元素类型 元素类型) bool { return 项标识(当前元素类型) <= 值.之后标识 })
	页 := 供应商包.页[元素类型]{项集: 项集}
	if len(项集) > 值.上限 {
		页.项集 = 项集[:值.上限]
		页.还有更多 = true
		页.下一个之后标识 = 项标识(页.项集[len(页.项集)-1])
	}
	return 页, nil
}
```
