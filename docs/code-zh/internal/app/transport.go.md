```go
package 应用模块

import (
	错误判断包 "errors"
	输入输出包 "io"
	HTTP包 "net/http"
)

// 限制 SDK 读取的原始响应，防止非流式响应在应用检查文本长度前无限分配内存。
// 此限制也适用于流式响应的累计传输字节，与文本大小和 词元 容量相互独立。
const 最大上游字节 = 16 << 20

var 响应上限错误 = 错误判断包.新建("模型原始响应超过 16 MiB 限制")

type 有界传输 struct{ 基准 HTTP包.轮次往返器 }

func (当前有界传输 有界传输) 轮次往返(请求 *HTTP包.请求) (*HTTP包.响应, error) {
	响应, 错误 := 当前有界传输.基准.轮次往返(请求)
	if 错误 == nil && 响应 != nil && 响应.正文 != nil {
		响应.正文 = &有界正文{读取关闭器: 响应.正文, 剩余: 最大上游字节}
	}
	return 响应, 错误
}

type 有界正文 struct {
	输入输出包.读取关闭器
	剩余 int
}

func (当前有界正文 *有界正文) 读取(字节数据 []byte) (int, error) {
	if len(字节数据) == 0 {
		return 0, nil
	}
	if 当前有界正文.剩余 == 0 {
		var 探测 [1]byte
		数量, 错误 := 当前有界正文.读取关闭器.读取(探测[:])
		if 数量 > 0 {
			return 0, 响应上限错误
		}
		return 0, 错误
	}
	if len(字节数据) > 当前有界正文.剩余 {
		字节数据 = 字节数据[:当前有界正文.剩余]
	}
	数量, 错误 := 当前有界正文.读取关闭器.读取(字节数据)
	当前有界正文.剩余 -= 数量
	return 数量, 错误
}
```
