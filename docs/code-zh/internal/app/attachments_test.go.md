func Test继续拒绝不兼容模型之前变化中执行(测试器 *测试包.测试器) {
	运行时, _ := 运行时测试夹具(测试器, 1)
	执行上下文 := 测试器.上下文()
	var 调用集 原子操作包.整数32
	上游 := HTTP测试包.新建服务端(HTTP包.处理器函数(func(当前响应写入器 HTTP包.响应写入器, 当前请求 *HTTP包.请求) {
		调用集.添加(1)
		正文, 错误 := 输入输出包.读取全部(当前请求.正文)
		if 错误 != nil || !字节包.包含(正文, []byte("data:image/png;base64,")) {
			测试器.格式化创建错误("继续请求缺少原图片: %v", 错误)
		}
		当前响应写入器.头部().设置("Content-Type", "application/json")
		_, _ = 输入输出包.写入字符串(当前响应写入器, `{"choices":[{"index":0,"message":{"role":"assistant","content":"已继续"},"finish_reason":"stop"}]}`)
	}))
	测试器.注册清理函数(上游.关闭)
	进程标识, 错误 := 运行时.获取供应商集().创建供应商(执行上下文, 供应商包.供应商输入{名称: "厂商", 协议: "chatcompletions", 基础URL: 上游.URL + "/v1"})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	添加模型 := func(名称 string, 类型集 ...string) string {
		测试器.标记测试辅助函数()
		标识, 错误 := 运行时.获取供应商集().创建模型(执行上下文, 供应商包.模型输入{供应商标识: 进程标识, 显示名称: 名称, 远程模型标识: 名称,
			已启用: true, 模型类型: 类型集, 上下文窗口: new(int64(1000000))})
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		return 标识
	}
	视觉, 文本 := 添加模型("vision", "image"), 添加模型("text", "text")
	var 原始 字节包.缓冲区
	if 错误 := PNG包.编码(&原始, 图片包.新建RGBA(图片包.矩形(0, 0, 2, 2))); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	路径 := 文件路径包.合并(测试器.创建临时目录(), "resume.png")
	if 错误 := 操作系统包.写入文件(路径, 原始.字节(), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}

	服务 := 运行时.获取会话集()
	var 初始 会话包.执行
	错误 = 服务.发送(执行上下文, 会话包.发送{模型标识: 视觉, 流: new(false), 内容: []模型包.内容块{{类型: "image", 路径: 路径}}}, func(本次执行 会话包.执行) error {
		初始 = 本次执行
		if !运行时.取消执行(本次执行.执行.标识) {
			测试器.报告致命错误("受理阶段无法取消任务")
		}
		return nil
	})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	标识, 运行 := 初始.会话.标识, 初始.执行.标识
	运行时.管理器.等待(运行)
	之前, 错误 := 服务.历史(执行上下文, 标识, "", 25)
	if 错误 != nil || len(之前.执行集) != 1 || 之前.执行集[0].状态 != 智能体包.已取消 {
		测试器.格式化报告致命错误("未准备好已停止的任务: %+v %v", 之前, 错误)
	}
	步骤集, 错误 := 运行时.执行集().步骤集(执行上下文, 运行, 0, 20)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	当前对象 := 共享通信对端(测试器, 运行时)
	参数 := map[string]any{"conversation_id": 标识, "run_id": 运行, "model_id": 文本}
	if 已拒绝 := 当前对象.调用(测试器, "conversation.continue", 参数); 已拒绝.错误 == nil || 已拒绝.错误.代码 != 1005 {
		测试器.格式化报告致命错误("不兼容的恢复未返回既有错误: %+v", 已拒绝)
	}
	之后, 错误 := 服务.历史(执行上下文, 标识, "", 25)
	if 错误 != nil || !反射包.深度相等(之前, 之后) || 调用集.加载() != 0 {
		测试器.格式化报告致命错误("拒绝恢复后原执行状态被修改或调用了模型: %v", 错误)
	}
	之后步骤集, 错误 := 运行时.执行集().步骤集(执行上下文, 运行, 0, 20)
	if 错误 != nil || !反射包.深度相等(步骤集, 之后步骤集) {
		测试器.格式化报告致命错误("拒绝恢复后改写了执行步骤: %v", 错误)
	}
	已保存, 错误 := 服务.获取(执行上下文, 标识)
	if 错误 != nil || 已保存.模型.标识 != 视觉 {
		测试器.格式化报告致命错误("拒绝恢复后改写了会话模型: %+v %v", 已保存.模型, 错误)
	}
	参数["model_id"] = 视觉
	if 实际值 := 结果字符串(测试器, 当前对象.调用(测试器, "conversation.continue", 参数), "run_id"); 实际值 != 运行 {
		测试器.格式化报告致命错误("继续使用了不同执行: %q", 实际值)
	}
	当前对象.事件(测试器, "run.completed")
	if 调用集.加载() != 1 {
		测试器.格式化报告致命错误("模型调用次数: %d", 调用集.加载())
	}
	当前对象.停止(测试器)
}

func Test附件历史编辑重发并且恢复(测试器 *测试包.测试器) {
	var 原始 字节包.缓冲区
	if 错误 := PNG包.编码(&原始, 图片包.新建RGBA(图片包.矩形(0, 0, 2, 2))); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	预期 := "data:image/png;base64," + 基准64包.标准编码.编码目标字符串(原始.字节())
	var 调用集 原子操作包.整数32
	上游 := HTTP测试包.新建服务端(HTTP包.处理器函数(func(当前响应写入器 HTTP包.响应写入器, 当前请求 *HTTP包.请求) {
		数据, 错误 := 输入输出包.读取全部(当前请求.正文)
		if 错误 != nil {
			测试器.错误文本(错误)
		}
		if !字节包.包含(数据, []byte(预期)) {
			测试器.错误文本("模型未收到原图的完整字节")
		}
		调用集.添加(1)
		当前响应写入器.头部().设置("Content-Type", "application/json")
		_, _ = 输入输出包.写入字符串(当前响应写入器, `{"choices":[{"index":0,"message":{"role":"assistant","content":"图片已收到"},"finish_reason":"stop"}]}`)
	}))
	defer 上游.关闭()
	启动配置, 错误 := 配置模块包.加载(文件路径包.合并(测试器.创建临时目录(), "data"))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	启动配置.最大并发执行集 = 1
	当前对象 := 开始应用(测试器, 启动配置, 输入输出包.丢弃)
	供应商标识 := 结果字符串(测试器, 当前对象.调用(测试器, "provider.create", map[string]any{
		"name": "附件测试", "protocol": "chatcompletions", "base_url": 上游.URL + "/v1",
	}), "id")
	模型标识 := 结果字符串(测试器, 当前对象.调用(测试器, "model.create", map[string]any{
		"provider_id": 供应商标识, "display_name": "附件模型", "remote_model_id": "vision",
		"enabled": true, "context_window": 1000000, "model_type": []string{"image"},
	}), "id")
	路径 := 文件路径包.合并(测试器.创建临时目录(), "original.png")
	if 错误 := 操作系统包.写入文件(路径, 原始.字节(), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	首个 := 当前对象.调用(测试器, "conversation.send", map[string]any{
		"model_id": 模型标识, "stream": false, "content": []map[string]string{{"type": "image", "path": 路径}},
	})
	会话 := 结果字符串(测试器, 首个, "conversation_id")
	运行 := 结果字符串(测试器, 首个, "run_id")
	用户 := 结果字符串(测试器, 首个, "user_message_id")
	当前对象.事件(测试器, "run.completed")
	历史 := 当前对象.调用(测试器, "conversation.history", map[string]any{"conversation_id": 会话})
	var 页 会话包.历史页
	要求成功(测试器, 历史)
	if 错误 := JSON包.反序列化(历史.结果, &页); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	内容块 := 页.消息集[0].内容[0]
	if 内容块.路径 != "" || 内容块.附件引用 == "" {
		测试器.格式化报告致命错误("持久化图片未使用相对引用: %+v", 内容块)
	}
	附件目录 := 文件路径包.合并(启动配置.数据目录, "attachments")
	自有 := 文件路径包.合并(附件目录, 文件路径包.来源斜杠(内容块.附件引用))
	已保存, 错误 := 操作系统包.读取文件(自有)
	if 错误 != nil || !字节包.相等(已保存, 原始.字节()) {
		测试器.格式化报告致命错误("副本不一致: %v", 错误)
	}
	if 错误 := 操作系统包.移除(路径); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	当前对象.停止(测试器)
	// 模拟已提交后尚未删除的标记，以及提交前中断留下的附件。
	待处理 := 文件路径包.合并(附件目录, ".pending")
	孤立标识 := UUID包.必须(UUID包.新建UUIDv7()).字符串()
	for _, 标识 := range []string{运行, 孤立标识} {
		if 错误 := 操作系统包.写入文件(文件路径包.合并(待处理, 会话+"_"+标识), nil, 0600); 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	孤立 := 文件路径包.合并(附件目录, 会话, 孤立标识)
	if 错误 := 操作系统包.创建目录(孤立, 0700); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 操作系统包.写入文件(文件路径包.合并(孤立, 文件路径包.基准(自有)), 原始.字节(), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	当前对象 = 开始应用(测试器, 启动配置, 输入输出包.丢弃)
	if _, 错误 := 操作系统包.状态信息(孤立); !操作系统包.是否不存在(错误) {
		测试器.格式化报告致命错误("未回收未受理的目录: %v", 错误)
	}
	条目集, 错误 := 操作系统包.读取目录(待处理)
	if 错误 != nil || len(条目集) != 0 {
		测试器.格式化报告致命错误("恢复标记未清理: %v %v", 条目集, 错误)
	}
	预览 := 当前对象.调用(测试器, "conversation.attachment_preview", map[string]any{
		"conversation_id": 会话, "message_id": 用户, "index": 0,
	})
	要求成功(测试器, 预览)
	var 查看 struct {
		MIME string `json:"mime"`
		数据   []byte `json:"data"`
	}
	if 错误 := JSON包.反序列化(预览.结果, &查看); 错误 != nil || 查看.MIME != "image/png" || !字节包.相等(查看.数据, 原始.字节()) {
		测试器.格式化报告致命错误("重启预览不一致: %v", 错误)
	}
	读取 := func(预期 string) 通信帧 {
		return 当前对象.调用(测试器, "conversation.attachment_read", map[string]any{
			"conversation_id": 会话, "message_id": 用户, "expected_run_id": 预期, "index": 0, "offset": 0,
		})
	}
	首个读取 := 读取(运行)
	要求成功(测试器, 首个读取)
	var 分片 会话包.附件分片
	摘要 := SHA256包.校验和256(原始.字节())
	if 错误 := JSON包.反序列化(首个读取.结果, &分片); 错误 != nil || 分片.执行标识 != 运行 ||
		分片.SHA256 != 十六进制包.编码目标字符串(摘要[:]) || !字节包.相等(分片.数据, 原始.字节()) {
		测试器.格式化报告致命错误("读取结果未绑定附件与消息版本: %+v %v", 分片, 错误)
	}
	编辑重发 := 当前对象.调用(测试器, "conversation.resend", map[string]any{
		"conversation_id": 会话, "message_id": 用户, "model_id": 模型标识, "stream": false,
		"expected_run_id": 运行, "retained_attachments": []int{0}, "content": []any{},
	})
	要求成功(测试器, 编辑重发)
	if 过时 := 读取(运行); 过时.错误 == nil || 过时.错误.代码 != 1003 {
		测试器.格式化报告致命错误("旧消息版本仍可另存: %+v", 过时)
	}
	要求成功(测试器, 读取(结果字符串(测试器, 编辑重发, "run_id")))
	当前对象.事件(测试器, "run.completed")
	if 调用集.加载() != 2 {
		测试器.格式化报告致命错误("期待原图和重发两次调用，得到 %d", 调用集.加载())
	}
	当前对象.停止(测试器)
}
