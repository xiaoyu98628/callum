func 等待操作结果(测试器 *测试包.测试器, 运行时 *运行时, 订阅 *事件集包.订阅, 执行标识 string, 开启审批 func(智能体包.审批)) 智能体包.事件 {
	测试器.标记测试辅助函数()
	执行上下文, 取消函数 := 上下文包.创建超时上下文(测试器.上下文(), 5*时间包.秒)
	defer 取消函数()
	for {
		事件, 错误 := 订阅.下一个(执行上下文)
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		if 事件.执行标识 != 执行标识 {
			continue
		}
		if 事件.类型 == "approval.requested" {
			if 开启审批 == nil {
				测试器.报告致命错误("出现非预期审批")
			}
			开启审批(事件.数据.(智能体包.审批))
		}
		if 事件.类型 == "run.completed" || 事件.类型 == "run.failed" || 事件.类型 == "run.cancelled" {
			运行时.管理器.等待(执行标识)
			return 事件
		}
	}
}

					测试器.报告致命错误("清理失败未保存")
				}
			}

				测试器.报告致命错误("缺少操作步骤")
			}

		})

			要求成功(测试器, 通信对端.调用(测试器, "run.cancel", map[string]string{"run_id": 执行标识}))
			通信对端.事件(测试器, "run.cancelled")
			首个, 输出 := 命令步骤输出(测试器, 通信对端, 执行标识)
			var 原始 struct{ 操作 工具包.操作结果记录 }
			if 错误 := JSON包.反序列化(输出, &原始); 错误 != nil || 原始.操作.操作结果 != 工具包.操作结果未知 {
				测试器.格式化报告致命错误("首次命令的结果未记录为未知: %s %v", 输出, 错误)
			}
			通信对端.停止(测试器)

			通信对端 = 开始应用(测试器, 启动配置, 输入输出包.丢弃)
			if 选项 == "never" {
				设置审批策略(测试器, 通信对端, 选项)
			}
			已继续 := 通信对端.调用(测试器, "conversation.continue", map[string]any{"conversation_id": 会话标识, "run_id": 执行标识, "resume_in_place": true})
			if 结果字符串(测试器, 已继续, "run_id") != 执行标识 {
				测试器.报告致命错误("继续任务未复用原执行")
			}
			var 步骤标识 string
			if 选项 != "never" {
				通知 := 通信对端.事件(测试器, "approval.requested")
				var 事件 struct{ 审批 智能体包.审批 }
				if 错误 := JSON包.反序列化(通知.参数, &事件); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				步骤标识 = 事件.审批.步骤标识
				if 事件.审批.执行标识 != 执行标识 || 步骤标识 == 首个.标识 || 事件.审批.请求哈希 == 原始.操作.请求哈希 {
					测试器.格式化报告致命错误("当前命令未使用自己独立的审批快照: %+v", 事件.审批)
				}
				if 选项 == "approved" {
					if 错误 := 操作系统包.写入文件(释放, nil, 0600); 错误 != nil {
						测试器.报告致命错误(错误)
					}
				}
				要求成功(测试器, 通信对端.调用(测试器, "approval.decide", map[string]string{"approval_id": 事件.审批.标识, "decision": 选项}))
			}
			通信对端.事件(测试器, "run.completed")
			列表 := 通信对端.调用(测试器, "step.list", map[string]string{"run_id": 执行标识})
			要求成功(测试器, 列表)
			var 步骤集 []智能体包.步骤摘要
			if 错误 := JSON包.反序列化(列表.结果, &步骤集); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			for _, 步骤 := range 步骤集 {
				if 步骤.名称 == "run_command" && 步骤.标识 != 首个.标识 {
					步骤标识 = 步骤.标识
				}
			}
			for 字段, 预期 := range map[string]string{"input": "previous_result_requires_confirmation", "output": string(工具包.操作结果不已生效)} {
				if 字段 == "output" && 选项 == "approved" {
					预期 = `"started":true`
				}
				数据 := 通信对端.调用(测试器, "step.data", map[string]string{"step_id": 步骤标识, "field": 字段})
				要求成功(测试器, 数据)
				if 片段 := 结果字符串(测试器, 数据, "fragment"); !字符串包.包含(片段, 预期) {
					测试器.格式化报告致命错误("命令 %s 缺少 %s: %s", 字段, 预期, 片段)
				}
			}
			if 选项 != "approved" {
				for _, 事件 := range 通信对端.待处理 {
					if 事件.方法 == "tool.output" || 选项 == "never" && 事件.方法 == "approval.requested" {
						测试器.格式化报告致命错误("禁止重试的命令产生多余事件: %s", 事件.参数)
					}
				}
				return
			}
			// 同一会话的后续调用在历史成功后恢复普通只读命令策略。
			下一个 := 通信对端.调用(测试器, "conversation.send", map[string]any{"conversation_id": 会话标识, "model_id": 模型标识, "stream": false, "content": 消息内容块集("再次读取")})
			下一个执行 := 结果字符串(测试器, 下一个, "run_id")
			通信对端.事件(测试器, "run.completed")
			_, 输出 = 命令步骤输出(测试器, 通信对端, 下一个执行)
			if !字符串包.包含(string(输出), `"started":true`) {
				测试器.格式化报告致命错误("成功核实后仍未自动运行: %s", 输出)
			}
		})
	}
}
