func Test上下文用量匹配集请求设置并且工作区(测试器 *测试包.测试器) {
	for _, 模式 := range []string{"unbound", "single_directory", "without_directory"} {
		测试器.执行(模式, func(测试器 *测试包.测试器) {
			type 请求 struct {
				消息集 []模型包.消息                `json:"messages"`
				工具集 []struct{ 函数 模型包.工具定义 } `json:"tools"`
			}
			请求集 := make(chan 请求, 1)
			服务端 := HTTP测试包.新建服务端(HTTP包.处理器函数(func(当前响应写入器 HTTP包.响应写入器, 当前请求 *HTTP包.请求) {
				var 输入 请求
				if 错误 := JSON包.新建解码器(当前请求.正文).解码(&输入); 错误 != nil {
					测试器.错误文本(错误)
					当前响应写入器.写入头部(HTTP包.状态无效请求)
					return
				}
				if len(输入.工具集) > 0 {
					select {
					case 请求集 <- 输入:
					default:
						测试器.错误文本("意外重复调用主模型")
						当前响应写入器.写入头部(HTTP包.状态内部服务端错误)
						return
					}
				}
				写入文件工具响应(测试器, 当前响应写入器, false, map[string]any{"content": "已核对"}, "stop")
			}))
			测试器.注册清理函数(服务端.关闭)
			启动配置, 错误 := 配置模块包.加载(文件路径包.合并(测试器.创建临时目录(), "data"))
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			启动配置.最大并发执行集 = 1
			当前对象 := 开始应用(测试器, 启动配置, 输入输出包.丢弃)
			defer 当前对象.停止(测试器)
			消息标识 := 初始化会话模型(测试器, 当前对象, 服务端.URL)
			执行 := 读取设置(测试器, 当前对象)
			参数 := map[string]any{"model_id": 消息标识}
			根 := ""
			if 模式 != "unbound" {
				if 模式 == "single_directory" {
					根 = 测试器.创建临时目录()
				}
				已创建 := 当前对象.调用(测试器, "workspace.create", map[string]any{"name": "上下文目录", "root_path": 根})
				参数["workspace_id"] = 结果字符串(测试器, 已创建, "id")
				var 项目 struct {
					根路径 string `json:"root_path"`
				}
				if 错误 := JSON包.反序列化(已创建.结果, &项目); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				根 = 项目.根路径
			}
			帧 := 当前对象.调用(测试器, "conversation.context_usage", 参数)
			要求成功(测试器, 帧)
			var 预览 会话包.上下文用量
			if 错误 := JSON包.反序列化(帧.结果, &预览); 错误 != nil || 预览.估算词元 == nil {
				测试器.格式化报告致命错误("预览失败: %s %v", 帧.结果, 错误)
			}
			const 问题 = "核对实际请求和只读预览"
			参数["content"], 参数["stream"] = 消息内容块集(问题), false
			结果字符串(测试器, 当前对象.调用(测试器, "conversation.send", 参数), "run_id")
			当前对象.事件(测试器, "run.completed")
			var 已捕获 请求
			select {
			case 已捕获 = <-请求集:
			case <-测试器.上下文().完成():
				测试器.报告致命错误(测试器.上下文().获取错误())
			}
			消息集 := 已捕获.消息集
			工具集 := make([]模型包.工具定义, 0, len(已捕获.工具集))
			for _, 定义 := range 已捕获.工具集 {
				工具集 = append(工具集, 定义.函数)
			}
			实际 := 词元包.估算(消息集, 工具集, 设置包.默认压缩().图片词元估算)
			用户词元 := 词元包.估算([]模型包.消息{{角色: 模型包.角色用户, 内容: 问题}}, nil, 0) - 16
			if 实际 != *预览.估算词元+用户词元 {
				测试器.格式化报告致命错误("预览与实际上下文不一致: actual=%d preview=%d user=%d", 实际, *预览.估算词元, 用户词元)
			}
			var 已投影 struct {
				来源  string `json:"source"`
				修订号 int64  `json:"settings_revision"`
				设置包.执行
			}
			系统 := 消息集[0].内容
			if JSON包.反序列化([]byte(系统[字符串包.最后索引(系统, "\n")+1:]), &已投影) != nil || 已投影.来源 != "server_execution_settings" || 已投影.修订号 != 执行.修订号 || 已投影.执行 != 设置包.默认执行() {
				测试器.格式化报告致命错误("主请求未使用受理设置: %+v", 已投影)
			}
			if 消息集[1].角色 != 模型包.角色用户 {
				测试器.报告致命错误("目录元数据不是数据消息")
			}
			if 根 == "" {
				if !字符串包.包含(消息集[1].内容, "必须使用绝对路径") {
					测试器.报告致命错误("缺少目录基准时未明确要求绝对路径")
				}
			} else {
				_, 原始, 已找到 := 字符串包.截断(消息集[1].内容, "\n")
				var 项目 struct {
					根路径 string `json:"root_path"`
				}
				if !已找到 || JSON包.反序列化([]byte(原始), &项目) != nil || 项目.根路径 != 根 {
					测试器.报告致命错误("单目录项目未发送主目录")
				}
			}
		})
	}
}

func Test上下文用量只读预算草稿并且实际保留重启(测试器 *测试包.测试器) {
	var 调用集 原子操作包.整数32
	服务端 := HTTP测试包.新建服务端(HTTP包.处理器函数(func(当前响应写入器 HTTP包.响应写入器, 当前请求 *HTTP包.请求) {

		当前响应写入器.头部().设置("Content-Type", "application/json")
		_, _ = 输入输出包.写入字符串(当前响应写入器, `{"choices":[{"index":0,"message":{"role":"assistant","content":"受控回答"},"finish_reason":"stop"}],"usage":{"prompt_tokens":201,"total_tokens":208,"custom":"unchanged"}}`)
	}))
	测试器.注册清理函数(服务端.关闭)
	启动配置, 错误 := 配置模块包.加载(文件路径包.合并(测试器.创建临时目录(), "data"))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	启动配置.最大并发执行集 = 1
	当前对象 := 开始应用(测试器, 启动配置, 输入输出包.丢弃)
	消息标识 := 初始化会话模型(测试器, 当前对象, 服务端.URL)
	基准 := 读取上下文用量(测试器, 当前对象, 消息标识, "", "")
	草稿 := 读取上下文用量(测试器, 当前对象, 消息标识, "", 字符串包.重复("草稿", 100))
	if 基准.状态 != "ready" || 基准.估算词元 == nil || *基准.估算词元 <= 100 || 基准.输入上限 == nil || *基准.输入上限 != 1_000_000-4096-1024 || 基准.实际输入词元 != nil {
		测试器.格式化报告致命错误("基础统计无效: %+v", 基准)
	}
	预期草稿 := 词元包.估算([]模型包.消息{{角色: 模型包.角色用户, 内容: 字符串包.重复("草稿", 100)}}, nil, 0) - 16
	if 草稿.草稿词元 != 预期草稿 || *草稿.估算词元 != *基准.估算词元+预期草稿 || 调用集.加载() != 0 {
		测试器.报告致命错误("草稿被调用模型或统计不同")
	}

	要求成功(测试器, 列表)

		测试器.格式化报告致命错误("只读预览创建了会话: %s", 列表.结果)
	}
	for _, 无效 := range []map[string]any{{"model_id": 消息标识, "unexpected": true}, {"model_id": "invalid"}, {"model_id": 消息标识, "conversation_id": 消息标识, "workspace_id": 消息标识}} {
		if 当前对象.调用(测试器, "conversation.context_usage", 无效).错误 == nil {
			测试器.报告致命错误("接受了非法输入")
		}
	}
	会话标识 := 压缩发送(测试器, 当前对象, 消息标识, "", "正式问题")
	空闲 := 读取上下文用量(测试器, 当前对象, 消息标识, 会话标识, "")
	if 空闲.估算词元 == nil || *空闲.估算词元 <= *基准.估算词元 || 空闲.实际输入词元 == nil || *空闲.实际输入词元 != 201 || 调用集.加载() != 1 {
		测试器.格式化报告致命错误("完成后统计无效: %+v calls=%d", 空闲, 调用集.加载())
	}
	当前对象.停止(测试器)
	当前对象 = 开始应用(测试器, 启动配置, 输入输出包.丢弃)
	已恢复 := 读取上下文用量(测试器, 当前对象, 消息标识, 会话标识, "")
	if 已恢复.实际输入词元 == nil || *已恢复.实际输入词元 != 201 || 已恢复.估算词元 == nil || *已恢复.估算词元 != *空闲.估算词元 {
		测试器.格式化报告致命错误("重启统计不一致: %+v", 已恢复)
	}
	供应商 := 结果标识(测试器, 当前对象.调用(测试器, "provider.create", map[string]any{"name": "未知容量", "protocol": "chatcompletions", "base_url": 服务端.URL + "/v1"}), "id")
	未知 := 结果标识(测试器, 当前对象.调用(测试器, "model.create", map[string]any{"provider_id": 供应商, "display_name": "未知", "remote_model_id": "unknown", "enabled": true}), "id")

	if 值.输入上限 != nil || 值.上下文窗口 != nil || 值.估算词元 == nil || 值.原因 != "capacity_unconfigured" {
		测试器.格式化报告致命错误("伪造了未知容量: %+v", 值)
	}
	压缩 := 设置包.默认压缩()
	压缩.词元预算 = 50000
	压缩设置(测试器, 当前对象, 压缩)
	值 = 读取上下文用量(测试器, 当前对象, 未知, "", "")
	if 值.输入上限 == nil || *值.输入上限 != 50000 || 值.上下文窗口 != nil {
		测试器.报告致命错误("未使用显式预算")
	}
}

func Test上下文用量流集通知集草稿隔离并且取消(测试器 *测试包.测试器) {
	下一个 := make(chan struct{})
	服务端 := HTTP测试包.新建服务端(HTTP包.处理器函数(func(当前响应写入器 HTTP包.响应写入器, 当前请求 *HTTP包.请求) {
		当前响应写入器.头部().设置("Content-Type", "text/event-stream")
		_, _ = 输入输出包.写入字符串(当前响应写入器, "data: {\"choices\":[{\"index\":0,\"delta\":{\"content\":\"短\"},\"finish_reason\":null}]}\n\n")
		当前响应写入器.(HTTP包.刷新器).刷新()
		select {
		case <-下一个:
		case <-当前请求.上下文().完成():
			return
		}
		分片, _ := JSON包.序列化(字符串包.重复("生成中", 300))
		_, _ = 输入输出包.写入字符串(当前响应写入器, "data: {\"choices\":[{\"index\":0,\"delta\":{\"content\":"+string(分片)+"},\"finish_reason\":null}]}\n\n")
		当前响应写入器.(HTTP包.刷新器).刷新()
		<-当前请求.上下文().完成()
	}))
	测试器.注册清理函数(服务端.关闭)
	启动配置, 错误 := 配置模块包.加载(文件路径包.合并(测试器.创建临时目录(), "data"))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	启动配置.最大并发执行集 = 1
	当前对象 := 开始应用(测试器, 启动配置, 输入输出包.丢弃)
	消息标识 := 初始化会话模型(测试器, 当前对象, 服务端.URL)
	已受理 := 当前对象.调用(测试器, "conversation.send", map[string]any{"model_id": 消息标识, "stream": true, "content": 消息内容块集("流式问题")})
	会话标识, 执行标识 := 结果字符串(测试器, 已受理, "conversation_id"), 结果字符串(测试器, 已受理, "run_id")
	当前对象.事件(测试器, "message.delta")
	首个 := 读取上下文用量(测试器, 当前对象, 消息标识, 会话标识, "")
	if 首个.状态 != "ready" || 首个.执行标识 != 执行标识 || 首个.估算词元 == nil {
		测试器.格式化报告致命错误("没有实时快照: %+v", 首个)
	}
	压缩 := 设置包.默认压缩()
	压缩.安全余量 = 4096
	压缩设置(测试器, 当前对象, 压缩)
	已冻结 := 读取上下文用量(测试器, 当前对象, 消息标识, 会话标识, "")
	if 已冻结.输入上限 == nil || *已冻结.输入上限 != *首个.输入上限 {
		测试器.报告致命错误("活动执行使用了修改后的全局设置")
	}
	if 当前对象.调用(测试器, "conversation.context_usage", map[string]any{"conversation_id": 会话标识, "model_id": UUID包.必须(UUID包.新建UUIDv7()).字符串()}).错误 == nil {
		测试器.报告致命错误("活动执行允许切换统计模型")
	}
	私有 := 读取上下文用量(测试器, 当前对象, 消息标识, 会话标识, "不应持久化的草稿")
	if 私有.草稿词元 <= 0 || *私有.估算词元 <= *首个.估算词元 {
		测试器.报告致命错误("草稿没有叠加")
	}
	close(下一个)
	当前对象.事件(测试器, "message.delta")
	秒 := 读取上下文用量(测试器, 当前对象, 消息标识, 会话标识, "")
	if 秒.估算词元 == nil || *秒.估算词元 <= *首个.估算词元+1000 || 秒.修订号 <= 首个.修订号 || 秒.草稿词元 != 0 {
		测试器.格式化报告致命错误("流式快照未更新或草稿污染: %+v", 秒)
	}
	其他 := 读取上下文用量(测试器, 当前对象, 消息标识, "", "")
	if 其他.执行标识 != "" || *其他.估算词元 >= *秒.估算词元 {
		测试器.报告致命错误("其他对话混入活动执行")
	}
	事件 := 当前对象.事件(测试器, "conversation.context_usage.updated")
	if 字符串包.包含(string(事件.参数), "不应持久化") || !字符串包.包含(string(事件.参数), `"context_usage"`) {
		测试器.格式化报告致命错误("通知包含草稿或丢失结构: %s", 事件.参数)
	}
	要求成功(测试器, 当前对象.调用(测试器, "run.cancel", map[string]any{"run_id": 执行标识}))
	当前对象.事件(测试器, "run.cancelled")
	最终 := 读取上下文用量(测试器, 当前对象, 消息标识, 会话标识, "")
	// 结束通知后查询恢复持久化历史，活动缓存不会继续持有取消的执行。
	if 最终.执行标识 != "" || 最终.状态 != "ready" || 最终.输入上限 == nil || *最终.输入上限 != 1_000_000-4096-4096 {
		测试器.格式化报告致命错误("活动快照未释放: %+v", 最终)
	}
	if len(读取消息集(测试器, 当前对象, 会话标识)) != 2 {
		测试器.报告致命错误("预览草稿写入了消息")
	}
}
