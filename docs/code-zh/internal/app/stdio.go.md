package 应用模块

import (
	上下文包 "context"
	错误判断包 "errors"
	输入输出包 "io"
	结构化日志包 "log/slog"

	事件集包 "github.com/xiaoyu98628/callum/internal/agent/events"
	配置模块包 "github.com/xiaoyu98628/callum/internal/config"
	凭据包 "github.com/xiaoyu98628/callum/internal/credentials"
	系统包 "github.com/xiaoyu98628/callum/internal/credentials/system"
	标准输入输出包 "github.com/xiaoyu98628/callum/internal/stdio"
	执行包 "github.com/xiaoyu98628/callum/internal/stdio/execution"
	MCP处理器包 "github.com/xiaoyu98628/callum/internal/stdio/mcp"
	传输模块包 "github.com/xiaoyu98628/callum/internal/stdio/transport"
)

// 执行 是独立 标准输入输出 进程入口，接管输入输出，并在连接结束后关闭整个运行时。
func 执行(执行上下文 上下文包.上下文, 启动配置 配置模块包.配置, 输入 输入输出包.读取关闭器, 输出 输入输出包.写入关闭器, 日志记录器 *结构化日志包.日志记录器) (返回错误 error) {
	return 运行(执行上下文, 启动配置, 输入, 输出, 日志记录器, 系统包.新建())
}

func 运行(执行上下文 上下文包.上下文, 启动配置 配置模块包.配置, 输入 输入输出包.读取关闭器, 输出 输入输出包.写入关闭器, 日志记录器 *结构化日志包.日志记录器, 系统凭据库 凭据包.系统凭据库) (返回错误 error) {
	if 输入 == nil || 输出 == nil {
		return 错误判断包.新建("输入输出不能为空")
	}
	运行时, 错误 := 新建运行时(执行上下文, 启动配置, 日志记录器, 系统凭据库)
	if 错误 != nil {
		return 错误判断包.合并(错误, 输入.关闭(), 输出.关闭())
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 运行时.关闭()) }()
	return 运行时.提供服务标准输入输出(执行上下文, 输入, 输出)
}

// 提供服务标准输入输出 接管一个连接及其发起的任务；文件结束 或 系统.关闭 只结束该连接。
// 共享运行时和其他调用方的执行仍然有效；独立进程通过 执行 关闭整个运行时。
func (当前运行时 *运行时) 提供服务标准输入输出(执行上下文 上下文包.上下文, 输入 输入输出包.读取关闭器, 输出 输入输出包.写入关闭器) (返回错误 error) {
	if 执行上下文 == nil || 输入 == nil || 输出 == nil {
		return 错误判断包.新建("通信初始化参数无效")
	}
	if 错误 := 当前运行时.开始会话连接(); 错误 != nil {
		return 错误判断包.合并(错误, 输入.关闭(), 输出.关闭())
	}
	defer 当前运行时.会话连接集.完成()
	会话连接执行上下文, 取消函数 := 上下文包.创建可取消上下文(执行上下文)
	停止 := 上下文包.之后函数(当前运行时.执行上下文, 取消函数)
	defer func() { 停止(); 取消函数() }()
	连接, 错误 := 传输模块包.新建(会话连接执行上下文, 输入, 输出)
	if 错误 != nil {
		return 错误判断包.合并(错误, 输入.关闭(), 输出.关闭())
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 连接.关闭()) }()
	订阅, 错误 := 当前运行时.订阅(事件集包.筛选{})
	if 错误 != nil {
		return 错误
	}
	defer 订阅.关闭()
	所属对象 := 新建会话连接执行集(连接.上下文(), 当前运行时.管理器)
	defer 所属对象.关闭()
	服务端, 错误 := 标准输入输出包.新建带有会话集(当前运行时.供应商集, 当前运行时.调用集, 当前运行时.工作区集, 当前运行时.会话集, 当前运行时.引擎, 当前运行时.审批集, 当前运行时.设置, 当前运行时.技能, 所属对象.注册, 当前运行时.日志记录器, MCP处理器包.新建(当前运行时.MCP))
	if 错误 != nil {
		return 错误
	}
	if 错误 := 服务端.就绪(连接); 错误 != nil {
		return 错误
	}
	已转发 := make(chan struct{})
	go func() {
		defer close(已转发)
		for {
			事件, 错误 := 订阅.下一个(连接.上下文())
			if 错误 != nil {
				if !错误判断包.是否属于(错误, 输入输出包.文件结束) && 连接.上下文().获取错误() == nil {
					连接.失败(错误)
				}
				return
			}
			if 错误 := 执行包.发布持久化(连接.通知, 事件); 错误 != nil {
				连接.失败(错误)
				return
			}
		}
	}()
	错误 = 服务端.提供服务请求集(连接)
	// 正常 文件结束 保留事件转发直到本连接的任务清理完毕，再排空订阅和协议队列。
	所属对象.关闭()
	订阅.关闭()
	<-已转发
	return 错误
}
