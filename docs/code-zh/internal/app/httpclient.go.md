```go
package 应用模块

import (
	网络包 "net"
	HTTP包 "net/http"
	时间包 "time"
)

// 新建HTTP客户端 创建应用持有的连接池，调用方在模型任务结束后关闭其空闲连接。
func 新建HTTP客户端(最大连接集 int) (*HTTP包.客户端, *HTTP包.传输) {
	传输模块 := &HTTP包.传输{
		代理:        HTTP包.代理来源环境,
		拨号上下文:     (&网络包.拨号器{超时: 30 * 时间包.秒, 保留存活: 30 * 时间包.秒}).拨号上下文,
		强制尝试HTTP2: true, 最大空闲连接集: 64, 最大空闲连接集每个主机: 最大连接集,
		最大连接集每个主机: 最大连接集, 空闲连接超时: 90 * 时间包.秒,
		// 响应头和响应体共用模型调用方的期限，非流式生成可能在完成后才发送响应头。
		TLS握手超时: 10 * 时间包.秒,
		预期继续超时:  时间包.秒,
	}
	客户端 := &HTTP包.客户端{传输: 有界传输{传输模块}, 检查重定向: func(*HTTP包.请求, []*HTTP包.请求) error { return HTTP包.使用最后响应错误 }}
	return 客户端, 传输模块
}
```
