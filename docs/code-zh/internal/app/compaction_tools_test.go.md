package 应用模块

func Test已压缩未知操作仍然需要审批重启后(测试器 *测试包.测试器) {
	运行时, 启动配置 := 运行时测试夹具(测试器, 1)
	记录, 错误 := 运行时.获取设置().获取(测试器.上下文(), 设置包.压缩键)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	配置模块 := 设置包.默认压缩()
	配置模块.词元预算, 配置模块.保留轮次集 = 8000, 1
	原始, 错误 := JSON包.序列化(配置模块)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 运行时.获取设置().更新(测试器.上下文(), 设置包.压缩键, 记录.修订号, 原始); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	实例 := &操作结果测试工具{路径: 文件路径包.合并(测试器.创建临时目录(), "evidence"), 操作结果: 工具包.操作结果未知}
	客户端 := 操作结果测试模型(func(_ 上下文包.上下文, 请求 模型包.请求) (模型包.响应, error) {
		if len(请求.工具集) == 0 {
			if 请求.消息集[0].内容 == 压缩包.提示词 {
				// 即便模型摘要错误地声称成功，也不能覆盖服务端未知操作事实。
				return 模型包.响应{文本: "操作已经成功。继续任务。"}, nil
			}
			return 模型包.响应{文本: "标题"}, nil
		}
		最后 := 请求.消息集[len(请求.消息集)-1]
		if 最后.角色 == 模型包.角色工具 {
			return 模型包.响应{文本: 字符串包.重复("回答正文", 200)}, nil
		}
		文本 := 最后.内容
		if len(最后.内容块集) > 0 {
			文本 = 最后.内容块集[0].文本
		}
		if 字符串包.存在前缀(文本, "执行操作") || 字符串包.存在前缀(文本, "重试操作") {
			return 模型包.响应{工具调用集: []模型包.工具调用{{标识: "fixed-call", 名称: "operation", 参数: JSON包.原始JSON消息(`{}`)}}}, nil
		}
		return 模型包.响应{文本: 字符串包.重复("中间回答", 200)}, nil
	})
	标识 := 策略测试标识(测试器)
	服务, _ := 操作结果服务(测试器, 运行时, 实例, 客户端, 标识, nil)
	订阅, 错误 := 运行时.订阅(事件集包.筛选{})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 订阅.关闭()
	首个 := 发送操作结果(测试器, 服务, 标识, "", "执行操作"+字符串包.重复("必须保留约束", 200), nil)
	if 终态 := 等待操作结果(测试器, 运行时, 订阅, 首个.执行.标识, nil); 终态.类型 != "run.completed" {
		测试器.报告致命错误("首个执行失败")
	}
	下一个 := 发送操作结果(测试器, 服务, 标识, 首个.会话.标识, 字符串包.重复("中间问题", 200), nil)
	if 终态 := 等待操作结果(测试器, 运行时, 订阅, 下一个.执行.标识, nil); 终态.类型 != "run.completed" {
		测试器.报告致命错误("中间执行失败")
	}
	// 历史工具结果及独立操作事实计入 词元 预算，第三轮触发压缩。
	已受理 := 发送操作结果(测试器, 服务, 标识, 首个.会话.标识, 字符串包.重复("第三个问题", 200), nil)
	if 终态 := 等待操作结果(测试器, 运行时, 订阅, 已受理.执行.标识, nil); 终态.类型 != "run.completed" {
		测试器.格式化报告致命错误("工具轨迹压缩失败: %+v", 终态.数据)
	}
	版本集, 错误 := 服务.上下文版本集(测试器.上下文(), 首个.会话.标识, 50, 0)
	if 错误 != nil || len(版本集) != 1 || 版本集[0].通过序号 != 2 {
		测试器.格式化报告致命错误("未覆盖工具轮次: %+v %v", 版本集, 错误)
	}
	if 错误 := 运行时.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	已重启, 错误 := 新建测试运行时(测试器.上下文(), 启动配置, 结构化日志包.新建(结构化日志包.新建JSON处理器(输入输出包.丢弃, nil)))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer func() {
		if 错误 := 已重启.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	}()
	重试服务, _ := 操作结果服务(测试器, 已重启, 实例, 客户端, 标识, nil)
	重试订阅, 错误 := 已重启.订阅(事件集包.筛选{})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 重试订阅.关闭()
	重试 := 发送操作结果(测试器, 重试服务, 标识, 首个.会话.标识, "重试操作", nil)
	审批集 := 0
	终态 := 等待操作结果(测试器, 已重启, 重试订阅, 重试.执行.标识, func(当前审批 智能体包.审批) {
		审批集++
		if 实例.调用集.加载() != 1 {
			测试器.报告致命错误("压缩或重启导致未知操作自动重试")
		}
		if _, 错误 := 已重启.获取审批集().决定(测试器.上下文(), 当前审批.标识, 智能体包.审批已拒绝, nil); 错误 != nil {
			测试器.报告致命错误(错误)
		}
	})
	if 终态.类型 != "run.completed" || 审批集 != 1 || 实例.调用集.加载() != 1 {
		测试器.报告致命错误("压缩后的未知操作未得到保护")
	}
}

func Test自动压缩之间工具轮次集保留当前调用集(测试器 *测试包.测试器) {
	运行时, _ := 运行时测试夹具(测试器, 1)
	记录, 错误 := 运行时.获取设置().获取(测试器.上下文(), 设置包.压缩键)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	启动配置 := 设置包.默认压缩()
	启动配置.词元预算, 启动配置.保留轮次集 = 7000, 1
	原始, 错误 := JSON包.序列化(启动配置)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 运行时.获取设置().更新(测试器.上下文(), 设置包.压缩键, 记录.修订号, 原始); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	实例 := &操作结果测试工具{路径: 文件路径包.合并(测试器.创建临时目录(), "evidence"), 操作结果: 工具包.操作结果已生效}
	已检查 := false
	客户端 := 操作结果测试模型(func(_ 上下文包.上下文, 请求 模型包.请求) (模型包.响应, error) {
		if len(请求.工具集) == 0 {
			if 请求.消息集[0].内容 == 压缩包.提示词 {
				return 模型包.响应{文本: "历史事实。继续当前任务。"}, nil
			}
			return 模型包.响应{文本: "标题"}, nil
		}
		最后 := 请求.消息集[len(请求.消息集)-1]
		if 最后.角色 == 模型包.角色工具 {
			数量 := 0
			摘要 := false
			for _, 消息 := range 请求.消息集 {
				if 消息.角色 == 模型包.角色工具 {
					数量++
				}
				if 字符串包.包含(消息.内容, "conversation_context_summary") {
					摘要 = true
				}
			}
			if 数量 != 8 || !摘要 {
				测试器.格式化创建错误("当前工具链在压缩后不完整: %d summary=%v", 数量, 摘要)
			}
			已检查 = true
			return 模型包.响应{文本: "工具轮次完成"}, nil
		}
		if len(最后.内容块集) > 0 && 最后.内容块集[0].文本 == "开始工具轮次" {
			调用集 := make([]模型包.工具调用, 8)
			for 索引 := range 调用集 {
				调用集[索引] = 模型包.工具调用{标识: string(rune('a' + 索引)), 名称: "operation", 参数: JSON包.原始JSON消息(`{}`)}
			}
			return 模型包.响应{工具调用集: 调用集}, nil
		}
		return 模型包.响应{文本: 字符串包.重复("历史回答", 200)}, nil
	})
	标识 := 策略测试标识(测试器)
	服务, _ := 操作结果服务(测试器, 运行时, 实例, 客户端, 标识, nil)
	订阅, 错误 := 运行时.订阅(事件集包.筛选{})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 订阅.关闭()
	会话标识 := ""
	for range 2 {
		本次执行 := 发送操作结果(测试器, 服务, 标识, 会话标识, 字符串包.重复("历史问题", 200), nil)
		会话标识 = 本次执行.会话.标识
		if 终态 := 等待操作结果(测试器, 运行时, 订阅, 本次执行.执行.标识, nil); 终态.类型 != "run.completed" {
			测试器.报告致命错误("历史执行失败")
		}
	}
	本次执行 := 发送操作结果(测试器, 服务, 标识, 会话标识, "开始工具轮次", nil)
	if 终态 := 等待操作结果(测试器, 运行时, 订阅, 本次执行.执行.标识, nil); 终态.类型 != "run.completed" {
		测试器.格式化报告致命错误("工具轮次失败: %+v", 终态)
	}
	// 终态通知先于调度任务完全退出，等待后再读取模型回调中的验证结果。
	运行时.管理器.等待(本次执行.执行.标识)
	if !已检查 || 实例.调用集.加载() != 8 {
		测试器.报告致命错误("未验证当前工具链")
	}
}
