const 压缩摘要 = "## 已完成进展\n- 保留用户约束\n- 历史操作需按审计核实\n\n下一步是继续完成任务。"

func 压缩端点(测试器 *测试包.测试器) (*压缩模型, *HTTP测试包.服务端) {
	测试器.标记测试辅助函数()
	映射 := &压缩模型{回答集: map[string]string{}, 已开始: make(chan struct{}, 8)}
	服务端 := HTTP测试包.新建服务端(HTTP包.处理器函数(func(当前响应写入器 HTTP包.响应写入器, 当前请求 *HTTP包.请求) {
		var 输入 struct {
			工具集 []struct {
				函数 模型包.工具定义 `json:"function"`
			} `json:"tools"`
			消息集 []struct{ 角色, 内容 string } `json:"messages"`
		}
		if 错误 := JSON包.新建解码器(当前请求.正文).解码(&输入); 错误 != nil || len(输入.消息集) == 0 {
			测试器.格式化创建错误("模型请求损坏: %v", 错误)
			当前响应写入器.写入头部(400)
			return
		}
		文本 := 字符串包.重复("已记录事实。", 50)
		if 输入.消息集[0].角色 == "system" && 输入.消息集[0].内容 == 压缩包.提示词 {
			if len(输入.工具集) != 0 {
				测试器.错误文本("摘要模型获得工具")
			}
			最后 := 输入.消息集[len(输入.消息集)-1]
			if 最后.角色 != "user" || 最后.内容 != 压缩包.请求 || 字符串包.包含(字符串包.目标小写(输入.消息集[0].内容+最后.内容), "codex") {
				测试器.错误文本("摘要请求未使用无品牌字样的纯文本交接指引")
			}
			if !字符串包.包含(最后.内容, "简体中文") {
				测试器.错误文本("摘要请求未要求使用简体中文")
			}
			映射.摘要集.添加(1)
			文本 = 压缩摘要
			switch 映射.模式.加载() {
			case 1:
				文本 = `{"goals":["继续完成任务"],"constraints":[],"decisions":[],"facts":[],"open_questions":[]}`
			case 3:
				文本 = 字符串包.重复("x", 压缩包.最大摘要字节+1)
			case 2:
				if 之后 := 映射.内容块之后.加载(); 之后 > 0 && 映射.摘要集.加载() < 之后 {
					break
				}
				select {
				case 映射.已开始 <- struct{}{}:
				default:
				}
				<-当前请求.上下文().完成()
				return
			}
		} else if len(输入.工具集) == 0 {
			文本 = "会话标题"
		} else {
			最后 := 输入.消息集[len(输入.消息集)-1].内容
			原始, 错误 := JSON包.序列化(输入.消息集)
			if 错误 != nil {
				测试器.错误文本(错误)
			}
			映射.互斥锁.加锁()
			映射.回答集[最后] = string(原始)
			映射.定义集 = make([]模型包.工具定义, len(输入.工具集))
			for 索引, 定义 := range 输入.工具集 {
				映射.定义集[索引] = 定义.函数
			}
			映射.互斥锁.解锁()
		}
		原始, 错误 := JSON包.序列化(文本)
		if 错误 != nil {
			测试器.错误文本(错误)
			return
		}
		当前响应写入器.头部().设置("Content-Type", "application/json")
		当前响应写入器.头部().设置("X-Request-ID", "compaction-fixture")
		_, _ = 格式化包.格式化输出(当前响应写入器, `{"choices":[{"index":0,"message":{"role":"assistant","content":%s},"finish_reason":"stop"}],"usage":{"total_tokens":7,"fixture":true}}`, 原始)
	}))
	测试器.注册清理函数(服务端.关闭)
	return 映射, 服务端
}

		会话标识 = 压缩发送(测试器, 当前对象, 消息标识, 会话标识, 格式化包.格式化字符串("长对话-%03d-", 索引)+字符串包.重复("必须保留目标。", 40))

	上游.互斥锁.加锁()

	上游.互斥锁.解锁()

	}
