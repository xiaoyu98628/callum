	for _, 已拒绝 := range []bool{false, true} {

			})
			标识 := 策略测试标识(测试器)

			订阅, 错误 := 运行时.订阅(事件集包.筛选{})
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			defer 订阅.关闭()
			var 本次执行 会话包.执行
			var 引导 会话包.后续消息

					会话标识: 本次执行.会话.标识, 预期执行标识: 本次执行.执行.标识, 模式: "steer",
					内容: []模型包.内容块{{类型: "text", 文本: "开跑前引导"}},
				})
				if 错误 != nil || 引导.回执 == nil {
					测试器.格式化报告致命错误("启动前引导未被受理: %v", 错误)
				}
				if 已拒绝 {
					return 拒绝
				}
				return nil
			})
			if 已拒绝 {
				if !错误判断包.是否属于(错误, 拒绝) || 已采样.加载() {
					测试器.格式化报告致命错误("受理响应失败后仍启动执行或未保留原错误: %v", 错误)
				}
				已重新排队 := 下一个业务(测试器, 订阅, "conversation.steering_requeued", 本次执行.执行.标识)

				return
			}

				测试器.报告致命错误(错误)
			}
			if 终态 := 等待操作结果(测试器, 运行时, 订阅, 本次执行.执行.标识, nil); 终态.类型 != "run.completed" || !已采样.加载() {
				测试器.格式化报告致命错误("执行未使用启动前引导完成: %+v", 终态)
			}
		})
	}
}

func Test引导允许已开始工具结束并且跳过剩余调用(测试器 *测试包.测试器) {
	运行时, _ := 运行时测试夹具(测试器, 1)
	已开始, 释放 := make(chan struct{}), make(chan struct{})
	系统信号已开始 := 同步包.一次函数(func() { close(已开始) })
	解除阻塞 := 同步包.一次函数(func() { close(释放) })
	defer 解除阻塞()
	实例 := &操作结果测试工具{路径: 文件路径包.合并(测试器.创建临时目录(), "applied"), 操作结果: 工具包.操作结果已生效, 之后: func() {
		系统信号已开始()
		<-释放
	}}
	客户端 := 操作结果测试模型(func(_ 上下文包.上下文, 请求 模型包.请求) (模型包.响应, error) {
		最后 := 请求.消息集[len(请求.消息集)-1]
		if !字符串包.包含(历史包.文本(最后.内容块集), "跳过剩余旧工具") {
			return 模型包.响应{工具调用集: []模型包.工具调用{
				{标识: "started", 名称: "operation", 参数: JSON包.原始JSON消息(`{}`)},
				{标识: "remaining", 名称: "operation", 参数: JSON包.原始JSON消息(`{}`)},
			}}, nil
		}
		var 已生效, 已跳过 bool
		for _, 消息 := range 请求.消息集 {
			已生效 = 已生效 || (消息.工具调用标识 == "started" && 字符串包.包含(消息.内容, `"outcome":"applied"`))
			已跳过 = 已跳过 || (消息.工具调用标识 == "remaining" && 字符串包.包含(消息.内容, `"kind":"steered"`))
		}
		if !已生效 || !已跳过 {
			return 模型包.响应{}, 错误判断包.新建("引导后的模型请求缺少完整工具轨迹")
		}
		return 模型包.响应{文本: "按引导完成"}, nil
	})
	标识 := 策略测试标识(测试器)
	服务, _ := 操作结果服务(测试器, 运行时, 实例, 客户端, 标识, nil)
	订阅, 错误 := 运行时.订阅(事件集包.筛选{})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 订阅.关闭()
	本次执行 := 发送操作结果(测试器, 服务, 标识, "", "执行两项操作", nil)
	等待系统信号(测试器, 已开始)
	if _, 错误 := 服务.添加后续消息(测试器.上下文(), 会话包.后续消息输入{
		会话标识: 本次执行.会话.标识, 预期执行标识: 本次执行.执行.标识, 模式: "steer",
		内容: []模型包.内容块{{类型: "text", 文本: "跳过剩余旧工具"}},
	}); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	解除阻塞()
	if 终态 := 等待操作结果(测试器, 运行时, 订阅, 本次执行.执行.标识, nil); 终态.类型 != "run.completed" || 实例.调用集.加载() != 1 {
		测试器.格式化报告致命错误("工具引导边界错误，终态 %+v，实际工具调用次数 %d", 终态, 实例.调用集.加载())
	}
}

func Test后续消息队列并且引导通过标准输入输出(测试器 *测试包.测试器) {

		}
		if 错误 := JSON包.新建解码器(当前请求.正文).解码(&输入); 错误 != nil || len(输入.消息集) == 0 {

				return
			case 字符串包.包含(最后.内容, "改用新要求"):

						已恢复.存储(true)
					}
				}
			}
		}
		if 输入.流 {
			当前响应写入器.头部().设置("Content-Type", "text/event-stream")
			_, _ = 格式化包.输出文本(当前响应写入器, "data: {\"choices\":[{\"index\":0,\"delta\":{\"content\":\"已处理\"},\"finish_reason\":null}]}\n\ndata: {\"choices\":[{\"index\":0,\"delta\":{},\"finish_reason\":\"stop\"}]}\n\ndata: [DONE]\n\n")

		_, _ = 输入输出包.写入字符串(当前响应写入器, `{"choices":[{"index":0,"message":{"role":"assistant","content":"已处理"},"finish_reason":"stop"}]}`)
