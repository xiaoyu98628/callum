```go
package 应用模块

import (
	上下文包 "context"
	错误判断包 "errors"
	HTTP包 "net/http"
	HTTP测试包 "net/http/httptest"
	测试包 "testing"
	时间包 "time"
)

func TestHTTP客户端使用请求截止时间用于响应头部集(测试器 *测试包.测试器) {
	const 响应延迟 = 100 * 时间包.毫秒
	服务端 := HTTP测试包.新建服务端(HTTP包.处理器函数(func(当前响应写入器 HTTP包.响应写入器, 当前请求 *HTTP包.请求) {
		计时器 := 时间包.新建计时器(响应延迟)
		defer 计时器.停止()
		select {
		case <-当前请求.上下文().完成():
		case <-计时器.字符:
			当前响应写入器.写入头部(HTTP包.状态无内容)
		}
	}))
	测试器.注册清理函数(服务端.关闭)
	客户端, 传输模块 := 新建HTTP客户端(1)
	测试器.注册清理函数(传输模块.关闭空闲连接)
	if 传输模块.响应头部超时 != 0 {
		测试器.格式化报告致命错误("响应头等待仍有独立期限: %v", 传输模块.响应头部超时)
	}
	for _, 测试用例 := range []struct {
		名称 string
		超时 时间包.时长
	}{
		{"可在请求期限内等待慢响应", 2 * 时间包.秒},
		{"超过请求期限时结束等待", 20 * 时间包.毫秒},
	} {
		测试器.执行(测试用例.名称, func(测试器 *测试包.测试器) {
			执行上下文, 取消函数 := 上下文包.创建超时上下文(测试器.上下文(), 测试用例.超时)
			defer 取消函数()
			请求, 错误 := HTTP包.新建请求带有上下文(执行上下文, HTTP包.方法提交, 服务端.URL, nil)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			响应, 错误 := 客户端.执行(请求)
			if 响应 != nil {
				defer 响应.正文.关闭()
			}
			if 测试用例.超时 < 响应延迟 {
				if !错误判断包.是否属于(错误, 上下文包.超出截止时间错误) {
					测试器.格式化报告致命错误("预期请求超时，收到 %v", 错误)
				}
				return
			}
			if 错误 != nil || 响应.状态代码 != HTTP包.状态无内容 {
				测试器.格式化报告致命错误("请求期限内未取得响应: %v, %v", 响应, 错误)
			}
		})
	}
}
```
