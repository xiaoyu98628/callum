```go
package 应用模块

import (
	JSON包 "encoding/json"
	错误判断包 "errors"
	输入输出包 "io"
	HTTP包 "net/http"
	HTTP测试包 "net/http/httptest"
	测试包 "testing"

	事件集包 "github.com/xiaoyu98628/callum/internal/agent/events"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	供应商包 "github.com/xiaoyu98628/callum/internal/model/provider"
)

func Test会话更新使用已解析模型并且保留协议(测试器 *测试包.测试器) {
	运行时, _ := 运行时测试夹具(测试器, 1)
	上游 := HTTP测试包.新建服务端(HTTP包.处理器函数(func(当前响应写入器 HTTP包.响应写入器, _ *HTTP包.请求) {
		当前响应写入器.头部().设置("Content-Type", "application/json")
		_, _ = 输入输出包.写入字符串(当前响应写入器, `{"choices":[{"index":0,"message":{"role":"assistant","content":"回答"},"finish_reason":"stop"}]}`)
	}))
	测试器.注册清理函数(上游.关闭)
	首个模型 := 直接模型(测试器, 运行时, 上游.URL)
	秒模型 := 直接模型(测试器, 运行时, 上游.URL)
	已停用 := 直接模型(测试器, 运行时, 上游.URL)
	if 错误 := 运行时.获取供应商集().设置模型已启用(测试器.上下文(), 已停用, false); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	订阅, 错误 := 运行时.订阅(事件集包.筛选{})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 订阅.关闭()
	var 已受理 会话包.执行
	if 错误 := 运行时.获取会话集().发送(测试器.上下文(), 会话包.发送{模型标识: 首个模型, 流: new(false), 内容: []模型包.内容块{{类型: "text", 文本: "问题"}}}, func(本次执行 会话包.执行) error {
		已受理 = 本次执行
		return nil
	}); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	下一个业务(测试器, 订阅, "run.completed", 已受理.执行.标识)
	标识 := 已受理.会话.标识
	之前, 错误 := 运行时.获取会话集().获取(测试器.上下文(), 标识)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	for _, 测试用例 := range []struct {
		名称, 模型标识 string
		预期       error
	}{
		{"invalid", "invalid", 会话包.无效错误},
		{"missing", 策略测试标识(测试器), 供应商包.未找到错误},
		{"disabled", 已停用, 供应商包.已停用错误},
	} {
		测试器.执行(测试用例.名称, func(测试器 *测试包.测试器) {
			_, 错误 := 运行时.获取会话集().更新(测试器.上下文(), 标识, 会话包.更新{标题: new("不应保存"), 置顶: new(true), 模型标识: 测试用例.模型标识})
			if !错误判断包.是否属于(错误, 测试用例.预期) {
				测试器.格式化报告致命错误("错误模型未被拒绝: %v", 错误)
			}
			之后, 错误 := 运行时.获取会话集().获取(测试器.上下文(), 标识)
			if 错误 != nil || 之后.模型 != 之前.模型 || 之后.标题 != 之前.标题 || 之后.置顶位于 != nil || 之后.更新时间 != 之前.更新时间 {
				测试器.格式化报告致命错误("失败更新发生了部分写入: %+v %v", 之后, 错误)
			}
		})
	}
	已更新, 错误 := 运行时.获取会话集().更新(测试器.上下文(), 标识, 会话包.更新{标题: new("  人工标题  "), 置顶: new(true)})
	if 错误 != nil || 已更新.标题 != "人工标题" || 已更新.标题来源 != "manual" || 已更新.置顶位于 == nil || 已更新.模型 != 之前.模型 {
		测试器.格式化报告致命错误("普通更新丢失字段或改写模型: %+v %v", 已更新, 错误)
	}
	已更新, 错误 = 运行时.获取会话集().更新(测试器.上下文(), 标识, 会话包.更新{模型标识: 秒模型})
	if 错误 != nil || 已更新.模型.标识 != 秒模型 || 已更新.模型.远程模型标识 != "remote" || 已更新.模型.供应商.标识 == 之前.模型.供应商.标识 {
		测试器.格式化报告致命错误("未使用新模型的服务端快照: %+v %v", 已更新, 错误)
	}
	通信对端 := 共享通信对端(测试器, 运行时)
	for _, 字段 := range []string{"model", "deleted"} {
		帧 := 通信对端.调用(测试器, "conversation.update", map[string]any{"id": 标识, 字段: map[string]string{"id": 首个模型}})
		if 帧.错误 == nil || 帧.错误.代码 != -32602 {
			测试器.格式化报告致命错误("协议接受内部写入字段 %s: %+v", 字段, 帧)
		}
	}
	帧 := 通信对端.调用(测试器, "conversation.update", map[string]any{"id": 标识, "title": "协议标题", "model_id": 首个模型, "pinned": false, "archived": true})
	要求成功(测试器, 帧)
	var 通信 会话包.会话
	if 错误 := JSON包.反序列化(帧.结果, &通信); 错误 != nil || 通信.标识 != 标识 || 通信.标题 != "协议标题" || 通信.模型 != 之前.模型 || 通信.置顶位于 != nil || 通信.已归档位于 == nil {
		测试器.格式化报告致命错误("更新协议不兼容: %s %v", 帧.结果, 错误)
	}
	for _, 方法 := range []string{"conversation.delete", "conversation.restore"} {
		帧 := 通信对端.调用(测试器, 方法, map[string]string{"id": 标识})
		要求成功(测试器, 帧)
		if 错误 := JSON包.反序列化(帧.结果, &通信); 错误 != nil || (通信.删除时间 != nil) != (方法 == "conversation.delete") {
			测试器.格式化报告致命错误("删除或恢复不兼容: %s %v", 帧.结果, 错误)
		}
	}
}
```
