```go
package 应用模块

import (
	上下文包 "context"
	同步包 "sync"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	执行器包 "github.com/xiaoyu98628/callum/internal/agent/executor"
)

// 会话连接执行集 只登记本连接发起的执行；登记发生在受理响应之前。
// 连接持有本次受理的租约。使用同一 标识 继续的另一个连接拥有独立租约。
type 会话连接执行集 struct {
	执行上下文 上下文包.上下文
	管理器   *执行器包.管理器
	互斥锁   同步包.互斥锁
	租约集   map[string]*执行器包.执行租约
	已关闭   bool
}

func 新建会话连接执行集(执行上下文 上下文包.上下文, 管理器 *执行器包.管理器) *会话连接执行集 {
	return &会话连接执行集{执行上下文: 执行上下文, 管理器: 管理器, 租约集: make(map[string]*执行器包.执行租约)}
}

func (当前会话连接执行集 *会话连接执行集) 注册(标识 string) error {
	当前会话连接执行集.互斥锁.加锁()
	defer 当前会话连接执行集.互斥锁.解锁()
	if 当前会话连接执行集.已关闭 {
		return 智能体包.已关闭错误
	}
	if 错误 := 当前会话连接执行集.执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	执行租约 := 当前会话连接执行集.管理器.活动执行租约(标识)
	if 执行租约 == nil {
		return 智能体包.未找到错误
	}
	for 前一个, 自有 := range 当前会话连接执行集.租约集 {
		if 当前会话连接执行集.管理器.活动执行租约(前一个) != 自有 {
			delete(当前会话连接执行集.租约集, 前一个)
		}
	}
	当前会话连接执行集.租约集[标识] = 执行租约
	return nil
}

func (当前会话连接执行集 *会话连接执行集) 关闭() {
	当前会话连接执行集.互斥锁.加锁()
	当前会话连接执行集.已关闭 = true
	租约集 := make([]*执行器包.执行租约, 0, len(当前会话连接执行集.租约集))
	for _, 执行租约 := range 当前会话连接执行集.租约集 {
		租约集 = append(租约集, 执行租约)
	}
	当前会话连接执行集.互斥锁.解锁()
	for _, 执行租约 := range 租约集 {
		执行租约.取消执行()
	}
	for _, 执行租约 := range 租约集 {
		执行租约.等待()
	}
}
```
