func 容量终态(测试器 *测试包.测试器, 运行时 *运行时, 订阅 *事件集包.订阅, 标识 string) 智能体包.事件 {
	测试器.标记测试辅助函数()
	// 最长场景执行 420 个实际工具步骤；并行竞态检查保留 30 秒等待预算。
	执行上下文, 取消函数 := 上下文包.创建超时上下文(测试器.上下文(), 30*时间包.秒)
	defer 取消函数()
	for {
		事件, 错误 := 订阅.下一个(执行上下文)
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		if 事件.执行标识 != 标识 {
			continue
		}
		if 事件.类型 == "approval.requested" {
			测试器.报告致命错误("出现非预期审批")
		}
		if 事件.类型 == "run.completed" || 事件.类型 == "run.failed" || 事件.类型 == "run.cancelled" {
			运行时.管理器.等待(标识)
			return 事件
		}
	}
}

func 要求已配对输入(测试器 *测试包.测试器, 请求 模型包.请求) {
	测试器.标记测试辅助函数()
	if 错误 := 历史包.检查(请求.消息集); 错误 != nil {
		测试器.格式化创建错误("超限请求发给模型: %v", 错误)
	}
	待处理 := map[string]bool{}
	for _, 消息 := range 请求.消息集 {
		if 消息.角色 == 模型包.角色工具 {
			if !待处理[消息.工具调用标识] {
				测试器.错误文本("工具结果缺少配对调用")
			}
			delete(待处理, 消息.工具调用标识)
		} else {
			if len(待处理) > 0 {
				测试器.错误文本("调用结果尚未齐全就插入其他消息")
			}
			for _, 调用 := range 消息.工具调用集 {
				待处理[调用.标识] = true
			}
		}
	}
	if len(待处理) > 0 {
		测试器.错误文本("模型请求含未完成工具批次")
	}
}

	}
}
