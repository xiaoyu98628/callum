```go
package 服务

import (
	上下文包 "context"
	操作系统包 "os"
	文件路径包 "path/filepath"
	切片包 "slices"
	字符串包 "strings"
	时间包 "time"

	UUID包 "github.com/google/uuid"

	标识符包 "github.com/xiaoyu98628/callum/internal/id"
	工作区包 "github.com/xiaoyu98628/callum/internal/workspace"
)

// 服务 校验工作区操作；数据库所有权属于装配层。
type 服务 struct{ 仓储 工作区包.仓储 }

func 新建(仓储 工作区包.仓储) (*服务, error) {
	if 仓储 == nil {
		return nil, 工作区包.无效错误
	}
	return &服务{仓储: 仓储}, nil
}

// 创建 保存项目。省略其他目录时，重新添加已删除的项目会保留原目录。
func (当前服务 *服务) 创建(执行上下文 上下文包.上下文, 名称, 根路径 string, 其他路径集 ...string) (工作区包.工作区, error) {
	名称 = 字符串包.裁剪空间(名称)
	if 名称 == "" || len(名称) > 1024 || (根路径 != "" && !有效路径(根路径)) || len(其他路径集) > 15 || (根路径 == "" && len(其他路径集) != 0) {
		return 工作区包.工作区{}, 工作区包.无效错误
	}
	if 根路径 != "" {
		根路径 = 文件路径包.清理(根路径)
	}
	if len(其他路径集) > 0 {
		实际根, 根信息, 错误 := 真实目录(根路径)
		if 错误 != nil {
			return 工作区包.工作区{}, 错误
		}
		根路径 = 实际根
		已见 := []操作系统包.文件信息{根信息}
		已规范化 := make([]string, 0, len(其他路径集))
		for _, 路径 := range 其他路径集 {
			if 错误 := 执行上下文.获取错误(); 错误 != nil {
				return 工作区包.工作区{}, 错误
			}
			实际, 信息, 错误 := 真实目录(路径)
			if 错误 != nil || 切片包.包含函数(已见, func(前一个 操作系统包.文件信息) bool { return 操作系统包.相同文件(前一个, 信息) }) {
				return 工作区包.工作区{}, 工作区包.无效错误
			}
			已见 = append(已见, 信息)
			已规范化 = append(已规范化, 实际)
		}
		其他路径集 = 已规范化
	}
	标识, 错误 := UUID包.新建UUIDv7()
	if 错误 != nil {
		return 工作区包.工作区{}, 错误
	}
	当前时间 := 时间包.当前时间().Unix毫秒()
	return 当前服务.仓储.创建(执行上下文, 工作区包.工作区{标识: 标识.字符串(), 名称: 名称, 根路径: 根路径, 其他路径集: 其他路径集, 创建时间: 当前时间, 更新时间: 当前时间})
}

func 有效路径(路径 string) bool {
	return len(路径) <= 8192 && !字符串包.包含字符(路径, 0) && 文件路径包.是否绝对路径(路径)
}

func 真实目录(路径 string) (string, 操作系统包.文件信息, error) {
	if !有效路径(路径) {
		return "", nil, 工作区包.无效错误
	}
	实际, 错误 := 文件路径包.求值符号链接集(路径)
	if 错误 != nil || !有效路径(实际) {
		return "", nil, 工作区包.无效错误
	}
	信息, 错误 := 操作系统包.状态信息(实际)
	if 错误 != nil || !信息.是否目录() {
		return "", nil, 工作区包.无效错误
	}
	return 实际, 信息, nil
}

func 有效标识(标识 string) bool { return 标识符包.有效版本7(标识) }

func (当前服务 *服务) 获取(执行上下文 上下文包.上下文, 标识 string) (工作区包.工作区, error) {
	if !有效标识(标识) {
		return 工作区包.工作区{}, 工作区包.无效错误
	}
	return 当前服务.仓储.获取(执行上下文, 标识)
}

func (当前服务 *服务) 列表(执行上下文 上下文包.上下文, 筛选 工作区包.筛选) ([]工作区包.工作区, error) {
	if 筛选.上限 == 0 {
		筛选.上限 = 100
	}
	if 筛选.上限 < 1 || 筛选.上限 > 200 || 筛选.偏移 < 0 {
		return nil, 工作区包.无效错误
	}
	return 当前服务.仓储.列表(执行上下文, 筛选)
}

func (当前服务 *服务) 更新(执行上下文 上下文包.上下文, 标识 string, 变更 工作区包.变更) (工作区包.工作区, error) {
	if !有效标识(标识) {
		return 工作区包.工作区{}, 工作区包.无效错误
	}
	if (变更.其他路径集 == nil) != (变更.预期文件夹集 == nil) {
		return 工作区包.工作区{}, 工作区包.无效错误
	}
	if 变更.名称 != nil {
		值 := 字符串包.裁剪空间(*变更.名称)
		if 值 == "" || len(值) > 1024 {
			return 工作区包.工作区{}, 工作区包.无效错误
		}
		变更.名称 = &值
	}
	if 变更.其他路径集 != nil {
		if 变更.根路径 == nil || 变更.预期文件夹集 == nil || len(*变更.其他路径集) > 15 || len(*变更.预期文件夹集) > 16 {
			return 工作区包.工作区{}, 工作区包.无效错误
		}
		for _, 路径 := range *变更.预期文件夹集 {
			if !有效路径(路径) {
				return 工作区包.工作区{}, 工作区包.无效错误
			}
		}
		根, 其他, 错误 := 规范化文件夹集(执行上下文, *变更.根路径, *变更.其他路径集)
		if 错误 != nil {
			return 工作区包.工作区{}, 错误
		}
		变更.根路径, 变更.其他路径集 = &根, &其他
	} else if 变更.根路径 != nil {
		路径, _, 错误 := 真实目录(*变更.根路径)
		if 错误 != nil {
			return 工作区包.工作区{}, 错误
		}
		变更.根路径 = &路径
	}
	return 当前服务.仓储.更新(执行上下文, 标识, 变更)
}

// 规范化文件夹集 允许无目录项目；绑定目录时按实际身份拒绝重复和符号链接别名。
func 规范化文件夹集(执行上下文 上下文包.上下文, 根 string, 其他 []string) (string, []string, error) {
	if 根 == "" {
		if len(其他) != 0 {
			return "", nil, 工作区包.无效错误
		}
		return "", []string{}, nil
	}
	路径集 := append([]string{根}, 其他...)
	已见 := make([]操作系统包.文件信息, 0, len(路径集))
	for 索引, 路径 := range 路径集 {
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			return "", nil, 错误
		}
		实际, 信息, 错误 := 真实目录(路径)
		if 错误 != nil || 切片包.包含函数(已见, func(前一个 操作系统包.文件信息) bool { return 操作系统包.相同文件(前一个, 信息) }) {
			return "", nil, 工作区包.无效错误
		}
		路径集[索引] = 实际
		已见 = append(已见, 信息)
	}
	return 路径集[0], 路径集[1:], nil
}
```
