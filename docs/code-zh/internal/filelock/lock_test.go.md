package 文件锁

import (
	缓冲输入输出包 "bufio"
	错误判断包 "errors"
	格式化包 "fmt"
	操作系统包 "os"
	进程执行包 "os/exec"
	文件路径包 "path/filepath"
	测试包 "testing"
)

func Test加锁辅助(测试器 *测试包.测试器) {
	路径 := 操作系统包.获取环境变量("CALLUM_TEST_LOCK_PATH")
	if 路径 == "" {
		return
	}
	文件, 错误 := 操作系统包.打开文件(路径, 操作系统包.创建文件标志|操作系统包.读写打开标志, 0600)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 文件.关闭()
	if 错误 := 尝试(文件); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	格式化包.打印并换行("locked")
	var 乙 [1]byte
	_, _ = 操作系统包.标准输入.读取(乙[:])
}

func Test加锁排除并且释放(测试器 *测试包.测试器) {
	路径 := 文件路径包.合并(测试器.创建临时目录(), "instance.lock")
	首个, 错误 := 操作系统包.打开文件(路径, 操作系统包.创建文件标志|操作系统包.读写打开标志, 0600)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 首个.关闭()
	秒, 错误 := 操作系统包.打开文件(路径, 操作系统包.读写打开标志, 0600)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 秒.关闭()
	if 错误 := 尝试(首个); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 尝试(秒); !错误判断包.是否属于(错误, 已加锁错误) {
		测试器.格式化报告致命错误("未互斥: %v", 错误)
	}
	if 错误 := 首个.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 尝试(秒); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 尝试(nil); !错误判断包.是否属于(错误, 操作系统包.无效错误) {
		测试器.报告致命错误(错误)
	}
}

func Test加锁已释放之后进程已终止(测试器 *测试包.测试器) {
	路径 := 文件路径包.合并(测试器.创建临时目录(), "crash.lock")
	可执行文件, 错误 := 操作系统包.可执行文件()
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	命令 := 进程执行包.命令上下文(测试器.上下文(), 可执行文件, "-test.run=^TestLockHelper$")
	命令.环境变量 = append(操作系统包.环境变量集(), "CALLUM_TEST_LOCK_PATH="+路径)
	输入, 错误 := 命令.标准输入管道()
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 输入.关闭()
	输出, 错误 := 命令.标准输出管道()
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 输出.关闭()
	if 错误 := 命令.启动(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer func() { _ = 命令.进程.终止(); _ = 命令.等待() }()
	if 行, 错误 := 缓冲输入输出包.新建读取器(输出).读取字符串('\n'); 错误 != nil || 行 != "locked\n" {
		测试器.格式化报告致命错误("子进程未持锁: %q %v", 行, 错误)
	}
	文件, 错误 := 操作系统包.打开文件(路径, 操作系统包.读写打开标志, 0600)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 文件.关闭()
	if 错误 := 尝试(文件); !错误判断包.是否属于(错误, 已加锁错误) {
		测试器.格式化报告致命错误("未保持互斥: %v", 错误)
	}
	if 错误 := 命令.进程.终止(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 命令.等待(); 错误 == nil {
		测试器.报告致命错误("强制退出未生效")
	}
	if 错误 := 尝试(文件); 错误 != nil {
		测试器.格式化报告致命错误("崩溃后未释放锁: %v", 错误)
	}
}
