// 包 文件锁 提供随文件句柄和进程退出释放的跨进程排他锁。
package 文件锁

import (
	错误判断包 "errors"
	操作系统包 "os"
)

// 已加锁错误 表示另一个句柄已持有排他锁，不代表路径或权限错误。
var 已加锁错误 = 错误判断包.新建("文件锁已被占用")

// 尝试 非阻塞地锁定普通文件。调用者持有文件，关闭句柄即释放锁；
// 不删除锁文件，避免等待者仍持有旧 inode 时另一个实例创建第二把锁。
func 尝试(文件 *操作系统包.文件) error {
	if 文件 == nil {
		return 操作系统包.无效错误
	}
	信息, 错误 := 文件.状态信息()
	if 错误 != nil {
		return 错误
	}
	if !信息.模式().是否普通文件() {
		return 操作系统包.无效错误
	}
	return 加锁(文件)
}
