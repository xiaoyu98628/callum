//go:build darwin || linux

// 包 进程组 统一独立 Unix 进程组的停止与有界收尾确认，不持有进程或管道。
package 进程组

import (
	上下文包 "context"
	错误判断包 "errors"
	运行时包 "runtime"
	系统调用包 "syscall"
	时间包 "time"
)

// 停止 只向调用者持有的独立进程组发送一次终止信号。
// 已退出 必须来自已经完成的主进程等待，不能用 进程标识 存在性推测。
func 停止(进程标识 int, 已退出 bool, 超时 时间包.时长) error {
	if 进程标识 <= 0 || 超时 <= 0 {
		return 系统调用包.无效参数错误
	}
	return 执行停止(已退出, 运行时包.目标操作系统 == "darwin", 超时, func(系统信号 系统调用包.系统信号) error { return 系统调用包.终止(-进程标识, 系统信号) })
}

func 执行停止(已退出, macOS bool, 超时 时间包.时长, 系统信号 func(系统调用包.系统信号) error) error {
	错误 := 系统信号(系统调用包.强制终止信号)
	if 错误判断包.是否属于(错误, 系统调用包.进程不存在错误) {
		return nil
	}
	if !macOS || !已退出 || !错误判断包.是否属于(错误, 系统调用包.操作不允许错误) {
		return 错误
	}
	// macOS 对退出中的空进程组也可能返回 操作不允许错误。之后只读确认消失，
	// 不再次向已回收的组号发送终止信号；无法确认时保留原始错误。
	if 确认错误 := 等待已移除(func() error { return 系统信号(0) }, 超时); 确认错误 != nil {
		return 错误判断包.合并(错误, 确认错误)
	}
	return nil
}

func 等待已移除(探测 func() error, 超时 时间包.时长) error {
	截止时间 := 时间包.新建计时器(超时)
	defer 截止时间.停止()
	周期计时器 := 时间包.新建周期计时器(10 * 时间包.毫秒)
	defer 周期计时器.停止()
	for {
		错误 := 探测()
		if 错误判断包.是否属于(错误, 系统调用包.进程不存在错误) {
			return nil
		}
		if 错误 != nil && !错误判断包.是否属于(错误, 系统调用包.操作不允许错误) {
			return 错误
		}
		select {
		case <-截止时间.字符:
			return 上下文包.超出截止时间错误
		case <-周期计时器.字符:
		}
	}
}
