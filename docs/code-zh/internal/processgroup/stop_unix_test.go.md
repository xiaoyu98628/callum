//go:build darwin || linux

package 进程组

import (
	上下文包 "context"
	错误判断包 "errors"
	系统调用包 "syscall"
	测试包 "testing"
	同步测试包 "testing/synctest"
	时间包 "time"
)

func Test停止确认(测试器 *测试包.测试器) {
	for _, 测试用例 := range []struct {
		名称         string
		已退出, macOS bool
		终止         error
		探测集        []error
		预期         error
	}{
		{名称: "absent", 终止: 系统调用包.进程不存在错误},
		{名称: "killed"},
		{名称: "live_permission_denied", macOS: true, 终止: 系统调用包.操作不允许错误, 预期: 系统调用包.操作不允许错误},
		{名称: "linux_permission_denied", 已退出: true, 终止: 系统调用包.操作不允许错误, 预期: 系统调用包.操作不允许错误},
		{名称: "other_kill_error", 已退出: true, macOS: true, 终止: 系统调用包.输入输出错误, 预期: 系统调用包.输入输出错误},
		{名称: "exiting", 已退出: true, macOS: true, 终止: 系统调用包.操作不允许错误, 探测集: []error{系统调用包.操作不允许错误, nil, 系统调用包.进程不存在错误}},
		{名称: "still_exists", 已退出: true, macOS: true, 终止: 系统调用包.操作不允许错误, 探测集: []error{nil}, 预期: 上下文包.超出截止时间错误},
		{名称: "permission_persists", 已退出: true, macOS: true, 终止: 系统调用包.操作不允许错误, 探测集: []error{系统调用包.操作不允许错误}, 预期: 上下文包.超出截止时间错误},
		{名称: "probe_error", 已退出: true, macOS: true, 终止: 系统调用包.操作不允许错误, 探测集: []error{系统调用包.输入输出错误}, 预期: 系统调用包.输入输出错误},
	} {
		测试器.执行(测试用例.名称, func(测试器 *测试包.测试器) {
			同步测试包.测试(测试器, func(测试器 *测试包.测试器) {
				终止, 探测集 := 0, 0
				错误 := 执行停止(测试用例.已退出, 测试用例.macOS, 100*时间包.毫秒, func(系统信号 系统调用包.系统信号) error {
					if 系统信号 == 系统调用包.强制终止信号 {
						终止++
						return 测试用例.终止
					}
					if 系统信号 != 0 || len(测试用例.探测集) == 0 {
						测试器.格式化报告致命错误("不应发送额外信号 %v", 系统信号)
					}
					值 := 测试用例.探测集[min(探测集, len(测试用例.探测集)-1)]
					探测集++
					return 值
				})
				if !错误判断包.是否属于(错误, 测试用例.预期) || 终止 != 1 {
					测试器.格式化报告致命错误("停止结果 err=%v kills=%d", 错误, 终止)
				}
				if len(测试用例.探测集) > 0 && 探测集 < len(测试用例.探测集) {
					测试器.格式化报告致命错误("缺少消失确认: %d", 探测集)
				}
				if 测试用例.预期 != nil && len(测试用例.探测集) > 0 && !错误判断包.是否属于(错误, 系统调用包.操作不允许错误) {
					测试器.格式化报告致命错误("丢失原始 EPERM: %v", 错误)
				}
			})
		})
	}
}

func Test停止拒绝无效身份(测试器 *测试包.测试器) {
	for _, 进程标识 := range []int{-1, 0} {
		if !错误判断包.是否属于(停止(进程标识, true, 时间包.秒), 系统调用包.无效参数错误) {
			测试器.报告致命错误("接受了无效进程组")
		}
	}
}
