package 执行器

import (
	上下文包 "context"
	错误判断包 "errors"
	字符串包 "strings"
	测试包 "testing"
	时间包 "time"

	调度器包 "github.com/xiaoyu98628/callum/internal/agent/executor"
	推理调用包 "github.com/xiaoyu98628/callum/internal/inference"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

type 模拟客户端 struct {
	生成回调 func(上下文包.上下文, 模型包.请求) (模型包.响应, error)
	流    func(上下文包.上下文, 模型包.请求, func(模型包.事件) error) error
}

func (当前模拟客户端 模拟客户端) 生成(执行上下文 上下文包.上下文, 请求 模型包.请求) (模型包.响应, error) {
	return 当前模拟客户端.生成回调(执行上下文, 请求)
}
func (当前模拟客户端 模拟客户端) 流式生成(执行上下文 上下文包.上下文, 请求 模型包.请求, 发出事件 func(模型包.事件) error) error {
	return 当前模拟客户端.流(执行上下文, 请求, 发出事件)
}

func 下一个事件(测试器 *测试包.测试器, 事件集 <-chan 推理调用包.事件) 推理调用包.事件 {
	测试器.标记测试辅助函数()
	select {
	case 事件 := <-事件集:
		return 事件
	case <-时间包.之后(3 * 时间包.秒):
		测试器.报告致命错误("未收到执行事件")
		return 推理调用包.事件{}
	}
}

func 新建测试管理器(测试器 *测试包.测试器, 客户端 模型包.客户端, 上限 int, 超时 时间包.时长, 事件集 chan<- 推理调用包.事件) (*管理器, *调度器包.管理器) {
	测试器.标记测试辅助函数()
	执行集, 错误 := 调度器包.新建(测试器.上下文(), 上限, 超时)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(执行集.关闭)
	管理器, 错误 := 新建带有调度器(执行集, func(上下文包.上下文, string) (模型包.客户端, string, error) {
		return 客户端, "remote", nil
	}, func(当前事件 推理调用包.事件) error {
		事件集 <- 当前事件
		return nil
	})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	return 管理器, 执行集
}

func Test并发执行取消并且容量(测试器 *测试包.测试器) {
	事件集 := make(chan 推理调用包.事件, 16)
	已开始 := make(chan string, 2)
	结束 := make(chan struct{})
	客户端 := 模拟客户端{生成回调: func(执行上下文 上下文包.上下文, 请求 模型包.请求) (模型包.响应, error) {
		已开始 <- 请求.消息集[0].内容
		select {
		case <-执行上下文.完成():
			return 模型包.响应{}, 执行上下文.获取错误()
		case <-结束:
			return 模型包.响应{文本: 请求.消息集[0].内容}, nil
		}
	}}
	管理器, 执行集 := 新建测试管理器(测试器, 客户端, 2, 时间包.秒, 事件集)
	var 标识集 []string
	for _, 文本 := range []string{"甲", "乙"} {
		输入 := 推理调用包.请求{模型标识: "01900000-0000-7000-8000-000000000001", 消息集: []模型包.消息{{角色: 模型包.角色用户, 内容: 文本}}}
		if 错误 := 管理器.启动(测试器.上下文(), 输入, func(标识 string) error { 标识集 = append(标识集, 标识); return nil }); 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	<-已开始
	<-已开始
	if 错误 := 管理器.启动(测试器.上下文(), 推理调用包.请求{模型标识: "01900000-0000-7000-8000-000000000001", 消息集: []模型包.消息{{角色: 模型包.角色用户}}}, func(string) error { return nil }); !错误判断包.是否属于(错误, 推理调用包.繁忙错误) {
		测试器.格式化报告致命错误("并发上限未生效: %v", 错误)
	}
	if !管理器.取消执行(标识集[0]) {
		测试器.报告致命错误("取消未生效")
	}
	if 事件 := 下一个事件(测试器, 事件集); 事件.执行标识 != 标识集[0] || 事件.类型 != "cancelled" {
		测试器.格式化报告致命错误("取消事件错误: %+v", 事件)
	}
	close(结束)
	if 事件 := 下一个事件(测试器, 事件集); 事件.执行标识 != 标识集[1] || 事件.类型 != "completed" || 事件.文本 == nil || *事件.文本 != "乙" {
		测试器.格式化报告致命错误("其他执行受到影响: %+v", 事件)
	}
	执行集.关闭()
	执行集.关闭()
	if 管理器.取消执行(标识集[0]) {
		测试器.报告致命错误("已结束执行不应再次取消")
	}
	if 错误 := 管理器.启动(测试器.上下文(), 推理调用包.请求{模型标识: "01900000-0000-7000-8000-000000000001", 消息集: []模型包.消息{{角色: 模型包.角色用户}}}, func(string) error { return nil }); !错误判断包.是否属于(错误, 推理调用包.已关闭错误) {
		测试器.格式化报告致命错误("关闭后仍接受执行: %v", 错误)
	}
}

func Test查找上下文不取消已受理执行(测试器 *测试包.测试器) {
	查找执行上下文, 取消查找 := 上下文包.创建可取消上下文(测试器.上下文())
	继续 := make(chan struct{})
	事件集 := make(chan 推理调用包.事件, 2)
	客户端 := 模拟客户端{生成回调: func(执行上下文 上下文包.上下文, 请求 模型包.请求) (模型包.响应, error) {
		select {
		case <-继续:
			return 模型包.响应{文本: 请求.消息集[0].内容}, 执行上下文.获取错误()
		case <-执行上下文.完成():
			return 模型包.响应{}, 执行上下文.获取错误()
		}
	}}
	管理器, _ := 新建测试管理器(测试器, 客户端, 1, 时间包.秒, 事件集)
	输入 := 推理调用包.请求{模型标识: "01900000-0000-7000-8000-000000000001", 消息集: []模型包.消息{{角色: 模型包.角色用户, 内容: "原文"}}}
	if 错误 := 管理器.启动(查找执行上下文, 输入, func(string) error { return nil }); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	取消查找()
	输入.消息集[0].内容 = "调用方已修改"
	close(继续)
	事件 := 下一个事件(测试器, 事件集)
	if 事件.类型 != "completed" || *事件.文本 != "原文" {
		测试器.格式化报告致命错误("上下文或消息快照隔离失败: %+v", 事件)
	}
}

func Test流终态状态使用返回值(测试器 *测试包.测试器) {
	for _, 测试 := range []struct {
		名称 string
		流  func(上下文包.上下文, 模型包.请求, func(模型包.事件) error) error
		预期 string
		原因 error
	}{
		{"完整", func(_ 上下文包.上下文, _ 模型包.请求, 发出事件 func(模型包.事件) error) error {
			if 错误 := 发出事件(模型包.事件{类型: 模型包.事件文本增量, 文本: "你好"}); 错误 != nil {
				return 错误
			}
			return 发出事件(模型包.事件{类型: 模型包.事件已完成})
		}, "completed", nil},
		{"完成回调后失败", func(_ 上下文包.上下文, _ 模型包.请求, 发出事件 func(模型包.事件) error) error {
			if 错误 := 发出事件(模型包.事件{类型: 模型包.事件已完成}); 错误 != nil {
				return 错误
			}
			return 模型包.不完整错误
		}, "failed", 模型包.不完整错误},
		{"输出超限", func(_ 上下文包.上下文, _ 模型包.请求, 发出事件 func(模型包.事件) error) error {
			return 发出事件(模型包.事件{类型: 模型包.事件文本增量, 文本: 字符串包.重复("x", 推理调用包.最大文本字节+1)})
		}, "failed", 推理调用包.输出上限错误},
		{"超时", func(执行上下文 上下文包.上下文, _ 模型包.请求, _ func(模型包.事件) error) error {
			<-执行上下文.完成()
			return 执行上下文.获取错误()
		}, "failed", 上下文包.超出截止时间错误},
	} {
		测试器.执行(测试.名称, func(测试器 *测试包.测试器) {
			事件集 := make(chan 推理调用包.事件, 4)
			超时 := 时间包.秒
			if 错误判断包.是否属于(测试.原因, 上下文包.超出截止时间错误) {
				超时 = 50 * 时间包.毫秒
			}
			管理器, 执行集 := 新建测试管理器(测试器, 模拟客户端{流: 测试.流}, 1, 超时, 事件集)
			if 错误 := 管理器.启动(测试器.上下文(), 推理调用包.请求{模型标识: "01900000-0000-7000-8000-000000000001", 流: true, 消息集: []模型包.消息{{角色: 模型包.角色用户}}}, func(string) error { return nil }); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			管理器事件 := 下一个事件(测试器, 事件集)
			if 管理器事件.类型 == "text_delta" {
				管理器事件 = 下一个事件(测试器, 事件集)
			}
			if 管理器事件.类型 != 测试.预期 || !错误判断包.是否属于(管理器事件.错误, 测试.原因) {
				测试器.格式化报告致命错误("终态错误: %+v", 管理器事件)
			}
			执行集.关闭()
			if len(事件集) != 0 {
				测试器.报告致命错误("产生了重复终态")
			}
		})
	}
}
