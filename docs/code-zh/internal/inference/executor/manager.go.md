```go
package 执行器

import (
	上下文包 "context"
	随机包 "crypto/rand"
	错误判断包 "errors"
	字符串包 "strings"
	时间包 "time"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	调度器包 "github.com/xiaoyu98628/callum/internal/agent/executor"
	推理调用包 "github.com/xiaoyu98628/callum/internal/inference"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

// 管理器 为兼容的无历史模型调用提供入口，与普通对话共享执行调度器。
// 发出事件 必须支持并发，已受理 必须在返回前将受理响应排在事件之前。
type 管理器 struct {
	调度器 *调度器包.管理器
	解析   推理调用包.解析器
	发出事件      func(推理调用包.事件) error
}

// 新建带有调度器 借用共享调度器，调用方负责关闭调度器。
func 新建带有调度器(管理器 *调度器包.管理器, 解析 推理调用包.解析器, 发出事件 func(推理调用包.事件) error) (*管理器, error) {
	if 管理器 == nil || 解析 == nil || 发出事件 == nil {
		return nil, 推理调用包.无效请求错误
	}
	return &管理器{调度器包: 管理器, 解析: 解析, 发出事件: 发出事件}, nil
}

func (当前管理器 *管理器) 上限集() (int, 时间包.时长) { return 当前管理器.调度器.上限集() }

// 启动 使用请求上下文解析配置，已接受的任务使用独立运行上下文。
func (当前管理器 *管理器) 启动(执行上下文 上下文包.上下文, 输入 推理调用包.请求, 已受理 func(string) error) error {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	if 已受理 == nil || 字符串包.裁剪空间(输入.模型标识) == "" || len(输入.消息集) == 0 || len(输入.消息集) > 256 {
		return 推理调用包.无效请求错误
	}
	for _, 消息 := range 输入.消息集 {
		if 消息.角色 != 模型包.角色系统 && 消息.角色 != 模型包.角色用户 && 消息.角色 != 模型包.角色助手 {
			return 推理调用包.无效请求错误
		}
	}
	输入.消息集 = append([]模型包.消息(nil), 输入.消息集...)
	for 索引 := range 输入.消息集 {
		输入.消息集[索引].内容块集 = append([]模型包.内容块(nil), 输入.消息集[索引].内容块集...)
	}
	标识 := 随机包.文本()
	执行租约, 错误 := 当前管理器.调度器.预留上下文(执行上下文, 标识, "")
	if 错误 != nil {
		switch {
		case 错误判断包.是否属于(错误, 智能体包.繁忙错误):
			return 推理调用包.繁忙错误
		case 错误判断包.是否属于(错误, 智能体包.已关闭错误):
			return 推理调用包.已关闭错误
		default:
			return 错误
		}
	}
	defer 执行租约.中止()
	查找执行上下文, 取消函数 := 上下文包.创建可取消上下文(执行上下文)
	停止 := 上下文包.之后函数(执行租约.上下文(), 取消函数)
	客户端, 远程, 错误 := 当前管理器.解析(查找执行上下文, 输入.模型标识)
	查找错误 := 查找执行上下文.获取错误()
	停止()
	取消函数()
	if 错误 := 执行租约.上下文().获取错误(); 错误 != nil {
		return 错误
	}
	if 错误 != nil {
		return 错误
	}
	if 查找错误 != nil {
		return 查找错误
	}
	if 客户端 == nil || 字符串包.裁剪空间(远程) == "" {
		return 推理调用包.无效请求错误
	}
	if 错误 := 已受理(标识); 错误 != nil {
		return 错误
	}
	执行租约.启动(func(执行上下文 上下文包.上下文) {
		当前管理器.执行(执行上下文, 标识, 客户端, 模型包.请求{模型: 远程, 消息集: 输入.消息集}, 输入.流, 执行租约)
	})
	return nil
}

func (当前管理器 *管理器) 取消执行(标识 string) bool { return 当前管理器.调度器.取消执行(标识) }
```
