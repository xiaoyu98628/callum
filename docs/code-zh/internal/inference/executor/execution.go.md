```go
package 执行器

import (
	上下文包 "context"
	错误判断包 "errors"

	调度器包 "github.com/xiaoyu98628/callum/internal/agent/executor"
	推理调用包 "github.com/xiaoyu98628/callum/internal/inference"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

func (当前管理器 *管理器) 执行(执行上下文 上下文包.上下文, 标识 string, 客户端 模型包.客户端, 请求 模型包.请求, 流 bool, 执行租约 *调度器包.执行租约) {
	事件 := 推理调用包.事件{类型: "completed", 执行标识: 标识}
	错误 := 当前管理器.调度器.调用(执行上下文, func() error {
		var 错误 error
		if 流 {
			字节 := 0
			错误 = 客户端.流式生成(执行上下文, 请求, func(增量 模型包.事件) error {
				if 错误 := 执行上下文.获取错误(); 错误 != nil {
					return 错误
				}
				事件.供应商请求标识 = 增量.请求标识
				if 增量.类型 != 模型包.事件文本增量 {
					return nil
				}
				字节 += len(增量.文本)
				if 字节 > 推理调用包.最大文本字节 {
					return 推理调用包.输出上限错误
				}
				return 当前管理器.发出事件(推理调用包.事件{类型: "text_delta", 执行标识: 标识, 文本: new(增量.文本), 供应商请求标识: 增量.请求标识})
			})
		} else {
			var 响应 模型包.响应
			响应, 错误 = 客户端.生成(执行上下文, 请求)
			事件.供应商请求标识 = 响应.请求标识
			if len(响应.文本) > 推理调用包.最大文本字节 {
				错误 = 错误判断包.合并(错误, 推理调用包.输出上限错误)
			} else {
				事件.文本 = new(响应.文本)
			}
		}
		return 错误
	})
	// 与取消操作串行确定终态，已进入终态的执行不能再被标记为取消。
	if 取消错误 := 执行租约.是否正在结束(); 取消错误 != nil {
		错误 = 错误判断包.合并(取消错误, 错误)
	}
	if 错误 != nil {
		事件.类型, 事件.错误 = "failed", 错误
		if 错误判断包.是否属于(错误, 上下文包.已取消错误) {
			事件.类型 = "cancelled"
		}
		if 调用, 成功 := 错误判断包.按类型匹配[*模型包.调用错误](错误); 成功 {
			事件.供应商请求标识 = 调用.请求标识
		}
	}
	// 只有客户端方法返回成功才发出 已完成；适配器内部的完成回调不直接成为执行终态。
	// 事件接收方拒绝发布后不递归报告错误，接收方负责处理自身的生命周期。
	if 发布错误 := 当前管理器.发出事件(事件); 发布错误 != nil {
		return
	}
}
```
