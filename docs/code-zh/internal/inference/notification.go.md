```go
package 推理调用

import (
	上下文包 "context"
	错误判断包 "errors"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	供应商包 "github.com/xiaoyu98628/callum/internal/model/provider"
)

// 失败 只包含可公开的错误描述，不保留原始错误链或供应商响应正文。
type 失败 struct {
	类别      string
	消息      string
	供应商请求标识 string
	HTTP状态  int
}

// 通知 是无历史模型调用的只读事件数据，协议编码由通信边界处理。
type 通知 struct {
	文本      *string
	供应商请求标识 string
	失败      *失败
}

// 描述错误 统一提取受控错误信息，防止事件队列保存凭据或大型错误正文。
func 描述错误(错误 error) 失败 {
	当前条目 := 失败{类别: "internal", 消息: "内部处理失败"}
	switch {
	case 错误判断包.是否属于(错误, 智能体包.存储错误):
		当前条目.消息, 当前条目.类别 = "保存执行数据失败", "internal"
	case 错误判断包.是否属于(错误, 上下文包.超出截止时间错误):
		当前条目.消息, 当前条目.类别 = "操作超时", "timeout"
	case 错误判断包.是否属于(错误, 上下文包.已取消错误):
		当前条目.消息, 当前条目.类别 = "操作已取消", "cancelled"
	case 错误判断包.是否属于(错误, 供应商包.无效配置错误), 错误判断包.是否属于(错误, 无效请求错误), 错误判断包.是否属于(错误, 智能体包.无效错误):
		当前条目.消息, 当前条目.类别 = "参数无效", "invalid_params"
	case 错误判断包.是否属于(错误, 供应商包.未找到错误):
		当前条目.消息, 当前条目.类别 = "配置不存在", "not_found"
	case 错误判断包.是否属于(错误, 智能体包.未找到错误):
		当前条目.消息, 当前条目.类别 = "执行或步骤不存在", "not_found"
	case 错误判断包.是否属于(错误, 供应商包.冲突错误):
		当前条目.消息, 当前条目.类别 = "同一厂商下的远端模型标识已存在", "conflict"
	case 错误判断包.是否属于(错误, 供应商包.被引用错误):
		当前条目.消息, 当前条目.类别 = "关联配置不存在或仍被引用", "referenced"
	case 错误判断包.是否属于(错误, 供应商包.已停用错误):
		当前条目.消息, 当前条目.类别 = "模型已禁用", "disabled"
	case 错误判断包.是否属于(错误, 供应商包.凭据错误):
		当前条目.消息, 当前条目.类别 = "系统凭据库不可用或凭据无法解密，请解锁凭据库后重试。", "credentials_unavailable"
	case 错误判断包.是否属于(错误, 不支持错误), 错误判断包.是否属于(错误, 模型包.不支持输出错误):
		当前条目.消息, 当前条目.类别 = "当前适配器不支持该模型功能或输出", "unsupported"
	case 错误判断包.是否属于(错误, 繁忙错误):
		当前条目.消息, 当前条目.类别 = "模型调用已达到并发上限", "busy"
	case 错误判断包.是否属于(错误, 已关闭错误):
		当前条目.消息, 当前条目.类别 = "服务正在关闭", "closing"
	case 错误判断包.是否属于(错误, 输出上限错误):
		当前条目.消息, 当前条目.类别 = "模型文本超过大小限制", "output_limit"
	case 错误判断包.是否属于(错误, 模型包.附件错误):
		当前条目.消息, 当前条目.类别 = 模型包.附件错误.错误文本(), "attachment_unavailable"
	case 错误判断包.是否属于(错误, 模型包.图片读取错误):
		当前条目.消息, 当前条目.类别 = "图片文件不可读取，请重新添加", "image_read"
	case 错误判断包.是否属于(错误, 模型包.图片格式错误):
		当前条目.消息, 当前条目.类别 = "图片损坏或格式不支持", "image_format"
	case 错误判断包.是否属于(错误, 模型包.图片上限错误):
		当前条目.消息, 当前条目.类别 = 模型包.图片上限错误.错误文本(), "image_limit"
	case 错误判断包.是否属于(错误, 模型包.输入上限错误):
		当前条目.消息, 当前条目.类别 = "模型请求超过字节上限，请减少图片或文字", "input_limit"
	case 错误判断包.是否属于(错误, 模型包.不完整错误):
		当前条目.消息, 当前条目.类别 = "模型输出未完整完成", "incomplete"
	case 错误判断包.是否属于(错误, 模型包.已拒绝错误):
		当前条目.消息, 当前条目.类别 = "模型拒绝生成内容", "refused"
	case 错误判断包.是否属于(错误, 模型包.无效响应错误):
		当前条目.消息, 当前条目.类别 = "模型响应格式无效", "invalid_response"
	default:
		当前条目.消息, 当前条目.类别 = "模型调用失败", "model_failed"
	}

	if 调用, 成功 := 错误判断包.按类型匹配[*模型包.调用错误](错误); 成功 {
		当前条目.供应商请求标识, 当前条目.HTTP状态 = 调用.请求标识, 调用.状态代码
	}
	return 当前条目
}

// 公开事件 在发布到共享订阅前移除错误链。
func 公开事件(事件 事件) 智能体包.事件 {
	数据 := 通知{文本: 事件.文本, 供应商请求标识: 事件.供应商请求标识}
	if 事件.错误 != nil {
		失败 := 描述错误(事件.错误)
		数据.失败 = &失败
	}
	return 智能体包.事件{类型: "run." + 事件.类型, 执行标识: 事件.执行标识, 数据: 数据}
}
```
