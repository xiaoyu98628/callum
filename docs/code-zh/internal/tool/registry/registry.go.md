package 注册表

import (
	JSON包 "encoding/json"
	正则表达式包 "regexp"
	切片包 "slices"
	同步包 "sync"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

var 名称模式 = 正则表达式包.必须编译(`^[a-zA-Z0-9_-]{1,64}$`)

// 注册表 持有工具实例，注册后不可替换同名工具，查找支持并发。
type 注册表 struct {
	互斥锁 同步包.读写互斥锁
	工具集 map[string]工具包.工具
}

func 新建() *注册表 { return &注册表{工具集: make(map[string]工具包.工具)} }

func (当前注册表 *注册表) 注册(当前工具 工具包.工具) error {
	if 当前工具 == nil {
		return 工具包.无效错误
	}
	丁 := 当前工具.获取定义()
	if !名称模式.匹配字符串(丁.名称) || 丁.说明 == "" || 丁.来源 == "" || 丁.版本 == "" || !JSON包.有效(丁.参数) {
		return 工具包.无效错误
	}
	for _, 影响 := range 丁.影响集 {
		if !影响.有效() {
			return 工具包.无效错误
		}
	}
	当前注册表.互斥锁.加锁()
	defer 当前注册表.互斥锁.解锁()
	if _, 成功 := 当前注册表.工具集[丁.名称]; 成功 {
		return 工具包.无效错误
	}
	当前注册表.工具集[丁.名称] = 当前工具
	return nil
}

func (当前注册表 *注册表) 获取(名称 string) (工具包.工具, error) {
	当前注册表.互斥锁.加读锁()
	defer 当前注册表.互斥锁.解读锁()
	当前项 := 当前注册表.工具集[名称]
	if 当前项 == nil {
		return nil, 工具包.不可用错误
	}
	return 当前项, nil
}

func (当前注册表 *注册表) 定义集() []工具包.定义 {
	当前注册表.互斥锁.加读锁()
	defer 当前注册表.互斥锁.解读锁()
	输出 := make([]工具包.定义, 0, len(当前注册表.工具集))
	for _, 当前项 := range 当前注册表.工具集 {
		丁 := 当前项.获取定义()
		丁.参数 = 切片包.复制(丁.参数)
		丁.影响集 = 切片包.复制(丁.影响集)
		输出 = append(输出, 丁)
	}
	切片包.排序函数(输出, func(当前定义, 当前定义 工具包.定义) int {
		if 当前定义.名称 < 当前定义.名称 {
			return -1
		}
		if 当前定义.名称 > 当前定义.名称 {
			return 1
		}
		return 0
	})
	return 输出
}
