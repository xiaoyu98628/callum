```go
// 包 工具 定义工具执行契约，权限和可选工作区来自服务端，不接受模型覆盖。
package 工具

import (
	上下文包 "context"
	JSON包 "encoding/json"
	错误判断包 "errors"

	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

const 最大结果字节 = 64 << 10
const 最大参数字节 = 16 << 10

const (
	最大输出分片字节 = 4 << 10
	最大输出字节      = 64 << 10
	最大输出事件集     = 128
)

var (
	无效错误        = 错误判断包.新建("工具参数无效")
	已拒绝错误         = 错误判断包.新建("操作超出授权工作区或权限范围")
	不可用错误    = 错误判断包.新建("工具不可用")
	超出上限错误          = 错误判断包.新建("工具输入或结果超过限制")
	路径基准错误       = 错误判断包.新建("相对路径需要工作区，请提供绝对路径或绑定工作区")
	未知操作结果错误 = 错误判断包.新建("操作结果未知，需要核实后再重试")
	部分操作结果错误 = 错误判断包.新建("操作仅部分生效，需要核实已完成的部分")
	冲突错误       = 错误判断包.新建("文件版本、路径或补丁上下文发生冲突，请重新读取后修改")
)

// 操作结果 描述实际副作用，与工具调用是否成功、执行是否取消分别记录。
// 只能由可信工具实现报告；未报告或无法确认时不能推断为未生效。
type 操作结果 string

const (
	操作结果无       操作结果 = "none"              // 操作没有修改或外发等副作用。
	操作结果不已生效 操作结果 = "not_applied"       // 确认尚未产生副作用。
	操作结果已生效    操作结果 = "applied"           // 确认请求的操作已经生效。
	操作结果部分    操作结果 = "partially_applied" // 确认只有部分操作生效。
	操作结果未知    操作结果 = "unknown"           // 无法确认操作是否或完全生效。
)

// 有效 判断是否为当前契约可表达的事实。
func (当前操作结果 操作结果) 有效() bool {
	return 当前操作结果 == 操作结果无 || 当前操作结果 == 操作结果不已生效 || 当前操作结果 == 操作结果已生效 || 当前操作结果 == 操作结果部分 || 当前操作结果 == 操作结果未知
}

// 可能拥有影响集 用于保留需要后续知晓的操作事实，不表示执行成功或得到授权。
func (当前操作结果 操作结果) 可能拥有影响集() bool {
	return 当前操作结果 == 操作结果已生效 || 当前操作结果 == 操作结果部分 || 当前操作结果 == 操作结果未知
}

// 结果 保留有界 JSON 输出及独立的副作用事实；返回错误也不得丢弃已经发生的操作。
// 输出 在调用返回后归调用方持有，工具不得继续修改；错误时可以没有可用输出。
type 结果 struct {
	输出  JSON包.原始JSON消息
	操作结果 操作结果
	图片集  []模型包.内容块 // 可信读图工具生成的托管图片引用，单独计入视觉上下文。
}

// 操作版本 标识步骤 JSON 中操作事实的兼容版本，不是数据库结构版本。
const 操作版本 = 1

// 操作结果记录 是步骤和工具上下文中的服务端事实，不能替代审批或权限配置。
// 版本缺失的旧记录保持原义；已恢复 表示从未完成的新步骤确认结果尚不可得。
type 操作结果记录 struct {
	版本     int     `json:"version"`
	操作结果     操作结果 `json:"outcome"`
	请求哈希 string  `json:"request_hash,omitempty"`
	已恢复   bool    `json:"recovered,omitempty"`
}

// 有效 同时检查契约版本和事实枚举，不推断旧记录的含义。
func (当前操作结果记录 操作结果记录) 有效() bool { return 当前操作结果记录.版本 == 操作版本 && 当前操作结果记录.操作结果.有效() }

// 未确认操作结果 只根据可信定义识别无副作用工具；其他已启动操作一律保持未知。
func 未确认操作结果(影响集 []影响) 操作结果 {
	if len(影响集) == 0 {
		return 操作结果未知
	}
	for _, 影响 := range 影响集 {
		if 影响 != 影响读取 {
			return 操作结果未知
		}
	}
	return 操作结果无
}

// 影响 描述工具可能产生的操作影响，由可信工具实现声明。
type 影响 string

const (
	影响读取        影响 = "read"        // 读取，不修改资源；文件访问范围另行校验。
	影响写入       影响 = "write"       // 新建或普通修改，不包含大范围覆盖。
	影响破坏性 影响 = "destructive" // 删除、大范围覆盖等破坏性操作。
	影响外部    影响 = "external"    // 向外部发送数据。
	影响未知     影响 = "unknown"     // 尚不能明确判断操作影响。
)

// 有效 判断操作类别是否受支持；缺少类别时由审批策略按未知操作处理。
func (当前影响 影响) 有效() bool {
	switch 当前影响 {
	case 影响读取, 影响写入, 影响破坏性, 影响外部, 影响未知:
		return true
	default:
		return false
	}
}

// 定义 由可信注册代码提供；影响集 不能由模型或调用参数声明。
// 一个工具可包含多个影响，不能用读取类别抵消删除或外发影响。
type 定义 struct {
	模型包.工具定义
	来源  string   `json:"source"`
	版本 string   `json:"version"`
	影响集 []影响 `json:"effects"`
}

// 范围 提供可选的工作区路径基准；零值表示会话未绑定工作区。
type 范围 struct {
	标识       string `json:"id"`
	根路径 string `json:"root_path"`
}

// 沙箱模式 与审批策略分开；空值仅保留旧执行快照的无沙箱语义。
type 沙箱模式 string

const (
	沙箱只读       沙箱模式 = "read-only"
	沙箱工作区写入 沙箱模式 = "workspace-write"
	沙箱完整访问     沙箱模式 = "danger-full-access"
)

func (当前沙箱模式 沙箱模式) 有效() bool {
	return 当前沙箱模式 == 沙箱只读 || 当前沙箱模式 == 沙箱工作区写入 || 当前沙箱模式 == 沙箱完整访问
}

// 执行权限 由受理时的服务端设置快照提供，模型不能指定沙箱模式。
type 执行权限 struct {
	沙箱模式     沙箱模式
	会话标识  string
	执行标识           string
	输入目录  string // 服务端持有的本轮附件输入，只读。
	工作目录   string // 服务端持有的本轮私有中间目录，可写且结束后回收。
	允许图片输入 bool   // 来自用户选择的模型类型，不能由工具参数扩大。
}

// 权限工具 为需要运行权限的工具显式传入执行快照；普通文件工具沿用 准备。
type 权限工具 interface {
	工具
	准备带有权限(上下文包.上下文, 范围, JSON包.原始JSON消息, 执行权限) (工具调用, error)
}

// 命令边界 记录固定命令实际使用的执行边界，不将未知命令伪装为只读。
type 命令边界 struct {
	沙箱内                 bool        `json:"sandboxed"`
	模式                      沙箱模式 `json:"sandbox_mode,omitempty"`
	提权                 bool        `json:"escalated"`
	风险                      string      `json:"risk,omitempty"`
	创建工作区目录 bool        `json:"-"` // 由固定的系统程序、参数和工作目录计算，不改变请求指纹。
}

type 命令工具调用 interface {
	工具调用
	命令边界() 命令边界
}

// 目标 记录文件工具固定的实际目标，目录操作的范围限于该目录。
// 范围内工作区 由可信实现依据实际路径判断，不能由模型参数指定。
type 目标 struct {
	请求的路径   string `json:"requested_path"`
	路径            string `json:"path"`
	类别            string `json:"kind"`
	范围内工作区 bool   `json:"within_workspace"`
	动作          string `json:"action,omitempty"`
	预期SHA256  string `json:"expected_sha256,omitempty"`
	恢复路径    string `json:"recovery_path,omitempty"`
	暂存路径     string `json:"staging_path,omitempty"`
}

// 计划 固定多目标操作及额外影响。由可信调用实现生成，不能由模型声明权限。
// 附加影响集 只能增加影响，不能抵消工具定义中已有的影响。
type 计划 struct {
	目标集           []目标 `json:"targets"`
	附加影响集 []影响 `json:"additional_effects,omitempty"`
}

// 已规划工具调用 为实际需要多目标或动态影响判定的工具提供审批计划。
// 计划 必须只返回元数据，在 准备、审批等待及执行开始前保持不变。
type 已规划工具调用 interface {
	工具调用
	获取计划() 计划
}

// 工具调用 持有一次调用的固定资源；执行 在使用前重新校验目标。
// 非文件工具的 目标 返回 nil。关闭 必须可重复调用；调用创建者负责关闭，
// 包括审批拒绝、取消以及 准备 同时返回调用对象和错误时。
type 工具调用 interface {
	获取目标() *目标
	执行(上下文包.上下文) (结果, error)
	关闭() error
}

// 快照工具调用 提供服务端生成的补充审批详情，不包含凭据或可变引用。
// 快照必须是非空 JSON 对象，不超过 最大参数字节；执行前重新比较其内容。
// 未实现此接口的工具保持现有审批快照格式。
type 快照工具调用 interface {
	工具调用
	快照() JSON包.原始JSON消息
}

// 输出分片 是一次工具调用的输出片段。序号 由执行器分配，从 1 开始。
// 数据 允许任意字节，回调只能在返回前借用；需要保存时必须复制。
type 输出分片 struct {
	序号 int64
	流   string
	数据     []byte
}

// 流式工具调用 在执行期间同步发布有界输出。回调失败必须停止执行并清理。
// 返回前必须等待全部生产者退出，不允许返回后继续调用 发出事件；发出事件 为 nil 时仅保存最终结果。
// 发出事件 必须及时返回，不得在回调内同步等待本次调用的 关闭。
type 流式工具调用 interface {
	工具调用
	执行流(上下文包.上下文, func(输出分片) error) (结果, error)
}

// 工具 的 校验 返回规范化参数；准备 固定目标，但不得读取正文或执行操作。
// 工具按自身需求使用工作区，不要求所有工具依赖文件系统或工作区。
type 工具 interface {
	获取定义() 定义
	校验(JSON包.原始JSON消息) (JSON包.原始JSON消息, error)
	准备(上下文包.上下文, 范围, JSON包.原始JSON消息) (工具调用, error)
}
```
