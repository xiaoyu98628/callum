```go
package 策略测试

import (
	测试包 "testing"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
	策略包 "github.com/xiaoyu98628/callum/internal/tool/policy"
)

func Test审批策略不展开操作权限(测试器 *测试包.测试器) {
	for _, 测试 := range []struct {
		名称  string
		影响集 []工具包.影响
		请求  策略包.动作
		从不  策略包.动作
	}{
		{"read", []工具包.影响{工具包.影响读取}, 策略包.允许, 策略包.允许},
		{"ordinary_write", []工具包.影响{工具包.影响写入}, 策略包.允许, 策略包.允许},
		{"read_write", []工具包.影响{工具包.影响读取, 工具包.影响写入}, 策略包.允许, 策略包.允许},
		{"delete_or_overwrite", []工具包.影响{工具包.影响破坏性}, 策略包.询问, 策略包.拒绝},
		{"external", []工具包.影响{工具包.影响外部}, 策略包.询问, 策略包.拒绝},
		{"unknown", []工具包.影响{工具包.影响未知}, 策略包.询问, 策略包.拒绝},
		{"missing", nil, 策略包.询问, 策略包.拒绝},
		{"read_and_delete", []工具包.影响{工具包.影响读取, 工具包.影响破坏性}, 策略包.询问, 策略包.拒绝},
		{"write_and_external", []工具包.影响{工具包.影响写入, 工具包.影响外部}, 策略包.询问, 策略包.拒绝},
		{"read_and_unknown", []工具包.影响{工具包.影响读取, 工具包.影响未知}, 策略包.询问, 策略包.拒绝},
	} {
		测试器.执行(测试.名称, func(测试器 *测试包.测试器) {
			for _, 选项 := range []struct {
				策略 策略包.审批策略
				预期 策略包.动作
			}{{策略包.开启请求, 测试.请求}, {策略包.总是, 策略包.询问}, {策略包.从不, 测试.从不}} {
				实际值 := 策略包.评估(选项.策略, 测试.影响集, false)
				if 实际值.动作 != 选项.预期 || 实际值.原因 == "" {
					测试器.格式化报告致命错误("%s: %+v，期望 %s", 选项.策略, 实际值, 选项.预期)
				}
			}
		})
	}
}

func Test命令边界是独立所属审批策略(测试器 *测试包.测试器) {
	for _, 测试 := range []struct {
		名称       string
		边界       工具包.命令边界
		开启请求, 从不 策略包.动作
	}{
		{"workspace", 工具包.命令边界{沙箱内: true, 模式: 工具包.沙箱工作区写入}, 策略包.询问, 策略包.拒绝},
		{"readonly", 工具包.命令边界{沙箱内: true, 模式: 工具包.沙箱只读}, 策略包.允许, 策略包.允许},
		{"elevated", 工具包.命令边界{提权: true, 模式: 工具包.沙箱工作区写入}, 策略包.询问, 策略包.拒绝},
		{"destructive", 工具包.命令边界{沙箱内: true, 模式: 工具包.沙箱工作区写入, 风险: "destructive_command"}, 策略包.询问, 策略包.拒绝},
		{"legacy", 工具包.命令边界{}, 策略包.询问, 策略包.拒绝},
		{"unsupported", 工具包.命令边界{模式: 工具包.沙箱工作区写入, 风险: "sandbox_backend_unsupported"}, 策略包.询问, 策略包.拒绝},
	} {
		测试器.执行(测试.名称, func(测试器 *测试包.测试器) {
			for _, 选项 := range []struct {
				当前审批策略 策略包.审批策略
				预期     策略包.动作
			}{{策略包.开启请求, 测试.开启请求}, {策略包.从不, 测试.从不}, {策略包.总是, 策略包.询问}} {
				if 实际值 := 策略包.评估命令(选项.当前对象, 测试.边界); 实际值.动作 != 选项.预期 || 实际值.原因 == "" {
					测试器.格式化报告致命错误("审批扩大了边界: %s %+v", 选项.当前对象, 实际值)
				}
			}
		})
	}
}

func Test无效策略并且操作失败已关闭(测试器 *测试包.测试器) {
	for _, 当前对象 := range []策略包.审批策略{"", "auto-read", "allow-all"} {
		if 实际值 := 策略包.评估(当前对象, []工具包.影响{工具包.影响读取}, false); 实际值.动作 != 策略包.拒绝 {
			测试器.格式化报告致命错误("未知审批策略被放行: %s %+v", 当前对象, 实际值)
		}
	}
	for _, 当前对象 := range []策略包.审批策略{策略包.开启请求, 策略包.总是, 策略包.从不} {
		if 实际值 := 策略包.评估(当前对象, []工具包.影响{工具包.影响读取, "invalid"}, false); 实际值.动作 != 策略包.拒绝 {
			测试器.格式化报告致命错误("无效类别被忽略或允许审批: %s %+v", 当前对象, 实际值)
		}
	}
}

func Test外部工作区写入要求审批(测试器 *测试包.测试器) {
	for _, 测试 := range []struct {
		策略 策略包.审批策略
		预期 策略包.动作
	}{
		{策略包.开启请求, 策略包.询问}, {策略包.总是, 策略包.询问}, {策略包.从不, 策略包.拒绝},
	} {
		for _, 影响集 := range [][]工具包.影响{{工具包.影响写入}, {工具包.影响读取, 工具包.影响写入}, {工具包.影响读取, 工具包.影响破坏性}, {工具包.影响读取, 工具包.影响外部}, {工具包.影响读取, 工具包.影响未知}, nil} {
			if 实际值 := 策略包.评估(测试.策略, 影响集, true); 实际值.动作 != 测试.预期 {
				测试器.格式化报告致命错误("工作区外访问被错误放行: %s %+v", 测试.策略, 实际值)
			}
		}
	}
}

func Test只读访问独立所属工作区(测试器 *测试包.测试器) {
	for _, 外部 := range []bool{false, true} {
		for _, 选项 := range []struct {
			策略 策略包.审批策略
			预期 策略包.动作
		}{
			{策略包.开启请求, 策略包.允许}, {策略包.从不, 策略包.允许}, {策略包.总是, 策略包.询问},
		} {
			实际值 := 策略包.评估(选项.策略, []工具包.影响{工具包.影响读取}, 外部)
			if 实际值.动作 != 选项.预期 {
				测试器.格式化报告致命错误("纯读取策略错误: outside=%v policy=%s got=%+v", 外部, 选项.策略, 实际值)
			}
			if 选项.预期 == 策略包.允许 && 实际值.原因 != "read_only_operation" {
				测试器.格式化报告致命错误("只读来源错误: %+v", 实际值)
			}
		}
	}
}
```
