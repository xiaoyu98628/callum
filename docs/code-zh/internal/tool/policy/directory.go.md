package 策略

import (
	文件路径包 "path/filepath"
	切片包 "slices"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

// 目录覆盖 只接受全部已分类为读取或普通修改的固定文件目标，检查辅助文件路径。
func 目录覆盖(当前目录范围 工具包.目录范围, 目标 工具包.目标, 影响集 []工具包.影响) bool {
	if !当前目录范围.有效() || !文件路径包.是否绝对路径(目标.路径) || len(影响集) == 0 || (目标.类别 != "file" && 目标.类别 != "directory") {
		return false
	}
	for _, 影响 := range 影响集 {
		if 影响 != 工具包.影响读取 && (影响 != 工具包.影响写入 || 当前目录范围.附件访问器 != "read_write") {
			return false
		}
	}
	for _, 路径 := range []string{目标.路径, 目标.恢复路径, 目标.暂存路径} {
		if 路径 == "" {
			continue
		}
		if !文件路径包.是否绝对路径(路径) {
			return false
		}
		相对路径, 错误 := 文件路径包.相对路径(当前目录范围.根路径, 路径)
		if 错误 != nil || !文件路径包.是否本地路径(相对路径) {
			return false
		}
	}
	return true
}

// 目录集覆盖 要求工作区外的所有固定目标及辅助路径都被授权覆盖。
// 工作区内目标已由 开启-请求 策略授权；无文件目标的命令不能继承目录权限。
func 目录集覆盖(范围集 []工具包.目录范围, 目标集 []工具包.目标, 影响集 []工具包.影响) bool {
	外部 := false
	for _, 目标 := range 目标集 {
		if 目标.范围内工作区 {
			continue
		}
		外部 = true
		if !切片包.包含函数(范围集, func(范围 工具包.目录范围) bool { return 目录覆盖(范围, 目标, 影响集) }) {
			return false
		}
	}
	return 外部
}

// 目录选项 从已固定并参与操作指纹的目标推导授权选项，不改变历史操作身份。
// 创建目录的计划必须包含所有缺失父目录；由它们共同确定现有父目录范围。
// 不为跨盘或隐式扩大到磁盘根目录的操作提供目录授权选项。
func 目录选项(目标集 []工具包.目标, 影响集 []工具包.影响) *工具包.目录范围 {
	if len(目标集) == 0 || len(影响集) == 0 {
		return nil
	}
	访问 := "read"
	for _, 影响 := range 影响集 {
		if 影响 != 工具包.影响读取 && 影响 != 工具包.影响写入 {
			return nil
		}
		if 影响 == 工具包.影响写入 {
			访问 = "read_write"
		}
	}
	var 根 string
	for _, 目标 := range 目标集 {
		if !文件路径包.是否绝对路径(目标.路径) || (目标.类别 != "file" && 目标.类别 != "directory") {
			return nil
		}
		目录 := 目标.路径
		if 目标.类别 != "directory" || 目标.动作 == "create" {
			目录 = 文件路径包.目录(目录)
		}
		if 根 == "" {
			根 = 目录
		}
		for {
			相对路径, 错误 := 文件路径包.相对路径(根, 目录)
			if 错误 != nil {
				return nil
			}
			if 文件路径包.是否本地路径(相对路径) {
				break
			}
			下一个 := 文件路径包.目录(根)
			if 下一个 == 根 {
				return nil
			}
			根 = 下一个
		}
	}
	if 文件路径包.目录(根) == 根 && !(len(目标集) == 1 && 目标集[0].类别 == "directory" && 目标集[0].路径 == 根) {
		return nil
	}
	范围 := 工具包.目录范围{根路径: 根, 附件访问器: 访问}
	if 切片包.包含函数(目标集, func(目标 工具包.目标) bool { return !目录覆盖(范围, 目标, 影响集) }) {
		return nil
	}
	return &范围
}
