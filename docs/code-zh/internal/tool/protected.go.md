```go
package 工具

import (
	错误判断包 "errors"
	操作系统包 "os"
	文件路径包 "path/filepath"
	字符串包 "strings"
)

// 受保护数据 固定服务端自己的数据目录和 SQLite 文件位置。空值用于没有应用存储的调用方。
type 受保护数据 struct {
	目录    string
	数据库   string
	只读目录集 []string // 装配层传入已解析链接的实际绝对路径。
}

// 写入受保护 拒绝修改私有数据或只读运行时；读取运行时仍按普通文件规则处理。
func (当前受保护数据 受保护数据) 写入受保护(路径 string) bool {
	return 当前受保护数据.包含(路径) || 当前受保护数据.只读包含(路径)
}

// 读取受保护 仅向装配层登记的只读子树开放私有数据读取；写保护和 SQLite 身份检查保持独立。
func (当前受保护数据 受保护数据) 读取受保护(路径 string) bool {
	if !当前受保护数据.包含(路径) {
		return false
	}
	for _, 根 := range 当前受保护数据.只读目录集 {
		资源 := 受保护数据{目录: 根}
		if 当前受保护数据.包含(根) && !资源.包含(当前受保护数据.目录) && 资源.包含(路径) {
			return false
		}
	}
	return true
}

// 只读包含 判断实际目标是否属于只读资源树，不限制读取。
func (当前受保护数据 受保护数据) 只读包含(路径 string) bool {
	for _, 根 := range 当前受保护数据.只读目录集 {
		if (受保护数据{目录: 根}).包含(路径) {
			return true
		}
	}
	return false
}

// 保护后代 也保护资源目录的祖先，防止通过重命名祖先改变固定路径。
func (当前受保护数据 受保护数据) 保护后代(路径 string) bool {
	for _, 根 := range append([]string{当前受保护数据.目录}, 当前受保护数据.只读目录集...) {
		if 根 != "" && (受保护数据{目录: 路径}).包含(根) {
			return true
		}
	}
	return false
}

// 包含 判断已解析的目标是否在数据目录内，包括大小写不敏感卷上的路径变体。
func (当前受保护数据 受保护数据) 包含(路径 string) bool {
	if 当前受保护数据.目录 == "" {
		return false
	}
	相对, 错误 := 文件路径包.相对路径(字符串包.目标小写(当前受保护数据.目录), 字符串包.目标小写(路径))
	return 错误 == nil && 文件路径包.是否本地路径(相对)
}

// SQLite文件集 检查数据库与附属文件的当前身份；附属文件可随连接活动创建或移除。
func (当前受保护数据 受保护数据) SQLite文件集() ([]操作系统包.文件信息, error) {
	if 当前受保护数据.数据库 == "" {
		return nil, nil
	}
	var 文件集 []操作系统包.文件信息
	for _, 后缀 := range []string{"", "-journal", "-wal", "-shm", ".lock"} {
		信息, 错误 := 操作系统包.状态信息(当前受保护数据.数据库 + 后缀)
		if 后缀 != "" && 错误判断包.是否属于(错误, 操作系统包.不存在错误) {
			continue
		}
		if 错误 != nil || !信息.模式().是否普通文件() {
			return nil, 已拒绝错误
		}
		文件集 = append(文件集, 信息)
	}
	return 文件集, nil
}

// 检查文件 拒绝使用其他路径访问受保护的 SQLite 文件身份。
func (当前受保护数据 受保护数据) 检查文件(信息 操作系统包.文件信息) error {
	if !信息.模式().是否普通文件() {
		return nil
	}
	文件集, 错误 := 当前受保护数据.SQLite文件集()
	if 错误 != nil {
		return 错误
	}
	for _, 文件 := range 文件集 {
		if 操作系统包.相同文件(信息, 文件) {
			return 已拒绝错误
		}
	}
	return nil
}
```
