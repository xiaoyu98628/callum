```go
// 包 参数 提供内置工具共用的 JSON 参数结构校验，不处理业务默认值或权限。
package 参数

import (
	字节包 "bytes"
	JSON包 "encoding/json"
	错误判断包 "errors"
	输入输出包 "io"
	反射包 "reflect"
	字符串转换包 "strconv"
	字符串包 "strings"
	UTF8包 "unicode/utf8"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

const 最大深度 = 8

// 解码 将单个参数对象解码到结构体，返回顶层字段的存在标记。
// 顶层及嵌套对象均按显式结构体 JSON 标签精确匹配字段名，类型由目标结构体校验。
// 所有层级均拒绝重复字段、空 和过深嵌套，错误时不得使用 目标 中的部分值。
func 解码(原始 JSON包.原始JSON消息, 目标 any) (map[string]bool, error) {
	if len(原始) > 工具包.最大参数字节 {
		return nil, 工具包.超出上限错误
	}
	if !UTF8包.有效(原始) {
		return nil, 工具包.无效错误
	}
	目标 := 反射包.获取类型(目标)
	if 目标 == nil || 目标.类别() != 反射包.指针 || 反射包.获取值(目标).是否空值() {
		return nil, 工具包.无效错误
	}
	解码器 := JSON包.新建解码器(字节包.新建读取器(原始))
	解码器.使用数量()
	首个, 错误 := 解码器.词元()
	if 错误 != nil || 首个 != JSON包.分隔符('{') {
		return nil, 工具包.无效错误
	}
	已见 := map[string]bool{}
	for 解码器.更多() {
		词元, 错误 := 解码器.词元()
		键, 成功 := 词元.(string)
		if 错误 != nil || !成功 || 已见[键] {
			return nil, 工具包.无效错误
		}
		已见[键] = true
		字段 := 字段类型(目标.元素(), 键)
		if 字段 == nil {
			return nil, 工具包.无效错误
		}
		if !值(解码器, 字段, 1) {
			return nil, 字段错误(键, 字段)
		}
	}
	最后, 错误 := 解码器.词元()
	if 错误 != nil || 最后 != JSON包.分隔符('}') {
		return nil, 工具包.无效错误
	}
	var 尾部 any
	if !错误判断包.是否属于(解码器.解码(&尾部), 输入输出包.文件结束) {
		return nil, 工具包.无效错误
	}
	解码器 = JSON包.新建解码器(字节包.新建读取器(原始))
	解码器.禁止未知字段集()
	if 错误 := 解码器.解码(目标); 错误 != nil {
		if 带类型, 成功 := 错误判断包.按类型匹配[*JSON包.反序列化类型错误](错误); 成功 {
			字段 := 目标.元素()
			for 键 := range 字符串包.拆分序号(带类型.字段, ".") {
				for 字段 != nil && 字段.类别() == 反射包.指针 {
					字段 = 字段.元素()
				}
				if 字段 == nil {
					break
				}
				if 字段.类别() == 反射包.切片 || 字段.类别() == 反射包.数组 {
					// Go 1.27 的错误路径包含数组下标，只按已声明的元素类型继续定位。
					索引, 错误 := 字符串转换包.字符串转整数(键)
					if 错误 != nil || 索引 < 0 {
						字段 = nil
						break
					}
					字段 = 字段.元素()
					continue
				}
				字段 = 字段类型(字段, 键)
			}
			if 字段 != nil {
				return nil, 字段错误(带类型.字段, 字段)
			}
		}
		return nil, 工具包.无效错误
	}
	return 已见, nil
}

func 值(解码器 *JSON包.解码器, 目标 反射包.类型, 深度 int) bool {
	if 深度 > 最大深度 {
		return false
	}
	词元, 错误 := 解码器.词元()
	if 错误 != nil || 词元 == nil {
		return false
	}
	分隔符, 成功 := 词元.(JSON包.分隔符)
	if !成功 {
		return true
	}
	for 目标.类别() == 反射包.指针 {
		目标 = 目标.元素()
	}
	switch 分隔符 {
	case '{':
		已见 := map[string]bool{}
		for 解码器.更多() {
			词元, 错误 := 解码器.词元()
			键, 成功 := 词元.(string)
			if 错误 != nil || !成功 || 已见[键] {
				return false
			}
			已见[键] = true
			字段 := 字段类型(目标, 键)
			if 字段 == nil || !值(解码器, 字段, 深度+1) {
				return false
			}
		}
	case '[':
		if 目标.类别() != 反射包.切片 && 目标.类别() != 反射包.数组 {
			return false
		}
		for 解码器.更多() {
			if !值(解码器, 目标.元素(), 深度+1) {
				return false
			}
		}
	default:
		return false
	}
	最后, 错误 := 解码器.词元()
	return 错误 == nil && ((分隔符 == '{' && 最后 == JSON包.分隔符('}')) || (分隔符 == '[' && 最后 == JSON包.分隔符(']')))
}

// 工具参数显式声明结构体字段，不依赖 编码/JSON 的字段名大小写兼容匹配。
func 字段类型(目标 反射包.类型, 键 string) 反射包.类型 {
	for 目标.类别() == 反射包.指针 {
		目标 = 目标.元素()
	}
	if 目标.类别() != 反射包.结构体 {
		return nil
	}
	for 字段 := range 目标.字段集() {
		if !字段.是导出() {
			continue
		}
		名称, _, _ := 字符串包.截断(字段.标签.获取("json"), ",")
		if 名称 == "" {
			名称 = 字段.名称
		}
		if 名称 != "-" && 名称 == 键 {
			return 字段.类型
		}
	}
	return nil
}

func 字段错误(键 string, 目标 反射包.类型) error {
	for 目标.类别() == 反射包.指针 {
		目标 = 目标.元素()
	}
	预期 := "符合工具定义的值"
	switch 目标.类别() {
	case 反射包.字符串:
		预期 = "字符串"
	case 反射包.布尔:
		预期 = "布尔值"
	case 反射包.整数, 反射包.整数64:
		预期 = "整数"
	case 反射包.结构体:
		预期 = "JSON 对象"
	case 反射包.切片, 反射包.数组:
		预期 = "数组"
		if 目标.元素().类别() == 反射包.字符串 {
			预期 = "字符串数组"
		}
	}
	消息 := "必须是" + 预期 + "，且不能为 null；请按工具定义重新提供"
	if 目标.类别() == 反射包.切片 || 目标.类别() == 反射包.数组 {
		消息 += "，不要把数组编码成字符串"
	}
	return &工具包.参数错误{字段: 键, 消息: 消息}
}
```
