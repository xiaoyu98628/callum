func Test工作区工具集并且权限边界(测试器 *测试包.测试器) {
	目录 := 测试器.创建临时目录()
	if 错误 := 操作系统包.写入文件(文件路径包.合并(目录, "a.txt"), []byte("first\nneedle here\nlast"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	外部 := 测试器.创建临时目录()
	if 错误 := 操作系统包.写入文件(文件路径包.合并(外部, "secret"), []byte("outside"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	当前读取器 := 注册表包.新建()
	for _, 实例 := range 文件系统包.工具集(工具包.受保护数据{}) {
		if 错误 := 当前读取器.注册(实例); 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	当前条目, 错误 := 执行器包.新建(当前读取器, 时间包.秒)
	if 错误 != nil {
		测试器.报告致命错误(错误)

	for _, 测试 := range []struct{ 名称, 启动参数 string }{{"list_directory", `{"path":"."}`}, {"read_file", `{"path":"a.txt","start_line":2,"end_line":2}`}, {"search_text", `{"path":".","query":"needle"}`}} {
		已准备, 错误 := 当前条目.准备(测试器.上下文(), 范围, 测试.名称, JSON包.原始JSON消息(测试.启动参数))
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		输出, 调用错误 := 当前条目.执行(测试器.上下文(), 已准备, 已准备.获取哈希())
		关闭错误 := 已准备.关闭()
		if 调用错误 != nil || 关闭错误 != nil || !JSON包.有效(输出.输出) || 输出.操作结果 != 工具包.操作结果无 {
			测试器.格式化报告致命错误("%s: %s %v %v", 测试.名称, 输出.输出, 调用错误, 关闭错误)
		}
	}
	for _, 参数 := range []string{`{"path":"../secret"}`, `{"path":"a.txt","unknown":true}`, `{"path":"a.txt","start_line":-1}`} {
		if 当前对象, 错误 := 当前条目.准备(测试器.上下文(), 范围, "read_file", JSON包.原始JSON消息(参数)); 错误 == nil {

			测试器.格式化报告致命错误("接受越界或无效参数: %s", 参数)
		}
	}
	if 错误 := 操作系统包.符号链接(文件路径包.合并(外部, "secret"), 文件路径包.合并(目录, "link")); 错误 != nil {
		测试器.格式化跳过("当前系统无法创建符号链接: %v", 错误)
	}
	当前对象, 错误 := 当前条目.准备(测试器.上下文(), 范围, "read_file", JSON包.原始JSON消息(`{"path":"link"}`))

		测试器.报告致命错误(错误)

		测试器.报告致命错误("符号链接没有保持实际目标分类或纯读取被错误要求审批")
	}

func Test读取版本能够授权补丁(测试器 *测试包.测试器) {
	目录 := 测试器.创建临时目录()
	路径 := 文件路径包.合并(目录, "note")
	if 错误 := 操作系统包.写入文件(路径, []byte("head\r\nold\r\ntail"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	当前读取器 := 注册表包.新建()
	for _, 实例 := range 文件系统包.工具集(工具包.受保护数据{}) {
		if 错误 := 当前读取器.注册(实例); 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	当前条目, 错误 := 执行器包.新建(当前读取器, 时间包.秒)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	范围 := 工具包.范围{标识: "w", 根路径: 目录}
	读取, 错误 := 当前条目.准备(测试器.上下文(), 范围, "read_file", JSON包.原始JSON消息(`{"path":"note","start_line":2,"end_line":2}`))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	结果, 调用错误 := 当前条目.执行(测试器.上下文(), 读取, 读取.获取哈希())
	if 错误 := 错误判断包.合并(调用错误, 读取.关闭()); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	var 版本 struct{ SHA256, 文本 string }
	if 错误 := JSON包.反序列化(结果.输出, &版本); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 版本.文本 != "old\r" || len(版本.SHA256) != 64 {
		测试器.格式化报告致命错误("读取范围或整文件版本错误: %s", 结果.输出)
	}
	参数, 错误 := JSON包.序列化(map[string]any{
		"patch":         "*** Begin Patch\n*** Update File: note\n@@\n head\n-old\n+new\n tail\n*** End Patch\n",
		"base_versions": []map[string]string{{"path": "note", "sha256": 版本.SHA256}},
	})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	补丁, 错误 := 当前条目.准备(测试器.上下文(), 范围, "apply_patch", 参数)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 补丁.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	结果, 错误 = 当前条目.执行(测试器.上下文(), 补丁, 补丁.获取哈希())
	if 错误 != nil || 结果.操作结果 != 工具包.操作结果已生效 {
		测试器.格式化报告致命错误("读取版本无法用于补丁: %s, %v", 结果.输出, 错误)
	}
	内容, 错误 := 操作系统包.读取文件(路径)
	if 错误 != nil || string(内容) != "head\r\nnew\r\ntail" {
		测试器.格式化报告致命错误("修改结果错误: %q, %v", 内容, 错误)
	}
}

func Test已准备目标无法被已重定向(测试器 *测试包.测试器) {
	for _, 类别 := range []string{"file", "directory", "symlink", "workspace"} {
		测试器.执行(类别, func(测试器 *测试包.测试器) {
			根 := 测试器.创建临时目录()
			路径 := 文件路径包.合并(根, "target")
			if 错误 := 操作系统包.写入文件(路径, []byte("original"), 0600); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			范围 := 工具包.范围{}
			操作 := "read_file"
			if 类别 == "directory" {
				路径 = 根
				操作 = "list_directory"
			}
			if 类别 == "workspace" {
				范围 = 工具包.范围{标识: "workspace", 根路径: 根}
			}
			if 类别 == "symlink" {
				链接 := 文件路径包.合并(根, "link")
				if 错误 := 操作系统包.符号链接(路径, 链接); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				路径 = 链接
			}
			当前读取器 := 注册表包.新建()
			for _, 实例 := range 文件系统包.工具集(工具包.受保护数据{}) {
				if 错误 := 当前读取器.注册(实例); 错误 != nil {
					测试器.报告致命错误(错误)
				}
			}
			当前条目, 错误 := 执行器包.新建(当前读取器, 时间包.秒)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			参数, 错误 := JSON包.序列化(map[string]string{"path": 路径})
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			当前对象, 错误 := 当前条目.准备(测试器.上下文(), 范围, 操作, 参数)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			defer func() {
				if 错误 := 当前对象.关闭(); 错误 != nil {
					测试器.错误文本(错误)
				}
			}()
			switch 类别 {
			case "directory", "workspace":
				if 错误 := 操作系统包.重命名(根, 根+"-original"); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				测试器.注册清理函数(func() { 操作系统包.移除全部(根 + "-original") })
				if 错误 := 操作系统包.创建目录(根, 0700); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				if 错误 := 操作系统包.写入文件(文件路径包.合并(根, "target"), []byte("replacement"), 0600); 错误 != nil {
					测试器.报告致命错误(错误)
				}
			case "file":
				if 错误 := 操作系统包.重命名(路径, 路径+"-original"); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				if 错误 := 操作系统包.写入文件(路径, []byte("replacement"), 0600); 错误 != nil {
					测试器.报告致命错误(错误)
				}
			case "symlink":
				其他 := 文件路径包.合并(根, "other")
				if 错误 := 操作系统包.写入文件(其他, []byte("replacement"), 0600); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				if 错误 := 操作系统包.移除(路径); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				if 错误 := 操作系统包.符号链接(其他, 路径); 错误 != nil {
					测试器.报告致命错误(错误)
				}
			}
			if _, 错误 := 当前条目.执行(测试器.上下文(), 当前对象, 当前对象.获取哈希()); !错误判断包.是否属于(错误, 工具包.已拒绝错误) {
				测试器.格式化报告致命错误("目标变化后仍继续执行: %v", 错误)
			}
			if 错误 := 当前对象.关闭(); 错误 != nil {
				测试器.报告致命错误(错误)
			}
		})
	}
}
