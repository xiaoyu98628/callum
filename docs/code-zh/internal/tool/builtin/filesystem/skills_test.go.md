```go
package 文件系统

import (
	JSON包 "encoding/json"
	错误判断包 "errors"
	操作系统包 "os"
	文件路径包 "path/filepath"
	字符串包 "strings"
	测试包 "testing"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

func Test技能读取例外不暴露应用数据(测试器 *测试包.测试器) {
	数据 := 测试器.创建临时目录()
	数据, 错误 := 文件路径包.求值符号链接集(数据)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	技能 := 文件路径包.合并(数据, "skills")
	if 错误 := 操作系统包.创建目录(技能, 0700); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	数据库 := 文件路径包.合并(数据, "callum.db")
	if 错误 := 操作系统包.写入文件(数据库, []byte("private database"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	条目 := 文件路径包.合并(技能, "SKILL.md")
	if 错误 := 操作系统包.写入文件(条目, []byte("skill-body"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	保护 := 工具包.受保护数据{目录: 数据, 数据库: 数据库, 只读目录集: []string{技能}}
	var 读取器 工具包.工具
	for _, 实例 := range 工具集(保护) {
		if 实例.获取定义().名称 == "read_file" {
			读取器 = 实例
		}
	}
	读取 := func(路径 string) (工具包.结果, error) {
		原始, 错误 := JSON包.序列化(map[string]any{"path": 路径})
		if 错误 != nil {
			return 工具包.结果{}, 错误
		}
		调用, 错误 := 读取器.准备(测试器.上下文(), 工具包.范围{}, 原始)
		if 错误 != nil {
			return 工具包.结果{}, 错误
		}
		结果, 执行错误 := 调用.执行(测试器.上下文())
		return 结果, 错误判断包.合并(执行错误, 调用.关闭())
	}
	结果, 错误 := 读取(条目)
	if 错误 != nil || !字符串包.包含(string(结果.输出), "skill-body") {
		测试器.格式化报告致命错误("技能不可读: %s %v", 结果.输出, 错误)
	}
	if _, 错误 := 读取(数据库); !错误判断包.是否属于(错误, 工具包.已拒绝错误) {
		测试器.格式化报告致命错误("数据库可直接读取: %v", 错误)
	}
	测试器.执行("symlink", func(测试器 *测试包.测试器) {
		路径 := 文件路径包.合并(技能, "escape.md")
		if 错误 := 操作系统包.符号链接(数据库, 路径); 错误 != nil {
			测试器.格式化跳过("平台无法创建符号链接: %v", 错误)
		}
		if _, 错误 := 读取(路径); !错误判断包.是否属于(错误, 工具包.已拒绝错误) {
			测试器.格式化报告致命错误("链接绕过私有数据边界: %v", 错误)
		}
	})
	测试器.执行("hardlink", func(测试器 *测试包.测试器) {
		路径 := 文件路径包.合并(技能, "alias.md")
		if 错误 := 操作系统包.链接(数据库, 路径); 错误 != nil {
			测试器.格式化跳过("平台无法创建硬链接: %v", 错误)
		}
		if _, 错误 := 读取(路径); !错误判断包.是否属于(错误, 工具包.已拒绝错误) {
			测试器.格式化报告致命错误("数据库身份校验失效: %v", 错误)
		}
	})
	if !保护.写入受保护(条目) {
		测试器.报告致命错误("技能目录意外开放写入")
	}
}
```
