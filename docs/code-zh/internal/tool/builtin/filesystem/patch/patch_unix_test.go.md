func Test补丁拒绝特殊文件集并且硬链接别名集(测试器 *测试包.测试器) {
	目录 := 测试器.创建临时目录()
	路径 := 文件路径包.合并(目录, "pipe")
	if 错误 := 系统调用包.创建命名管道(路径, 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	当前条目 := 补丁测试执行器(测试器)
	范围 := 工具包.范围{标识: "w", 根路径: 目录}
	if _, 错误 := 当前条目.准备(测试器.上下文(), 范围, "apply_patch", 补丁测试参数(测试器, "*** Update File: pipe\n@@\n-a\n+b", 补丁版本{"pipe", 文本SHA256([]byte("a\n"))})); 错误 == nil {
		测试器.报告致命错误("打开了特殊文件")
	}
	文件 := 文件路径包.合并(目录, "file")
	补丁测试写入(测试器, 文件, "a\n")
	if 错误 := 操作系统包.链接(文件, 文件路径包.合并(测试器.创建临时目录(), "alias")); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 当前条目.准备(测试器.上下文(), 范围, "apply_patch", 补丁测试参数(测试器, "*** Update File: file\n@@\n-a\n+b", 补丁版本{"file", 文本SHA256([]byte("a\n"))})); !错误判断包.是否属于(错误, 工具包.已拒绝错误) {
		测试器.格式化报告致命错误("硬链接可能修改未批准路径: %v", 错误)
	}
}

func Test补丁保留Unix权限并且ACL(测试器 *测试包.测试器) {
	目录 := 测试器.创建临时目录()
	路径 := 文件路径包.合并(目录, "note")
	之前 := "header\nold\ntail\n"
	补丁测试写入(测试器, 路径, 之前)
	if 错误 := 操作系统包.修改权限(路径, 0750); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	var ACL之前 string
	if 运行时包.目标操作系统 == "darwin" {
		// 仅对临时文件设置 ACL。禁止执行这一项不影响读写和测试目录清理。
		if 输出, 错误 := 进程执行包.命令上下文(测试器.上下文(), "/bin/chmod", "+a", "everyone deny execute", 路径).已合并输出(); 错误 != nil {
			测试器.格式化报告致命错误("设置临时 ACL: %s %v", 输出, 错误)
		}
		输出, 错误 := 进程执行包.命令上下文(测试器.上下文(), "/bin/ls", "-le", 路径).输出()
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		ACL之前 = 字符串包.合并(字符串包.拆分(string(输出), "\n")[1:], "\n")
	}
	当前条目 := 补丁测试执行器(测试器)
	当前对象 := 补丁测试准备(测试器, 当前条目, 工具包.范围{标识: "w", 根路径: 目录}, 补丁测试参数(测试器, "*** Update File: note\n@@\n header\n-old\n+new\n tail", 补丁版本{"note", 文本SHA256([]byte(之前))}))
	if 结果, 错误 := 当前条目.执行(测试器.上下文(), 当前对象, 当前对象.获取哈希()); 错误 != nil || 结果.操作结果 != 工具包.操作结果已生效 {
		测试器.格式化报告致命错误("修改失败: %s %v", 结果.输出, 错误)
	}
	信息, 错误 := 操作系统包.状态信息(路径)
	if 错误 != nil || 信息.模式().权限() != 0750 {
		测试器.格式化报告致命错误("权限改变: %v %v", 信息, 错误)
	}
	if 运行时包.目标操作系统 == "darwin" {
		输出, 错误 := 进程执行包.命令上下文(测试器.上下文(), "/bin/ls", "-le", 路径).输出()
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		ACL之后 := 字符串包.合并(字符串包.拆分(string(输出), "\n")[1:], "\n")
		if ACL之前 != ACL之后 {
			测试器.格式化报告致命错误("改变了原 ACL: %q -> %q", ACL之前, ACL之后)
		}
	}
}

type 拒绝补丁清理 struct {
	上下文包.上下文
	路径, 目录 string
	已生效    bool
}

func (当前拒绝补丁清理 *拒绝补丁清理) 获取错误() error {
	if !当前拒绝补丁清理.已生效 {
		数据, _ := 操作系统包.读取文件(当前拒绝补丁清理.路径)
		if string(数据) == "head\nnew\ntail\n" {
			当前拒绝补丁清理.已生效 = true
			_ = 操作系统包.修改权限(当前拒绝补丁清理.目录, 0500)
		}
	}
	return 当前拒绝补丁清理.上下文.获取错误()
}
func Test补丁清理错误保留已生效输出并且副本集(测试器 *测试包.测试器) {
	if 操作系统包.获取有效用户标识() == 0 {
		测试器.跳过("特权用户不受目录权限拒绝约束")
	}
	目录 := 测试器.创建临时目录()
	测试器.注册清理函数(func() {
		if 错误 := 操作系统包.修改权限(目录, 0700); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	路径 := 文件路径包.合并(目录, "note")
	之前 := "head\nold\ntail\n"
	补丁测试写入(测试器, 路径, 之前)
	实例 := 新建(工具包.受保护数据{})
	参数, 错误 := 实例.校验(补丁测试参数(测试器, "*** Update File: note\n@@\n head\n-old\n+new\n tail", 补丁版本{"note", 文本SHA256([]byte(之前))}))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	工具调用, 错误 := 实例.准备(测试器.上下文(), 工具包.范围{标识: "w", 根路径: 目录}, 参数)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 工具调用.关闭()
	执行上下文 := &拒绝补丁清理{上下文: 测试器.上下文(), 路径: 路径, 目录: 目录}
	结果, 错误 := 工具调用.执行(执行上下文)
	if 错误 == nil || 结果.操作结果 != 工具包.操作结果已生效 {
		测试器.格式化报告致命错误("清理错误掩盖已生效结果: %s %v", 结果.输出, 错误)
	}
	var 报告 补丁报告
	if JSON包.反序列化(结果.输出, &报告) != nil || len(报告.恢复文件集) != 2 || 报告.变更集[0].已观测SHA256 != 文本SHA256([]byte("head\nnew\ntail\n")) {
		测试器.格式化报告致命错误("清理失败丢失审计: %s", 结果.输出)
	}
	if 补丁测试读取(测试器, 路径) != "head\nnew\ntail\n" {
		测试器.报告致命错误("实际修改被回滚")
	}
}

func Test补丁私有副本集不继承macOS读取ACL(测试器 *测试包.测试器) {
	if 运行时包.目标操作系统 != "darwin" {
		测试器.跳过("Darwin ACL 继承边界")
	}
	目录 := 测试器.创建临时目录()
	if 输出, 错误 := 进程执行包.命令上下文(测试器.上下文(), "/bin/chmod", "+a", "everyone allow read,file_inherit,directory_inherit", 目录).已合并输出(); 错误 != nil {
		测试器.格式化报告致命错误("设置临时继承 ACL: %s %v", 输出, 错误)
	}
	路径 := 文件路径包.合并(目录, "private-copy")
	文件, 错误 := 操作系统包.打开文件(路径, 操作系统包.创建文件标志|操作系统包.独占创建标志|操作系统包.读写打开标志, 0600)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	安全错误 := 保护补丁文件(文件)
	关闭错误 := 文件.关闭()
	if 安全错误 != nil || 关闭错误 != nil {
		测试器.格式化报告致命错误("收敛新文件权限: %v %v", 安全错误, 关闭错误)
	}
	输出, 错误 := 进程执行包.命令上下文(测试器.上下文(), "/bin/ls", "-le", 路径).输出()
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if len(字符串包.拆分(字符串包.裁剪空间(string(输出)), "\n")) != 1 {
		测试器.格式化报告致命错误("私有副本仍有继承 ACL: %s", 输出)
	}
	新建目录 := 文件路径包.合并(目录, "child")
	if 错误 := 操作系统包.创建目录(新建目录, 0700); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	目录, 错误 := 操作系统包.打开(新建目录)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	安全错误 = 保护补丁目录(目录)
	关闭错误 = 目录.关闭()
	if 安全错误 != nil || 关闭错误 != nil {
		测试器.格式化报告致命错误("收敛新目录权限: %v %v", 安全错误, 关闭错误)
	}
	输出, 错误 = 进程执行包.命令上下文(测试器.上下文(), "/bin/ls", "-led", 新建目录).输出()
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if len(字符串包.拆分(字符串包.裁剪空间(string(输出)), "\n")) != 1 {
		测试器.格式化报告致命错误("私有目录仍有继承 ACL: %s", 输出)
	}
}
