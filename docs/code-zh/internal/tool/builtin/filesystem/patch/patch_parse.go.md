```go
package 补丁

import (
	字节包 "bytes"
	字符串包 "strings"
	UTF8包 "unicode/utf8"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

const 最大补丁文件集 = 8

// 补丁文件 只描述创建和修改，删除和移动标记不能退化成其他操作。
type 补丁文件 struct {
	路径, 动作, 内容 string
	补丁块集       []补丁补丁块
	已移除大写      int64
}
type 补丁补丁块 struct {
	锚点 string
	行集 []string
	结束 bool
}

func 规范化补丁(文件集 []补丁文件) string {
	var 输出 字符串包.构建器
	输出.写入字符串("*** Begin Patch\n")
	for _, 文件 := range 文件集 {
		if 文件.动作 == "create" {
			输出.写入字符串("*** Add File: " + 文件.路径 + "\n")
			if 文件.内容 != "" {
				for 行 := range 字符串包.拆分序号(字符串包.裁剪后缀(文件.内容, "\n"), "\n") {
					输出.写入字符串("+" + 行 + "\n")
				}
			}
		} else {
			输出.写入字符串("*** Update File: " + 文件.路径 + "\n")
			for _, 处理器 := range 文件.补丁块集 {
				输出.写入字符串("@@")
				if 处理器.锚点 != "" {
					输出.写入字符串(" " + 处理器.锚点)
				}
				输出.写入字节('\n')
				for _, 行 := range 处理器.行集 {
					输出.写入字符串(行)
					输出.写入字节('\n')
				}
				if 处理器.结束 {
					输出.写入字符串("*** End of File\n")
				}
			}
		}
	}
	输出.写入字符串("*** End Patch\n")
	return 输出.字符串()
}

func 解析补丁(值 string) ([]补丁文件, error) {
	if !UTF8包.有效字符串(值) || 字符串包.包含字符(值, 0) || len(值) > 工具包.最大参数字节 {
		return nil, 工具包.无效错误
	}
	行集 := 字符串包.拆分(字符串包.替换全部(值, "\r\n", "\n"), "\n")
	if 行集[len(行集)-1] == "" {
		行集 = 行集[:len(行集)-1]
	}
	if len(行集) < 3 || 行集[0] != "*** Begin Patch" || 行集[len(行集)-1] != "*** End Patch" {
		return nil, 工具包.无效错误
	}
	var 文件集 []补丁文件
	已见 := map[string]bool{}
	for 索引 := 1; 索引 < len(行集)-1; {
		行 := 行集[索引]
		文件 := 补丁文件{}
		switch {
		case 字符串包.存在前缀(行, "*** Add File: "):
			文件.路径, 文件.动作 = 字符串包.裁剪前缀(行, "*** Add File: "), "create"
		case 字符串包.存在前缀(行, "*** Update File: "):
			文件.路径, 文件.动作 = 字符串包.裁剪前缀(行, "*** Update File: "), "update"
		default:
			return nil, 工具包.无效错误
		}
		路径, 错误 := 补丁路径(文件.路径)
		if 错误 != nil || 已见[路径] {
			return nil, 工具包.无效错误
		}
		文件.路径 = 路径
		已见[路径] = true
		索引++
		if 文件.动作 == "create" {
			var 内容 字符串包.构建器
			for 索引 < len(行集)-1 && 字符串包.存在前缀(行集[索引], "+") {
				内容.写入字符串(行集[索引][1:])
				内容.写入字节('\n')
				索引++
			}
			文件.内容 = 内容.字符串() // 空文件允许没有正文行。
		} else {
			for 索引 < len(行集)-1 && (行集[索引] == "@@" || 字符串包.存在前缀(行集[索引], "@@ ")) {
				处理器 := 补丁补丁块{锚点: 字符串包.裁剪前缀(行集[索引], "@@")}
				处理器.锚点 = 字符串包.裁剪前缀(处理器.锚点, " ")
				索引++
				变化通知 := false
				for 索引 < len(行集)-1 && len(行集[索引]) > 0 && (行集[索引][0] == ' ' || 行集[索引][0] == '+' || 行集[索引][0] == '-') {
					值 := 行集[索引]
					处理器.行集 = append(处理器.行集, 值)
					变化通知 = 变化通知 || 值[0] != ' '
					if 值[0] == '-' {
						文件.已移除大写 += int64(len(值) - 1 + 2)
					}
					索引++
				}
				if !变化通知 {
					return nil, 工具包.无效错误
				}
				if 索引 < len(行集)-1 && 行集[索引] == "*** End of File" {
					处理器.结束 = true
					索引++
				}
				文件.补丁块集 = append(文件.补丁块集, 处理器)
				if 处理器.结束 {
					break
				}
			}
			if len(文件.补丁块集) == 0 {
				return nil, 工具包.无效错误
			}
		}
		文件集 = append(文件集, 文件)
		if len(文件集) > 最大补丁文件集 {
			return nil, 工具包.超出上限错误
		}
	}
	if len(文件集) == 0 {
		return nil, 工具包.无效错误
	}
	return 文件集, nil
}

type 文本行 struct {
	文本, 结束 string
	已插入    bool
}

func 拆分文本(数据 []byte) []文本行 {
	var 行集 []文本行
	for len(数据) > 0 {
		索引 := 字节包.索引字节(数据, '\n')
		if 索引 < 0 {
			行集 = append(行集, 文本行{文本: string(数据)})
			break
		}
		文本, 结束 := 数据[:索引], "\n"
		if len(文本) > 0 && 文本[len(文本)-1] == '\r' {
			文本 = 文本[:len(文本)-1]
			结束 = "\r\n"
		}
		行集 = append(行集, 文本行{文本: string(文本), 结束: 结束})
		数据 = 数据[索引+1:]
	}
	return 行集
}

// 应用文本 只接受唯一的精确匹配，不做空白、大小写或标点模糊匹配。
// 未修改的行保留原始换行；插入行沿用邻近换行，文件尾原来无换行时仍保持。
func 应用文本(数据 []byte, 文件 补丁文件) ([]byte, error) {
	原始 := 拆分文本(数据)
	结束 := "\n"
	for _, 行 := range 原始 {
		if 行.结束 != "" {
			结束 = 行.结束
			break
		}
	}
	var 输出 []文本行
	游标 := 0
	for _, 处理器 := range 文件.补丁块集 {
		启动 := 游标
		if 处理器.锚点 != "" {
			已找到 := -1
			for 索引 := 游标; 索引 < len(原始); 索引++ {
				if 原始[索引].文本 == 处理器.锚点 {
					if 已找到 >= 0 {
						return nil, 工具包.冲突错误
					}
					已找到 = 索引
				}
			}
			if 已找到 < 0 {
				return nil, 工具包.冲突错误
			}
			启动 = 已找到 + 1
		}
		var 旧 []string
		for _, 行 := range 处理器.行集 {
			if 行[0] != '+' {
				旧 = append(旧, 行[1:])
			}
		}
		匹配 := -1
		if len(旧) == 0 {
			switch {
			case 处理器.结束:
				匹配 = len(原始)
			case len(原始) == 0:
				匹配 = 0
			case 处理器.锚点 != "":
				匹配 = 启动
			default:
				return nil, 工具包.冲突错误
			}
		} else {
			for 索引 := 启动; 索引+len(旧) <= len(原始); 索引++ {
				if 处理器.结束 && 索引+len(旧) != len(原始) {
					continue
				}
				相同 := true
				for 次索引, 值 := range 旧 {
					if 原始[索引+次索引].文本 != 值 {
						相同 = false
						break
					}
				}
				if 相同 {
					if 匹配 >= 0 {
						return nil, 工具包.冲突错误
					}
					匹配 = 索引
				}
			}
		}
		if 匹配 < 游标 {
			return nil, 工具包.冲突错误
		}
		输出 = append(输出, 原始[游标:匹配]...)
		样式 := 结束
		if 匹配 < len(原始) && 原始[匹配].结束 != "" {
			样式 = 原始[匹配].结束
		}
		位置 := 匹配
		for _, 行 := range 处理器.行集 {
			switch 行[0] {
			case ' ':
				输出 = append(输出, 原始[位置])
				位置++
			case '-':
				位置++
			case '+':
				输出 = append(输出, 文本行{文本: 行[1:], 结束: 样式, 已插入: true})
			}
		}
		游标 = 位置
	}
	输出 = append(输出, 原始[游标:]...)
	if len(原始) > 0 && len(输出) > 0 && 原始[len(原始)-1].结束 == "" && 输出[len(输出)-1].已插入 {
		输出[len(输出)-1].结束 = ""
	}
	var 结果 字节包.缓冲区
	for 索引, 行 := range 输出 {
		结果.写入字符串(行.文本)
		if 行.结束 == "" && 索引 < len(输出)-1 {
			结果.写入字符串(结束)
		} else {
			结果.写入字符串(行.结束)
		}
		if 结果.长度() > 1<<20 {
			return nil, 工具包.超出上限错误
		}
	}
	if 字节包.相等(数据, 结果.字节()) {
		return nil, 工具包.无效错误
	}
	return 结果.字节(), nil
}
```
