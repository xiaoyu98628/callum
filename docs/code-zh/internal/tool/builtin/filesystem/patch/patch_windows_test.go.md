```go
package 补丁

import (
	操作系统包 "os"
	文件路径包 "path/filepath"
	测试包 "testing"

	Windows包 "golang.org/x/sys/windows"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

func Test补丁保留WindowsDACL并且保护新建文件集(测试器 *测试包.测试器) {
	目录 := 测试器.创建临时目录()
	路径 := 文件路径包.合并(目录, "note")
	之前 := "header\r\nold\r\ntail\r\n"
	补丁测试写入(测试器, 路径, 之前)
	文件, 错误 := 操作系统包.打开文件(路径, 操作系统包.读写打开标志, 0)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 保护补丁文件(文件); 错误 != nil {
		文件.关闭()
		测试器.报告致命错误(错误)
	}
	安全描述符, 错误 := Windows包.获取安全信息(Windows包.句柄(文件.文件描述符()), Windows包.步骤错误文件对象, Windows包.DACL安全信息)
	if 错误 != nil {
		文件.关闭()
		测试器.报告致命错误(错误)
	}
	原始 := 安全描述符.字符串()
	if 错误 := 文件.关闭(); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	当前条目 := 补丁测试执行器(测试器)
	当前对象 := 补丁测试准备(测试器, 当前条目, 工具包.范围{标识: "w", 根路径: 目录}, 补丁测试参数(测试器, "*** Update File: note\n@@\n header\n-old\n+new\n tail\n*** Add File: nested/new.txt\n+created", 补丁版本{"note", 文本SHA256([]byte(之前))}))
	if 结果, 错误 := 当前条目.执行(测试器.上下文(), 当前对象, 当前对象.获取哈希()); 错误 != nil || 结果.操作结果 != 工具包.操作结果已生效 {
		测试器.格式化报告致命错误("修改失败: %s %v", 结果.输出, 错误)
	}
	for _, 目标 := range []string{路径, 文件路径包.合并(目录, "nested", "new.txt")} {
		文件, 错误 := 操作系统包.打开(目标)
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		安全描述符, 安全错误 := Windows包.获取安全信息(Windows包.句柄(文件.文件描述符()), Windows包.步骤错误文件对象, Windows包.DACL安全信息)
		关闭错误 := 文件.关闭()
		if 安全错误 != nil || 关闭错误 != nil {
			测试器.格式化报告致命错误("读取权限失败: %v %v", 安全错误, 关闭错误)
		}
		if 安全描述符.字符串() != 原始 {
			测试器.格式化报告致命错误("DACL 未保留或新文件权限不私有: %s != %s", 安全描述符.字符串(), 原始)
		}
	}
}
```
