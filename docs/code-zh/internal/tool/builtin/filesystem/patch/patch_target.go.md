package 补丁

import (
	上下文包 "context"
	SHA256包 "crypto/sha256"
	十六进制包 "encoding/hex"
	JSON包 "encoding/json"
	错误判断包 "errors"
	操作系统包 "os"
	文件路径包 "path/filepath"
	切片包 "slices"
	字符串包 "strings"
	同步包 "sync"
	原子操作包 "sync/atomic"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
	文件访问包 "github.com/xiaoyu98628/callum/internal/tool/builtin/filesystem/internal/fileaccess"
)

type 补丁目标 struct {
	补丁       补丁文件
	目标       工具包.目标
	固定       *文件访问包.句柄
	基准路径     string
	基准信息     操作系统包.文件信息
	根        *操作系统包.根
	父级       *操作系统包.根
	父级相对, 名称 string
	缺失       []string
}

type 补丁调用 struct {
	所属对象  *补丁工具
	范围    工具包.范围
	工作区   string
	工作区信息 操作系统包.文件信息
	文件集   []*补丁目标
	计划    工具包.计划
	已开始   原子操作包.布尔
	关闭一次  同步包.一次
	关闭错误  error
	操作种子  string
}

func (*补丁调用) 获取目标() *工具包.目标 { return nil }
func (当前补丁调用 *补丁调用) 获取计划() 工具包.计划 {
	return 工具包.计划{目标集: 切片包.复制(当前补丁调用.计划.目标集), 附加影响集: 切片包.复制(当前补丁调用.计划.附加影响集)}
}
func (当前补丁调用 *补丁调用) 准备模块(执行上下文 上下文包.上下文, 参数 补丁参数, 文件集 []补丁文件) error {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	规范化, 错误 := JSON包.序列化(参数)
	if 错误 != nil {
		return 错误
	}
	当前补丁调用.操作种子 = string(规范化)
	if 当前补丁调用.范围 != (工具包.范围{}) {
		if 当前补丁调用.范围.标识 == "" || !文件路径包.是否绝对路径(当前补丁调用.范围.根路径) {
			return 工具包.已拒绝错误
		}
		var 错误 error
		当前补丁调用.工作区, 错误 = 文件路径包.求值符号链接集(当前补丁调用.范围.根路径)
		if 错误 != nil {
			return 工具包.不可用错误
		}
		当前补丁调用.工作区信息, 错误 = 操作系统包.状态信息(当前补丁调用.工作区)
		if 错误 != nil || !当前补丁调用.工作区信息.是否目录() {
			return 工具包.不可用错误
		}
	}
	版本集 := map[string]string{}
	for _, 值 := range 参数.基准版本集 {
		版本集[值.路径] = 值.SHA256
	}
	目录集 := map[string]bool{}
	for _, 文件 := range 文件集 {
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			return 错误
		}
		条目 := &补丁目标{补丁: 文件}
		当前补丁调用.文件集 = append(当前补丁调用.文件集, 条目) // 此后任何失败由调用对象统一关闭已经固定的资源。
		if 错误 := 当前补丁调用.准备目标(执行上下文, 条目, 版本集[文件.路径]); 错误 != nil {
			return 错误
		}
		for _, 其他 := range 当前补丁调用.文件集[:len(当前补丁调用.文件集)-1] {
			if 字符串包.相等折叠(其他.目标.路径, 条目.目标.路径) || 相同补丁文件(其他, 条目) {
				return 工具包.冲突错误
			}
		}
		当前补丁调用.计划.目标集 = append(当前补丁调用.计划.目标集, 条目.目标)
		for _, 目录 := range 条目.缺失 {
			if 目录集[目录] {
				continue
			}
			目录集[目录] = true
			if len(目录集) > 16 {
				return 工具包.超出上限错误
			}
			当前补丁调用.计划.目标集 = append(当前补丁调用.计划.目标集, 工具包.目标{请求的路径: 目录, 路径: 目录, 类别: "directory", 动作: "create", 范围内工作区: 条目.目标.范围内工作区})
		}
		// 不读取正文即可做保守分类：可能覆盖全部内容，或删除上限达到
		// 原文件一半且至少 4 KiB 时要求审批。真正内容和版本在批准后校验。
		if 条目.固定 != nil && 文件.已移除大写 > 0 && (文件.已移除大写 >= 条目.固定.获取信息().大小() || (文件.已移除大写 >= 4096 && 文件.已移除大写*2 >= 条目.固定.获取信息().大小())) {
			if !切片包.包含(当前补丁调用.计划.附加影响集, 工具包.影响破坏性) {
				当前补丁调用.计划.附加影响集 = append(当前补丁调用.计划.附加影响集, 工具包.影响破坏性)
			}
		}
		for _, 路径 := range []string{条目.目标.路径, 条目.目标.请求的路径} {
			for 部分 := range 字符串包.拆分序号(文件路径包.目标斜杠(路径), "/") {
				部分 = 字符串包.目标小写(部分)
				if 部分 == ".git" || 部分 == ".codex" || 部分 == ".agents" {
					if !切片包.包含(当前补丁调用.计划.附加影响集, 工具包.影响未知) {
						当前补丁调用.计划.附加影响集 = append(当前补丁调用.计划.附加影响集, 工具包.影响未知)
					}
				}
			}
		}
	}
	已编码, 错误 := JSON包.序列化(当前补丁调用.计划)
	if 错误 != nil {
		return 错误
	}
	if len(已编码) > 24<<10 {
		return 工具包.超出上限错误
	}
	return 当前补丁调用.检查工作区(执行上下文)
}

func (当前补丁调用 *补丁调用) 准备目标(执行上下文 上下文包.上下文, 条目 *补丁目标, 哈希 string) error {
	请求的 := 条目.补丁.路径
	if !文件路径包.是否绝对路径(请求的) {
		if 当前补丁调用.范围.标识 == "" {
			return 工具包.路径基准错误
		}
		请求的 = 文件路径包.合并(当前补丁调用.范围.根路径, 请求的)
	}
	if 条目.补丁.动作 == "update" {
		固定, 错误 := 文件访问包.准备(执行上下文, 当前补丁调用.范围, 请求的, 文件访问包.文件, 当前补丁调用.所属对象.保护)
		if 错误 != nil {
			return 错误
		}
		条目.固定 = 固定
		条目.目标 = *条目.固定.获取目标()
		条目.基准路径 = 文件路径包.目录(条目.目标.路径)
		条目.基准信息, 错误 = 操作系统包.状态信息(条目.基准路径)
		if 错误 != nil {
			return 工具包.不可用错误
		}
		if 条目.固定.获取信息().大小() > 1<<20 {
			return 工具包.超出上限错误
		}
		if 错误 := 补丁文件就绪(条目.固定.获取文件()); 错误 != nil {
			return 错误
		}
	} else {
		实际, 基准, 信息, 缺失, 错误 := 创建目标(请求的)
		if 错误 != nil {
			return 错误
		}
		条目.目标 = 工具包.目标{请求的路径: 请求的, 路径: 实际, 类别: "file"}
		条目.基准路径, 条目.基准信息, 条目.缺失 = 基准, 信息, 缺失
	}
	if 当前补丁调用.所属对象.保护.写入受保护(条目.目标.路径) {
		return 工具包.已拒绝错误
	}
	if len(条目.目标.路径) > 1024 {
		return 工具包.超出上限错误
	}
	条目.目标.范围内工作区 = false
	if 当前补丁调用.范围.标识 != "" {
		相对路径, 错误 := 文件路径包.相对路径(当前补丁调用.工作区, 条目.目标.路径)
		条目.目标.范围内工作区 = 错误 == nil && 文件路径包.是否本地路径(相对路径)
	}
	条目.目标.动作, 条目.目标.预期SHA256 = 条目.补丁.动作, 哈希
	// 辅助路径属于固定操作，必须可重复计算。随机名字会改变 请求哈希，
	// 使同一未知操作的重试绕过重新确认；已有辅助文件用 独占创建标志 拒绝覆盖。
	校验和 := SHA256包.校验和256([]byte(当前补丁调用.操作种子 + "\x00" + 当前补丁调用.范围.标识 + "\x00" + 条目.目标.路径))
	标识 := 十六进制包.编码目标字符串(校验和[:16])
	父级路径 := 文件路径包.目录(条目.目标.路径)
	条目.目标.恢复路径 = 文件路径包.合并(父级路径, ".callum-patch-"+标识+"-before")
	条目.目标.暂存路径 = 文件路径包.合并(父级路径, ".callum-patch-"+标识+"-after")
	条目.名称 = 文件路径包.基准(条目.目标.路径)
	var 错误 error
	条目.父级相对, 错误 = 文件路径包.相对路径(条目.基准路径, 父级路径)
	if 错误 != nil || !文件路径包.是否本地路径(条目.父级相对) {
		return 工具包.已拒绝错误
	}
	条目.根, 错误 = 当前补丁调用.打开基准(条目)
	if 错误 != nil {
		return 错误
	}
	return 条目.检查(执行上下文, 当前补丁调用)
}

// 创建目标 固定尚不存在目标的最近现存祖先。读取链接自身状态 将悬空链接也视为
// 已有目标，不能把它当作空闲路径；返回的缺失目录只允许在执行阶段创建。
func 创建目标(请求的 string) (string, string, 操作系统包.文件信息, []string, error) {
	if _, 错误 := 操作系统包.读取链接自身状态(请求的); 错误 == nil {
		return "", "", nil, nil, 工具包.冲突错误
	} else if !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		return "", "", nil, nil, 工具包.不可用错误
	}
	父级 := 文件路径包.目录(请求的)
	for 深度 := 0; ; 深度++ {
		if 深度 > 16 {
			return "", "", nil, nil, 工具包.超出上限错误
		}
		信息, 错误 := 操作系统包.状态信息(父级)
		if 错误 == nil {
			if !信息.是否目录() {
				return "", "", nil, nil, 工具包.冲突错误
			}
			已解析, 错误 := 文件路径包.求值符号链接集(父级)
			if 错误 != nil {
				return "", "", nil, nil, 工具包.不可用错误
			}
			信息, 错误 = 操作系统包.状态信息(已解析)
			if 错误 != nil {
				return "", "", nil, nil, 工具包.不可用错误
			}
			相对路径, 错误 := 文件路径包.相对路径(父级, 请求的)
			if 错误 != nil || !文件路径包.是否本地路径(相对路径) {
				return "", "", nil, nil, 工具包.已拒绝错误
			}
			实际 := 文件路径包.合并(已解析, 相对路径)
			var 缺失 []string
			for 目录 := 文件路径包.目录(实际); 目录 != 已解析; 目录 = 文件路径包.目录(目录) {
				缺失 = append(缺失, 目录)
			}
			切片包.反向(缺失)
			return 实际, 已解析, 信息, 缺失, nil
		}
		if !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
			return "", "", nil, nil, 工具包.不可用错误
		}
		// 不沿尚未解析的悬空符号链接创建父目录。
		if _, 错误 := 操作系统包.读取链接自身状态(父级); 错误 == nil {
			return "", "", nil, nil, 工具包.冲突错误
		}
		下一个 := 文件路径包.目录(父级)
		if 下一个 == 父级 {
			return "", "", nil, nil, 工具包.不可用错误
		}
		父级 = 下一个
	}
}

func (当前补丁调用 *补丁调用) 打开基准(条目 *补丁目标) (_ *操作系统包.根, 返回错误 error) {
	var 根 *操作系统包.根
	var 打开错误 error
	if 条目.目标.范围内工作区 {
		工作区, 错误 := 操作系统包.打开根(当前补丁调用.工作区)
		if 错误 != nil {
			return nil, 工具包.不可用错误
		}
		defer func() { 返回错误 = 错误判断包.合并(返回错误, 工作区.关闭()) }()
		信息, 错误 := 工作区.状态信息(".")
		if 错误 != nil || !操作系统包.相同文件(信息, 当前补丁调用.工作区信息) {
			return nil, 工具包.已拒绝错误
		}
		相对路径, 错误 := 文件路径包.相对路径(当前补丁调用.工作区, 条目.基准路径)
		if 错误 != nil || !文件路径包.是否本地路径(相对路径) {
			return nil, 工具包.已拒绝错误
		}
		根, 打开错误 = 工作区.打开根(相对路径)
	} else {
		根, 打开错误 = 操作系统包.打开根(条目.基准路径)
	}
	if 打开错误 != nil {
		return nil, 错误判断包.合并(工具包.已拒绝错误, 打开错误)
	}
	信息, 错误 := 根.状态信息(".")
	if 错误 != nil || !操作系统包.相同文件(信息, 条目.基准信息) {
		return nil, 错误判断包.合并(工具包.已拒绝错误, 根.关闭())
	}
	return 根, nil
}
func (当前补丁调用 *补丁调用) 检查工作区(执行上下文 上下文包.上下文) error {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	if 当前补丁调用.范围.标识 == "" {
		return nil
	}
	路径, 错误 := 文件路径包.求值符号链接集(当前补丁调用.范围.根路径)
	if 错误 != nil || 路径 != 当前补丁调用.工作区 {
		return 工具包.已拒绝错误
	}
	信息, 错误 := 操作系统包.状态信息(路径)
	if 错误 != nil || !操作系统包.相同文件(信息, 当前补丁调用.工作区信息) {
		return 工具包.已拒绝错误
	}
	return nil
}
func (当前补丁目标 *补丁目标) 检查(执行上下文 上下文包.上下文, 当前补丁调用 *补丁调用) error {
	if 当前补丁调用.所属对象.保护.写入受保护(当前补丁目标.目标.路径) {
		return 工具包.已拒绝错误
	}
	if 错误 := 当前补丁调用.检查工作区(执行上下文); 错误 != nil {
		return 错误
	}
	信息, 错误 := 操作系统包.状态信息(当前补丁目标.基准路径)
	if 错误 != nil || !操作系统包.相同文件(信息, 当前补丁目标.基准信息) {
		return 工具包.已拒绝错误
	}
	if 当前补丁目标.固定 != nil {
		if 错误 := 当前补丁目标.固定.检查(执行上下文); 错误 != nil {
			return 错误
		}
		当前, 错误 := 当前补丁目标.固定.获取文件().状态信息()
		if 错误 != nil || 当前.大小() != 当前补丁目标.固定.获取信息().大小() || 当前.模式() != 当前补丁目标.固定.获取信息().模式() || !当前.模块时间().相等(当前补丁目标.固定.获取信息().模块时间()) {
			return 工具包.冲突错误
		}
		return 补丁文件就绪(当前补丁目标.固定.获取文件())
	}
	实际, _, _, _, 错误 := 创建目标(当前补丁目标.目标.请求的路径)
	if 错误 != nil {
		return 错误
	}
	if 实际 != 当前补丁目标.目标.路径 {
		return 工具包.已拒绝错误
	}
	return nil
}
func 相同补丁文件(当前补丁目标, 当前补丁目标 *补丁目标) bool {
	return 当前补丁目标.固定 != nil && 当前补丁目标.固定 != nil && 操作系统包.相同文件(当前补丁目标.固定.获取信息(), 当前补丁目标.固定.获取信息())
}
func (当前补丁调用 *补丁调用) 重叠(其他 *补丁调用) bool {
	for _, 甲 := range 当前补丁调用.文件集 {
		for _, 乙 := range 其他.文件集 {
			if 字符串包.相等折叠(甲.目标.路径, 乙.目标.路径) || 相同补丁文件(甲, 乙) {
				return true
			}
		}
	}
	return false
}
func (当前补丁调用 *补丁调用) 关闭() error {
	当前补丁调用.关闭一次.执行(func() {
		for _, 条目 := range 当前补丁调用.文件集 {
			if 条目.父级 != nil {
				当前补丁调用.关闭错误 = 错误判断包.合并(当前补丁调用.关闭错误, 条目.父级.关闭())
			}
			if 条目.根 != nil {
				当前补丁调用.关闭错误 = 错误判断包.合并(当前补丁调用.关闭错误, 条目.根.关闭())
			}
			if 条目.固定 != nil {
				当前补丁调用.关闭错误 = 错误判断包.合并(当前补丁调用.关闭错误, 条目.固定.关闭())
			}
		}
	})
	return 当前补丁调用.关闭错误
}
