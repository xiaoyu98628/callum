```go
package 补丁

import (
	上下文包 "context"
	SHA256包 "crypto/sha256"
	十六进制包 "encoding/hex"
	JSON包 "encoding/json"
	错误判断包 "errors"
	输入输出包 "io"
	操作系统包 "os"
	文件路径包 "path/filepath"
	字符串包 "strings"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
	文件访问包 "github.com/xiaoyu98628/callum/internal/tool/builtin/filesystem/internal/fileaccess"
)

type 补丁变更 struct {
	路径           string       `json:"path"`
	动作         string       `json:"action"`
	操作结果        工具包.操作结果 `json:"outcome"`
	之前SHA256   string       `json:"before_sha256,omitempty"`
	预期SHA256 string       `json:"intended_sha256,omitempty"`
	已观测SHA256 string       `json:"observed_sha256,omitempty"`
}
type 补丁报告 struct {
	变更集            []补丁变更 `json:"changes"`
	已创建目录集 []string      `json:"created_directories"`
	恢复文件集      []string      `json:"recovery_files"`
}
type 补丁交付物 struct {
	根       *操作系统包.根
	名称, 路径 string
	信息       操作系统包.文件信息
	已移除    bool
}

func 文本SHA256(数据 []byte) string { 校验和 := SHA256包.校验和256(数据); return 十六进制包.编码目标字符串(校验和[:]) }
func 补丁快照(执行上下文 上下文包.上下文, 文件 *操作系统包.文件) ([]byte, error) {
	if _, 错误 := 文件.定位(0, 输入输出包.定位开始); 错误 != nil {
		return nil, 工具包.不可用错误
	}
	return 文件访问包.读取文本(执行上下文, 文件)
}

func (当前补丁调用 *补丁调用) 执行(执行上下文 上下文包.上下文) (结果 工具包.结果, 返回错误 error) {
	结果.操作结果 = 工具包.操作结果不已生效
	if !当前补丁调用.已开始.比较并且交换(false, true) {
		return 结果, 工具包.已拒绝错误
	}
	释放, 错误 := 当前补丁调用.所属对象.获取(执行上下文, 当前补丁调用)
	if 错误 != nil {
		return 结果, 错误
	}
	defer 释放()
	报告 := 补丁报告{变更集: make([]补丁变更, len(当前补丁调用.文件集)), 已创建目录集: []string{}, 恢复文件集: []string{}}
	for 索引, 条目 := range 当前补丁调用.文件集 {
		报告.变更集[索引] = 补丁变更{路径: 条目.目标.路径, 动作: 条目.补丁.动作, 操作结果: 工具包.操作结果不已生效}
	}
	defer func() {
		结果.操作结果 = 报告.操作结果(返回错误)
		输出, 错误 := JSON包.序列化(报告)
		结果.输出 = 输出
		返回错误 = 错误判断包.合并(返回错误, 错误)
	}()
	// 全部版本及上下文通过后才创建目录、暂存文件或修改任何目标。
	之前, 之后 := make([][]byte, len(当前补丁调用.文件集)), make([][]byte, len(当前补丁调用.文件集))
	for 索引, 条目 := range 当前补丁调用.文件集 {
		if 错误 := 条目.检查(执行上下文, 当前补丁调用); 错误 != nil {
			return 结果, 错误
		}
		if 条目.固定 != nil {
			数据, 错误 := 补丁快照(执行上下文, 条目.固定.获取文件())
			if 错误 != nil {
				return 结果, 错误
			}
			报告.变更集[索引].之前SHA256 = 文本SHA256(数据)
			if 报告.变更集[索引].之前SHA256 != 条目.目标.预期SHA256 {
				报告.变更集[索引].已观测SHA256 = 报告.变更集[索引].之前SHA256
				return 结果, 工具包.冲突错误
			}
			之前[索引] = 数据
			之后[索引], 错误 = 应用文本(数据, 条目.补丁)
			if 错误 != nil {
				return 结果, 错误
			}
		} else {
			之后[索引] = []byte(条目.补丁.内容)
		}
		报告.变更集[索引].预期SHA256 = 文本SHA256(之后[索引])
	}
	for 索引, 条目 := range 当前补丁调用.文件集 {
		if 错误 := 条目.检查(执行上下文, 当前补丁调用); 错误 != nil {
			return 结果, 错误
		}
		if 错误 := 条目.打开父级(执行上下文, 当前补丁调用, &报告); 错误 != nil {
			return 结果, 错误
		}
		if 错误 := 当前补丁调用.写入目标(执行上下文, 条目, 之前[索引], 之后[索引], &报告.变更集[索引], &报告); 错误 != nil {
			return 结果, 错误
		}
	}
	return 结果, 执行上下文.获取错误()
}

func (当前补丁报告 补丁报告) 操作结果(错误 error) 工具包.操作结果 {
	if 错误判断包.是否属于(错误, 工具包.未知操作结果错误) {
		return 工具包.操作结果未知
	}
	已生效, 部分 := 0, false
	for _, 变更 := range 当前补丁报告.变更集 {
		switch 变更.操作结果 {
		case 工具包.操作结果未知:
			return 工具包.操作结果未知
		case 工具包.操作结果部分:
			部分 = true
		case 工具包.操作结果已生效:
			已生效++
		}
	}
	if 已生效 == len(当前补丁报告.变更集) && !部分 {
		return 工具包.操作结果已生效
	}
	if 部分 || 已生效 > 0 || (错误 != nil && (len(当前补丁报告.已创建目录集) > 0 || len(当前补丁报告.恢复文件集) > 0)) {
		return 工具包.操作结果部分
	}
	return 工具包.操作结果不已生效
}

func (当前补丁目标 *补丁目标) 打开父级(执行上下文 上下文包.上下文, 当前补丁调用 *补丁调用, 报告 *补丁报告) (返回错误 error) {
	当前, 自有 := 当前补丁目标.根, false
	前缀 := 当前补丁目标.基准路径
	defer func() {
		if 自有 {
			返回错误 = 错误判断包.合并(返回错误, 当前.关闭())
		}
	}()
	if 当前补丁目标.父级相对 != "." {
		for 组件 := range 字符串包.拆分序号(当前补丁目标.父级相对, string(文件路径包.分隔符)) {
			if 错误 := 当前补丁目标.检查(执行上下文, 当前补丁调用); 错误 != nil {
				return 错误
			}
			if 错误 := 检查补丁目录(当前, 前缀); 错误 != nil {
				return 错误
			}
			// 每一级都从已经固定的父句柄继续，不能按复合路径重走可能被替换的祖先。
			已创建 := false
			if 错误 := 当前.创建目录(组件, 0700); 错误 == nil {
				已创建 = true
				报告.已创建目录集 = append(报告.已创建目录集, 文件路径包.合并(前缀, 组件))
			} else if !错误判断包.是否属于(错误, 操作系统包.存在错误) {
				return 错误
			}
			信息, 错误 := 当前.读取链接自身状态(组件)
			if 错误 != nil || !信息.是否目录() || 信息.模式()&操作系统包.模式符号链接 != 0 {
				return 工具包.已拒绝错误
			}
			下一个, 错误 := 当前.打开根(组件)
			if 错误 != nil {
				return 工具包.已拒绝错误
			}
			已打开, 错误 := 下一个.状态信息(".")
			if 错误 != nil || !操作系统包.相同文件(已打开, 信息) {
				return 错误判断包.合并(工具包.已拒绝错误, 下一个.关闭())
			}
			if 已创建 {
				文件, 错误 := 下一个.打开(".")
				if 错误 != nil {
					return 错误判断包.合并(错误, 下一个.关闭())
				}
				安全错误 := 保护补丁目录(文件)
				关闭错误 := 文件.关闭()
				if 错误 := 错误判断包.合并(安全错误, 关闭错误); 错误 != nil {
					return 错误判断包.合并(错误, 下一个.关闭())
				}
			}
			if 自有 {
				自有 = false
				if 错误 := 当前.关闭(); 错误 != nil {
					return 错误判断包.合并(错误, 下一个.关闭())
				}
			}
			当前, 自有 = 下一个, true
			前缀 = 文件路径包.合并(前缀, 组件)
		}
	}
	if !自有 {
		var 错误 error
		当前, 错误 = 当前补丁目标.根.打开根(".")
		if 错误 != nil {
			return 工具包.已拒绝错误
		}
	}
	当前补丁目标.父级, 自有 = 当前, false // 句柄交给调用对象，所有退出分支由 关闭 统一释放。
	return 当前补丁目标.检查父级()
}
func (当前补丁目标 *补丁目标) 检查父级() error {
	return 检查补丁目录(当前补丁目标.父级, 文件路径包.目录(当前补丁目标.目标.路径))
}
func 检查补丁目录(根 *操作系统包.根, 路径 string) error {
	已打开, 错误 := 根.状态信息(".")
	if 错误 != nil {
		return 工具包.已拒绝错误
	}
	当前, 错误 := 操作系统包.状态信息(路径)
	if 错误 != nil || !操作系统包.相同文件(已打开, 当前) {
		return 工具包.已拒绝错误
	}
	return nil
}

func (当前补丁交付物 *补丁交付物) 移除() error {
	if 当前补丁交付物.信息 == nil {
		return 工具包.未知操作结果错误
	}
	当前, 错误 := 当前补丁交付物.根.读取链接自身状态(当前补丁交付物.名称)
	if 错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		当前补丁交付物.已移除 = true
		return 工具包.冲突错误
	}
	if 错误 != nil {
		return 错误
	}
	if !操作系统包.相同文件(当前, 当前补丁交付物.信息) {
		return 工具包.冲突错误
	}
	if 错误 := 当前补丁交付物.根.移除(当前补丁交付物.名称); 错误 != nil {
		return 错误
	}
	当前补丁交付物.已移除 = true
	return nil
}

func 私有补丁复制(执行上下文 上下文包.上下文, 根 *操作系统包.根, 路径 string, 数据 []byte) (_ *补丁交付物, 返回错误 error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return nil, 错误
	}
	名称 := 文件路径包.基准(路径)
	文件, 错误 := 文件访问包.打开文件(根, 名称, 操作系统包.创建文件标志|操作系统包.独占创建标志|操作系统包.读写打开标志, 0600)
	if 错误判断包.是否属于(错误, 操作系统包.存在错误) {
		return nil, 工具包.冲突错误
	}
	if 错误 != nil {
		return nil, 错误
	}
	信息, 错误 := 文件.状态信息()
	if 错误 != nil {
		return &补丁交付物{根: 根, 名称: 名称, 路径: 路径}, 错误判断包.合并(工具包.未知操作结果错误, 错误, 文件.关闭())
	}
	甲 := &补丁交付物{根: 根, 名称: 名称, 路径: 路径, 信息: 信息}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 文件.关闭()) }()
	if 错误 := 保护补丁文件(文件); 错误 != nil {
		return 甲, 错误
	}
	if _, 错误 := 文件.写入(数据); 错误 != nil {
		return 甲, 错误
	}
	if 错误 := 文件.同步(); 错误 != nil {
		return 甲, 错误
	}
	return 甲, nil
}

func (当前补丁调用 *补丁调用) 写入目标(执行上下文 上下文包.上下文, 当前补丁目标 *补丁目标, 之前, 之后 []byte, 变更 *补丁变更, 报告 *补丁报告) (返回错误 error) {
	if 错误 := 当前补丁目标.检查(执行上下文, 当前补丁调用); 错误 != nil {
		return 错误
	}
	if 错误 := 当前补丁目标.检查父级(); 错误 != nil {
		return 错误
	}
	var 交付物集 []*补丁交付物
	defer func() {
		// 无目标修改时清理辅助文件；成功也清理。发生部分写入或结果未知时
		// 保留同步过的原文和预期内容，位置已经绑定在审批快照中。
		for _, 甲 := range 交付物集 {
			if 返回错误 == nil || 变更.操作结果 == 工具包.操作结果不已生效 {
				if 错误 := 甲.移除(); 错误 != nil {
					返回错误 = 错误判断包.合并(返回错误, 错误)
				}
			}
			if !甲.已移除 {
				报告.恢复文件集 = append(报告.恢复文件集, 甲.路径)
			}
		}
	}()
	复制文件 := func(路径 string, 数据 []byte) error {
		甲, 错误 := 私有补丁复制(执行上下文, 当前补丁目标.父级, 路径, 数据)
		if 甲 != nil {
			交付物集 = append(交付物集, 甲)
		}
		return 错误
	}
	if 当前补丁目标.固定 != nil {
		if 错误 := 复制文件(当前补丁目标.目标.恢复路径, 之前); 错误 != nil {
			return 错误
		}
	}
	if 错误 := 复制文件(当前补丁目标.目标.暂存路径, 之后); 错误 != nil {
		return 错误
	}
	if 错误 := 当前补丁目标.检查(执行上下文, 当前补丁调用); 错误 != nil {
		return 错误
	}
	if 错误 := 当前补丁目标.检查父级(); 错误 != nil {
		return 错误
	}
	参数集 := 操作系统包.读写打开标志
	if 当前补丁目标.固定 == nil {
		参数集 |= 操作系统包.创建文件标志 | 操作系统包.独占创建标志
	}
	文件, 错误 := 文件访问包.打开文件(当前补丁目标.父级, 当前补丁目标.名称, 参数集, 0600)
	if 错误判断包.是否属于(错误, 操作系统包.存在错误) {
		return 工具包.冲突错误
	}
	if 错误 != nil {
		return 错误
	}
	// 文件创建本身已经生效，空目标或后续权限失败也不能标成未执行。
	if 当前补丁目标.固定 == nil {
		变更.操作结果 = 工具包.操作结果部分
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 文件.关闭()) }()
	信息, 错误 := 文件.状态信息()
	if 错误 != nil {
		变更.操作结果 = 工具包.操作结果未知
		return 错误
	}
	if 当前补丁目标.固定 != nil && !操作系统包.相同文件(信息, 当前补丁目标.固定.获取信息()) {
		return 工具包.已拒绝错误
	}
	if 错误 := 补丁文件就绪(文件); 错误 != nil {
		return 错误
	}
	if 当前补丁目标.固定 == nil {
		if 错误 := 保护补丁文件(文件); 错误 != nil {
			return 错误
		}
	} else {
		当前, 错误 := 补丁快照(执行上下文, 文件)
		if 错误 != nil {
			return 错误
		}
		变更.已观测SHA256 = 文本SHA256(当前)
		if 变更.已观测SHA256 != 当前补丁目标.目标.预期SHA256 {
			return 工具包.冲突错误
		}
	}
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	// 开始覆盖后完成这一有界文件的写入和截断，再观察取消；不在两者之间
	// 主动放弃，避免把取消扩大为本可避免的损坏。系统错误仍如实报告。
	数量, 写入错误 := 文件.写入位于(之后, 0)
	if 数量 > 0 {
		变更.操作结果 = 工具包.操作结果部分
	}
	if 写入错误 == nil && 数量 != len(之后) {
		写入错误 = 输入输出包.短写入错误
	}
	if 写入错误 == nil {
		写入错误 = 文件.截取(int64(len(之后)))
	}
	if 写入错误 == nil {
		写入错误 = 文件.同步()
	}
	已观测, 读取错误 := 补丁快照(上下文包.无取消(执行上下文), 文件)
	if 读取错误 != nil {
		变更.操作结果 = 工具包.操作结果未知
		return 错误判断包.合并(写入错误, 读取错误)
	}
	变更.已观测SHA256 = 文本SHA256(已观测)
	switch 变更.已观测SHA256 {
	case 变更.预期SHA256:
		变更.操作结果 = 工具包.操作结果已生效
	case 变更.之前SHA256:
		// 即使内容恰好仍与原文一致，已经写过的前缀也可能更新元数据。
		// 不将已发生的写入重新解释为未执行。
		if 变更.操作结果 != 工具包.操作结果不已生效 {
			变更.操作结果 = 工具包.操作结果部分
		}
	default:
		变更.操作结果 = 工具包.操作结果部分
	}
	if 写入错误 != nil {
		return 写入错误
	}
	if 变更.操作结果 != 工具包.操作结果已生效 {
		return 工具包.冲突错误
	}
	已解析, 错误 := 文件路径包.求值符号链接集(当前补丁目标.目标.请求的路径)
	当前, 状态信息错误 := 操作系统包.状态信息(当前补丁目标.目标.路径)
```
