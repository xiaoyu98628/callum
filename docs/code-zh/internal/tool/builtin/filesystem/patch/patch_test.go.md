func Test补丁创建并且更新集带有边界版本集(测试器 *测试包.测试器) {
	目录 := 测试器.创建临时目录()
	路径 := 文件路径包.合并(目录, "old.txt")
	之前 := "header\r\nold\r\ntail"
	补丁测试写入(测试器, 路径, 之前)
	if 错误 := 操作系统包.修改权限(路径, 0640); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	当前条目 := 补丁测试执行器(测试器)
	范围 := 工具包.范围{标识: "workspace", 根路径: 目录}
	参数 := 补丁测试参数(测试器, "*** Update File: old.txt\n@@\n header\n-old\n+new\n tail\n*** Add File: nested/note.txt\n+创建的内容", 补丁版本{"old.txt", 文本SHA256([]byte(之前))})
	当前对象 := 补丁测试准备(测试器, 当前条目, 范围, 参数)
	if 当前对象.外部工作区() || 策略包.评估(策略包.开启请求, 当前对象.获取定义().影响集, false).动作 != 策略包.允许 {
		测试器.报告致命错误("普通修改未使用工作区策略")
	}
	if _, 错误 := 操作系统包.状态信息(文件路径包.合并(目录, "nested")); !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		测试器.报告致命错误("准备阶段创建了目录")
	}
	结果, 错误 := 当前条目.执行(测试器.上下文(), 当前对象, 当前对象.获取哈希())
	if 错误 != nil || 结果.操作结果 != 工具包.操作结果已生效 {
		测试器.格式化报告致命错误("补丁执行失败: %s %+v %v", 结果.输出, 结果, 错误)
	}
	if 补丁测试读取(测试器, 路径) != "header\r\nnew\r\ntail" || 补丁测试读取(测试器, 文件路径包.合并(目录, "nested", "note.txt")) != "创建的内容\n" {
		测试器.报告致命错误("文件内容错误")
	}
	信息, 错误 := 操作系统包.状态信息(路径)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 信息.模式().权限() != 0640 && 信息.模式().权限() != 0666 {
		测试器.格式化报告致命错误("改变了已有权限: %v", 信息.模式())
	} // Windows 权限由平台测试验证 DACL。
	var 报告 补丁报告
	if JSON包.反序列化(结果.输出, &报告) != nil || len(报告.变更集) != 2 || len(报告.已创建目录集) != 1 || len(报告.恢复文件集) != 0 {
		测试器.格式化报告致命错误("结果审计错误: %s", 结果.输出)
	}
	条目集, 错误 := 操作系统包.读取目录(目录)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	for _, 条目 := range 条目集 {
		if 字符串包.存在前缀(条目.名称(), ".callum-patch-") {
			测试器.报告致命错误("成功后遗留恢复副本")
		}
	}
}

func Test补丁执行前检查拒绝整体批次开启冲突(测试器 *测试包.测试器) {
	for _, 类别 := range []string{"version", "context", "existing_create", "ambiguous"} {
		测试器.执行(类别, func(测试器 *测试包.测试器) {
			目录 := 测试器.创建临时目录()
			一个, 两个 := 文件路径包.合并(目录, "one"), 文件路径包.合并(目录, "two")
			补丁测试写入(测试器, 一个, "header\nold\ntail\n")
			补丁测试写入(测试器, 两个, "a\na\n")
			哈希 := 文本SHA256([]byte("a\na\n"))
			正文 := "*** Update File: two\n@@\n-a\n+b"
			if 类别 == "version" {
				哈希 = 字符串包.重复("0", 64)
				正文 = "*** Update File: two\n@@\n a\n-a\n+b"
			}
			if 类别 == "context" {
				正文 = "*** Update File: two\n@@\n-missing\n+b"
			}
			版本集 := []补丁版本{{"one", 文本SHA256([]byte(补丁测试读取(测试器, 一个)))}, {"two", 哈希}}
			if 类别 == "existing_create" {
				正文 = "*** Add File: two\n+x"
				版本集 = 版本集[:1]
			}
			当前条目 := 补丁测试执行器(测试器)
			参数 := 补丁测试参数(测试器, "*** Update File: one\n@@\n header\n-old\n+new\n tail\n"+正文, 版本集...)
			当前对象, 错误 := 当前条目.准备(测试器.上下文(), 工具包.范围{标识: "w", 根路径: 目录}, "apply_patch", 参数)
			if 错误 == nil {
				defer 当前对象.关闭()
				var 结果 工具包.结果
				结果, 错误 = 当前条目.执行(测试器.上下文(), 当前对象, 当前对象.获取哈希())
				if 类别 == "version" {
					var 报告 补丁报告
					实际哈希 := 文本SHA256([]byte(补丁测试读取(测试器, 两个)))
					if JSON包.反序列化(结果.输出, &报告) != nil || 报告.变更集[1].之前SHA256 != 实际哈希 || 报告.变更集[1].已观测SHA256 != 实际哈希 {
						测试器.格式化报告致命错误("把请求前提伪造成实际文件版本: %s", 结果.输出)
					}
				}
				if 结果.操作结果 != 工具包.操作结果不已生效 {
					测试器.格式化报告致命错误("冲突前已产生副作用: %+v", 结果)
				}
			}
			if !错误判断包.是否属于(错误, 工具包.冲突错误) {
				测试器.格式化报告致命错误("没有拒绝冲突: %v", 错误)
			}
			if 补丁测试读取(测试器, 一个) != "header\nold\ntail\n" || 补丁测试读取(测试器, 两个) != "a\na\n" {
				测试器.报告致命错误("冲突导致部分覆盖")
			}
			条目集, _ := 操作系统包.读取目录(目录)
			if len(条目集) != 2 {
				测试器.报告致命错误("冲突遗留辅助文件")
			}
		})
	}
}

func Test补丁计划并且策略绑定每个目标(测试器 *测试包.测试器) {
	目录, 外部 := 测试器.创建临时目录(), 测试器.创建临时目录()
	外部 := 文件路径包.合并(外部, "new.txt")
	当前条目 := 补丁测试执行器(测试器)
	原始 := 补丁测试参数(测试器, "*** Add File: inside.txt\n+x\n*** Add File: "+外部+"\n+y")
	当前对象 := 补丁测试准备(测试器, 当前条目, 工具包.范围{标识: "w", 根路径: 目录}, 原始)
	var 请求 执行器包.操作
	if JSON包.反序列化(当前对象.获取请求(), &请求) != nil || 请求.计划 == nil || len(请求.计划.目标集) != 2 {
		测试器.格式化报告致命错误("缺少多目标快照: %s", 当前对象.获取请求())
	}
	if !当前对象.外部工作区() || 请求.计划.目标集[0].范围内工作区 == 请求.计划.目标集[1].范围内工作区 {
		测试器.报告致命错误("未按全部实际目标判断范围")
	}
	if 策略包.评估(策略包.开启请求, 当前对象.获取定义().影响集, 当前对象.外部工作区()).动作 != 策略包.询问 || 策略包.评估(策略包.从不, 当前对象.获取定义().影响集, 当前对象.外部工作区()).动作 != 策略包.拒绝 {
		测试器.报告致命错误("工作区外目标绕过审批")
	}
	// 相同固定操作不能因为辅助名字或参数表顺序变化而改变指纹。
	再次 := 补丁测试准备(测试器, 当前条目, 工具包.范围{标识: "w", 根路径: 目录}, 原始)
	if 再次.获取哈希() != 当前对象.获取哈希() || !反射包.深度相等(再次.获取请求(), 当前对象.获取请求()) {
		测试器.报告致命错误("相同操作的重试指纹不稳定")
	}
	请求.计划.目标集[0].路径 = 外部
	var 新建 执行器包.操作
	_ = JSON包.反序列化(当前对象.获取请求(), &新建)
	if 新建.计划.目标集[0].路径 == 外部 {
		测试器.报告致命错误("调用方修改了固定计划")
	}
	if _, 错误 := 当前条目.执行(测试器.上下文(), 当前对象, "changed"); !错误判断包.是否属于(错误, 工具包.已拒绝错误) {
		测试器.报告致命错误("未核对审批指纹")
	}
}

func Test补丁拒绝已替换目标集并且版本变更集(测试器 *测试包.测试器) {
	for _, 类别 := range []string{"file", "parent", "content", "new_target"} {
		测试器.执行(类别, func(测试器 *测试包.测试器) {
			目录 := 测试器.创建临时目录()
			路径 := 文件路径包.合并(目录, "note")
			之前 := "head\nold\ntail\n"
			补丁测试写入(测试器, 路径, 之前)
			当前条目 := 补丁测试执行器(测试器)
			正文 := "*** Update File: note\n@@\n head\n-old\n+new\n tail"
			版本集 := []补丁版本{{"note", 文本SHA256([]byte(之前))}}
			if 类别 == "new_target" {
				正文 = "*** Add File: future\n+x"
				版本集 = nil
			}
			当前对象 := 补丁测试准备(测试器, 当前条目, 工具包.范围{标识: "w", 根路径: 目录}, 补丁测试参数(测试器, 正文, 版本集...))
			switch 类别 {
			case "file":
				if 错误 := 操作系统包.重命名(路径, 路径+"-old"); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				补丁测试写入(测试器, 路径, "user changes")
			case "parent":
				if 错误 := 操作系统包.重命名(目录, 目录+"-old"); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				测试器.注册清理函数(func() { _ = 操作系统包.移除全部(目录 + "-old") })
				if 错误 := 操作系统包.创建目录(目录, 0700); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				补丁测试写入(测试器, 路径, "user changes")
			case "content":
				补丁测试写入(测试器, 路径, "user changes")
			case "new_target":
				补丁测试写入(测试器, 文件路径包.合并(目录, "future"), "user changes")
			}
			结果, 错误 := 当前条目.执行(测试器.上下文(), 当前对象, 当前对象.获取哈希())
			if 错误 == nil || 结果.操作结果 != 工具包.操作结果不已生效 {
				测试器.格式化报告致命错误("替换后仍执行: %s %v", 结果.输出, 错误)
			}
			if 类别 == "new_target" {
				路径 = 文件路径包.合并(目录, "future")
			}
			if 补丁测试读取(测试器, 路径) != "user changes" {
				测试器.报告致命错误("覆盖了用户改动")
			}
		})
	}
}

func (当前取消之后补丁 *取消之后补丁) 获取错误() error {
	数据, _ := 操作系统包.读取文件(当前取消之后补丁.路径)
	if string(数据) == 当前取消之后补丁.值 {
		当前取消之后补丁.取消函数()
	}
	return 当前取消之后补丁.上下文.获取错误()
}
func Test补丁取消保留已生效文件集并且恢复副本集(测试器 *测试包.测试器) {
	目录 := 测试器.创建临时目录()
	首个, 秒 := 文件路径包.合并(目录, "one"), 文件路径包.合并(目录, "two")
	之前 := "head\nold\ntail\n"
	for _, 路径 := range []string{首个, 秒} {
		补丁测试写入(测试器, 路径, 之前)
	}
	实例 := 新建(工具包.受保护数据{})
	原始 := 补丁测试参数(测试器, "*** Update File: one\n@@\n head\n-old\n+new\n tail\n*** Update File: two\n@@\n head\n-old\n+new\n tail", 补丁版本{"one", 文本SHA256([]byte(之前))}, 补丁版本{"two", 文本SHA256([]byte(之前))})
	已规范化, 错误 := 实例.校验(原始)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	工具调用, 错误 := 实例.准备(测试器.上下文(), 工具包.范围{标识: "w", 根路径: 目录}, 已规范化)

		测试器.报告致命错误(错误)
	}
	defer 工具调用.关闭()
	基准, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	defer 取消函数()
	执行上下文 := &取消之后补丁{上下文: 基准, 路径: 首个, 值: "head\nnew\ntail\n", 取消函数: 取消函数}
	结果, 错误 := 工具调用.执行(执行上下文)

		测试器.格式化报告致命错误("取消覆盖操作事实: %s %s %v", 结果.操作结果, 结果.输出, 错误)
	}
	if 补丁测试读取(测试器, 首个) != "head\nnew\ntail\n" || 补丁测试读取(测试器, 秒) != 之前 {
		测试器.报告致命错误("取消后继续修改其他文件")
	}
	var 报告 补丁报告
	if JSON包.反序列化(结果.输出, &报告) != nil || 报告.变更集[0].操作结果 != 工具包.操作结果已生效 || 报告.变更集[1].操作结果 != 工具包.操作结果不已生效 {
		测试器.格式化报告致命错误("缺少逐文件事实: %s", 结果.输出)
	}
	if len(报告.恢复文件集) != 2 {
		测试器.格式化报告致命错误("取消后缺少恢复内容: %s", 结果.输出)
	}
	内容 := map[string]bool{}
	for _, 路径 := range 报告.恢复文件集 {
		内容[补丁测试读取(测试器, 路径)] = true
	}
	if !内容[之前] || !内容["head\nnew\ntail\n"] {
		测试器.报告致命错误("恢复副本不包含原文及预期内容")
	}
}

func Test并发补丁集无法覆盖每个其他(测试器 *测试包.测试器) {
	目录 := 测试器.创建临时目录()
	路径 := 文件路径包.合并(目录, "note")
	之前 := "head\nold\ntail\n"
	补丁测试写入(测试器, 路径, 之前)
	当前条目 := 补丁测试执行器(测试器)
	原始 := 补丁测试参数(测试器, "*** Update File: note\n@@\n head\n-old\n+new\n tail", 补丁版本{"note", 文本SHA256([]byte(之前))})
	首个 := 补丁测试准备(测试器, 当前条目, 工具包.范围{标识: "w", 根路径: 目录}, 原始)
	秒 := 补丁测试准备(测试器, 当前条目, 工具包.范围{标识: "w", 根路径: 目录}, 原始)
	type 回答 struct {
		结果 工具包.结果
		错误    error
	}
	完成 := make(chan 回答, 2)
	for _, 已准备 := range []*执行器包.已准备调用{首个, 秒} {
		go func() { 结果, 错误 := 当前条目.执行(测试器.上下文(), 已准备, 已准备.获取哈希()); 完成 <- 回答{结果, 错误} }()
	}
	成功, 冲突集 := 0, 0
	for range 2 {
		回答 := <-完成
		if 回答.错误 == nil && 回答.结果.操作结果 == 工具包.操作结果已生效 {
			成功++
		} else if 错误判断包.是否属于(回答.错误, 工具包.冲突错误) && 回答.结果.操作结果 == 工具包.操作结果不已生效 {
			冲突集++
		} else {
			测试器.格式化报告致命错误("并发结果错误: %+v %v", 回答.结果, 回答.错误)
		}
	}
	if 成功 != 1 || 冲突集 != 1 || 补丁测试读取(测试器, 路径) != "head\nnew\ntail\n" {
		测试器.格式化报告致命错误("并发覆盖或重复执行: success=%d conflict=%d", 成功, 冲突集)
	}
}

			补丁测试写入(测试器, 其他, "unrelated")

					测试器.报告致命错误("链接实际目标在工作区外却自动放行")
				}
				if 错误 := 操作系统包.移除(链接); 错误 != nil {
					测试器.报告致命错误(错误)
				}

					测试器.格式化跳过("无法创建符号链接: %v", 错误)
				}
			}
			结果, 错误 := 当前条目.执行(测试器.上下文(), 当前对象, 当前对象.获取哈希())

type 替换已创建补丁父级 struct {
	上下文包.上下文
	目录     string
	变化通知 bool
	错误     error
}

func (当前替换已创建补丁父级 *替换已创建补丁父级) 获取错误() error {
	if !当前替换已创建补丁父级.变化通知 {
		路径 := 文件路径包.合并(当前替换已创建补丁父级.目录, "created")
		if 信息, 错误 := 操作系统包.状态信息(路径); 错误 == nil && 信息.是否目录() {
			当前替换已创建补丁父级.变化通知 = true
			if 错误 := 操作系统包.重命名(路径, 路径+"-original"); 错误 != nil {
				当前替换已创建补丁父级.错误 = 错误
				return 错误
			}
			当前替换已创建补丁父级.错误 = 操作系统包.创建目录(路径, 0700)
		}
	}
	if 当前替换已创建补丁父级.错误 != nil {
		return 当前替换已创建补丁父级.错误
	}
	return 当前替换已创建补丁父级.上下文.获取错误()
}
