package 文件访问

import (
	字节包 "bytes"
	上下文包 "context"
	错误判断包 "errors"
	输入输出包 "io"
	操作系统包 "os"
	UTF8包 "unicode/utf8"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

// 打开文件 为固定目录内的打开操作添加平台非阻塞标志，避免特殊文件阻塞取消。
// 调用方负责检查目标类型并关闭返回的文件；底层打开错误保持原样。
func 打开文件(根 *操作系统包.根, 路径 string, 参数集 int, 权限 操作系统包.文件模式) (*操作系统包.文件, error) {
	return 根.打开文件(路径, 参数集|非阻塞, 权限)
}

// 打开读取 使用非阻塞标志打开路径，避免命名管道阻塞取消；只读普通文件和目录才可使用。
func 打开读取(根 *操作系统包.根, 路径 string, 预期 操作系统包.文件信息) (*操作系统包.文件, error) {
	文件, 错误 := 打开文件(根, 路径, 操作系统包.只读打开标志, 0)
	if 错误 != nil {
		return nil, 工具包.已拒绝错误
	}
	// 搜索先用 读取链接自身状态 跳过符号链接；打开后复核身份，拒绝遍历期间的路径替换。
	if 预期 != nil {
		已打开, 错误 := 文件.状态信息()
		if 错误 != nil || !操作系统包.相同文件(预期, 已打开) {
			return nil, 错误判断包.合并(工具包.已拒绝错误, 文件.关闭())
		}
	}
	return 文件, nil
}

// 读取文本文件 读取固定目录内的文本，并在返回前关闭本次打开的文件。
func 读取文本文件(执行上下文 上下文包.上下文, 根 *操作系统包.根, 路径 string, 预期 操作系统包.文件信息, 保护 工具包.受保护数据) (数据 []byte, 返回错误 error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return nil, 错误
	}
	文件, 错误 := 打开读取(根, 路径, 预期)
	if 错误 != nil {
		return nil, 错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 文件.关闭()) }()
	if 错误 := 保护.检查文件(预期); 错误 != nil {
		return nil, 错误
	}
	return 读取文本(执行上下文, 文件)
}

// 读取文本 从借用句柄的当前位置读取不超过 1 MiB 的 UTF-8 文本，不关闭或重置句柄。
func 读取文本(执行上下文 上下文包.上下文, 当前文件 *操作系统包.文件) ([]byte, error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return nil, 错误
	}
	状态信息, 错误 := 当前文件.状态信息()
	if 错误 != nil {
		return nil, 工具包.不可用错误
	}
	if !状态信息.模式().是否普通文件() {
		return nil, 工具包.不可用错误
	}
	if 状态信息.大小() > 1<<20 {
		return nil, 工具包.超出上限错误
	}
	数据, 错误 := 输入输出包.读取全部(输入输出包.上限读取器(当前文件, (1<<20)+1))
	if 错误 != nil {
		return nil, 工具包.不可用错误
	}
	if len(数据) > 1<<20 {
		return nil, 工具包.超出上限错误
	}
	if !UTF8包.有效(数据) || 字节包.索引字节(数据, 0) >= 0 {
		return nil, 工具包.不可用错误
	}
	return 数据, 执行上下文.获取错误()
}
