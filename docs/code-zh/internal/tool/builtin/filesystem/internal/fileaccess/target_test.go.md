```go
package 文件访问测试

import (
	上下文包 "context"
	错误判断包 "errors"
	操作系统包 "os"
	文件路径包 "path/filepath"
	测试包 "testing"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
	文件访问包 "github.com/xiaoyu98628/callum/internal/tool/builtin/filesystem/internal/fileaccess"
)

func Test固定句柄所有权并且取消(测试器 *测试包.测试器) {
	目录 := 测试器.创建临时目录()
	if 错误 := 操作系统包.写入文件(文件路径包.合并(目录, "note"), []byte("text"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	范围 := 工具包.范围{标识: "w", 根路径: 目录}
	for _, 测试 := range []struct {
		名称, 路径 string
		类别     文件访问包.类别
	}{
		{"file", "note", 文件访问包.文件},
		{"directory", ".", 文件访问包.目录},
	} {
		测试器.执行(测试.名称, func(测试器 *测试包.测试器) {
			句柄, 错误 := 文件访问包.准备(测试器.上下文(), 范围, 测试.路径, 测试.类别, 工具包.受保护数据{})
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			测试器.注册清理函数(func() {
				if 错误 := 句柄.关闭(); 错误 != nil {
					测试器.错误文本(错误)
				}
			})
			原始 := *句柄.获取目标()
			句柄.获取目标().路径 = "changed"
			if *句柄.获取目标() != 原始 || !原始.范围内工作区 {
				测试器.报告致命错误("目标副本可修改固定身份，或工作区分类丢失")
			}
			执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
			取消函数()
			if 错误 := 句柄.检查(执行上下文); !错误判断包.是否属于(错误, 上下文包.已取消错误) {
				测试器.格式化报告致命错误("目标检查未保留取消: %v", 错误)
			}
			if 错误 := 句柄.检查(测试器.上下文()); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			if 测试.类别 == 文件访问包.文件 {
				if _, 错误 := 文件访问包.读取文本(执行上下文, 句柄.获取文件()); !错误判断包.是否属于(错误, 上下文包.已取消错误) {
					测试器.格式化报告致命错误("读取未保留取消: %v", 错误)
				}
				if _, 错误 := 句柄.获取文件().状态信息(); 错误 != nil {
					测试器.格式化报告致命错误("取消读取关闭了借用句柄: %v", 错误)
				}
			}
			if 错误 := 句柄.关闭(); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			if 错误 := 句柄.关闭(); 错误 != nil {
				测试器.格式化报告致命错误("重复关闭失败: %v", 错误)
			}
			if 测试.类别 == 文件访问包.文件 {
				if _, 错误 := 句柄.获取文件().状态信息(); !错误判断包.是否属于(错误, 操作系统包.已关闭错误) {
					测试器.格式化报告致命错误("固定文件未关闭: %v", 错误)
				}
			} else if _, 错误 := 句柄.获取根().状态信息("."); 错误 == nil {
				测试器.报告致命错误("固定目录未关闭")
			}
		})
	}
	for _, 测试 := range []struct {
		路径 string
		类别 文件访问包.类别
	}{
		{"note", 文件访问包.目录},
		{".", 文件访问包.文件},
	} {
		句柄, 错误 := 文件访问包.准备(测试器.上下文(), 范围, 测试.路径, 测试.类别, 工具包.受保护数据{})
		if 句柄 != nil {
			if 错误 := 句柄.关闭(); 错误 != nil {
				测试器.错误文本(错误)
			}
		}
		if !错误判断包.是否属于(错误, 工具包.不可用错误) {
			测试器.格式化报告致命错误("接受不匹配的目标类型: %s, %v", 测试.路径, 错误)
		}
	}
}
```
