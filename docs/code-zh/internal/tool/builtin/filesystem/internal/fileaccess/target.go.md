```go
// 包 文件访问 管理文件工具共享的固定目标、句柄与有界文本读取。
package 文件访问

import (
	上下文包 "context"
	错误判断包 "errors"
	操作系统包 "os"
	文件路径包 "path/filepath"
	同步包 "sync"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

// 句柄 在审批前只固定句柄和元数据，不读取文件正文或目录内容。
// 创建者负责 关闭；借用的文件与目录句柄不得由调用方单独关闭。
// 单文件持有文件句柄，目录持有仅覆盖目标目录的 根，不授权相邻资源。
type 句柄 struct {
	目标    工具包.目标
	信息    操作系统包.文件信息
	保护    工具包.受保护数据
	根     *操作系统包.根
	文件    *操作系统包.文件
	范围    工具包.范围
	工作区   string
	工作区信息 操作系统包.文件信息
	关闭一次  同步包.一次
	关闭错误  error
}

// 类别 限定允许固定的目标类型。
type 类别 uint8

const (
	文件     类别 = iota // 仅普通文件。
	目录               // 仅目录。
	文件或者目录           // 普通文件或目录。
)

// 准备 固定已校验路径的目标及可选工作区身份，不读取正文。成功后由调用方关闭句柄。
func 准备(执行上下文 上下文包.上下文, 范围 工具包.范围, 路径 string, 类别 类别, 保护 工具包.受保护数据) (_ *句柄, 返回错误 error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return nil, 错误
	}
	if 类别 > 文件或者目录 {
		return nil, 工具包.无效错误
	}
	请求的 := 路径
	if !文件路径包.是否绝对路径(请求的) {
		if 范围.标识 == "" {
			return nil, 工具包.路径基准错误
		}
		请求的 = 文件路径包.合并(范围.根路径, 请求的)
	}
	已解析, 错误 := 文件路径包.求值符号链接集(请求的)
	if 错误 != nil {
		return nil, 工具包.不可用错误
	}
	if 保护.读取受保护(已解析) {
		return nil, 工具包.已拒绝错误
	}
	信息, 错误 := 操作系统包.状态信息(已解析)
	if 错误 != nil {
		return nil, 工具包.不可用错误
	}
	当前对象 := &句柄{信息: 信息, 范围: 范围, 保护: 保护,
		目标: 工具包.目标{请求的路径: 请求的, 路径: 已解析, 类别: "file"}}
	defer func() {
		if 返回错误 != nil {
			返回错误 = 错误判断包.合并(返回错误, 当前对象.关闭())
		}
	}()
	if 信息.是否目录() {
		当前对象.目标.类别 = "directory"
	} else if !信息.模式().是否普通文件() {
		return nil, 工具包.不可用错误
	}
	if (类别 == 文件 && !信息.模式().是否普通文件()) || (类别 == 目录 && !信息.是否目录()) {
		return nil, 工具包.不可用错误
	}
	if 范围.标识 != "" {
		当前对象.工作区, 错误 = 文件路径包.求值符号链接集(范围.根路径)
		if 错误 != nil {
			return nil, 工具包.不可用错误
		}
		当前对象.工作区信息, 错误 = 操作系统包.状态信息(当前对象.工作区)
		if 错误 != nil || !当前对象.工作区信息.是否目录() {
			return nil, 工具包.不可用错误
		}
		相对路径, 错误 := 文件路径包.相对路径(当前对象.工作区, 已解析)
		当前对象.目标.范围内工作区 = 错误 == nil && 文件路径包.是否本地路径(相对路径)
	}
	// 工作区内目标必须经工作区 根 打开；即使路径在此时被替换为越界链接，
	// 也不能因先前的路径分类而获得自动放行后的越界读取能力。
	基准, 相对 := 文件路径包.目录(已解析), 文件路径包.基准(已解析)
	if 信息.是否目录() {
		基准, 相对 = 已解析, "."
	}
	if 当前对象.目标.范围内工作区 {
		基准 = 当前对象.工作区
		相对, 错误 = 文件路径包.相对路径(基准, 已解析)
		if 错误 != nil {
			return nil, 工具包.已拒绝错误
		}
	}
	根, 错误 := 操作系统包.打开根(基准)
	if 错误 != nil {
		return nil, 工具包.不可用错误
	}
	defer func() { 返回错误 = 错误判断包.合并(返回错误, 根.关闭()) }()
	if 当前对象.目标.范围内工作区 {
		已打开, 错误 := 根.状态信息(".")
		if 错误 != nil || !操作系统包.相同文件(已打开, 当前对象.工作区信息) {
			return nil, 工具包.已拒绝错误
		}
	}
	var 已打开 操作系统包.文件信息
	if 信息.是否目录() {
		当前对象.根, 错误 = 根.打开根(相对)
		if 错误 == nil {
			已打开, 错误 = 当前对象.根.状态信息(".")
		}
	} else {
		当前对象.文件, 错误 = 打开文件(根, 相对, 操作系统包.只读打开标志, 0)
		if 错误 == nil {
			已打开, 错误 = 当前对象.文件.状态信息()
		}
	}
	if 错误 != nil || !操作系统包.相同文件(信息, 已打开) {
		return nil, 工具包.已拒绝错误
	}
	if 错误 := 当前对象.检查(执行上下文); 错误 != nil {
		return nil, 错误
	}
	return 当前对象, nil
}

// 目标 返回固定目标元数据的副本。
func (当前句柄 *句柄) 获取目标() *工具包.目标 { return new(当前句柄.目标) }

// 信息 返回准备阶段记录的目标身份和元数据。
func (当前句柄 *句柄) 获取信息() 操作系统包.文件信息 { return 当前句柄.信息 }

// 文件 借出固定普通文件句柄；目录目标返回 nil。生命周期归 句柄 所有。
func (当前句柄 *句柄) 获取文件() *操作系统包.文件 { return 当前句柄.文件 }

// 根 借出固定目录句柄；文件目标返回 nil。生命周期归 句柄 所有。
func (当前句柄 *句柄) 获取根() *操作系统包.根 { return 当前句柄.根 }

// 检查 检查原始路径、目标身份及工作区是否仍与审批时一致。
// 读取始终使用固定句柄；检查之后发生的路径替换也不能把读取重定向到别的文件。
func (当前句柄 *句柄) 检查(执行上下文 上下文包.上下文) error {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	已解析, 错误 := 文件路径包.求值符号链接集(当前句柄.目标.请求的路径)
	if 错误 != nil || 已解析 != 当前句柄.目标.路径 {
		return 工具包.已拒绝错误
	}
	当前, 错误 := 操作系统包.状态信息(已解析)
	if 错误 != nil || !操作系统包.相同文件(当前, 当前句柄.信息) {
		return 工具包.已拒绝错误
	}
	if 错误 := 当前句柄.保护.检查文件(当前); 错误 != nil {
		return 错误
	}
	if 当前句柄.范围.标识 != "" {
		工作区, 错误 := 文件路径包.求值符号链接集(当前句柄.范围.根路径)
		if 错误 != nil || 工作区 != 当前句柄.工作区 {
			return 工具包.已拒绝错误
		}
		当前, 错误 := 操作系统包.状态信息(工作区)
		if 错误 != nil || !操作系统包.相同文件(当前, 当前句柄.工作区信息) {
			return 工具包.已拒绝错误
		}
	}
	return nil
}

// 关闭 释放持有的句柄，可重复调用。
func (当前句柄 *句柄) 关闭() error {
	当前句柄.关闭一次.执行(func() {
		if 当前句柄.根 != nil {
			当前句柄.关闭错误 = 当前句柄.根.关闭()
		}
		if 当前句柄.文件 != nil {
			当前句柄.关闭错误 = 错误判断包.合并(当前句柄.关闭错误, 当前句柄.文件.关闭())
		}
	})
	return 当前句柄.关闭错误
}
```
