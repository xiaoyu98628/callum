package 列表测试

import (
	JSON包 "encoding/json"
	错误判断包 "errors"
	操作系统包 "os"
	文件路径包 "path/filepath"
	测试包 "testing"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
	列表包 "github.com/xiaoyu98628/callum/internal/tool/builtin/filesystem/list"
)

func Test目录参数(测试器 *测试包.测试器) {
	实例 := 列表包.新建(工具包.受保护数据{})
	for _, 测试 := range []struct{ 输入, 预期 string }{
		{`{"path":"."}`, `{"path":".","limit":100}`},
		{`{"path":"a/../notes","limit":0}`, `{"path":"notes","limit":100}`},
		{`{"path":".","limit":200}`, `{"path":".","limit":200}`},
	} {
		实际值, 错误 := 实例.校验(JSON包.原始JSON消息(测试.输入))
		if 错误 != nil || string(实际值) != 测试.预期 {
			测试器.格式化报告致命错误("参数规范化改变: %s => %s, %v", 测试.输入, 实际值, 错误)
		}
	}
	for _, 原始 := range []string{
		`null`, `[]`, `{"path":""}`, `{"path":"../outside"}`,
		`{"path":".","limit":null}`, `{"path":".","limit":-1}`,
		`{"path":".","limit":201}`, `{"path":".","limit":1.5}`,
		`{"path":".","query":"text"}`, `{"path":".","start_line":1}`,
		`{"path":".","Path":"other"}`,
		`{"path":".","path":"other"}`,
		`{"path":".","limit":1,"limit":2}`,
		`{"path":".","Limit":1}`,
		`{"path":"."} {}`,
		`{"path":".","limit":"1"}`,
	} {
		if _, 错误 := 实例.校验(JSON包.原始JSON消息(原始)); !错误判断包.是否属于(错误, 工具包.无效错误) {
			测试器.格式化报告致命错误("接受目录工具的无效参数: %s, %v", 原始, 错误)
		}
		if 工具调用, 错误 := 实例.准备(测试器.上下文(), 工具包.范围{}, JSON包.原始JSON消息(原始)); !错误判断包.是否属于(错误, 工具包.无效错误) {
			if 工具调用 != nil {
				if 关闭错误 := 工具调用.关闭(); 关闭错误 != nil {
					测试器.错误文本(关闭错误)
				}
			}
			测试器.格式化报告致命错误("准备阶段接受无效参数: %s, %v", 原始, 错误)
		}
	}
}

func Test目录上限并且目标类型(测试器 *测试包.测试器) {
	目录 := 测试器.创建临时目录()
	for _, 名称 := range []string{"a", "b"} {
		if 错误 := 操作系统包.写入文件(文件路径包.合并(目录, 名称), []byte("text"), 0600); 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	实例 := 列表包.新建(工具包.受保护数据{})
	范围 := 工具包.范围{标识: "w", 根路径: 目录}
	参数, 错误 := 实例.校验(JSON包.原始JSON消息(`{"path":".","limit":1}`))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	工具调用, 错误 := 实例.准备(测试器.上下文(), 范围, 参数)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 工具调用.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	结果, 错误 := 工具调用.执行(测试器.上下文())
	if 错误 != nil || 结果.操作结果 != 工具包.操作结果无 {
		测试器.格式化报告致命错误("目录读取失败: %s, %v", 结果.输出, 错误)
	}
	var 输出 struct {
		条目集 []struct{ 名称 string } `json:"entries"`
		已截取 bool                  `json:"truncated"`
	}
	if 错误 := JSON包.反序列化(结果.输出, &输出); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if len(输出.条目集) != 1 || !输出.已截取 {
		测试器.格式化报告致命错误("目录上限未保持: %s", 结果.输出)
	}
	if 工具调用, 错误 := 实例.准备(测试器.上下文(), 范围, JSON包.原始JSON消息(`{"path":"a","limit":1}`)); !错误判断包.是否属于(错误, 工具包.不可用错误) {
		if 工具调用 != nil {
			测试器.注册清理函数(func() {
				if 错误 := 工具调用.关闭(); 错误 != nil {
					测试器.错误文本(错误)
				}
			})
		}
		测试器.格式化报告致命错误("目录工具接受了文件目标: %v", 错误)
	}
}
