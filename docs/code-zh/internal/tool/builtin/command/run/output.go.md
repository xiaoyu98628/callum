```go
package 运行

import (
	基准64包 "encoding/base64"
	JSON包 "encoding/json"
	切片包 "slices"
	同步包 "sync"
	UTF8包 "unicode/utf8"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

const 已保留字节 = 16 << 10
const 已编码流字节 = 24 << 10

type 输出数据 struct {
	编码  string `json:"encoding"`
	数据      string `json:"data"`
	字节     int64  `json:"bytes"`
	已截取 bool   `json:"truncated"`
}

type 输出缓冲区 struct {
	数据  []byte
	总计 int64
}

type 捕获 struct {
	互斥锁              同步包.互斥锁
	标准输出, 标准错误输出  输出缓冲区
	发出事件            func(工具包.输出分片) error
	取消函数          func()
	发布错误         error
	事件集, 已发布 int
	流已截取 bool
}

// 写入 持续排空管道；保留和推送分别有界，截断不会让子进程阻塞在输出上。
func (当前捕获 *捕获) 写入(流 string, 数据 []byte) {
	当前捕获.互斥锁.加锁()
	defer 当前捕获.互斥锁.解锁()
	缓冲区 := &当前捕获.标准输出
	if 流 == "stderr" {
		缓冲区 = &当前捕获.标准错误输出
	}
	缓冲区.总计 += int64(len(数据))
	保留 := min(len(数据), 已保留字节-len(缓冲区.数据))
	缓冲区.数据 = append(缓冲区.数据, 数据[:保留]...)
	if 当前捕获.发出事件 == nil || 当前捕获.发布错误 != nil {
		return
	}
	保留 = min(len(数据), 工具包.最大输出字节-当前捕获.已发布)
	if 当前捕获.事件集 >= 工具包.最大输出事件集 {
		保留 = 0
	}
	if 保留 < len(数据) {
		当前捕获.流已截取 = true
	}
	if 保留 == 0 {
		return
	}
	当前捕获.事件集++
	当前捕获.已发布 += 保留
	if 错误 := 当前捕获.发出事件(工具包.输出分片{流: 流, 数据: 数据[:保留]}); 错误 != nil {
		当前捕获.发布错误 = 错误
		当前捕获.取消函数()
	}
}

func (当前捕获 *捕获) 失败() error { 当前捕获.互斥锁.加锁(); defer 当前捕获.互斥锁.解锁(); return 当前捕获.发布错误 }

func (当前输出缓冲区 输出缓冲区) 结果() 输出数据 {
	文本 := UTF8包.有效(当前输出缓冲区.数据)
	创建结果 := func(数量 int) 输出数据 {
		数据 := 当前输出缓冲区.数据[:数量]
		编码, 值 := "base64", ""
		if 文本 {
			for !UTF8包.有效(数据) {
				数据 = 数据[:len(数据)-1]
			}
			编码, 值 = "utf-8", string(数据)
		} else {
			值 = 基准64包.标准编码.编码目标字符串(数据)
		}
		return 输出数据{编码, 值, 当前输出缓冲区.总计, int64(len(数据)) < 当前输出缓冲区.总计}
	}
	// 按实际 JSON 编码长度限制，转义字符或二进制输出也不能突破工具结果上限。
	低, 高 := 0, len(当前输出缓冲区.数据)
	for 低 < 高 {
		中间 := (低 + 高 + 1) / 2
		原始, _ := JSON包.序列化(创建结果(中间)) // 仅包含字符串、布尔与整数。
		if len(原始) <= 已编码流字节 {
			低 = 中间
		} else {
			高 = 中间 - 1
		}
	}
	return 创建结果(低)
}

func (当前捕获 *捕获) 结果集() (输出数据, 输出数据, bool) {
	当前捕获.互斥锁.加锁()
	defer 当前捕获.互斥锁.解锁()
	return (输出缓冲区{切片包.复制(当前捕获.标准输出.数据), 当前捕获.标准输出.总计}).结果(), (输出缓冲区{切片包.复制(当前捕获.标准错误输出.数据), 当前捕获.标准错误输出.总计}).结果(), 当前捕获.流已截取
}
```
