func TestSeatbelt清理失败是已报告无私有路径(测试器 *测试包.测试器) {

			测试器.错误文本(错误)

			return nil
		}

			return 错误

		测试器.格式化报告致命错误("清理失败诊断不符或泄漏路径: %s", 结果.输出)
	}
