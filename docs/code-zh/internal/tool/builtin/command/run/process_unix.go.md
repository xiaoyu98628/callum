//go:build darwin || linux

package 运行

import (
	错误判断包 "errors"
	格式化包 "fmt"
	操作系统包 "os"
	系统调用包 "syscall"

	进程组包 "github.com/xiaoyu98628/callum/internal/processgroup"
	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

type 进程 struct {
	进程标识 int
	已结束  bool
	退出   进程退出
}

func (当前进程 *进程) 获取进程标识() int { return 当前进程.进程标识 }

// 元数据检查与打开之间可能发生目标替换，非阻塞打开避免 先进先出 挂住审批准备。
func 打开目标路径(路径 string) (*操作系统包.文件, error) {
	return 操作系统包.打开文件(路径, 操作系统包.只读打开标志|系统调用包.非阻塞标志, 0)
}

func 开始进程(固定 *固定命令, 标准输入, 标准输出, 标准错误输出 *操作系统包.文件) (*进程, error) {
	可执行文件, 参数 := 固定.可执行文件.实际, 固定.命令
	if 固定.边界.沙箱内 {
		可执行文件, 参数 = 固定.沙箱可执行文件.实际, 固定.启动参数
	}
	进程标识, 错误 := 系统调用包.派生子进程进程执行(可执行文件, 参数, &系统调用包.进程属性{目录: 固定.目录.实际, 环境变量: 固定.环境, 文件集: []uintptr{标准输入.文件描述符(), 标准输出.文件描述符(), 标准错误输出.文件描述符()}, 系统: &系统调用包.系统进程属性{设置进程组: true}})
	if 错误 != nil {
		return nil, 错误
	}
	return &进程{进程标识: 进程标识}, nil
}

func (当前进程 *进程) 轮询() (进程退出, bool, error) {
	if 当前进程.已结束 {
		return 当前进程.退出, true, nil
	}
	var 状态 系统调用包.等待状态
	进程标识, 错误 := 系统调用包.等待4(当前进程.进程标识, &状态, 系统调用包.不阻塞等待, nil)
	if 错误判断包.是否属于(错误, 系统调用包.被信号中断错误) {
		return 进程退出{}, false, nil
	}
	if 错误 != nil {
		return 进程退出{}, false, 错误
	}
	if 进程标识 == 0 {
		return 进程退出{}, false, nil
	}
	当前进程.已结束, 当前进程.退出.代码 = true, 状态.退出状态()
	if 状态.被信号终止() {
		当前进程.退出.系统信号 = 状态.系统信号().字符串()
	}
	return 当前进程.退出, true, nil
}

// 停止 只作用于本次独立进程组，不使用进程名或遍历全局进程表。
// 主动 setsid/设置进程组 脱离的后代不在此边界内；这不是操作系统沙箱。
func (当前进程 *进程) 停止() error {
	return 进程组包.停止(当前进程.进程标识, 当前进程.已结束, 清理超时)
}

func (*进程) close() error { return nil }

func 文件身份(_ *操作系统包.文件, 信息 操作系统包.文件信息) (string, error) {
	状态信息, 成功 := 信息.系统().(*系统调用包.文件状态结构)
	if !成功 {
		return "", 工具包.不可用错误
	}
	return 格式化包.格式化字符串("%d:%d", 状态信息.开发, 状态信息.文件节点号), nil
}
