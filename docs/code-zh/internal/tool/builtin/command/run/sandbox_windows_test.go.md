func 准备Windows沙箱(测试器 *测试包.测试器, 根, 数据 string, 命令 []string, 写入 bool) *工具调用 {
	测试器.标记测试辅助函数()
	甲 := 参数{命令: 命令, 工作目录: 根}
	if 写入 {
		甲.沙箱权限, 甲.理由 = 沙箱项目写入, "验证临时项目的原生隔离"
	}
	原始, 错误 := JSON包.序列化(甲)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	实例 := 新建(操作系统包.环境变量集(), 工具包.受保护数据{目录: 数据}).(*命令工具)
	调用, 错误 := 实例.准备带有权限(测试器.上下文(), 工具包.范围{标识: "windows-fixture", 根路径: 根}, 原始, 工具包.执行权限{沙箱模式: 工具包.沙箱工作区写入})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	当前对象 := 调用.(*工具调用)
	测试器.注册清理函数(func() {
		if 错误 := 当前对象.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	if _, 错误 := 操作系统包.状态信息(当前对象.固定.临时路径); !错误判断包.是否属于(错误, 操作系统包.不存在错误) {
		测试器.报告致命错误("审批准备创建了运行资源")
	}
	return 当前对象
}

func Test应用容器文件网络并且子级边界集(测试器 *测试包.测试器) {
	辅助 := 构建辅助(测试器)
	根, 外部, 数据 := 测试器.创建临时目录(), 测试器.创建临时目录(), 测试器.创建临时目录()

		测试器.报告致命错误(错误)
	}
	for _, 测试用例 := range []struct {
		名称, 动作, 路径 string
		写入, 成功     bool
	}{
		{"read-only-write", "write", 文件路径包.合并(根, "blocked"), false, false},
		{"project-write", "write", 文件路径包.合并(根, "allowed"), true, true},
		{"outside-write", "write", 文件路径包.合并(外部, "blocked"), true, false},
		{"child-outside-write", "spawn-write", 文件路径包.合并(外部, "child-blocked"), true, false},
		{"git-write", "write", 文件路径包.合并(根, ".git", "blocked"), true, false},
		{"codex-write", "write", 文件路径包.合并(根, ".codex", "blocked"), true, false},
		{"agents-write", "write", 文件路径包.合并(根, ".agents", "blocked"), true, false},
		{"aws-write", "write", 文件路径包.合并(根, ".aws", "blocked"), true, false},
		{"app-data-read", "read", 密钥, false, false},
		{"app-data-write", "write", 密钥, true, false},
		{"outside-acl", "relax-acl", 外部, true, false},
		{"git-acl", "relax-acl", 文件路径包.合并(根, ".git"), true, false},
	} {
		测试器.执行(测试用例.名称, func(测试器 *测试包.测试器) {
			当前对象 := 准备Windows沙箱(测试器, 根, 数据, []string{辅助, 测试用例.动作, 测试用例.路径}, 测试用例.写入)
			预期 := 策略包.允许
			if 测试用例.写入 {
				预期 = 策略包.询问
			}
			if 决定 := 策略包.评估命令(策略包.开启请求, 当前对象.命令边界()); 决定.动作 != 预期 {
				测试器.格式化报告致命错误("审批判定错误: %+v", 决定)
			}
			结果, 错误 := 当前对象.执行(测试器.上下文())
			if 错误 != nil {
				测试器.格式化报告致命错误("原生沙箱执行失败: %v %s", 错误, 结果.输出)
			}
			报告 := 解码报告(测试器, 结果)
			if 报告.退出码 == nil || (*报告.退出码 == 0) != 测试用例.成功 {
				测试器.格式化报告致命错误("沙箱边界错误: %s", 结果.输出)
			}
			if !报告.边界.沙箱内 || !报告.清理完成 {
				测试器.格式化报告致命错误("沙箱或清理状态错误: %s", 结果.输出)
			}
		})
	}
	if 内容, 错误 := 操作系统包.读取文件(密钥); 错误 != nil || string(内容) != "private" {
		测试器.格式化报告致命错误("应用数据被修改: %q %v", 内容, 错误)
	}
	监听器, 错误 := 网络包.监听("tcp", "127.0.0.1:0")
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		if 错误 := 监听器.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	})
	当前对象 := 准备Windows沙箱(测试器, 根, 数据, []string{辅助, "connect", 监听器.地址().字符串()}, false)
	结果, 错误 := 当前对象.执行(测试器.上下文())
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 报告 := 解码报告(测试器, 结果); 报告.退出码 == nil || *报告.退出码 == 0 {
		测试器.格式化报告致命错误("沙箱允许连接回环网络: %s", 结果.输出)
	}
	if 错误 := 监听器.(*网络包.TCP监听器).设置截止时间(时间包.当前时间().添加(50 * 时间包.毫秒)); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 连接, 错误 := 监听器.接受(); 错误 == nil {
		if 错误 := 连接.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
		测试器.报告致命错误("受限连接到达本机监听器")
	}
}

	for _, 当前对象 := range []*工具调用{首个, 秒} {

			测试器.报告致命错误(错误)
		}
	}

	if 错误 := 操作系统包.移除(路径); 错误 != nil {
		测试器.报告致命错误(错误)

			测试器.格式化报告致命错误("其他执行的 SID 被清除: %s", 目标)
