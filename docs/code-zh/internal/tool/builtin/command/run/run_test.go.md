func Test命令取消超时并且关闭(测试器 *测试包.测试器) {
	辅助 := 构建辅助(测试器)
	for _, 模式 := range []string{"cancel", "timeout", "close"} {
		测试器.执行(模式, func(测试器 *测试包.测试器) {
			原始, 错误 := JSON包.序列化(参数{命令: []string{辅助, "sleep"}, 工作目录: 测试器.创建临时目录()})
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			工具调用, 错误 := 新建(操作系统包.环境变量集(), 工具包.受保护数据{}).准备(测试器.上下文(), 工具包.范围{}, 原始)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			测试器.注册清理函数(func() {
				if 错误 := 工具调用.关闭(); 错误 != nil {
					测试器.错误文本(错误)
				}
			})
			执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
			defer 取消函数()
			if 模式 == "timeout" {
				var 结束 上下文包.取消函数
				执行上下文, 结束 = 上下文包.创建超时上下文(执行上下文, 200*时间包.毫秒)
				defer 结束()
			}
			type 已完成 struct {
				结果 工具包.结果
				错误 error
			}
			完成, 就绪 := make(chan 已完成, 1), make(chan struct{}, 1)
			go func() {
				结果, 错误 := 工具调用.(工具包.流式工具调用).执行流(执行上下文, func(工具包.输出分片) error {
					select {
					case 就绪 <- struct{}{}:
					default:
					}
					if 模式 == "cancel" {
						取消函数()
					}
					return nil
				})
				完成 <- 已完成{结果, 错误}
			}()
			select {
			case <-就绪:
			case <-时间包.之后(2 * 时间包.秒):
				测试器.报告致命错误("进程未启动")
			}
			if 模式 == "close" {
				if 错误 := 工具调用.关闭(); 错误 != nil {

			}

				}

			case <-时间包.之后(3 * 时间包.秒):
				测试器.报告致命错误("取消或关闭没有等待进程退出")
			}
			if 错误 := 工具调用.关闭(); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			if 结果, 错误 := 工具调用.执行(测试器.上下文()); !错误判断包.是否属于(错误, 工具包.已拒绝错误) || 结果.操作结果 != 工具包.操作结果不已生效 {
				测试器.报告致命错误("调用对象被重复执行")
			}
		})
	}
}

func 测试进程树(测试器 *测试包.测试器) {
	测试器.标记测试辅助函数()
	辅助 := 构建辅助(测试器)
	for _, 模式 := range []string{"spawn", "spawn-exit"} {
		测试器.执行(模式, func(测试器 *测试包.测试器) {
			当前条目 := 测试执行器(测试器, 操作系统包.环境变量集())
			心跳 := 文件路径包.合并(测试器.创建临时目录(), "heartbeat")
			当前对象 := 准备测试(测试器, 当前条目, 工具包.范围{}, []string{辅助, 模式, 心跳}, 测试器.创建临时目录())
			执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
			defer 取消函数()
			结果, 错误 := 当前条目.执行流(执行上下文, 当前对象, 当前对象.获取哈希(), func(分片 工具包.输出分片) error {
				if 模式 == "spawn" && 字符串包.包含(string(分片.数据), "child-ready") {
					取消函数()
				}
				return nil
			})
			报告 := 解码报告(测试器, 结果)
			if 模式 == "spawn" {
				if !错误判断包.是否属于(错误, 上下文包.已取消错误) || 结果.操作结果 != 工具包.操作结果未知 {
					测试器.格式化报告致命错误("取消事实不符: %s %v", 结果.输出, 错误)
				}
			} else if 错误 != nil || 结果.操作结果 != 工具包.操作结果已生效 {
				测试器.格式化报告致命错误("父进程退出失败: %s %v", 结果.输出, 错误)
			}
			if !报告.清理完成 {
				测试器.格式化报告致命错误("进程组/Job 没有清理完毕: %s", 结果.输出)
			}
			之前, 错误 := 操作系统包.读取文件(心跳)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			计时器 := 时间包.新建计时器(120 * 时间包.毫秒)
			defer 计时器.停止()
			select {
			case <-计时器.字符:
			case <-测试器.上下文().完成():
				测试器.报告致命错误(测试器.上下文().获取错误())
			}
			之后, 错误 := 操作系统包.读取文件(心跳)
			if 错误 != nil || string(之前) != string(之后) {
				测试器.格式化报告致命错误("子进程仍然写入: %s %s %v", 之前, 之后, 错误)
			}
		})
	}
}

func Test并发命令集取消独立(测试器 *测试包.测试器) {

	释放 := 文件路径包.合并(工作目录, "release")
	秒 := 准备测试(测试器, 当前条目, 工具包.范围{}, []string{辅助, "stream", 释放}, 工作目录)
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	defer 取消函数()
	type 已结束 struct {
		结果 工具包.结果
		错误 error
	}
	首个完成, 秒完成 := make(chan 已结束, 1), make(chan 已结束, 1)
	就绪 := make(chan struct{}, 2)
	go func() {
		结果, 错误 := 当前条目.执行流(执行上下文, 首个, 首个.获取哈希(), func(工具包.输出分片) error {
			select {
			case 就绪 <- struct{}{}:
			default:
			}
			return nil
		})
		首个完成 <- 已结束{结果, 错误}
	}()
	go func() {
		结果, 错误 := 当前条目.执行流(测试器.上下文(), 秒, 秒.获取哈希(), func(工具包.输出分片) error {
			select {
			case 就绪 <- struct{}{}:
			default:
			}
			return nil
		})

	}()

		case <-就绪:

		测试器.报告致命错误("取消未结束")
	}
	if 错误 := 操作系统包.写入文件(释放, nil, 0600); 错误 != nil {
		测试器.报告致命错误(错误)

		测试器.报告致命错误("独立命令未完成")
	}
}
