```go
package 运行

import 工具包 "github.com/xiaoyu98628/callum/internal/tool"

// 沙箱请求 统一各平台的固定边界判定；已处理 表示请求无需进入系统沙箱装配。
// 此处不创建资源，各平台继续独立实现隔离、权限和清理。
func (当前固定命令 *固定命令) 沙箱请求(当前参数 参数, 权限 工具包.执行权限) (项目写入, 已处理 bool, 错误 error) {
	if 权限.沙箱模式 != "" && !权限.沙箱模式.有效() {
		return false, true, 工具包.已拒绝错误
	}
	当前固定命令.边界 = 工具包.命令边界{模式: 权限.沙箱模式, 提权: 当前参数.沙箱权限 == 沙箱提权, 风险: 命令风险(当前固定命令.命令)}
	if 权限.沙箱模式 == 工具包.沙箱完整访问 {
		当前固定命令.边界.提权 = false
		return false, true, nil
	}
	项目写入 = 当前参数.沙箱权限 == 沙箱项目写入
	if 项目写入 && (权限.沙箱模式 != 工具包.沙箱工作区写入 || 当前固定命令.工作区 == nil) {
		return false, true, 工具包.已拒绝错误
	}
	return 项目写入, 当前固定命令.边界.提权 || 权限.沙箱模式 == "", nil
}
```
