```go
package 运行

import (
	上下文包 "context"
	JSON包 "encoding/json"
	错误判断包 "errors"
	切片包 "slices"
	同步包 "sync"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

type 工具调用 struct {
	固定           *固定命令
	互斥锁              同步包.互斥锁
	已开始, 已关闭 bool
	取消函数          上下文包.取消函数
	完成            chan struct{}
	关闭一次       同步包.一次
	关闭错误        error
}

func (*工具调用) 获取目标() *工具包.目标                    { return nil }
func (当前工具调用 *工具调用) 快照() JSON包.原始JSON消息             { return 切片包.复制(当前工具调用.固定.快照) }
func (当前工具调用 *工具调用) 命令边界() 工具包.命令边界 { return 当前工具调用.固定.边界 }
func (当前工具调用 *工具调用) 执行(执行上下文 上下文包.上下文) (工具包.结果, error) {
	return 当前工具调用.执行流(执行上下文, nil)
}

func (当前工具调用 *工具调用) 执行流(执行上下文 上下文包.上下文, 发出事件 func(工具包.输出分片) error) (工具包.结果, error) {
	当前工具调用.互斥锁.加锁()
	if 当前工具调用.已开始 || 当前工具调用.已关闭 {
		当前工具调用.互斥锁.解锁()
		return 工具包.结果{操作结果: 工具包.操作结果不已生效}, 工具包.已拒绝错误
	}
	当前工具调用.已开始 = true
	执行执行上下文, 取消函数 := 上下文包.创建可取消上下文(执行上下文)
	当前工具调用.取消函数, 当前工具调用.完成 = 取消函数, make(chan struct{})
	当前工具调用.互斥锁.解锁()
	defer func() { 取消函数(); close(当前工具调用.完成) }()
	if 错误 := 当前工具调用.固定.检查(执行执行上下文); 错误 != nil {
		return 工具包.结果{操作结果: 工具包.操作结果不已生效}, 错误
	}
	清理, 错误 := 当前工具调用.固定.初始化执行(执行执行上下文)
	if 错误 != nil {
		操作结果 := 工具包.操作结果不已生效
		if 错误判断包.是否属于(错误, 工具包.未知操作结果错误) {
			操作结果 = 工具包.操作结果未知
		}
		return 工具包.结果{操作结果: 操作结果}, 错误
	}
	报告, 错误 := 执行进程(执行执行上下文, 当前工具调用.固定, 发出事件)
	if 清理错误 := 报告.清理错误("execution_cleanup", 清理()); 清理错误 != nil {
		错误 = 错误判断包.合并(错误, 清理错误)
	}
	输出, 序列化错误 := JSON包.序列化(报告)
	操作结果 := 工具包.操作结果不已生效
	if 报告.已启动 {
		操作结果 = 工具包.操作结果未知
		if 错误 == nil && 报告.清理完成 && 报告.退出码 != nil && 报告.系统信号 == "" {
			// 已生效 只表示本次命令已执行至正常退出；业务成功由退出码和输出表达。
			操作结果 = 工具包.操作结果已生效
		}
	}
	return 工具包.结果{输出: 输出, 操作结果: 操作结果}, 错误判断包.合并(错误, 序列化错误)
}

func (当前工具调用 *工具调用) 关闭() error {
	当前工具调用.互斥锁.加锁()
	当前工具调用.已关闭 = true
	if 当前工具调用.取消函数 != nil {
		当前工具调用.取消函数()
	}
	完成 := 当前工具调用.完成
	当前工具调用.互斥锁.解锁()
	if 完成 != nil {
		<-完成
	}
	当前工具调用.关闭一次.执行(func() { 当前工具调用.关闭错误 = 当前工具调用.固定.close() })
	return 当前工具调用.关闭错误
}
```
