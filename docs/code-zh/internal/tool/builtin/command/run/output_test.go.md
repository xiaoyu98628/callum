package 运行

import (
	字节包 "bytes"
	基准64包 "encoding/base64"
	JSON包 "encoding/json"
	错误判断包 "errors"
	字符串包 "strings"
	测试包 "testing"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

func Test输出编码并且已转义大小(测试器 *测试包.测试器) {
	for _, 数据 := range [][]byte{[]byte(字符串包.重复("\x00\"\\", 10000)), []byte(字符串包.重复("中文", 10000)), 字节包.重复([]byte{255}, 20000)} {
		缓冲区 := 输出缓冲区{数据: 数据[:min(len(数据), 已保留字节)], 总计: int64(len(数据))}
		值 := 缓冲区.结果()
		原始, 错误 := JSON包.序列化(值)
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		if len(原始) > 已编码流字节 || !值.已截取 || 值.字节 != int64(len(数据)) {
			测试器.格式化报告致命错误("编码后越界或丢失截断标识: size=%d value=%+v", len(原始), 值)
		}
		if 值.编码 == "base64" {
			已解码, 错误 := 基准64包.标准编码.解码字符串(值.数据)
			if 错误 != nil || !字节包.存在前缀(数据, 已解码) {
				测试器.报告致命错误("二进制输出损坏")
			}
		} else if !字符串包.存在前缀(string(数据), 值.数据) {
			测试器.报告致命错误("文本前缀改变")
		}
	}
}

func Test输出事件数量并且失败保持有界(测试器 *测试包.测试器) {
	调用集 := 0
	收集器 := &捕获{发出事件: func(工具包.输出分片) error { 调用集++; return nil }, 取消函数: func() {}}
	for range 工具包.最大输出事件集 + 10 {
		收集器.写入("stdout", []byte("x"))
	}
	if 调用集 != 工具包.最大输出事件集 || !收集器.流已截取 || 收集器.标准输出.总计 != int64(工具包.最大输出事件集+10) {
		测试器.报告致命错误("碎片事件未限制或未继续排空")
	}
	失败 := 错误判断包.新建("emit failure")
	已取消 := false
	收集器 = &捕获{发出事件: func(工具包.输出分片) error { return 失败 }, 取消函数: func() { 已取消 = true }}
	收集器.写入("stderr", []byte("first"))
	收集器.写入("stderr", []byte("second"))
	if !已取消 || !错误判断包.是否属于(收集器.失败(), 失败) || string(收集器.标准错误输出.数据) != "firstsecond" {
		测试器.报告致命错误("发布失败没有取消或丢失已读取内容")
	}
}
