package 运行

import (
	上下文包 "context"
	错误判断包 "errors"
	输入输出包 "io"
	操作系统包 "os"
	系统调用包 "syscall"
	时间包 "time"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

const 清理超时 = 2 * 时间包.秒

type 进程报告 struct {
	边界     工具包.命令边界 `json:"execution_boundary"`
	已启动    bool     `json:"started"`
	进程标识   *int     `json:"pid"`
	退出码    *int     `json:"exit_code"`
	系统信号   string   `json:"signal,omitempty"`
	已超时    bool     `json:"timed_out"`
	已取消    bool     `json:"cancelled"`
	清理完成   bool     `json:"cleanup_complete"`
	清理错误判断 []清理失败   `json:"cleanup_errors,omitempty"`
	标准输出   输出数据     `json:"stdout"`
	标准错误输出 输出数据     `json:"stderr"`
	流已截取   bool     `json:"stream_truncated"`
}

type 清理失败 struct {
	阶段   string `json:"stage"`
	类别   string `json:"kind"`
	系统代码 uint64 `json:"system_code,omitempty"`
}

// 清理错误 只保留固定阶段、错误类别及系统码，避免将路径、命令或环境写入诊断。
// 原始错误仍沿调用链返回；每次资源收尾只记录对应步骤，条目数量由固定资源数量限定。
func (当前进程报告 *进程报告) 清理错误(阶段 string, 错误 error) error {
	if 错误 == nil {
		return nil
	}
	失败 := 清理失败{阶段: 阶段, 类别: "unknown"}
	if 代码, 成功 := 错误判断包.按类型匹配[系统调用包.系统错误码](错误); 成功 {
		失败.类别, 失败.系统代码 = "system", uint64(代码)
	} else {
		switch {
		case 错误判断包.是否属于(错误, 上下文包.超出截止时间错误):
			失败.类别 = "timeout"
		case 错误判断包.是否属于(错误, 上下文包.已取消错误):
			失败.类别 = "cancelled"
		case 错误判断包.是否属于(错误, 操作系统包.已关闭错误):
			失败.类别 = "closed"
		case 错误判断包.是否属于(错误, 工具包.不可用错误):
			失败.类别 = "unavailable"
		}
	}
	当前进程报告.清理完成 = false
	当前进程报告.清理错误判断 = append(当前进程报告.清理错误判断, 失败)
	return 错误
}

type 进程退出 struct {
	代码   int
	系统信号 string
}

func 执行进程(执行上下文 上下文包.上下文, 固定 *固定命令, 发出事件 func(工具包.输出分片) error) (报告 进程报告, 返回错误 error) {
	报告.边界 = 固定.边界
	执行执行上下文, 取消函数 := 上下文包.创建可取消上下文(执行上下文)
	defer 取消函数()
	输出 := &捕获{发出事件: 发出事件, 取消函数: 取消函数}
	defer func() { 报告.标准输出, 报告.标准错误输出, 报告.流已截取 = 输出.结果集() }()
	var 文件集 []*操作系统包.文件
	defer func() {
		var 关闭错误 error
		阶段集 := [...]string{"stdin_close", "stdout_close", "stderr_close"}
		for 索引, 文件 := range 文件集 {
			关闭错误 = 错误判断包.合并(关闭错误, 报告.清理错误(阶段集[索引], 文件.关闭()))
		}
		返回错误 = 错误判断包.合并(返回错误, 关闭错误)
	}()
	标准输入, 错误 := 操作系统包.打开(操作系统包.开发空)
	if 错误 != nil {
		return 报告, 错误判断包.合并(工具包.不可用错误, 错误)
	}
	文件集 = append(文件集, 标准输入)
	标准输出, 标准输出写入器, 错误 := 操作系统包.管道()
	if 错误 != nil {
		return 报告, 错误判断包.合并(工具包.不可用错误, 错误)
	}
	文件集 = append(文件集, 标准输出)
	标准错误输出, 标准错误输出写入器, 错误 := 操作系统包.管道()
	if 错误 != nil {
		return 报告, 错误判断包.合并(工具包.不可用错误, 错误, 报告.清理错误("stdout_writer_close", 标准输出写入器.关闭()))
	}
	文件集 = append(文件集, 标准错误输出)
	if 错误 := 固定.检查(执行上下文); 错误 != nil {
		return 报告, 错误判断包.合并(错误, 报告.清理错误("stdout_writer_close", 标准输出写入器.关闭()), 报告.清理错误("stderr_writer_close", 标准错误输出写入器.关闭()))
	}
	进程, 开始错误 := 开始进程(固定, 标准输入, 标准输出写入器, 标准错误输出写入器)
	// 仅子进程持有写端；父进程必须及时关闭，否则读取者永远等不到 文件结束。
	关闭写入器集 := 错误判断包.合并(报告.清理错误("stdout_writer_close", 标准输出写入器.关闭()), 报告.清理错误("stderr_writer_close", 标准错误输出写入器.关闭()))
	if 进程 == nil {
		return 报告, 错误判断包.合并(工具包.不可用错误, 开始错误, 关闭写入器集)
	}
	报告.已启动 = true
	报告.进程标识 = new(进程.获取进程标识())
	defer func() {
		关闭错误 := 报告.清理错误("process_close", 进程.close())
		返回错误 = 错误判断包.合并(返回错误, 关闭错误)
	}()
	type 读取结果 struct {
		阶段 string
		错误 error
	}
	读取器集 := make(chan 读取结果, 2)
	go func() { 读取器集 <- 读取结果{"stdout_read", 排空(标准输出, "stdout", 输出)} }()
	go func() { 读取器集 <- 读取结果{"stderr_read", 排空(标准错误输出, "stderr", 输出)} }()
	// 两个读取任务只等待本次管道；终止进程后仍不结束时关闭读端并等待它们返回。
	退出, 等待错误 := 等待进程(执行执行上下文, 进程)
	停止错误 := 报告.清理错误("process_stop", 进程.停止())
	返回错误 = 错误判断包.合并(开始错误, 关闭写入器集, 等待错误, 停止错误)
	报告.清理错误("process_start", 开始错误)
	if 等待错误 != nil {
		// 调用方取消、超时或输出回调失败均结束本次进程组/作业，再有界等待主进程。
		清理, 结束 := 上下文包.创建超时上下文(上下文包.无取消(执行上下文), 清理超时)
		退出, 错误 = 等待进程(清理, 进程)
		结束()
		返回错误 = 错误判断包.合并(返回错误, 错误)
		报告.清理错误("process_wait", 错误)
		if 错误 == nil {
			报告.退出码, 报告.系统信号 = new(退出.代码), 退出.系统信号
		}
	} else {
		报告.退出码, 报告.系统信号 = new(退出.代码), 退出.系统信号
	}
	if 输出错误 := 输出.失败(); 输出错误 != nil {
		// 内部取消只是停止进程的手段，不将发布失败伪装成用户取消。
		返回错误 = 错误判断包.合并(输出错误, 开始错误, 关闭写入器集, 停止错误, 错误)
	}
	报告.已超时 = 错误判断包.是否属于(执行上下文.获取错误(), 上下文包.超出截止时间错误)
	报告.已取消 = 错误判断包.是否属于(执行上下文.获取错误(), 上下文包.已取消错误)
	if 退出.系统信号 != "" && 返回错误 == nil {
		返回错误 = 工具包.未知操作结果错误
	}
	计时器 := 时间包.新建计时器(清理超时)
	defer 计时器.停止()
	for 剩余 := 2; 剩余 > 0; {
		select {
		case 读取 := <-读取器集:
			返回错误 = 错误判断包.合并(返回错误, 报告.清理错误(读取.阶段, 读取.错误))
			剩余--
		case <-计时器.字符:
			// 后代保留管道或主动脱离进程组时，不让读取任务拖住运行时关闭。
			返回错误 = 错误判断包.合并(返回错误, 工具包.不可用错误)
			报告.清理错误("output_wait", 上下文包.超出截止时间错误)
			返回错误 = 错误判断包.合并(返回错误, 报告.清理错误("stdout_close", 标准输出.关闭()), 报告.清理错误("stderr_close", 标准错误输出.关闭()))
			文件集 = 文件集[:1] // 读端已经关闭，由此处收回所有权，避免重复关闭掩盖错误。
			for ; 剩余 > 0; 剩余-- {
				读取 := <-读取器集
				返回错误 = 错误判断包.合并(返回错误, 报告.清理错误(读取.阶段, 读取.错误))
			}
		}
	}
	报告.清理完成 = 报告.退出码 != nil && len(报告.清理错误判断) == 0
	return 报告, 返回错误
}

func 等待进程(执行上下文 上下文包.上下文, 进程 *进程) (进程退出, error) {
	周期计时器 := 时间包.新建周期计时器(10 * 时间包.毫秒)
	defer 周期计时器.停止()
	for {
		退出, 完成, 错误 := 进程.轮询()
		if 错误 != nil || 完成 {
			return 退出, 错误
		}
		select {
		case <-执行上下文.完成():
			return 进程退出{}, 执行上下文.获取错误()
		case <-周期计时器.字符:
		}
	}
}

func 排空(文件 *操作系统包.文件, 流 string, 输出 *捕获) error {
	缓冲区 := make([]byte, 工具包.最大输出分片字节)
	for {
		数量, 错误 := 文件.读取(缓冲区)
		if 数量 > 0 {
			输出.写入(流, 缓冲区[:数量])
		}
		if 错误判断包.是否属于(错误, 输入输出包.文件结束) {
			return nil
		}
		if 错误 != nil {
			return 错误判断包.合并(工具包.不可用错误, 错误)
		}
	}
}
