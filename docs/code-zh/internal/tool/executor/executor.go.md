```go
package 执行器

import (
	字节包 "bytes"
	上下文包 "context"
	SHA256包 "crypto/sha256"
	十六进制包 "encoding/hex"
	JSON包 "encoding/json"
	错误判断包 "errors"
	文件路径包 "path/filepath"
	切片包 "slices"
	同步包 "sync"
	时间包 "time"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
	注册表包 "github.com/xiaoyu98628/callum/internal/tool/registry"
)

// 操作 是服务端生成的审批快照，与实际执行使用相同的规范化参数。
type 操作 struct {
	工具      工具包.定义 `json:"tool"`
	工作区 *工具包.范围     `json:"workspace"`
	目标    *工具包.目标    `json:"target,omitempty"`
	计划      *工具包.计划      `json:"plan,omitempty"`
	详情   JSON包.原始JSON消息 `json:"details,omitempty"`
	参数 JSON包.原始JSON消息 `json:"arguments"`
	摘要   string          `json:"summary"`
}

// 已准备调用 持有固定操作和调用资源，创建者必须 关闭；等待审批期间不能替换操作。
type 已准备调用 struct {
	工具调用 工具包.工具调用
	目标     *工具包.目标
	实例   工具包.工具
	定义 工具包.定义
	计划       *工具包.计划
	详情    JSON包.原始JSON消息
	影响集    []工具包.影响
	请求    JSON包.原始JSON消息
	哈希       string
	关闭一次  同步包.一次
	关闭错误   error
	边界   *工具包.命令边界
}

func (当前已准备调用 *已准备调用) 获取请求() JSON包.原始JSON消息 { return 切片包.复制(当前已准备调用.请求) }
func (当前已准备调用 *已准备调用) 获取哈希() string             { return 当前已准备调用.哈希 }
func (当前已准备调用 *已准备调用) 命令边界() *工具包.命令边界 {
	if 当前已准备调用.边界 == nil {
		return nil
	}
	return new(*当前已准备调用.边界)
}
func (当前已准备调用 *已准备调用) 获取定义() 工具包.定义 {
	丁 := 当前已准备调用.定义
	丁.参数 = 切片包.复制(丁.参数)
	丁.影响集 = 切片包.复制(当前已准备调用.影响集)
	return 丁
}
func (当前已准备调用 *已准备调用) 外部工作区() bool {
	if 当前已准备调用.目标 != nil && !当前已准备调用.目标.范围内工作区 {
		return true
	}
	if 当前已准备调用.计划 != nil {
		for _, 目标 := range 当前已准备调用.计划.目标集 {
			if !目标.范围内工作区 {
				return true
			}
		}
	}
	return false
}
func (当前已准备调用 *已准备调用) 关闭() error {
	当前已准备调用.关闭一次.执行(func() { 当前已准备调用.关闭错误 = 当前已准备调用.工具调用.关闭() })
	return 当前已准备调用.关闭错误
}

// 执行器 在审批前校验参数并固定调用，具体资源由工具持有，不强制要求工作区。
type 执行器 struct {
	注册表 *注册表包.注册表
	超时  时间包.时长
}

func 新建(当前注册表 *注册表包.注册表, 超时 时间包.时长) (*执行器, error) {
	if 当前注册表 == nil || 超时 <= 0 {
		return nil, 工具包.无效错误
	}
	return &执行器{注册表包: 当前注册表, 超时: 超时}, nil
}

func (当前执行器 *执行器) 定义集() []工具包.定义 { return 当前执行器.注册表.定义集() }

func (当前执行器 *执行器) 准备(执行上下文 上下文包.上下文, 范围 工具包.范围, 名称 string, 参数 JSON包.原始JSON消息) (*已准备调用, error) {
	return 当前执行器.准备带有权限(执行上下文, 范围, 名称, 参数, 工具包.执行权限{})
}

func (当前执行器 *执行器) 准备带有权限(执行上下文 上下文包.上下文, 范围 工具包.范围, 名称 string, 参数 JSON包.原始JSON消息, 权限 工具包.执行权限) (*已准备调用, error) {
	return 当前执行器.准备模块(执行上下文, 范围, 名称, nil, 参数, 权限)
}

// 准备外部 用本轮固定的工具实例，避免通过同名目录重定向已获批请求。
func (当前执行器 *执行器) 准备外部(执行上下文 上下文包.上下文, 范围 工具包.范围, 当前工具 工具包.工具, 参数 JSON包.原始JSON消息, 权限 工具包.执行权限) (*已准备调用, error) {
	if 当前工具 == nil || 当前工具.获取定义().来源 != "mcp" {
		return nil, 工具包.无效错误
	}
	return 当前执行器.准备模块(执行上下文, 范围, "", 当前工具, 参数, 权限)
}

func (当前执行器 *执行器) 准备模块(执行上下文 上下文包.上下文, 范围 工具包.范围, 名称 string, 当前工具 工具包.工具, 参数 JSON包.原始JSON消息, 权限 工具包.执行权限) (*已准备调用, error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return nil, 错误
	}
	if 范围 != (工具包.范围{}) && (范围.标识 == "" || !文件路径包.是否绝对路径(范围.根路径)) {
		return nil, 工具包.已拒绝错误
	}
	if len(参数) > 工具包.最大参数字节 {
		return nil, 工具包.超出上限错误
	}
	if 当前工具 == nil {
		var 错误 error
		当前工具, 错误 = 当前执行器.注册表.获取(名称)
		if 错误 != nil {
			return nil, 错误
		}
	}
	已规范化, 错误 := 当前工具.校验(参数)
	if 错误 != nil {
		return nil, 错误
	}
	var 工具调用 工具包.工具调用
	if 感知, 成功 := 当前工具.(工具包.权限工具); 成功 {
		工具调用, 错误 = 感知.准备带有权限(执行上下文, 范围, 已规范化, 权限)
	} else {
		工具调用, 错误 = 当前工具.准备(执行上下文, 范围, 已规范化)
	}
	if 错误 != nil {
		if 工具调用 != nil {
			错误 = 错误判断包.合并(错误, 工具调用.关闭())
		}
		return nil, 错误
	}
	if 工具调用 == nil {
		return nil, 工具包.无效错误
	}
	var 工作区 *工具包.范围
	if 范围.标识 != "" {
		工作区 = new(范围)
	}
	目标 := 工具调用.获取目标()
	if 目标 != nil {
		目标 = new(*目标)
	}
	定义 := 当前工具.获取定义()
	定义.参数 = 切片包.复制(定义.参数)
	定义.影响集 = 切片包.复制(定义.影响集)
	计划, 错误 := 工具调用计划(工具调用)
	if 错误 != nil {
		return nil, 错误判断包.合并(错误, 工具调用.关闭())
	}
	详情, 错误 := 工具调用快照(工具调用)
	if 错误 != nil {
		return nil, 错误判断包.合并(错误, 工具调用.关闭())
	}
	影响集 := 切片包.复制(定义.影响集)
	if 计划 != nil {
		// 未分类工具不能借额外的 写入/读取 声明降低原有审批要求。
		if len(影响集) == 0 {
			影响集 = append(影响集, 工具包.影响未知)
		}
		for _, 影响 := range 计划.附加影响集 {
			if !影响.有效() {
				return nil, 错误判断包.合并(工具包.无效错误, 工具调用.关闭())
			}
			if !切片包.包含(影响集, 影响) {
				影响集 = append(影响集, 影响)
			}
		}
	}
	生效 := 定义
	生效.影响集 = 影响集
	请求, 错误 := JSON包.序列化(操作{工具: 生效, 工作区: 工作区, 目标: 目标, 计划: 计划, 详情: 详情, 参数: 已规范化, 摘要: 定义.说明})
	if 错误 != nil {
		return nil, 错误判断包.合并(错误, 工具调用.关闭())
	}
	哈希 := SHA256包.校验和256(请求)
	var 边界 *工具包.命令边界
	if 命令, 成功 := 工具调用.(工具包.命令工具调用); 成功 {
		边界 = new(命令.命令边界())
	}
	return &已准备调用{工具调用: 工具调用, 目标: 目标, 计划: 计划, 详情: 详情, 影响集: 影响集, 实例: 当前工具, 定义: 定义, 请求: 请求, 哈希: 十六进制包.编码目标字符串(哈希[:]), 边界: 边界}, nil
}

func 工具调用快照(工具调用 工具包.工具调用) (JSON包.原始JSON消息, error) {
	详细, 成功 := 工具调用.(工具包.快照工具调用)
	if !成功 {
		return nil, nil
	}
	原始 := 详细.快照()
	var 值 map[string]JSON包.原始JSON消息
	if len(原始) > 工具包.最大参数字节 || JSON包.反序列化(原始, &值) != nil || len(值) == 0 {
		return nil, 工具包.无效错误
	}
	// 归一化对象顺序并持有独立副本；执行前比较同一规范化结果。
	return JSON包.序列化(值)
}

func 工具调用计划(工具调用 工具包.工具调用) (*工具包.计划, error) {
	已规划, 成功 := 工具调用.(工具包.已规划工具调用)
	if !成功 {
		return nil, nil
	}
	计划 := 已规划.获取计划()
	计划.目标集 = 切片包.复制(计划.目标集)
	计划.附加影响集 = 切片包.复制(计划.附加影响集)
	if len(计划.目标集) > 32 {
		return nil, 工具包.超出上限错误
	}
	for _, 目标 := range 计划.目标集 {
		if !文件路径包.是否绝对路径(目标.路径) || !文件路径包.是否绝对路径(目标.请求的路径) || 目标.类别 == "" {
			return nil, 工具包.无效错误
		}
	}
	return &计划, nil
}

// 执行 的 已批准哈希 来自权限判定或已批准记录，工具结果必须是有界有效 JSON。
func (当前执行器 *执行器) 执行(执行上下文 上下文包.上下文, 当前已准备调用 *已准备调用, 已批准哈希 string) (工具包.结果, error) {
	return 当前执行器.执行流(执行上下文, 当前已准备调用, 已批准哈希, nil)
}

// 执行流 沿用普通工具的权限边界，并串行转交输出；旧工具无需实现流式接口。
func (当前执行器 *执行器) 执行流(执行上下文 上下文包.上下文, 当前已准备调用 *已准备调用, 已批准哈希 string, 发出事件 func(工具包.输出分片) error) (工具包.结果, error) {
	不已生效 := 工具包.结果{操作结果: 工具包.操作结果不已生效}
	if 当前已准备调用 == nil || 当前已准备调用.哈希 != 已批准哈希 {
		return 不已生效, 工具包.已拒绝错误
	}
	目标 := 当前已准备调用.工具调用.获取目标()
	if (目标 == nil) != (当前已准备调用.目标 == nil) || (目标 != nil && *目标 != *当前已准备调用.目标) {
		return 不已生效, 工具包.已拒绝错误
	}
	计划, 错误 := 工具调用计划(当前已准备调用.工具调用)
	if 错误 != nil {
		return 不已生效, 错误
	}
	if (计划 == nil) != (当前已准备调用.计划 == nil) || (计划 != nil && (!切片包.相等(计划.目标集, 当前已准备调用.计划.目标集) || !切片包.相等(计划.附加影响集, 当前已准备调用.计划.附加影响集))) {
		return 不已生效, 工具包.已拒绝错误
	}
	详情, 错误 := 工具调用快照(当前已准备调用.工具调用)
	if 错误 != nil {
		return 不已生效, 错误
	}
	if !字节包.相等(详情, 当前已准备调用.详情) {
		return 不已生效, 工具包.已拒绝错误
	}
	if 当前已准备调用.边界 != nil {
		命令, 成功 := 当前已准备调用.工具调用.(工具包.命令工具调用)
		if !成功 || 命令.命令边界() != *当前已准备调用.边界 {
			return 不已生效, 工具包.已拒绝错误
		}
	}
	定义 := 当前已准备调用.实例.获取定义()
	最新, 错误 := JSON包.序列化(定义)
	if 错误 != nil {
		return 不已生效, 错误
	}
	原始, 错误 := JSON包.序列化(当前已准备调用.定义)
	if 错误 != nil {
		return 不已生效, 错误
	}
	if string(最新) != string(原始) {
		return 不已生效, 工具包.已拒绝错误
	}
	调用, 取消函数 := 上下文包.创建超时上下文(执行上下文, 当前执行器.超时)
	defer 取消函数()
	if 错误 := 调用.获取错误(); 错误 != nil {
		return 不已生效, 错误
	}
	var 结果 工具包.结果
	var 流错误 error
	var 输出互斥锁 同步包.互斥锁
	var 数量, 输出字节 int
	if 流式, 成功 := 当前已准备调用.工具调用.(工具包.流式工具调用); 成功 && 发出事件 != nil {
		结果, 错误 = 流式.执行流(调用, func(分片 工具包.输出分片) error {
			输出互斥锁.加锁()
			defer 输出互斥锁.解锁()
			if 流错误 != nil {
				return 流错误
			}
			if 调用.获取错误() != nil {
				流错误 = 调用.获取错误()
			} else if (分片.流 != "stdout" && 分片.流 != "stderr") || len(分片.数据) == 0 || len(分片.数据) > 工具包.最大输出分片字节 {
				流错误 = 工具包.无效错误
			} else if 数量 >= 工具包.最大输出事件集 || 输出字节+len(分片.数据) > 工具包.最大输出字节 {
				流错误 = 工具包.超出上限错误
			} else {
				数量++
				输出字节 += len(分片.数据)
				分片.序号 = int64(数量)
				流错误 = 发出事件(分片)
			}
			return 流错误
		})
	} else {
		结果, 错误 = 当前已准备调用.工具调用.执行(调用)
	}
	错误 = 错误判断包.合并(错误, 流错误)
	if !结果.操作结果.有效() {
		结果.操作结果 = 工具包.未确认操作结果(当前已准备调用.影响集)
		错误 = 错误判断包.合并(错误, 工具包.无效错误)
	}
	错误 = 错误判断包.合并(错误, 调用.获取错误())
	if 错误 == nil && 结果.操作结果 == 工具包.操作结果未知 {
		错误 = 工具包.未知操作结果错误
	} else if 错误 == nil && 结果.操作结果 == 工具包.操作结果部分 {
		错误 = 工具包.部分操作结果错误
	}
	// 输出不可保存时仍保留独立事实，不能把已发生的操作改成未执行。
	if len(结果.输出) > 工具包.最大结果字节 {
		结果.输出 = nil
		错误 = 错误判断包.合并(错误, 工具包.超出上限错误)
	} else if (结果.输出 != nil || 错误 == nil) && !JSON包.有效(结果.输出) {
		结果.输出 = nil
		错误 = 错误判断包.合并(错误, 工具包.无效错误)
	}
	if len(结果.图片集) > 8 {
		结果.图片集 = nil
		错误 = 错误判断包.合并(错误, 工具包.超出上限错误)
	}
	if len(结果.图片集) > 0 {
		元数据, 元数据错误 := JSON包.序列化(结果.图片集)
		if 元数据错误 != nil || len(元数据)+len(结果.输出) > 工具包.最大结果字节 {
			结果.图片集 = nil
			错误 = 错误判断包.合并(错误, 工具包.超出上限错误, 元数据错误)
		}
	}
	for _, 内容块 := range 结果.图片集 {
		if 内容块.类型 != "image" || 内容块.附件引用 == "" || 内容块.路径 != "" {
			结果.图片集 = nil
			错误 = 错误判断包.合并(错误, 工具包.无效错误)
			break
		}
	}
	return 结果, 错误
}
```
