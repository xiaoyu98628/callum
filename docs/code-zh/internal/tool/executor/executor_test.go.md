func Test操作影响集为已冻结之前审批(测试器 *测试包.测试器) {
	实例 := &可修改工具{定义: 工具包.定义{
		工具定义: 模型包.工具定义{名称: "operation", 说明: "测试工具", 参数: JSON包.原始JSON消息(`{"type":"object"}`)},
		来源:         "test", 版本: "1", 影响集: []工具包.影响{工具包.影响破坏性},
	}}
	当前读取器 := 注册表包.新建()
	if 错误 := 当前读取器.注册(实例); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	// 返回值不能借用注册工具的可变类别切片。
	已列出 := 当前读取器.定义集()
	已列出[0].影响集[0] = 工具包.影响读取
	if 实例.定义.影响集[0] != 工具包.影响破坏性 {
		测试器.报告致命错误("查询结果修改了注册工具权限")
	}
	当前条目, 错误 := 执行器包.新建(当前读取器, 时间包.秒)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	当前对象, 错误 := 当前条目.准备(测试器.上下文(), 工具包.范围{标识: "test-workspace", 根路径: 测试器.创建临时目录()}, "operation", JSON包.原始JSON消息(`{}`))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer func() {
		if 错误 := 当前对象.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
	}()
	定义 := 当前对象.获取定义()
	定义.影响集[0] = 工具包.影响读取
	if 当前对象.获取定义().影响集[0] != 工具包.影响破坏性 {
		测试器.报告致命错误("调用方修改了固定操作的类别")
	}
	// 注册工具在等待审批时改变类别，不能覆盖快照，也不能继续执行。
	实例.定义.影响集[0] = 工具包.影响读取
	if 当前对象.获取定义().影响集[0] != 工具包.影响破坏性 {
		测试器.报告致命错误("工具变化修改了已经固定的操作")
	}
	if _, 错误 := 当前条目.执行(测试器.上下文(), 当前对象, 当前对象.获取哈希()); !错误判断包.是否属于(错误, 工具包.已拒绝错误) || 实例.已执行 {
		测试器.格式化报告致命错误("定义变化后仍执行工具: %v", 错误)
	}
}
