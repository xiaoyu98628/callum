```go
package 工具测试

import (
	文件路径包 "path/filepath"
	测试包 "testing"

	工具包 "github.com/xiaoyu98628/callum/internal/tool"
	策略包 "github.com/xiaoyu98628/callum/internal/tool/policy"
)

func Test目录范围边界集(测试器 *测试包.测试器) {
	根 := 测试器.创建临时目录()
	读取 := 工具包.目录范围{根路径: 根, 附件访问器: "read"}
	写入 := 工具包.目录范围{根路径: 根, 附件访问器: "read_write"}
	for _, 测试 := range []struct {
		名称  string
		范围  工具包.目录范围
		目标  工具包.目标
		影响集 []工具包.影响
		预期  bool
	}{
		{"descendant", 读取, 工具包.目标{路径: 文件路径包.合并(根, "a", "b.txt"), 类别: "file"}, []工具包.影响{工具包.影响读取}, true},
		{"same_directory", 读取, 工具包.目标{路径: 根, 类别: "directory"}, []工具包.影响{工具包.影响读取}, true},
		{"prefix_sibling", 读取, 工具包.目标{路径: 根 + "-other", 类别: "directory"}, []工具包.影响{工具包.影响读取}, false},
		{"parent", 读取, 工具包.目标{路径: 文件路径包.目录(根), 类别: "directory"}, []工具包.影响{工具包.影响读取}, false},
		{"read_does_not_write", 读取, 工具包.目标{路径: 文件路径包.合并(根, "a"), 类别: "file"}, []工具包.影响{工具包.影响写入}, false},
		{"write_includes_read", 写入, 工具包.目标{路径: 文件路径包.合并(根, "a"), 类别: "file"}, []工具包.影响{工具包.影响读取}, true},
		{"ordinary_write", 写入, 工具包.目标{路径: 文件路径包.合并(根, "a"), 类别: "file"}, []工具包.影响{工具包.影响写入}, true},
		{"destructive", 写入, 工具包.目标{路径: 文件路径包.合并(根, "a"), 类别: "file"}, []工具包.影响{工具包.影响读取, 工具包.影响破坏性}, false},
		{"unknown", 写入, 工具包.目标{路径: 文件路径包.合并(根, "a"), 类别: "file"}, []工具包.影响{工具包.影响未知}, false},
		{"external", 写入, 工具包.目标{路径: 文件路径包.合并(根, "a"), 类别: "file"}, []工具包.影响{工具包.影响外部}, false},
		{"unclassified", 写入, 工具包.目标{路径: 文件路径包.合并(根, "a"), 类别: "file"}, nil, false},
		{"auxiliary_outside", 写入, 工具包.目标{路径: 文件路径包.合并(根, "a"), 类别: "file", 恢复路径: 文件路径包.合并(文件路径包.目录(根), "backup")}, []工具包.影响{工具包.影响写入}, false},
	} {
		测试器.执行(测试.名称, func(测试器 *测试包.测试器) {
			if 实际值 := 策略包.目录覆盖(测试.范围, 测试.目标, 测试.影响集); 实际值 != 测试.预期 {
				测试器.格式化报告致命错误("目录范围错误: got=%v want=%v", 实际值, 测试.预期)
			}
		})
	}
	目标集 := []工具包.目标{{路径: 文件路径包.合并(根, "a"), 类别: "file"}, {路径: 根 + "-other", 类别: "file"}}
	if 策略包.目录集覆盖([]工具包.目录范围{读取}, 目标集, []工具包.影响{工具包.影响读取}) {
		测试器.报告致命错误("多目标操作只覆盖一部分即被放行")
	}
	if 策略包.目录集覆盖([]工具包.目录范围{写入}, nil, []工具包.影响{工具包.影响读取}) {
		测试器.报告致命错误("无文件目标操作继承了目录权限")
	}
}
```
