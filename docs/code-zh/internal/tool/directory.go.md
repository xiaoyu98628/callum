```go
package 工具

import 文件路径包 "path/filepath"

// 目录范围 是用户可明确授予的真实目录范围，读写授权包含读取。
// 它授权路径范围；每次操作仍须固定并复核实际目标，不授权命令或其他影响。
type 目录范围 struct {
	根路径   string `json:"root_path"`
	附件访问器 string `json:"access"`
}

// 有效 检查服务端规范化的绝对路径与受支持的授权类别。
func (当前目录范围 目录范围) 有效() bool {
	return 文件路径包.是否绝对路径(当前目录范围.根路径) && 文件路径包.清理(当前目录范围.根路径) == 当前目录范围.根路径 && (当前目录范围.附件访问器 == "read" || 当前目录范围.附件访问器 == "read_write")
}
```
