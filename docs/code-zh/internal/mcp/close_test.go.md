```go
package MCP

import (
	字节包 "bytes"
	上下文包 "context"
	JSON包 "encoding/json"
	错误判断包 "errors"
	格式化包 "fmt"
	结构化日志包 "log/slog"
	字符串包 "strings"
	测试包 "testing"

	SDK包 "github.com/modelcontextprotocol/go-sdk/mcp"
	设置包 "github.com/xiaoyu98628/callum/internal/settings"
)

type 关闭错误传输 struct {
	SDK包.传输
	错误 error
}

func (当前关闭错误传输 关闭错误传输) 连接(执行上下文 上下文包.上下文) (SDK包.连接, error) {
	连接, 错误 := 当前关闭错误传输.传输.连接(执行上下文)
	if 错误 != nil {
		return nil, 错误
	}
	return 关闭错误连接{连接: 连接, 错误: 当前关闭错误传输.错误}, nil
}

type 关闭错误连接 struct {
	SDK包.连接
	错误 error
}

func (当前关闭错误连接 关闭错误连接) 关闭() error {
	return 错误判断包.合并(当前关闭错误连接.连接.关闭(), 当前关闭错误连接.错误)
}

func Test关闭连接报告仅非预期失败集(测试器 *测试包.测试器) {
	const 私有 = "受控第三方错误原文"
	for _, 测试用例 := range []struct {
		名称 string
		错误 error
		类别 string
		退役 bool
	}{
		{名称: "正常关闭"},
		{名称: "预期取消", 错误: 格式化包.格式化创建错误("传输结束: %w", 上下文包.已取消错误), 退役: true},
		{名称: "旧会话失效", 错误: 错误判断包.合并(变化通知错误, SDK包.会话连接缺失错误)},
		{名称: "重连关闭失败", 错误: 错误判断包.新建(私有), 类别: "close_failed"},
		{名称: "退出清理失败", 错误: 格式化包.格式化创建错误("清理: %w", 错误判断包.合并(上下文包.已取消错误, 错误判断包.新建(私有))), 类别: "close_failed", 退役: true},
		{名称: "关闭超时", 错误: 上下文包.超出截止时间错误, 类别: "close_timeout", 退役: true},
	} {
		测试器.执行(测试用例.名称, func(测试器 *测试包.测试器) {
			执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
			defer 取消函数()
			客户端传输, 服务端传输 := SDK包.新建内存传输集()
			服务端 := SDK包.新建服务端(&SDK包.实现{名称: "fixture", 版本: "1"}, nil)
			通信对端, 错误 := 服务端.连接(执行上下文, 服务端传输, nil)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			defer func() {
				if 错误 := 通信对端.关闭(); 错误 != nil {
					测试器.错误文本(错误)
				}
			}()
			客户端 := SDK包.新建客户端(&SDK包.实现{名称: "fixture", 版本: "1"}, nil)
			会话连接, 错误 := 客户端.连接(执行上下文, 关闭错误传输{传输: 客户端传输, 错误: 测试用例.错误}, nil)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}

			var 日志集 字节包.缓冲区
			当前状态 := &服务{执行上下文: 执行上下文, 取消函数: 取消函数, 条目集: make(map[string]*条目),
				日志记录器: 结构化日志包.新建(结构化日志包.新建JSON处理器(&日志集, nil)).附加字段("module", "mcp")}
			const 服务端标识 = "fixture-server"
			当前状态.添加(设置包.MCP服务端{标识: 服务端标识})
			当前条目 := 当前状态.条目集[服务端标识]
			当前条目.连接 = &连接{会话连接: 会话连接}
			当前条目.公开连接.存储(当前条目.连接)
			if !测试用例.退役 {
				if 错误 := 当前条目.获取(执行上下文); 错误 != nil {
					测试器.报告致命错误(错误)
				}
				当前状态.关闭连接(当前条目)
				当前条目.释放()
			}
			当前状态.关闭()
			当前状态.关闭()

			if 当前条目.连接 != nil || 当前条目.公开连接.加载() != nil {
				测试器.报告致命错误("关闭后仍保留旧连接")
			}
			if 测试用例.类别 == "" {
				if 日志集.长度() != 0 {
					测试器.格式化报告致命错误("预期关闭产生了诊断: %s", 日志集.字符串())
				}
				return
			}
			if 字符串包.包含(日志集.字符串(), 私有) {
				测试器.报告致命错误("诊断包含第三方错误原文")
			}
			var 条目 struct {
				级别    string `json:"level"`
				模块    string `json:"module"`
				服务端标识 string `json:"server_id"`
				类别    string `json:"kind"`
			}
			if 错误 := JSON包.反序列化(日志集.字节(), &条目); 错误 != nil {
				测试器.格式化报告致命错误("关闭失败应产生一条结构化诊断: %v", 错误)
			}
			if 条目.级别 != "WARN" || 条目.模块 != "mcp" || 条目.服务端标识 != 服务端标识 || 条目.类别 != 测试用例.类别 {
				测试器.格式化报告致命错误("关闭诊断字段错误: %+v", 条目)
			}
		})
	}
}
```
