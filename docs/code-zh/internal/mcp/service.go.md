// 凭据 只在内部传递，由仓储在写入前加密。
type 凭据 struct {
	词元    string            `json:"token,omitempty"`
	刷新  string            `json:"refresh,omitempty"`
	过期时间  时间包.时间点         `json:"expires,omitzero"`
	客户端标识 string            `json:"client_id,omitempty"`
	词元URL string            `json:"token_url,omitempty"`
	资源 string            `json:"resource,omitempty"`
	环境变量      map[string]string `json:"env,omitempty"`
}

func (当前凭据 凭据) 空() bool {
	return 当前凭据.词元 == "" && len(当前凭据.环境变量) == 0
}

type 仓储 interface {
	加载(上下文包.上下文) (设置包.MCP, int64, error)
	保存(上下文包.上下文, 设置包.MCP, int64, string, *凭据) error
	凭据(上下文包.上下文, string) (凭据, error)
	保存凭据(上下文包.上下文, string, 凭据) error
}

type 条目 struct {

	启动配置               设置包.MCP服务端

	执行上下文        上下文包.上下文
	取消函数     上下文包.取消函数
	加锁       chan struct{}
	词元门控  chan struct{}
	连接 *连接    // 仅持有 加锁 时访问。
	流程       *授权 // 由 服务.互斥锁 保护。
}

func (当前条目 *条目) 获取(执行上下文 上下文包.上下文) error {
	select {
	case 当前条目.加锁 <- struct{}{}:
		if 错误 := 当前条目.执行上下文.获取错误(); 错误 != nil {
			当前条目.释放()
			return 变化通知错误
		}
		return nil
	case <-当前条目.执行上下文.完成():
		return 变化通知错误
	case <-执行上下文.完成():
		return 执行上下文.获取错误()
	}
}
func (当前条目 *条目) 释放() { <-当前条目.加锁 }

// 服务 由 Go 运行时持有。连接代际在保存、禁用和删除后立即取消。
type 服务 struct {
	执行上下文       上下文包.上下文
	取消函数    上下文包.取消函数
	仓储      仓储
	HTTP      *HTTP包.客户端
	OAuthHTTP *HTTP包.客户端
	日志记录器    *结构化日志包.日志记录器
	互斥锁        同步包.互斥锁
	等待组        同步包.等待组
	启动配置       设置包.MCP
	修订号  int64

	条目集 map[string]*条目
	已关闭  bool
}

// 新建 加载保存的配置并创建运行期服务，使用调用方注入的日志记录器。
func 新建(初始化执行上下文, 执行上下文 上下文包.上下文, 仓储 仓储, 客户端, OAuth客户端 *HTTP包.客户端, 日志记录器 *结构化日志包.日志记录器) (*服务, error) {
	if 日志记录器 == nil {
		return nil, 错误判断包.新建("MCP 服务日志未配置")
	}
	启动配置, 修订号, 错误 := 仓储.加载(初始化执行上下文)
	if 错误 != nil {
		return nil, 错误
	}
	启动配置 = 启动配置.复制()
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(执行上下文)
	当前状态 := &服务{执行上下文: 执行上下文, 取消函数: 取消函数, 仓储: 仓储, HTTP包: 客户端, OAuthHTTP: OAuth客户端,
		日志记录器: 日志记录器.附加字段("module", "mcp"), 启动配置: 启动配置, 修订号: 修订号, 条目集: make(map[string]*条目)}
	for _, 项 := range 启动配置.服务集 {
		当前状态.添加(项)
	}
	return 当前状态, nil
}

func (当前服务 *服务) 添加(启动配置 设置包.MCP服务端) {
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(当前服务.执行上下文)
	当前服务.条目集[启动配置.标识] = &条目{启动配置: 启动配置, 执行上下文: 执行上下文, 取消函数: 取消函数, 加锁: make(chan struct{}, 1), 词元门控: make(chan struct{}, 1)}
}

func (当前服务 *服务) 获取(标识 string) (*条目, error) {
	当前服务.互斥锁.加锁()
	defer 当前服务.互斥锁.解锁()
	if 当前服务.已关闭 {
		return nil, 连接错误
	}
	当前条目 := 当前服务.条目集[标识]
	if 当前条目 == nil {
		return nil, 未找到错误
	}
	return 当前条目, nil
}

func (当前服务 *服务) 退役(当前条目 *条目) {
	当前条目.取消函数()
	if 当前条目.流程 != nil {
		当前条目.流程.取消函数()
	}
	当前服务.等待组.Go(func() {
		当前条目.加锁 <- struct{}{}
		defer 当前条目.释放()
		当前服务.关闭连接(当前条目)
	})
}

// 关闭 停止受理请求并等待本服务持有的会话和授权任务结束，可重复调用。
func (当前服务 *服务) 关闭() {
	当前服务.互斥锁.加锁()
	if !当前服务.已关闭 {
		当前服务.已关闭 = true
		当前服务.取消函数()
		for _, 当前条目 := range 当前服务.条目集 {
			当前服务.退役(当前条目)
		}
	}
	当前服务.互斥锁.解锁()
	当前服务.等待组.等待()
}

func (当前服务 *服务) 列表(执行上下文 上下文包.上下文) (目录管理, error) {
	当前服务.互斥锁.加锁()
	修订号 := 当前服务.修订号
	条目集 := make([]*条目, 0, len(当前服务.条目集))
	for _, 启动配置 := range 当前服务.启动配置.服务集 {
		条目集 = append(条目集, 当前服务.条目集[启动配置.标识])
	}
	当前服务.互斥锁.解锁()
	输出 := 目录管理{修订号: 修订号, 服务集: make([]查看, 0, len(条目集)), 范围更新: true}
	for _, 当前条目 := range 条目集 {
		密钥, 错误 := 当前服务.仓储.凭据(执行上下文, 当前条目.启动配置.标识)
		if 错误 != nil {
			return 目录管理{}, 错误
		}
		查看 := 查看{MCP服务端: 当前条目.启动配置.复制(), 已授权: 密钥.词元 != "" && (密钥.过期时间.是否零值() || 时间包.当前时间().之前(密钥.过期时间) || 密钥.刷新 != ""), 密钥环境变量键集: []string{}, 工具集: []工具信息{}}
		if !密钥.过期时间.是否零值() {
			查看.过期时间 = 密钥.过期时间.Unix毫秒()
		}
		查看.密钥环境变量键集 = append(查看.密钥环境变量键集, 切片包.已排序(映射包.键集(密钥.环境变量))...)
		if 连接 := 当前条目.公开连接.加载(); 连接 != nil && 当前条目.执行上下文.获取错误() == nil && 连接.凭据版本 == 当前条目.凭据版本.加载() {
			查看.已连接 = true
			if 目录管理 := 连接.目录管理.加载(); 目录管理 != nil && !连接.脏.加载() {
				查看.工具集 = 切片包.复制(目录管理.公开)
			}
		}
		输出.服务集 = append(输出.服务集, 查看)
	}
	return 输出, nil
}

func (当前服务 *服务) 保存(执行上下文 上下文包.上下文, 修订号 int64, 启动配置 设置包.MCP服务端, 密钥集 密钥输入, 选项集 保存选项集) (string, error) {
	启动配置 = 启动配置.复制()
	当前服务.互斥锁.加锁()
	defer 当前服务.互斥锁.解锁()
	if 当前服务.已关闭 {
		return "", 连接错误
	}
	if 修订号 != 当前服务.修订号 {
		return "", 设置包.冲突错误
	}
	旧 := 当前服务.条目集[启动配置.标识]
	if 启动配置.标识 == "" {
		标识, 错误 := UUID包.新建UUIDv7()
		if 错误 != nil {
			return "", 错误
		}
		启动配置.标识 = 标识.字符串()
	} else if 旧 == nil {
		return "", 未找到错误
	}
	if 启动配置.允许工具集 == nil {
		启动配置.允许工具集 = []string{}
	}
	if 选项集.清空范围 && (启动配置.范围 != "" || 启动配置.传输 != 设置包.MCPHTTP || 启动配置.认证 != 设置包.MCPOAuth) {
		return "", 设置包.无效错误
	}
	if !选项集.清空范围 && 启动配置.范围 == "" && 旧 != nil && 旧.启动配置.范围 != "" {
		前一个 := 旧.启动配置
		前一个.范围 = ""
		if 相同身份(前一个, 启动配置) {
			启动配置.范围 = 旧.启动配置.范围
		}
	}
	if !启动配置.有效() {
		return "", 设置包.无效错误
	}
	if 启动配置.范围 != "" {
		启动配置.范围, _ = 设置包.规范化MCPOAuth范围(启动配置.范围) // 有效 已校验权限范围。
	}
	var 凭据 *凭据
	变化通知 := 旧 != nil && !相同身份(旧.启动配置, 启动配置)
	if 启动配置.传输 == 设置包.MCP标准输入输出 && 启动配置.已启用 && (旧 == nil || !旧.启动配置.已启用 || 变化通知 || 密钥集.环境变量 != nil) && !选项集.已确认 {
		return "", 设置包.无效错误
	}
	if 变化通知 {
		启动配置.允许工具集 = []string{}
	}
	if 变化通知 || 密钥集.词元 != nil || 密钥集.环境变量 != nil {
		值 := 凭据{}
		if 旧 != nil && !变化通知 {
			var 错误 error
			值, 错误 = 当前服务.仓储.凭据(执行上下文, 启动配置.标识)
			if 错误 != nil {
				return "", 错误
			}
		}
		if 密钥集.词元 != nil {
			if 启动配置.认证 != 设置包.MCP词元 || len(*密钥集.词元) > 8192 || 字符串包.包含任一字符(*密钥集.词元, "\x00\r\n \t") {
				return "", 设置包.无效错误
			}
			值.词元 = *密钥集.词元
		}
		if 密钥集.环境变量 != nil {
			if 启动配置.传输 != 设置包.MCP标准输入输出 || len(*密钥集.环境变量) > 64 {
				return "", 设置包.无效错误
			}
			for 键, 值 := range *密钥集.环境变量 {
				_, 重复 := 启动配置.环境变量[键]
				if 重复 || !设置包.有效MCP环境变量(键, 值) {
					return "", 设置包.无效错误
				}
			}
			值.环境变量 = 映射包.复制(*密钥集.环境变量)
		}
		凭据 = &值
	}
	if 旧 != nil && 凭据 == nil && 启动配置.传输 == 设置包.MCP标准输入输出 {
		值, 错误 := 当前服务.仓储.凭据(执行上下文, 启动配置.标识)
		if 错误 != nil {
			return "", 错误
		}
		for 键 := range 启动配置.环境变量 {
			if _, 成功 := 值.环境变量[键]; 成功 {
				return "", 设置包.无效错误
			}
		}
	}
	下一个 := 设置包.MCP{服务集: 切片包.复制(当前服务.启动配置.服务集)}
	if 旧 == nil {
		下一个.服务集 = append(下一个.服务集, 启动配置)
	} else {
		索引 := 切片包.索引函数(下一个.服务集, func(当前MCP服务端 设置包.MCP服务端) bool { return 当前MCP服务端.标识 == 启动配置.标识 })
		下一个.服务集[索引] = 启动配置
	}
	if 错误 := 当前服务.仓储.保存(执行上下文, 下一个.复制(), 修订号, 启动配置.标识, 凭据); 错误 != nil {
		return "", 错误
	}
	当前服务.启动配置, 当前服务.修订号 = 下一个, 修订号+1
	if 旧 != nil {
		当前服务.退役(旧)
	}
	当前服务.添加(启动配置)
	return 启动配置.标识, nil
}

func 相同身份(当前MCP服务端, 当前MCP服务端 设置包.MCP服务端) bool {
	当前MCP服务端.标识, 当前MCP服务端.名称, 当前MCP服务端.已启用, 当前MCP服务端.允许工具集 = "", "", false, nil
	当前MCP服务端.标识, 当前MCP服务端.名称, 当前MCP服务端.已启用, 当前MCP服务端.允许工具集 = "", "", false, nil
	甲甲, _ := JSON包.序列化(当前MCP服务端)
	乙乙, _ := JSON包.序列化(当前MCP服务端)
	return string(甲甲) == string(乙乙)
}

func (当前服务 *服务) 删除(执行上下文 上下文包.上下文, 修订号 int64, 标识 string) error {
	当前服务.互斥锁.加锁()
	defer 当前服务.互斥锁.解锁()
	if 当前服务.已关闭 {
		return 连接错误
	}
	if 修订号 != 当前服务.修订号 {
		return 设置包.冲突错误
	}
	当前条目 := 当前服务.条目集[标识]
	if 当前条目 == nil {
		return 未找到错误
	}
	下一个 := 设置包.MCP{服务集: 切片包.删除函数(切片包.复制(当前服务.启动配置.服务集), func(当前MCP服务端 设置包.MCP服务端) bool { return 当前MCP服务端.标识 == 标识 })}
	if 错误 := 当前服务.仓储.保存(执行上下文, 下一个.复制(), 修订号, 标识, &凭据{}); 错误 != nil {
		return 错误
	}
	当前服务.启动配置, 当前服务.修订号 = 下一个, 修订号+1
	delete(当前服务.条目集, 标识)
	当前服务.退役(当前条目)
	return nil
}

func (当前服务 *服务) 断开连接(执行上下文 上下文包.上下文, 标识 string) error {
	当前服务.互斥锁.加锁()
	defer 当前服务.互斥锁.解锁()
	if 当前服务.已关闭 {
		return 连接错误
	}
	当前条目 := 当前服务.条目集[标识]
	if 当前条目 == nil {
		return 未找到错误
	}
	if 当前条目.启动配置.传输 != 设置包.MCPHTTP || 当前条目.启动配置.认证 == 设置包.MCP无 {
		return 设置包.无效错误
	}
	if 错误 := 当前服务.仓储.保存凭据(执行上下文, 标识, 凭据{}); 错误 != nil {
		return 错误
	}
	当前服务.退役(当前条目)
	当前服务.添加(当前条目.启动配置)
	return nil
}

func (当前服务 *服务) 测试(执行上下文 上下文包.上下文, 标识 string, 已确认 bool, 修订号 int64) ([]工具信息, error) {
	当前服务.互斥锁.加锁()
	当前条目, 当前 := 当前服务.条目集[标识], 当前服务.修订号
	当前服务.互斥锁.解锁()
	if 当前条目 == nil {
		return nil, 未找到错误
	}
	if 当前条目.启动配置.传输 == 设置包.MCP标准输入输出 {
		if !已确认 {
			return nil, 设置包.无效错误
		}
		if 修订号 != 当前 {
			return nil, 设置包.冲突错误
		}
	}
	if 错误 := 当前条目.获取(执行上下文); 错误 != nil {
		return nil, 错误
	}
	defer 当前条目.释放()
	执行上下文, 取消函数 := 条目上下文(执行上下文, 当前条目)
	defer 取消函数()
	if 当前条目.连接 != nil && 当前条目.启动配置.范围 != "" {
		// 复用会话时提前刷新，以便把 范围 错误直接返回给设置页。
		if _, 错误 := 当前服务.访问词元(执行上下文, 当前条目); 错误 != nil {
			当前服务.关闭连接(当前条目)
			return nil, 错误
		}
	}
	丙, 错误 := 当前服务.连接(执行上下文, 当前条目)
	if 错误 != nil {
		return nil, 错误
	}
	if 错误 := 当前服务.发现(执行上下文, 当前条目, 丙); 错误 != nil {
		当前服务.关闭连接(当前条目)
		return nil, 错误
	}
	输出 := 切片包.复制(丙.目录管理.加载().公开)
	if !当前条目.启动配置.已启用 {
		当前服务.关闭连接(当前条目)
	}
	return 输出, nil
}

func 条目上下文(执行上下文 上下文包.上下文, 当前条目 *条目) (上下文包.上下文, 上下文包.取消函数) {
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(执行上下文)
	停止 := 上下文包.之后函数(当前条目.执行上下文, 取消函数)
	return 执行上下文, func() { 停止(); 取消函数() }
}

// 关闭连接 在持有条目锁时撤销快照并关闭会话，只记录脱敏的清理诊断。
func (当前服务 *服务) 关闭连接(当前条目 *条目) {
	当前条目.公开连接.存储(nil)
	丙 := 当前条目.连接
	当前条目.连接 = nil
	if 丙 == nil {
		return
	}
	if 错误 := 丙.会话连接.关闭(); 非预期关闭错误(错误) {
		类别 := "close_failed"
		if 错误判断包.是否属于(错误, 上下文包.超出截止时间错误) {
			类别 = "close_timeout"
		}
		当前服务.日志记录器.警告("关闭 MCP 会话失败", "server_id", 当前条目.启动配置.标识, "kind", 类别)
	}
}

func 非预期关闭错误(错误 error) bool {
	if 错误 == nil {
		return false
	}
	// 本地进程等传输可能合并清理错误，预期取消不能遮蔽同批次的其他失败。
	if 已合并, 成功 := 错误.(interface{ 解包() []error }); 成功 {
		return 切片包.包含函数(已合并.解包(), 非预期关闭错误)
	}
	if 原因 := 错误判断包.解包(错误); 原因 != nil {
		return 非预期关闭错误(原因)
	}
	return !错误判断包.是否属于(错误, 上下文包.已取消错误) && !错误判断包.是否属于(错误, 变化通知错误) &&
		!错误判断包.是否属于(错误, SDK包.连接已关闭错误) && !错误判断包.是否属于(错误, SDK包.会话连接缺失错误)
}

func (当前服务 *服务) 连接(执行上下文 上下文包.上下文, 当前条目 *条目) (*连接, error) {
	if 当前条目.连接 != nil {
		if 当前条目.连接.凭据版本 == 当前条目.凭据版本.加载() {
			return 当前条目.连接, nil
		}
		当前服务.关闭连接(当前条目)
	}
	丙 := &连接{凭据版本: 当前条目.凭据版本.加载()}
	var 传输模块 SDK包.传输
	switch 当前条目.启动配置.传输 {
	case 设置包.MCP标准输入输出:
		凭据, 错误 := 当前服务.仓储.凭据(执行上下文, 当前条目.启动配置.标识)
		if 错误 != nil {
			return nil, 错误
		}
		传输模块 = &进程传输{启动配置: 当前条目.启动配置, 密钥集: 凭据.环境变量}
	case 设置包.MCPHTTP:
		客户端 := *当前服务.HTTP
		基准 := 客户端.传输
		客户端.传输 = 轮次往返器(func(请求 *HTTP包.请求) (*HTTP包.响应, error) {
			if 当前条目.凭据版本.加载() != 丙.凭据版本 {
				return nil, 变化通知错误
			}
			if 当前条目.启动配置.认证 != 设置包.MCP无 {
				词元, 错误 := 当前服务.访问词元(请求.上下文(), 当前条目)
				if 错误 != nil {
					return nil, 错误
				}
				if 当前条目.凭据版本.加载() != 丙.凭据版本 {
					return nil, 变化通知错误
				}
				请求 = 请求.复制(请求.上下文())
				请求.头部.设置("Authorization", "Bearer "+词元)
			}
			响应, 错误 := 基准.轮次往返(请求)
			if 错误 != nil {
				return nil, 错误
			}
			if 响应.状态代码 == HTTP包.状态未授权 {
				_ = 响应.正文.关闭()
				return nil, 授权错误
			}
			if !字符串包.存在前缀(响应.头部.获取("Content-Type"), "text/event-stream") {
				响应.正文 = 上限正文(响应.正文)
			}
			return 响应, nil
		})
		传输模块 = &SDK包.可流式客户端传输{端点: 当前条目.启动配置.URL, HTTP客户端: &客户端, 最大事件大小: 1 << 20}
	}
	客户端 := SDK包.新建客户端(&SDK包.实现{名称: "Callum", 版本: "1.0"}, &SDK包.客户端选项集{
		工具列表变化通知处理器: func(上下文包.上下文, *SDK包.工具列表变化通知请求) { 丙.脏.存储(true) },
	})
	会话连接, 错误 := 客户端.连接(执行上下文, 传输模块, &SDK包.客户端会话连接选项集{协议版本: "2025-11-25"})
	if 错误 != nil {
		if 错误判断包.是否属于(错误, 授权错误) {
			return nil, 授权错误
		}
		if 错误判断包.是否属于(错误, OAuth范围错误) {
			return nil, OAuth范围错误
		}
		if 错误判断包.是否属于(错误, OAuth范围不匹配错误) {
			return nil, OAuth范围不匹配错误
		}
		return nil, 连接错误
	}
	丙.会话连接 = 会话连接
	当前条目.连接 = 丙
	当前条目.公开连接.存储(丙)
	当前服务.互斥锁.加锁()
	if 当前服务.已关闭 || 当前条目.执行上下文.获取错误() != nil {
		当前服务.互斥锁.解锁()
		当前服务.关闭连接(当前条目)
		return nil, 变化通知错误
	}
	当前服务.等待组.Go(func() {
		// 等待 用于感知连接结束，显式 关闭 读取 SDK 单独保留的关闭错误。
		_ = 会话连接.等待()
		丙.脏.存储(true)
		当前条目.公开连接.比较并且交换(丙, nil)
		select {
		case 当前条目.加锁 <- struct{}{}:
			if 当前条目.连接 == 丙 {
				当前服务.关闭连接(当前条目)
			}
			当前条目.释放()
		case <-当前条目.执行上下文.完成():
		}
	})
	当前服务.互斥锁.解锁()
	return 丙, nil
}

// 进程环境 避免把 Go 服务继承的其他凭据传给第三方进程。
func 进程环境(值集, 密钥集 map[string]string) []string {
	环境变量 := map[string]string{}
	for _, 键 := range []string{"PATH", "HOME", "USERPROFILE", "TMP", "TEMP", "TMPDIR", "SystemRoot", "WINDIR", "APPDATA", "LOCALAPPDATA", "LANG"} {
		if 值, 成功 := 操作系统包.查找环境变量(键); 成功 {
			环境变量[键] = 值
		}
	}
	映射包.复制(环境变量, 值集)
	映射包.复制(环境变量, 密钥集)
	输出 := make([]string, 0, len(环境变量))
	for _, 键 := range 切片包.已排序(映射包.键集(环境变量)) {
		输出 = append(输出, 键+"="+环境变量[键])
	}
	return 输出
}
