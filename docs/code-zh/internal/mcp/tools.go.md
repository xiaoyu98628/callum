package MCP

import (
	字节包 "bytes"
	上下文包 "context"
	SHA256包 "crypto/sha256"
	十六进制包 "encoding/hex"
	JSON包 "encoding/json"
	错误判断包 "errors"
	URL包 "net/url"
	切片包 "slices"
	排序包 "sort"
	字符串包 "strings"
	同步包 "sync"
	UTF8包 "unicode/utf8"

	JSON模式包 "github.com/google/jsonschema-go/jsonschema"
	SDK包 "github.com/modelcontextprotocol/go-sdk/mcp"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	设置包 "github.com/xiaoyu98628/callum/internal/settings"
	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

const 最大工具集每个服务端 = 128
const 最大发现工作线程集 = 4

// 工具目录管理 发布后只读，可以在其他调用占用连接时读取。
type 工具目录管理 struct {
	依据名称 map[string]*远程工具
	公开 []工具信息
}

type 工具信息 struct {
	名称        string `json:"name"`
	说明 string `json:"description"`
}

type 远程工具 struct {
	凭据版本 uint64
	服务           *服务
	条目             *条目
	名称              string
	版本           string
	定义               工具包.定义
	校验器         *JSON模式包.已解析
}

func (当前远程工具 *远程工具) 信息() 工具信息 {
	说明 := 当前远程工具.定义.说明
	// 列表一次返回所有服务，缩短展示文案以保留 RPC 帧余量。
	if len(说明) > 256 {
		结束 := 256
		for !UTF8包.字符开始(说明[结束]) {
			结束--
		}
		说明 = 说明[:结束] + "…"
	}
	return 工具信息{名称: 当前远程工具.名称, 说明: 说明}
}
func (当前远程工具 *远程工具) 获取定义() 工具包.定义 { return 当前远程工具.定义 }

func (当前远程工具 *远程工具) 校验(原始 JSON包.原始JSON消息) (JSON包.原始JSON消息, error) {
	if len(原始) > 工具包.最大参数字节 {
		return nil, 工具包.超出上限错误
	}
	if !JSON包.有效(原始) {
		return nil, 工具包.无效错误
	}
	var 对象 map[string]any
	解码器 := JSON包.新建解码器(字节包.新建读取器(原始))
	解码器.使用数量()
	if 解码器.解码(&对象) != nil || 对象 == nil || 当前远程工具.校验器.校验(对象) != nil {
		return nil, 工具包.无效错误
	}
	return JSON包.序列化(对象)
}

func (当前远程工具 *远程工具) 准备(执行上下文 上下文包.上下文, _ 工具包.范围, 参数 JSON包.原始JSON消息) (工具包.工具调用, error) {
	if 当前远程工具.条目.执行上下文.获取错误() != nil || 当前远程工具.凭据版本 != 当前远程工具.条目.凭据版本.加载() {
		return nil, 工具包.不可用错误
	}
	当前状态 := 当前远程工具.条目.启动配置
	目标 := 当前状态.命令
	if 当前状态.传输 == 设置包.MCPHTTP {
		地址, _ := URL包.解析(当前状态.URL)
		地址.原始查询 = ""
		目标 = 地址.字符串()
	}
	详情, _ := JSON包.序列化(map[string]any{
		"kind": "mcp", "server_id": 当前状态.标识, "server_name": 当前状态.名称, "tool_name": 当前远程工具.名称,
		"transport": 当前状态.传输, "target": 目标, "definition_version": 当前远程工具.版本,
	})
	return &工具调用{工具包: 当前远程工具, 参数: 参数, 详情: 详情}, nil
}

type 工具调用 struct {
	工具    *远程工具
	参数    JSON包.原始JSON消息
	详情 JSON包.原始JSON消息
}

func (*工具调用) 获取目标() *工具包.目标        { return nil }
func (*工具调用) 关闭() error                { return nil }
func (当前工具调用 *工具调用) 快照() JSON包.原始JSON消息 { return 当前工具调用.详情 }
func (当前工具调用 *工具调用) 执行(执行上下文 上下文包.上下文) (工具包.结果, error) {
	return 当前工具调用.工具.服务.调用(执行上下文, 当前工具调用.工具, 当前工具调用.参数)
}

// 工具集快照 冻结供本轮模型选择的工具。预览只读取已发现的定义。
func (当前服务 *服务) 工具集快照(执行上下文 上下文包.上下文, 连接 bool, 修订号 int64) []工具包.工具 {
	当前服务.互斥锁.加锁()
	if 当前服务.修订号 != 修订号 || 当前服务.已关闭 {
		当前服务.互斥锁.解锁()
		return nil
	}
	var 条目集 []*条目
	for _, 启动配置 := range 当前服务.启动配置.服务集 {
		if 启动配置.已启用 && len(启动配置.允许工具集) > 0 {
			条目集 = append(条目集, 当前服务.条目集[启动配置.标识])
		}
	}
	当前服务.互斥锁.解锁()
	if 执行上下文.获取错误() != nil {
		return nil
	}
	// 先识别可用目录，再用有界并发发现其余服务。等待某个服务不占用缓存读取。
	待处理 := make(chan *条目, len(条目集))
	for _, 当前条目 := range 条目集 {
		if 连接 && 已缓存目录管理(当前条目) == nil {
			待处理 <- 当前条目
		}
	}
	close(待处理)
	var 工作线程集 同步包.等待组
	for range min(最大发现工作线程集, len(待处理)) {
		工作线程集.Go(func() {
			for 当前条目 := range 待处理 {
				if 执行上下文.获取错误() != nil {
					return
				}
				当前服务.确保工具目录管理(执行上下文, 当前条目)
			}
		})
	}
	工作线程集.等待()

	当前服务.互斥锁.加锁()
	当前 := !当前服务.已关闭 && 当前服务.修订号 == 修订号
	当前服务.互斥锁.解锁()
	if !当前 {
		return nil
	}
	var 输出 []工具包.工具
	for _, 当前条目 := range 条目集 {
		if 目录管理 := 已缓存目录管理(当前条目); 目录管理 != nil {
			for _, 名称 := range 当前条目.启动配置.允许工具集 {
				if 当前项 := 目录管理.依据名称[名称]; 当前项 != nil {
					输出 = append(输出, 当前项)
				}
			}
		}
	}
	排序包.切片(输出, func(甲, 乙 int) bool { return 输出[甲].获取定义().名称 < 输出[乙].获取定义().名称 })
	return 输出
}

func 已缓存目录管理(当前条目 *条目) *工具目录管理 {
	if 丙 := 当前条目.公开连接.加载(); 丙 != nil {
		if 目录管理 := 丙.目录管理.加载(); !丙.脏.加载() && 当前条目.执行上下文.获取错误() == nil && 丙.凭据版本 == 当前条目.凭据版本.加载() {
			return 目录管理
		}
	}
	return nil
}

func (当前服务 *服务) 确保工具目录管理(执行上下文 上下文包.上下文, 当前条目 *条目) {
	if 当前条目.获取(执行上下文) != nil {
		return
	}
	defer 当前条目.释放()
	if 执行上下文.获取错误() != nil || 已缓存目录管理(当前条目) != nil {
		return
	}
	执行上下文, 取消函数 := 条目上下文(执行上下文, 当前条目)
	defer 取消函数()
	丙, 错误 := 当前服务.连接(执行上下文, 当前条目)
	if 错误 == nil && 当前服务.发现(执行上下文, 当前条目, 丙) != nil {
		当前服务.关闭连接(当前条目)
	}
}

func (当前服务 *服务) 发现(执行上下文 上下文包.上下文, 当前条目 *条目, 当前连接 *连接) error {
	当前连接.目录管理.存储(nil)
	当前连接.脏.存储(false)
	下一个 := map[string]*远程工具{}
	if cap := 当前连接.会话连接.初始化结果().能力集; cap == nil || cap.工具集 == nil {
		发布工具集(当前连接, 下一个)
		return nil
	}
	已见 := map[string]bool{}
	游标 := ""
	for range 32 {
		结果, 错误 := 当前连接.会话连接.列表工具集(执行上下文, &SDK包.列表工具集参数{游标: 游标})
		if 错误 != nil {
			return 连接错误
		}
		for _, 当前读取器 := range 结果.工具集 {
			if 当前读取器 == nil || 已见[当前读取器.名称] || len(已见) >= 最大工具集每个服务端 {
				return 连接错误
			}
			已见[当前读取器.名称] = true
			if 当前项 := 当前服务.适配(当前条目, 当前读取器); 当前项 != nil {
				下一个[当前读取器.名称] = 当前项
			}
		}
		if 游标 = 结果.下一个游标; 游标 == "" {
			发布工具集(当前连接, 下一个)
			return nil
		}
	}
	return 连接错误
}

func 发布工具集(当前连接 *连接, 下一个 map[string]*远程工具) {
	工具集 := make([]工具信息, 0, len(下一个))
	for _, 当前项 := range 下一个 {
		工具集 = append(工具集, 当前项.信息())
	}
	切片包.排序函数(工具集, func(当前工具信息, 当前工具信息 工具信息) int { return 字符串包.比较(当前工具信息.名称, 当前工具信息.名称) })
	当前连接.目录管理.存储(&工具目录管理{依据名称: 下一个, 公开: 工具集})
}

func (当前服务 *服务) 适配(当前条目 *条目, 当前工具 *SDK包.工具) *远程工具 {
	if 当前工具.名称 == "" || len(当前工具.名称) > 256 || 字符串包.包含任一字符(当前工具.名称, "\x00\r\n") || !UTF8包.有效字符串(当前工具.名称) {
		return nil
	}
	输入, 错误 := JSON包.序列化(当前工具.输入表结构)
	if 错误 != nil || len(输入) > 工具包.最大参数字节 {
		return nil
	}
	var 表结构 JSON模式包.表结构
	if JSON包.反序列化(输入, &表结构) != nil || 表结构.类型 != "object" {
		return nil
	}
	已解析, 错误 := 表结构.解析(nil)
	if 错误 != nil {
		return nil
	}
	说明 := 字符串包.裁剪空间(当前工具.说明)
	if 说明 == "" {
		说明 = "调用 " + 当前工具.名称
	}
	if len(说明) > 4096 {
		return nil
	}
	指纹 := SHA256包.校验和256(切片包.拼接([]byte(说明), []byte{0}, 输入))
	名称校验和 := SHA256包.校验和256([]byte(当前条目.启动配置.标识 + "\x00" + 当前工具.名称))
	版本 := 十六进制包.编码目标字符串(指纹[:])
	return &远程工具{凭据版本: 当前条目.凭据版本.加载(), 服务: 当前服务, 条目: 当前条目, 名称: 当前工具.名称, 版本: 版本, 校验器: 已解析, 定义: 工具包.定义{
		工具定义: 模型包.工具定义{名称: "mcp_" + 十六进制包.编码目标字符串(名称校验和[:14]), 说明: 当前条目.启动配置.名称 + " · " + 说明, 参数: 输入},
		来源:         "mcp", 版本: 版本, 影响集: []工具包.影响{工具包.影响未知},
	}}
}

func (当前服务 *服务) 调用(执行上下文 上下文包.上下文, 当前远程工具 *远程工具, 参数 JSON包.原始JSON消息) (工具包.结果, error) {
	当前条目 := 当前远程工具.条目
	不已生效 := 工具包.结果{操作结果: 工具包.操作结果不已生效}
	if !当前条目.启动配置.已启用 || !切片包.包含(当前条目.启动配置.允许工具集, 当前远程工具.名称) || 当前条目.凭据版本.加载() != 当前远程工具.凭据版本 {
		return 不已生效, 工具包.已拒绝错误
	}
	if 错误 := 当前条目.获取(执行上下文); 错误 != nil {
		return 不已生效, 错误判断包.合并(工具包.不可用错误, 错误)
	}
	defer 当前条目.释放()
	执行上下文, 取消函数 := 条目上下文(执行上下文, 当前条目)
	defer 取消函数()
	丙, 错误 := 当前服务.连接(执行上下文, 当前条目)
	if 错误 != nil {
		return 不已生效, 工具包.不可用错误
	}
	// 获批后重新获取定义，避免静默更新导致旧参数发往不同工具。
	if 错误 := 当前服务.发现(执行上下文, 当前条目, 丙); 错误 != nil {
		当前服务.关闭连接(当前条目)
		return 不已生效, 工具包.不可用错误
	}
	当前 := 丙.目录管理.加载().依据名称[当前远程工具.名称]
	if 当前 == nil || 当前.版本 != 当前远程工具.版本 || 当前远程工具.凭据版本 != 丙.凭据版本 || 丙.凭据版本 != 当前条目.凭据版本.加载() || 丙.脏.加载() || 执行上下文.获取错误() != nil {
		return 不已生效, 工具包.已拒绝错误
	}
	结果, 错误 := 丙.会话连接.调用工具(执行上下文, &SDK包.调用工具参数{名称: 当前远程工具.名称, 参数: 参数})
	if 错误 != nil {
		return 工具包.结果{操作结果: 工具包.操作结果未知}, 工具包.未知操作结果错误
	}
	输出, 错误 := JSON包.序列化(结果)
	if 错误 != nil {
		return 工具包.结果{操作结果: 工具包.操作结果未知}, 工具包.未知操作结果错误
	}
	if 结果.是否错误 {
		// 工具错误可能发生在副作用之后，保留远端结果并明确待核实。
		return 工具包.结果{输出: 输出, 操作结果: 工具包.操作结果未知}, 工具包.未知操作结果错误
	}
	return 工具包.结果{输出: 输出, 操作结果: 工具包.操作结果已生效}, nil
}
