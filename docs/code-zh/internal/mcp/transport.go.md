type 轮次往返器 func(*HTTP包.请求) (*HTTP包.响应, error)

func (当前轮次往返器 轮次往返器) 轮次往返(当前请求 *HTTP包.请求) (*HTTP包.响应, error) { return 当前轮次往返器(当前请求) }

type 有界正文 struct {
	输入输出包.读取关闭器
	剩余 int
}

func (当前有界正文 *有界正文) 读取(字节数据 []byte) (int, error) {
	if 当前有界正文.剩余 < 0 {
		return 0, 输入输出包.短缓冲区错误
	}
	if len(字节数据) > 当前有界正文.剩余+1 {
		字节数据 = 字节数据[:当前有界正文.剩余+1]
	}
	数量, 错误 := 当前有界正文.读取关闭器.读取(字节数据)
	当前有界正文.剩余 -= 数量
	if 当前有界正文.剩余 < 0 {
		return 0, 错误判断包.新建("MCP 响应过大")
	}
	return 数量, 错误
}
func 上限正文(当前读取关闭器 输入输出包.读取关闭器) 输入输出包.读取关闭器 { return &有界正文{读取关闭器: 当前读取关闭器, 剩余: 1 << 20} }

// HTTP客户端 为持久 SSE 流保留调用方取消，限制连接和响应头阶段。
func HTTP客户端(已发现 bool) (*HTTP包.客户端, *HTTP包.传输) {
	当前项 := HTTP包.默认传输.(*HTTP包.传输).复制()
	当前项.代理 = HTTP包.代理来源环境
	拨号器 := &网络包.拨号器{超时: 5 * 时间包.秒, 保留存活: 30 * 时间包.秒}
	if 已发现 {
		// 授权服务地址由远程元数据提供。校验实际拨号地址，避免 DNS 变化后访问内网。
		当前项.代理 = nil
		拨号器.控制 = func(_, 地址 string, _ 系统调用包.原始连接) error {
			主机, _, 错误 := 网络包.拆分主机端口(地址)
			if 错误 != nil {
				return OAuth错误
			}
			IP, 错误 := 网络IP包.解析地址(主机)
			if 错误 != nil {
				return OAuth错误
			}
			IP = IP.解除映射()
			if !IP.是全局单播() || IP.是否私有地址() || IP.是否回环地址() || 网络IP包.必须解析前缀("100.64.0.0/10").包含(IP) {
				return OAuth错误
			}
			return nil
		}
	}
	当前项.拨号上下文 = 拨号器.拨号上下文
	当前项.TLS握手超时 = 5 * 时间包.秒
	当前项.响应头部超时 = 8 * 时间包.秒
	当前项.最大连接集每个主机 = 24
	当前项.最大空闲连接集 = 32
	return &HTTP包.客户端{传输: 当前项, 检查重定向: func(*HTTP包.请求, []*HTTP包.请求) error { return HTTP包.使用最后响应错误 }}, 当前项
}

type 进程传输 struct {
	启动配置     设置包.MCP服务端
	密钥集 map[string]string
}

func (当前进程传输 *进程传输) 连接(执行上下文 上下文包.上下文) (SDK包.连接, error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return nil, 错误
	}
	命令 := 进程执行包.命令(当前进程传输.启动配置.命令, 当前进程传输.启动配置.启动参数...)
	命令.目录 = 当前进程传输.启动配置.目录
	if 命令.目录 == "" {
		命令.目录 = 操作系统包.创建临时目录()
	}
	命令.环境变量 = 进程环境(当前进程传输.启动配置.环境变量, 当前进程传输.密钥集)
	读取输入, 写入输入, 错误 := 操作系统包.管道()
	if 错误 != nil {
		return nil, 错误
	}
	读取输出, 写入输出, 错误 := 操作系统包.管道()
	if 错误 != nil {
		return nil, 错误判断包.合并(错误, 读取输入.关闭(), 写入输入.关闭())
	}
	命令.标准输入, 命令.标准输出 = 读取输入, 写入输出
	// 第三方 标准错误输出 可包含令牌，沿用 操作系统/进程执行 默认丢弃。
	完成 := make(chan struct{})
	停止, 错误 := 开始进程(命令, 完成)
	父级错误 := 错误判断包.合并(读取输入.关闭(), 写入输出.关闭())
	if 错误 != nil {
		return nil, 错误判断包.合并(错误, 父级错误, 写入输入.关闭(), 读取输出.关闭())
	}
	当前对象 := &子级进程{输入: 写入输入, 输出: 读取输出, 停止: 停止, 完成: 完成}
	go func() { _ = 命令.等待(); close(当前对象.完成) }()
	if 父级错误 != nil {
		return nil, 错误判断包.合并(父级错误, 当前对象.关闭())
	}
	传输模块 := SDK包.输入输出传输{读取器: 当前对象, 写入器: 当前对象, 最大行长度: 1 << 20}
	连接, 错误 := 传输模块.连接(执行上下文)
	if 错误 != nil {
		return nil, 错误判断包.合并(错误, 当前对象.关闭())
	}
	return 连接, nil
}

type 子级进程 struct {
	输入, 输出 *操作系统包.文件
	停止    func() error
	完成    chan struct{}
	一次    同步包.一次
	错误     error
}

const (
	进程退出宽限 = 250 * 时间包.毫秒
	进程停止等待  = 2 * 时间包.秒
)

func (当前子级进程 *子级进程) 读取(字节数据 []byte) (int, error)  { return 当前子级进程.输出.读取(字节数据) }
func (当前子级进程 *子级进程) 写入(字节数据 []byte) (int, error) { return 当前子级进程.输入.写入(字节数据) }
func (当前子级进程 *子级进程) 关闭() error {
	当前子级进程.一次.执行(func() {
		输入错误 := 当前子级进程.输入.关闭()
		select {
		case <-当前子级进程.完成:
		case <-时间包.之后(进程退出宽限):
		}
		// 即使主进程已经退出，仍要清理它的进程组或 作业。
		当前子级进程.错误 = 错误判断包.合并(输入错误, 当前子级进程.输出.关闭(), 当前子级进程.停止())
		select {
		case <-当前子级进程.完成:
		case <-时间包.之后(进程停止等待):
			// 停止失败时，保留 命令.等待 的回收责任，允许会话及应用继续关闭。
			当前子级进程.错误 = 错误判断包.合并(当前子级进程.错误, 上下文包.超出截止时间错误)
		}
	})
	return 当前子级进程.错误
}
