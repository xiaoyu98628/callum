package MCP

import (
	上下文包 "context"
	错误判断包 "errors"
	操作系统包 "os"
	原子操作包 "sync/atomic"
	测试包 "testing"
	同步测试包 "testing/synctest"
	时间包 "time"
)

func Test子级进程关闭(测试器 *测试包.测试器) {
	停止错误 := 错误判断包.新建("停止子进程失败")
	for _, 测试用例 := range []struct {
		名称   string
		已经完成 bool
		停止错误 error
		已用时  时间包.时长
	}{
		{名称: "自然退出后仍清理进程组", 已经完成: true},
		{名称: "强制停止成功", 已用时: 进程退出宽限},
		{名称: "停止失败且未退出", 停止错误: 停止错误, 已用时: 进程退出宽限 + 进程停止等待},
	} {
		测试器.执行(测试用例.名称, func(测试器 *测试包.测试器) {
			同步测试包.测试(测试器, func(测试器 *测试包.测试器) {
				完成 := make(chan struct{})
				if 测试用例.已经完成 {
					close(完成)
				}
				var 停止 int
				当前对象 := &子级进程{输入: 管道结束(测试器, true), 输出: 管道结束(测试器, false), 完成: 完成, 停止: func() error {
					停止++
					if !测试用例.已经完成 && 测试用例.停止错误 == nil {
						close(完成)
					}
					return 测试用例.停止错误
				}}
				已开始 := 时间包.当前时间()
				错误 := 当前对象.关闭()
				if 已用时 := 时间包.自从(已开始); 已用时 != 测试用例.已用时 {
					测试器.格式化创建错误("关闭耗时 %v，期望 %v", 已用时, 测试用例.已用时)
				}
				if 测试用例.停止错误 != nil {
					if !错误判断包.是否属于(错误, 停止错误) || !错误判断包.是否属于(错误, 上下文包.超出截止时间错误) {
						测试器.格式化创建错误("关闭应保留停止失败和超时，得到 %v", 错误)
					}
					select {
					case <-完成:
						测试器.错误文本("测试进程应仍在运行")
					default:
					}
				} else if 错误 != nil {
					测试器.格式化创建错误("关闭失败: %v", 错误)
				}
				if 当前对象.关闭() != 错误 || 停止 != 1 {
					测试器.格式化创建错误("重复关闭应复用结果，停止调用次数 %d", 停止)
				}
				for _, 文件 := range []*操作系统包.文件{当前对象.输入, 当前对象.输出} {
					if _, 错误 := 文件.状态信息(); !错误判断包.是否属于(错误, 操作系统包.已关闭错误) {
						测试器.格式化创建错误("关闭后管道仍可访问: %v", 错误)
					}
				}
			})
		})
	}
}

func Test子级进程并发关闭(测试器 *测试包.测试器) {
	完成 := make(chan struct{})
	close(完成)
	停止错误 := 错误判断包.新建("清理进程组失败")
	var 停止 原子操作包.整数32
	当前对象 := &子级进程{输入: 管道结束(测试器, true), 输出: 管道结束(测试器, false), 完成: 完成, 停止: func() error {
		停止.添加(1)
		return 停止错误
	}}
	启动 := make(chan struct{})
	结果集 := make(chan error, 8)
	for range cap(结果集) {
		go func() { <-启动; 结果集 <- 当前对象.关闭() }()
	}
	close(启动)
	首个 := <-结果集
	if !错误判断包.是否属于(首个, 停止错误) {
		测试器.格式化创建错误("关闭应返回停止错误，得到 %v", 首个)
	}
	for range cap(结果集) - 1 {
		if 错误 := <-结果集; 错误 != 首个 {
			测试器.格式化创建错误("并发关闭返回了不同的结果: %v", 错误)
		}
	}
	if 停止.加载() != 1 {
		测试器.格式化创建错误("并发关闭多次调用停止操作: %d", 停止.加载())
	}
}

func 管道结束(测试器 *测试包.测试器, 写入 bool) *操作系统包.文件 {
	测试器.标记测试辅助函数()
	当前读取器, 写入器, 错误 := 操作系统包.管道()
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(func() {
		_ = 当前读取器.关闭()
		_ = 写入器.关闭()
	})
	if 写入 {
		return 写入器
	}
	return 当前读取器
}
