package MCP

import (
	上下文包 "context"
	输入输出包 "io"
	结构化日志包 "log/slog"
	HTTP包 "net/http"
	文件路径包 "path/filepath"
	测试包 "testing"

	设置包 "github.com/xiaoyu98628/callum/internal/settings"
)

// 快照仓储 故意持有输入引用，用于检查服务是否把自己的可变配置交给外部持有者。
type 快照仓储 struct {
	仓储
	已加载 设置包.MCP
	已保存 设置包.MCP
	密钥  *凭据
}

func (当前快照仓储 *快照仓储) 加载(上下文包.上下文) (设置包.MCP, int64, error) {
	return 当前快照仓储.已加载, 1, nil
}

func (当前快照仓储 *快照仓储) 凭据(上下文包.上下文, string) (凭据, error) {
	return 凭据{}, nil
}

func (当前快照仓储 *快照仓储) 保存(_ 上下文包.上下文, 启动配置 设置包.MCP, _ int64, _ string, 密钥 *凭据) error {
	当前快照仓储.已保存, 当前快照仓储.密钥 = 启动配置, 密钥
	return nil
}

func Test配置不共享可修改字段集(测试器 *测试包.测试器) {
	命令 := 文件路径包.合并(测试器.创建临时目录(), "command")
	启动配置 := 设置包.MCP服务端{标识: "01901234-1234-7234-8234-123456789abc", 名称: "配置隔离", 传输: 设置包.MCP标准输入输出,
		命令: 命令, 认证: 设置包.MCP无, 启动参数: []string{"original"}, 环境变量: map[string]string{"PUBLIC": "original"}, 允许工具集: []string{"safe"}}
	仓储 := &快照仓储{已加载: 设置包.MCP{服务集: []设置包.MCP服务端{启动配置}}}
	当前状态, 错误 := 新建(测试器.上下文(), 测试器.上下文(), 仓储, HTTP包.默认客户端, HTTP包.默认客户端, 结构化日志包.新建(结构化日志包.新建文本处理器(输入输出包.丢弃, nil)))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 当前状态.关闭()
	修改 := func(启动配置 设置包.MCP服务端) {
		启动配置.启动参数[0], 启动配置.环境变量["PUBLIC"], 启动配置.允许工具集[0] = "changed", "changed", "changed"
	}
	检查 := func(修订号 int64, 索引 int) {
		测试器.标记测试辅助函数()
		查看, 错误 := 当前状态.列表(测试器.上下文())
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		实际值 := 查看.服务集[索引]
		if 查看.修订号 != 修订号 || 实际值.启动参数[0] != "original" || 实际值.环境变量["PUBLIC"] != "original" || 实际值.允许工具集[0] != "safe" {
			测试器.格式化报告致命错误("配置被外部引用改写: %+v", 查看)
		}
		修改(实际值.MCP服务端)
	}
	修改(启动配置) // 仓储加载值不能继续修改服务已持有的快照。
	检查(1, 0)
	检查(1, 0) // 查询返回值也必须独立。

	输入 := 设置包.MCP服务端{名称: "保存隔离", 传输: 设置包.MCP标准输入输出, 命令: 命令, 认证: 设置包.MCP无,
		启动参数: []string{"original"}, 环境变量: map[string]string{"PUBLIC": "original"}, 允许工具集: []string{"safe"}}
	密钥 := map[string]string{"PRIVATE": "secret"}
	if _, 错误 := 当前状态.保存(测试器.上下文(), 1, 输入, 密钥输入{环境变量: &密钥}, 保存选项集{}); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	修改(输入)
	密钥["PRIVATE"] = "changed"
	if 仓储.密钥 == nil || 仓储.密钥.环境变量["PRIVATE"] != "secret" {
		测试器.报告致命错误("保存凭据仍借用调用方的环境映射")
	}
	修改(仓储.已保存.服务集[1])
	检查(2, 1)
	检查(2, 1)
}
