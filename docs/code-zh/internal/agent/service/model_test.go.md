func Test附件失败保留取消(测试器 *测试包.测试器) {
	for _, 测试用例 := range []struct {
		错误 error
		类别 string
	}{
		{模型包.附件错误, "attachment_unavailable"},
		{错误判断包.合并(模型包.附件错误, 上下文包.已取消错误), "cancelled"},
		{错误判断包.合并(模型包.附件错误, 上下文包.超出截止时间错误), "timeout"},
	} {
		var 输出 struct {
			错误 struct {
				类别 string `json:"kind"`
			} `json:"error"`
		}
		if 错误 := JSON包.反序列化(工具失败(测试用例.错误), &输出); 错误 != nil {
			测试器.报告致命错误(错误)
		}
		_, 类别, _ := 失败(测试用例.错误)
		if 输出.错误.类别 != 测试用例.类别 || 类别 != 测试用例.类别 {
			测试器.格式化报告致命错误("附件错误覆盖了执行边界: tool=%s run=%s want=%s", 输出.错误.类别, 类别, 测试用例.类别)
		}
	}
	if _, 类别, _ := 失败(错误判断包.合并(智能体包.存储错误, 模型包.附件错误, 上下文包.已取消错误)); 类别 != "internal" {
		测试器.格式化报告致命错误("持久化故障被附件错误掩盖: %s", 类别)
	}
}

func Test已取消增量保存不成为存储失败(测试器 *测试包.测试器) {
	写入失败 := 错误判断包.新建("controlled storage failure")
	for _, 测试 := range []struct {
		名称               string
		取消函数, 失败写入, 失败清理 bool
		预期状态             string
		预期存储             bool
	}{
		{"cancelled_write", true, false, false, 智能体包.已取消, false},
		{"real_write_failure", false, true, false, 智能体包.失败, true},
		{"real_write_failure_with_cancel", true, true, false, 智能体包.失败, true},
		{"cancelled_write_with_cleanup_failure", true, false, true, 智能体包.失败, true},
	} {
		测试器.执行(测试.名称, func(测试器 *测试包.测试器) {
			执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
			defer 取消函数()
			写入 := 0
			仓储 := &已中断消息仓储{写入: func(执行上下文 上下文包.上下文) error {
				写入++
				// 确定性地让取消发生在保存正文时，不依赖网络与数据库的调度时机。
				if 测试.取消函数 {
					取消函数()
				}
				if 测试.失败写入 {
					return 写入失败
				}
				return 执行上下文.获取错误()
			}}
			if 测试.失败清理 {
				仓储.清理错误 = 写入失败
			}
			管理器, 错误 := 执行器包.新建(测试器.上下文(), 1, 时间包.秒)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			defer 管理器.关闭()
			引擎 := &引擎{仓储: 仓储, 管理器: 管理器, 发出事件: func(智能体包.事件) error { return nil }, 选项集: 选项集{调用超时: 时间包.秒}}
			本次执行 := 会话包.执行{执行: 智能体包.执行{流: true}, 助手: 会话包.消息{标识: "assistant"}}
			已解析 := 智能体包.已解析{客户端: 前导流{}}
			总计 := 0
			_, 错误 = 引擎.调用模型(执行上下文, 执行上下文, 本次执行, 已解析, 智能体包.步骤{输入: []byte(`{}`)}, nil, nil, func() {}, &总计)
			状态, _, _ := 失败(错误)
			if 写入 != 1 || 状态 != 测试.预期状态 || 错误判断包.是否属于(错误, 智能体包.存储错误) != 测试.预期存储 {
				测试器.格式化报告致命错误("增量保存错误分类不符: writes=%d status=%s err=%v", 写入, 状态, 错误)
			}
			if 测试.取消函数 && !错误判断包.是否属于(错误, 上下文包.已取消错误) {
				测试器.报告致命错误("取消原因丢失")
			}
			if (测试.失败写入 || 测试.失败清理) && !错误判断包.是否属于(错误, 写入失败) {
				测试器.报告致命错误("真实存储错误链丢失")
			}
		})
	}
}

func Test引导期间流式写入(测试器 *测试包.测试器) {
	for _, 测试 := range []struct {
		名称, 类别 string
		写入错误   error
	}{
		{名称: "write completes", 类别: "steered"},
		{名称: "real storage failure", 类别: "internal", 写入错误: 错误判断包.新建("write failed")},
	} {
		测试器.执行(测试.名称, func(测试器 *测试包.测试器) {
			采样执行上下文, 引导 := 上下文包.带有取消原因(测试器.上下文())
			defer 引导(nil)
			仓储 := &已中断消息仓储{写入: func(写入执行上下文 上下文包.上下文) error {
				引导(会话包.已引导错误)
				if 写入执行上下文.获取错误() != nil || !错误判断包.是否属于(上下文包.原因(采样执行上下文), 会话包.已引导错误) {
					测试器.报告致命错误("引导取消必须限定在模型请求")
				}
				return 测试.写入错误
			}}
			管理器, 错误 := 执行器包.新建(测试器.上下文(), 1, 时间包.秒)
			if 错误 != nil {
				测试器.报告致命错误(错误)
			}
			defer 管理器.关闭()
			引擎 := &引擎{仓储: 仓储, 管理器: 管理器, 发出事件: func(智能体包.事件) error { return nil }, 选项集: 选项集{调用超时: 时间包.秒}}
			本次执行 := 会话包.执行{执行: 智能体包.执行{流: true}, 助手: 会话包.消息{标识: "assistant"}}
			总计 := 0
			_, 错误 = 引擎.调用模型(测试器.上下文(), 采样执行上下文, 本次执行, 智能体包.已解析{客户端: 前导流{}}, 智能体包.步骤{输入: []byte(`{}`)}, nil, nil, func() {}, &总计)
			_, 类别, _ := 失败(错误)
			if 类别 != 测试.类别 || !错误判断包.是否属于(错误, 会话包.已引导错误) ||
				(测试.写入错误 != nil && !错误判断包.是否属于(错误, 测试.写入错误)) {
				测试器.格式化报告致命错误("引导或真实存储错误分类不符: %s, %v", 类别, 错误)
			}
		})
	}
}

// 测试同时覆盖供应商未交付最终结果的取消，以及超过累计内容边界的分片。
type 推理流 struct {
	模型包.客户端
	取消函数 bool
	拒绝   bool
}

func (当前推理流 推理流) 流式生成(执行上下文 上下文包.上下文, _ 模型包.请求, 发出事件 func(模型包.事件) error) error {
	if 错误 := 发出事件(模型包.事件{类型: 模型包.事件推理增量, 文本: "已收到的推理"}); 错误 != nil {
		return 错误
	}
	if 当前推理流.拒绝 {
		return 发出事件(模型包.事件{类型: 模型包.事件推理增量, 文本: 字符串包.重复("x", 推理调用包.最大文本字节)})
	}
	if 当前推理流.取消函数 {
		return 上下文包.已取消错误
	}
	return nil
}

type 推理仓储 struct {
	智能体包.仓储
	最终 智能体包.步骤
	部分 bool
}

func (当前推理仓储 *推理仓储) 保存步骤(_ 上下文包.上下文, 步骤 智能体包.步骤) error {
	if 步骤.状态 == 智能体包.运行中 && len(步骤.输出) > 0 {
		当前推理仓储.部分 = true
	}
	if 步骤.状态 != 智能体包.运行中 {
		当前推理仓储.最终 = 步骤
	}
	return nil
}
func Test推理持久化开启取消并且上限(测试器 *测试包.测试器) {
	for _, 拒绝 := range []bool{false, true} {
		仓储 := &推理仓储{}
		管理器, 错误 := 执行器包.新建(测试器.上下文(), 1, 时间包.秒)
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		测试器.注册清理函数(管理器.关闭)
		数量 := 0
		引擎 := &引擎{仓储: 仓储, 管理器: 管理器, 选项集: 选项集{调用超时: 时间包.秒}, 发出事件: func(事件 智能体包.事件) error {
			if 事件.类型 == "reasoning.delta" {
				数量++
			}
			return nil
		}}
		总计 := 0
		结果, 错误 := 引擎.调用模型(测试器.上下文(), 测试器.上下文(), 会话包.执行{执行: 智能体包.执行{流: true}}, 智能体包.已解析{客户端: 推理流{取消函数: !拒绝, 拒绝: 拒绝}}, 智能体包.步骤{输入: []byte(`{}`)}, nil, nil, func() {}, &总计)
		预期 := 上下文包.已取消错误
		if 拒绝 {
			预期 = 智能体包.输出上限错误
		}
		if !错误判断包.是否属于(错误, 预期) || 数量 != 1 || !仓储.部分 {
			测试器.格式化报告致命错误("推理取消或限额行为错误: count=%d partial=%v err=%v", 数量, 仓储.部分, 错误)
		}
		var 字段集 map[string]JSON包.原始JSON消息

			测试器.报告致命错误(错误)
		}

		if 错误 := JSON包.反序列化(字段集["reasoning"], &文本); 错误 != nil {
			测试器.报告致命错误(错误)
		}
		if 文本 != "已收到的推理" || 结果.推理 == nil || *结果.推理 != 文本 || 字段集["context_message"] != nil {
			测试器.格式化报告致命错误("部分推理保存错误: %s", 仓储.最终.输出)
		}
	}
}

func Test实际输入投影保留未知并且原始输出(测试器 *测试包.测试器) {
	for _, 输入 := range []JSON包.原始JSON消息{nil, JSON包.原始JSON消息(`null`), JSON包.原始JSON消息(`{"provider_result":{"text":"原始结果"}}`)} {
		输出, 错误 := 带有实际输入词元(输入, nil)
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		var 值 map[string]JSON包.原始JSON消息
		if 错误 := JSON包.反序列化(输出, &值); 错误 != nil {
			测试器.报告致命错误(错误)
		}
		if string(值["context_usage"]) != `{"actual_input_tokens":null}` {
			测试器.报告致命错误("缺失实际用量被补造")
		}
		if len(输入) > 4 && string(值["provider_result"]) != `{"text":"原始结果"}` {
			测试器.报告致命错误("原始输出被改写")
		}
	}
	输出, 错误 := 带有实际输入词元(nil, new(int64(42)))
	if 错误 != nil || string(输出) != `{"context_usage":{"actual_input_tokens":42}}` {
		测试器.格式化报告致命错误("实际输入投影丢失: %s %v", 输出, 错误)
	}
}
