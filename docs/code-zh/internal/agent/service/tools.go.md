func (当前引擎 *引擎) 调用工具(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 已解析 智能体包.已解析, 引导 *执行引导, 引导内容 []模型包.消息, 父级 智能体包.步骤, 调用 模型包.工具调用, 已拒绝, 待核实 map[string]bool, 外部 工具包.工具) (消息 模型包.消息, 返回错误 error) {
	范围, 操作错误 := 当前引擎.工具范围(执行上下文, 本次执行)
	var 已准备 *工具执行器包.已准备调用
	if 操作错误 == nil {
		权限 := 工具包.执行权限{会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 允许图片输入: 已解析.图片输入}
		if 当前引擎.选项集.附件集 != nil {
			if 文件集, 成功 := 当前引擎.选项集.附件集.文件集(本次执行.会话.标识, 本次执行.执行.标识); 成功 {
				权限.输入目录, 权限.工作目录 = 文件集.输入集, 文件集.工作
			}
		}
		if 快照 := 本次执行.执行.设置快照; 快照 != nil {
			权限.沙箱模式 = 快照.沙箱模式
		}
		if 外部 != nil {
			已准备, 操作错误 = 当前引擎.选项集.工具集.准备外部(执行上下文, 范围, 外部, 调用.参数, 权限)
		} else {
			已准备, 操作错误 = 当前引擎.选项集.工具集.准备带有权限(执行上下文, 范围, 调用.名称, 调用.参数, 权限)
		}
	}
	var 步骤 智能体包.步骤
	var 历史键 string
	var 命令 bool
	结果 := 工具包.结果{操作结果: 工具包.操作结果不已生效}
	// 先关闭固定调用资源，再保存操作结果；关闭失败也保留在受控错误和执行状态中。
	defer func() {
		if 已准备 != nil {
			关闭错误 := 已准备.关闭()
			操作错误 = 错误判断包.合并(操作错误, 关闭错误)
			if 结果.操作结果 == 工具包.操作结果未知 || 结果.操作结果 == 工具包.操作结果部分 || (结果.操作结果 == 工具包.操作结果已生效 && 操作错误 != nil) {
				待核实[历史键] = true
			} else if 操作错误 == nil && 待核实[历史键] && 确认工具结果(结果, 命令) {
				delete(待核实, 历史键)
			}
			if 步骤.标识 == "" {
				返回错误 = 错误判断包.合并(返回错误, 关闭错误)
			}
		}
		if 步骤.标识 != "" {
			var 结束错误 error
			消息, 结束错误 = 当前引擎.结束工具(执行上下文, 本次执行, 步骤, 调用, 结果, 操作错误)
			返回错误 = 错误判断包.合并(返回错误, 结束错误)
		}
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			返回错误 = 错误判断包.合并(返回错误, 错误)
		}
	}()
	if 已准备 != nil {
		历史键 = 已准备.获取哈希()
		命令 = 已准备.命令边界() != nil
		if 命令 {
			历史键, 操作错误 = 当前引擎.仓储.命令历史键(执行上下文, 已准备.获取请求(), 已准备.获取哈希())
			if 操作错误 != nil {
				return 消息, 操作存储错误(执行上下文, 操作错误)
			}
		}
	}
	决定 := 工具决定(已准备, 已拒绝[历史键], 待核实[历史键], 本次执行.执行.设置快照)
	var 错误 error
	var 目录标识集 []string
	if 决定.动作 == 策略包.询问 && 决定.原因 == "outside_workspace" && 当前引擎.选项集.审批集 != nil {
		目录标识集, 错误 = 当前引擎.选项集.审批集.匹配目录集(执行上下文, 本次执行.会话.标识, 已准备.目标集(), 已准备.获取定义().影响集)
		if 错误 != nil {
			return 消息, 操作存储错误(执行上下文, 错误)
		}
		决定 = 策略包.应用目录授权(决定, len(目录标识集) > 0)
	}
	步骤, 错误 = 当前引擎.创建工具步骤(执行上下文, 本次执行, 父级, 调用, 已准备, 决定, 目录标识集)
	if 错误 != nil {
		return 消息, 错误
	}
	if 错误 := 当前引擎.事件(本次执行, "step.created", 步骤.标识, "", 步骤); 错误 != nil {
		操作错误 = 错误
		return 消息, 错误
	}
	if 操作错误 != nil {
		return
	}
	审批标识, 错误 := 当前引擎.授权工具(执行上下文, 本次执行, 已解析, 引导, 引导内容, &步骤, 已准备, 决定, 历史键, 已拒绝)
	if 错误 != nil {
		操作错误 = 错误
		if 错误判断包.是否属于(错误, 工具包.已拒绝错误) || 错误判断包.是否属于(错误, 会话包.已引导错误) {
			return
		}
		return 消息, 错误
	}
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		操作错误 = 错误
		return 消息, 错误
	}
	当前, 错误 := 当前引擎.工具范围(执行上下文, 本次执行)
	if 错误 != nil || 当前 != 范围 {
		操作错误 = 工具包.已拒绝错误
		return
	}
	var 已认领 智能体包.步骤
	错误 = 当前引擎.之前工具(执行上下文, 引导, func() error {
		var 错误 error
		已认领, 错误 = 当前引擎.仓储.开始工具(执行上下文, 本次执行.执行.标识, 步骤.标识, 审批标识, 已准备.获取哈希())
		return 错误
	})
	if 错误 != nil {
		操作错误 = 错误
		if 错误判断包.是否属于(错误, 工具包.已拒绝错误) || 错误判断包.是否属于(错误, 会话包.已引导错误) {
			return
		}
		return 消息, 操作存储错误(执行上下文, 错误)
	}
	步骤 = 已认领
	// 先保存可能已发生操作的边界，保存失败则不调用工具。终态写入失败或崩溃时
	// 依然有保守的未知记录；正常返回再用工具报告的事实替换它。
	检查点 := 工具包.操作结果记录{版本: 工具包.操作版本, 操作结果: 工具包.未确认操作结果(已准备.获取定义().影响集), 请求哈希: 已准备.获取哈希()}
	步骤.输出, 错误 = JSON包.序列化(map[string]any{"operation": 检查点})
	if 错误 == nil {
		错误 = 当前引擎.仓储.保存步骤(执行上下文, 步骤)
	}
	if 错误 != nil {
		操作错误 = 错误
		return 消息, 操作存储错误(执行上下文, 错误)
	}
	if 错误 := 当前引擎.事件(本次执行, "run.updated", "", "", 本次执行.执行); 错误 != nil {
		操作错误 = 错误
		return 消息, 错误
	}
	if 错误 := 当前引擎.事件(本次执行, "step.updated", 步骤.标识, "", 步骤); 错误 != nil {
		操作错误 = 错误
		return 消息, 错误
	}
	结果, 操作错误 = 当前引擎.选项集.工具集.执行流(执行上下文, 已准备, 已准备.获取哈希(), func(分片 工具包.输出分片) error {
		// 转为独立字符串后再发布，订阅队列不借用工具的读取缓冲区。
		return 当前引擎.事件(本次执行, "tool.output", 步骤.标识, "", 智能体包.工具输出{序号: 分片.序号, 流: 分片.流, 编码: "base64", 数据: 基准64包.标准编码.编码目标字符串(分片.数据)})
	})
	return
}

func 确认工具结果(结果 工具包.结果, 命令 bool) bool {
	if 命令 {
		// 命令的 已生效 表示已运行至正常退出，仍需核验退出码才能清除上次的待核实状态。
		var 报告 struct {
			退出码 *int `json:"exit_code"`
		}
		return 结果.操作结果 == 工具包.操作结果已生效 && JSON包.反序列化(结果.输出, &报告) == nil &&
			报告.退出码 != nil && *报告.退出码 == 0
	}
	return 结果.操作结果 == 工具包.操作结果已生效 || 结果.操作结果 == 工具包.操作结果无
}

func 工具决定(已准备 *工具执行器包.已准备调用, 已拒绝, 待核实 bool, 快照 *设置包.快照) 策略包.决定 {
	if 快照 == nil || !快照.有效() {
		return 策略包.决定{动作: 策略包.拒绝, 原因: "invalid_settings_snapshot"}
	}
	if 已准备 == nil {
		return 策略包.决定{动作: 策略包.拒绝, 原因: "operation_validation_failed"}
	}
	return 策略包.评估操作(快照.审批策略, 策略包.操作{
		影响集: 已准备.获取定义().影响集, 外部工作区: 已准备.外部工作区(),
		沙箱模式: 快照.沙箱模式, 命令: 已准备.命令边界(),
		先前已拒绝: 已拒绝, 结果未确认: 待核实,
	})
}

func (当前引擎 *引擎) 创建工具步骤(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 父级 智能体包.步骤, 调用 模型包.工具调用, 已准备 *工具执行器包.已准备调用, 决定 策略包.决定, 目录标识集 []string) (智能体包.步骤, error) {
	输入 := map[string]any{"arguments": 调用.参数, "authorization": 决定, "operation_version": 工具包.操作版本}
	if len(目录标识集) > 0 {
		输入["directory_grant_ids"] = 目录标识集
	}
	if 快照 := 本次执行.执行.设置快照; 快照 != nil {
		输入["approval_policy"], 输入["approvals_reviewer"], 输入["settings_revision"] = 快照.审批策略, 快照.审批审核方, 快照.修订号
		输入["sandbox_mode"] = 快照.沙箱模式
	}
	var 定义 JSON包.原始JSON消息
	if 已准备 != nil {
		输入["request"], 输入["request_hash"] = 已准备.获取请求(), 已准备.获取哈希()
		var 错误 error
		定义, 错误 = JSON包.序列化(已准备.获取定义())
		if 错误 != nil {
			return 智能体包.步骤{}, 错误
		}
	}
	输入JSON, 错误 := JSON包.序列化(输入)
	if 错误 != nil {
		return 智能体包.步骤{}, 错误
	}
	标识, 错误 := UUID包.新建UUIDv7()
	if 错误 != nil {
		return 智能体包.步骤{}, 错误
	}
	当前时间 := 时间包.当前时间().Unix毫秒()
	步骤, 错误 := 当前引擎.仓储.创建步骤(执行上下文, 智能体包.步骤{标识: 标识.字符串(), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 父级步骤标识: new(父级.标识), 类型: "tool_call", 名称: 调用.名称, 定义: 定义, 状态: 智能体包.待处理, 输入: 输入JSON, 工具调用标识: new(调用.标识), 创建时间: 当前时间, 更新时间: 当前时间})
	if 错误 != nil {
		return 步骤, 操作存储错误(执行上下文, 错误)
	}
	return 步骤, nil
}

func (当前引擎 *引擎) 授权工具(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 已解析 智能体包.已解析, 引导 *执行引导, 引导内容 []模型包.消息, 步骤 *智能体包.步骤, 已准备 *工具执行器包.已准备调用, 决定 策略包.决定, 历史键 string, 已拒绝 map[string]bool) (string, error) {
	switch 决定.动作 {
	case 策略包.允许:
		return "", nil
	case 策略包.询问:
		// 审批策略只决定是否询问；审批人独立校验，不能回退成自动放行。
		if 本次执行.执行.设置快照 == nil || !本次执行.执行.设置快照.有效() {
			return "", 工具包.已拒绝错误
		}
	default:
		return "", 工具包.已拒绝错误
	}
	var 审核 *智能体包.审批审核
	if 本次执行.执行.设置快照.审批审核方 == 策略包.自动审核 && 能够审核工具(已准备, 决定) {
		var 错误 error
		审核, 错误 = 当前引擎.审核工具(执行上下文, 本次执行, 已解析, 引导内容, 步骤, 已准备)
		if 错误 != nil {
			return "", 错误
		}
	}
	var 甲 智能体包.审批
	错误 := 当前引擎.之前工具(执行上下文, 引导, func() error {
		var 错误 error
		甲, 错误 = 当前引擎.选项集.审批集.请求(执行上下文, 本次执行.执行.标识, 步骤.标识, 已准备.获取请求(), 已准备.获取哈希(), 审核)
		return 错误
	})
	if 错误判断包.是否属于(错误, 会话包.已引导错误) {
		return "", 错误
	}
	if 错误 != nil {
		return "", 操作存储错误(执行上下文, 错误)
	}
	步骤.状态, 步骤.更新时间 = 智能体包.等待中审批, 时间包.当前时间().Unix毫秒()
	已决定 := 甲
	if 甲.状态 == 智能体包.审批待处理 {
		if 错误 := 当前引擎.事件(本次执行, "step.updated", 步骤.标识, "", *步骤); 错误 != nil {
			return "", 错误
		}
		已暂停 := 本次执行.执行
		已暂停.状态, 已暂停.更新时间 = 智能体包.等待中审批, 步骤.更新时间
		if 错误 := 当前引擎.事件(本次执行, "run.updated", "", "", 已暂停); 错误 != nil {
			return "", 错误
		}
		已决定, 错误 = 当前引擎.选项集.审批集.等待(执行上下文, 甲.标识)
		if 错误 != nil {
			return "", 错误
		}
	}
	if 已决定.状态 == 智能体包.审批已批准 {
		return 已决定.标识, nil
	}
	已拒绝[历史键] = true
	if 错误 := 当前引擎.仓储.继续(执行上下文, 本次执行.执行.标识); 错误 != nil {
		return "", 操作存储错误(执行上下文, 错误)
	}
	if 错误 := 当前引擎.事件(本次执行, "run.updated", "", "", 本次执行.执行); 错误 != nil {
		return "", 错误
	}
	return "", 工具包.已拒绝错误
}

func (当前引擎 *引擎) 结束工具(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 步骤 智能体包.步骤, 调用 模型包.工具调用, 结果 工具包.结果, 操作错误 error) (模型包.消息, error) {
	var 输入 struct {
		请求哈希 string `json:"request_hash"`
	}
	if 错误 := JSON包.反序列化(步骤.输入, &输入); 错误 != nil {
		return 模型包.消息{}, 错误判断包.合并(智能体包.存储错误, 错误)
	}
	事实集 := 工具包.操作结果记录{版本: 工具包.操作版本, 操作结果: 结果.操作结果, 请求哈希: 输入.请求哈希}
	字段集 := map[string]any{"operation": 事实集, "result": 结果.输出}
	var 失败 struct {
		错误 struct {
			类别 string `json:"kind"`
			消息 string `json:"message"`
		} `json:"error"`
	}
	if 操作错误 != nil {
		受控 := 工具失败(操作错误)
		if 错误 := JSON包.反序列化(受控, &失败); 错误 != nil {
			return 模型包.消息{}, 错误
		}
		字段集["error"] = 失败.错误
		if 结果.输出 == nil {
			// 保留原有失败时 结果.error 的结构；实际输出存在时与错误分别保存。
			字段集["result"] = 受控
		}
	}
	信封, 错误 := JSON包.序列化(字段集)
	if 错误 != nil {
		return 模型包.消息{}, 错误
	}
	消息 := 模型包.消息{角色: 模型包.角色工具, 内容: string(信封), 工具调用标识: 调用.标识}
	if 操作错误 == nil && len(结果.图片集) > 0 {
		消息.内容块集 = append([]模型包.内容块{{类型: "text", 文本: 消息.内容}}, 结果.图片集...)
	}
	信封, 错误 = 带有上下文消息(信封, 消息)
	if 错误 != nil {
		return 消息, 错误
	}
	当前时间 := 时间包.当前时间().Unix毫秒()
	步骤.状态, 步骤.输出, 步骤.更新时间, 步骤.结束时间 = 智能体包.已完成, 信封, 当前时间, &当前时间
	if 操作错误 != nil {
		步骤.状态 = 智能体包.失败
		if 错误判断包.是否属于(操作错误, 上下文包.已取消错误) {
			步骤.状态 = 智能体包.已取消
		}
		步骤.错误类别, 步骤.错误消息 = &失败.错误.类别, &失败.错误.消息
	}
	清理, 取消函数 := 清理上下文(执行上下文)
	defer 取消函数()
	if 错误 := 当前引擎.仓储.保存步骤(清理, 步骤); 错误 != nil {
		return 消息, 错误判断包.合并(智能体包.存储错误, 错误)
	}
	return 消息, 当前引擎.事件(本次执行, "step.updated", 步骤.标识, "", 步骤)
}

func (当前引擎 *引擎) 工具范围(执行上下文 上下文包.上下文, 本次执行 会话包.执行) (工具包.范围, error) {
	if 本次执行.会话.工作区标识 == nil {
		return 工具包.范围{}, nil
	}
	工作区, 错误 := 当前引擎.选项集.工作区(执行上下文, *本次执行.会话.工作区标识)
	if 错误 != nil {
		return 工具包.范围{}, 工具包.不可用错误
	}
	if 工作区.删除时间 != nil || 工作区.已归档位于 != nil {
		return 工具包.范围{}, 工具包.已拒绝错误
	}
	if 工作区.根路径 == "" {
		return 工具包.范围{}, nil
	}
	return 工具包.范围{标识: 工作区.标识, 根路径: 工作区.根路径}, nil
}
