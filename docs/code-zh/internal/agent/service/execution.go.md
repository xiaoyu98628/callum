package 服务

import (
	上下文包 "context"
	错误判断包 "errors"
	字符串包 "strings"
	同步包 "sync"
	时间包 "time"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	执行器包 "github.com/xiaoyu98628/callum/internal/agent/executor"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	历史包 "github.com/xiaoyu98628/callum/internal/conversation/history"
)

// 准备执行 为已提交的普通对话登记引导状态。调用方必须启动返回的任务或调用 放弃。
func (当前引擎 *引擎) 准备执行(执行租约 *执行器包.执行租约, 本次执行 会话包.执行, 已解析 智能体包.已解析) func(上下文包.上下文) {
	引导 := 当前引擎.注册引导(执行租约, 本次执行.执行.标识)
	return func(执行上下文 上下文包.上下文) {
		defer 当前引擎.移除引导(执行租约)
		当前引擎.执行(执行上下文, 执行租约, 本次执行, 已解析, 引导)
	}
}

// 执行 在受理响应排队后运行，标题任务在终态前取消并等待退出。
func (当前引擎 *引擎) 执行(执行上下文 上下文包.上下文, 执行租约 *执行器包.执行租约, 本次执行 会话包.执行, 已解析 智能体包.已解析, 引导 *执行引导) {
	错误 := 当前引擎.初始事件集(本次执行)
	if 错误 == nil && 当前引擎.选项集.附件集 != nil {
		_, 错误 = 当前引擎.选项集.附件集.开始(执行上下文, 本次执行.会话.标识, 本次执行.执行.标识, 本次执行.已发布交付物集...)
	}
	当前引擎.开始上下文用量(本次执行, 已解析)
	defer 当前引擎.结束上下文用量(本次执行)
	if 本次执行.执行.设置快照 == nil || !本次执行.执行.设置快照.有效() {
		错误 = 错误判断包.合并(错误, 智能体包.无效错误)
	}
	var 标题等待组 同步包.等待组
	标题执行上下文, 标题取消 := 上下文包.创建可取消上下文(执行上下文)
	if 错误 == nil {
		错误 = 当前引擎.循环(执行上下文, 本次执行, 已解析, 引导, func() {
			if 文本 := 历史包.文本(本次执行.用户.内容); !本次执行.已继续 && 本次执行.用户.序号 == 1 && 本次执行.会话.标题来源 == "default" && 字符串包.裁剪空间(文本) != "" {
				if 释放, 成功, _ := 当前引擎.管理器.尝试获取调用(标题执行上下文); 成功 {
					标题等待组.Go(func() { defer 释放(); 当前引擎.标题(标题执行上下文, 本次执行, 已解析, 文本) })
				}
			}
		})
	}
	标题取消()
	标题等待组.等待()
	if 当前引擎.选项集.附件集 != nil {
		if 文件集, 成功 := 当前引擎.选项集.附件集.文件集(本次执行.会话.标识, 本次执行.执行.标识); 成功 && len(文件集.交付物集) > len(本次执行.已发布交付物集) {
			清理, 取消函数 := 清理上下文(执行上下文)
			消息, 读取错误 := 当前引擎.最新消息(清理, 本次执行)
			取消函数()
			if 读取错误 == nil {
				读取错误 = 当前引擎.保存回答(执行上下文, 本次执行, append(消息.内容, 文件集.交付物集[len(本次执行.已发布交付物集):]...), "")
			}
			错误 = 错误判断包.合并(错误, 读取错误)
		}
		错误 = 错误判断包.合并(错误, 当前引擎.选项集.附件集.结束(本次执行.执行.标识))
	}
	if 取消错误 := 执行租约.是否正在结束(); 取消错误 != nil {
		错误 = 错误判断包.合并(取消错误, 错误)
	}
	状态 := 智能体包.已完成
	var 类别, 消息 *string
	if 错误 != nil {
		var 键, 映射 string
		状态, 键, 映射 = 失败(错误)
		类别, 消息 = &键, &映射
	}
	当前时间 := 时间包.当前时间().Unix毫秒()
	本次执行.执行.状态, 本次执行.执行.错误类别, 本次执行.执行.错误消息, 本次执行.执行.更新时间, 本次执行.执行.结束时间 = 状态, 类别, 消息, 当前时间, &当前时间
	清理, 取消函数 := 清理上下文(执行上下文)
	defer 取消函数()
	待处理 := 当前引擎.加密封装引导(清理, 引导)
	结束错误 := 当前引擎.仓储.结束(清理, 本次执行.执行)
	if 结束错误 != nil {
		当前引擎.日志记录器.记录错误("保存执行终态失败", "module", "run", "run_id", 本次执行.执行.标识)
		本次执行.执行.状态, 本次执行.执行.错误类别, 本次执行.执行.错误消息 = 智能体包.失败, new("internal"), new("保存执行状态失败")
		// 无法保存终态时只报告执行失败及持久化未确认，不能宣称数据库已进入终态。
		当前引擎.结束上下文用量(本次执行)
		_ = 当前引擎.发出事件(智能体包.事件{类型: "run.failed", 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 数据: 本次执行.执行, 状态已持久化: new(false)})
		return
	} else if 错误判断包.是否属于(错误, 智能体包.存储错误) {
		当前引擎.日志记录器.记录错误("执行数据保存失败", "module", "run", "run_id", 本次执行.执行.标识)
	}
	当前引擎.引导已重新排队(本次执行, 待处理)
	if 结束错误 == nil {
		if 消息, 读取错误 := 当前引擎.最新消息(清理, 本次执行); 读取错误 == nil {
			_ = 当前引擎.事件(本次执行, "message.updated", "", 消息.标识, 消息)
		} else {
			当前引擎.日志记录器.记录错误("读取最终消息状态失败", "module", "run", "run_id", 本次执行.执行.标识)
		}
	}
	// 终态已持久化；发布器关闭后不递归发布错误，查询仍可获取已保存状态。
	// 用量刷新先于终态通知，连接可在终态后排空并结束。
	当前引擎.结束上下文用量(本次执行)
	_ = 当前引擎.事件(本次执行, "run."+本次执行.执行.状态, "", "", 本次执行.执行)
}

func (当前引擎 *引擎) 初始事件集(本次执行 会话包.执行) error {
	if 本次执行.已创建 {
		if 错误 := 当前引擎.事件(本次执行, "conversation.created", "", "", 本次执行.会话); 错误 != nil {
			return 错误
		}
	} else {
		if 错误 := 当前引擎.事件(本次执行, "conversation.updated", "", "", 本次执行.会话); 错误 != nil {
			return 错误
		}
	}
	if 本次执行.已继续 {
		if 错误 := 当前引擎.事件(本次执行, "run.resumed", "", "", 本次执行.执行); 错误 != nil {
			return 错误
		}
		if 错误 := 当前引擎.事件(本次执行, "message.updated", "", 本次执行.助手.标识, 本次执行.助手); 错误 != nil {
			return 错误
		}
		return 当前引擎.事件(本次执行, "step.created", 本次执行.步骤.标识, "", 本次执行.步骤)
	}
	if 错误 := 当前引擎.事件(本次执行, "run.started", "", "", 本次执行.执行); 错误 != nil {
		return 错误
	}
	用户事件 := "message.created"
	if 本次执行.执行.替换执行标识 != nil {
		用户事件 = "message.updated"
	}
	if 错误 := 当前引擎.事件(本次执行, 用户事件, "", 本次执行.用户.标识, 本次执行.用户); 错误 != nil {
		return 错误
	}
	if 错误 := 当前引擎.事件(本次执行, "message.created", "", 本次执行.助手.标识, 本次执行.助手); 错误 != nil {
		return 错误
	}
	return 当前引擎.事件(本次执行, "step.created", 本次执行.步骤.标识, "", 本次执行.步骤)
}

func (当前引擎 *引擎) 最新消息(执行上下文 上下文包.上下文, 本次执行 会话包.执行) (会话包.消息, error) {
	行集, 错误 := 当前引擎.会话集.消息集(执行上下文, 本次执行.会话.标识, 0, 1)
	if 错误 != nil {
		return 会话包.消息{}, 错误
	}
	if len(行集) != 1 || 行集[0].标识 != 本次执行.助手.标识 {
		return 会话包.消息{}, 会话包.未找到错误
	}
	return 行集[0], nil
}
