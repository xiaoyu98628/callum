package 服务

import (
	上下文包 "context"
	JSON包 "encoding/json"
	错误判断包 "errors"
	时间包 "time"

	UUID包 "github.com/google/uuid"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

// 摘要模型 使用统一调用槽位和超时，原始结果、用量 只保存到具体模型步骤。
func (当前引擎 *引擎) 摘要模型(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 已解析 智能体包.已解析, 父级 智能体包.步骤, 消息集 []模型包.消息, 版本标识 string) (结果 模型包.响应, 返回错误 error) {
	if 错误 := 摘要容量(执行上下文, 本次执行, 已解析, 消息集); 错误 != nil {
		return 结果, 错误
	}
	标识, 错误 := UUID包.新建UUIDv7()
	if 错误 != nil {
		return 结果, 错误
	}
	元数据 := map[string]any{"purpose": "context_summary", "protocol": 已解析.协议}
	var 父级输入 struct {
		来源执行标识 string `json:"source_run_id"`
	}
	if 错误 := JSON包.反序列化(父级.输入, &父级输入); 错误 != nil {
		return 结果, 错误
	}
	if 父级输入.来源执行标识 != "" {
		元数据["generated_checkpoint_id"] = 版本标识
	} else {
		元数据["generated_context_version_id"] = 版本标识
	}
	if 本次执行.上下文 != nil {
		元数据["context_version_id"] = 本次执行.上下文.标识
	}
	输入, 错误 := JSON包.序列化(元数据)
	if 错误 != nil {
		return 结果, 错误
	}
	已开始 := 时间包.当前时间().Unix毫秒()
	步骤 := 智能体包.步骤{标识: 标识.字符串(), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 父级步骤标识: &父级.标识,
		类型: "model_call", 名称: "生成历史摘要", 状态: 智能体包.运行中, 输入: 输入, 模型: &已解析.快照, 创建时间: 已开始, 开始时间: &已开始, 更新时间: 已开始}
	if 错误 := 模型元数据(&步骤, 消息集, nil, 已解析, false); 错误 != nil {
		return 结果, 错误
	}
	if 错误 := 词元元数据(&步骤, 本次执行, 已解析, 消息集, nil); 错误 != nil {
		return 结果, 错误
	}
	步骤, 错误 = 当前引擎.仓储.创建步骤(执行上下文, 步骤)
	if 错误 != nil {
		return 结果, 操作存储错误(执行上下文, 错误)
	}
	if 错误 := 当前引擎.事件(本次执行, "step.created", 步骤.标识, "", 步骤); 错误 != nil {
		return 结果, 错误
	}
	调用执行上下文, 取消函数 := 上下文包.创建超时上下文(执行上下文, 当前引擎.选项集.调用超时)
	defer 取消函数()
	返回错误 = 当前引擎.管理器.调用(调用执行上下文, func() error {
		var 错误 error
		结果, 错误 = 已解析.客户端.生成(调用执行上下文, 模型包.请求{模型: 已解析.快照.远程模型标识, 捕获结果: true, 允许图片输入: 已解析.图片输入, 消息集: 消息集})
		return 错误
	})
	if 错误 := 调用执行上下文.获取错误(); 错误 != nil {
		返回错误 = 错误判断包.合并(返回错误, 错误)
	}
	if 错误判断包.是否属于(返回错误, 上下文包.超出截止时间错误) {
		返回错误 = 错误判断包.合并(摘要超时错误, 返回错误)
	}
	填充失败(&结果, 已解析.协议, 返回错误)
	步骤.状态 = 智能体包.已完成
	if 返回错误 != nil {
		状态, 类别, 消息 := 失败(返回错误)
		步骤.状态, 步骤.错误类别, 步骤.错误消息 = 状态, &类别, &消息
	}
	已结束 := 时间包.当前时间().Unix毫秒()
	步骤.输出, 步骤.用量, 步骤.更新时间, 步骤.结束时间 = 结果.输出, 结果.用量, 已结束, &已结束
	if 结果.请求标识 != "" {
		步骤.供应商请求标识 = &结果.请求标识
	}
	清理, 结束 := 清理上下文(执行上下文)
	defer 结束()
	if 错误 := 当前引擎.仓储.保存步骤(清理, 步骤); 错误 != nil {
		return 结果, 错误判断包.合并(返回错误, 智能体包.存储错误, 错误)
	}
	if 错误 := 当前引擎.事件(本次执行, "step.updated", 步骤.标识, "", 步骤); 错误 != nil {
		return 结果, 错误判断包.合并(返回错误, 错误)
	}
	return 结果, 返回错误
}
