package 服务

import (
	上下文包 "context"
	JSON包 "encoding/json"
	错误判断包 "errors"
	字符串包 "strings"
	时间包 "time"

	UUID包 "github.com/google/uuid"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	历史包 "github.com/xiaoyu98628/callum/internal/conversation/history"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

// 标题超时 限制辅助标题请求，主回答结束时还会提前取消它。
const 标题超时 = 10 * 时间包.秒

func (当前引擎 *引擎) 标题(父级 上下文包.上下文, 本次执行 会话包.执行, 已解析 智能体包.已解析, 文本 string) {
	执行上下文, 取消函数 := 上下文包.创建超时上下文(父级, min(标题超时, 当前引擎.选项集.调用超时))
	defer 取消函数()
	错误 := 当前引擎.标题步骤(执行上下文, 本次执行, 已解析, 文本)
	if 错误 != nil && 执行上下文.获取错误() == nil {
		当前引擎.日志记录器.警告("辅助标题任务未完成", "module", "run", "run_id", 本次执行.执行.标识)
	}
}

func (当前引擎 *引擎) 标题步骤(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 已解析 智能体包.已解析, 文本 string) error {
	标识, 错误 := UUID包.新建UUIDv7()
	if 错误 != nil {
		return 错误
	}
	当前时间 := 时间包.当前时间().Unix毫秒()
	输入, 错误 := JSON包.序列化(map[string]any{"purpose": "conversation_title", "protocol": 已解析.协议, "message_id": 本次执行.用户.标识, "fingerprint": 历史包.指纹(本次执行.用户.内容)})
	if 错误 != nil {
		return 错误
	}
	步骤 := 智能体包.步骤{标识: 标识.字符串(), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 类型: "model_call", 名称: "生成会话标题", 状态: 智能体包.运行中, 输入: 输入, 模型: &已解析.快照, 创建时间: 当前时间, 开始时间: new(当前时间), 更新时间: 当前时间}
	步骤, 错误 = 当前引擎.仓储.创建步骤(执行上下文, 步骤)
	if 错误 != nil {
		return 错误
	}
	if 错误 := 当前引擎.事件(本次执行, "step.created", 步骤.标识, "", 步骤); 错误 != nil {
		return 错误
	}
	结果, 调用错误 := 已解析.客户端.生成(执行上下文, 模型包.请求{模型: 已解析.快照.远程模型标识, 捕获结果: true, 消息集: []模型包.消息{{角色: 模型包.角色系统, 内容: "请为用户的问题生成一个简短会话标题，不超过30个Unicode字符。使用用户的语言概括核心问题或任务，保留必要的工单编号和专有名称；问答可使用主题短语，不强制使用祈使句。只返回一行标题，不加引号、Markdown或末尾标点，不要回答问题、不要附加解释。"}, {角色: 模型包.角色用户, 内容: 文本}}})
	if 执行上下文.获取错误() != nil {
		调用错误 = 错误判断包.合并(执行上下文.获取错误(), 调用错误)
	}
	填充失败(&结果, 已解析.协议, 调用错误)
	状态 := 智能体包.已完成
	if 调用错误 != nil {
		var 类别, 消息 string
		状态, 类别, 消息 = 失败(调用错误)
		步骤.错误类别, 步骤.错误消息 = &类别, &消息
	}
	当前时间 = 时间包.当前时间().Unix毫秒()
	步骤.状态, 步骤.输出, 步骤.用量, 步骤.更新时间, 步骤.结束时间 = 状态, 结果.输出, 结果.用量, 当前时间, &当前时间
	if 结果.请求标识 != "" {
		步骤.供应商请求标识 = &结果.请求标识
	}
	清理, 结束取消 := 清理上下文(执行上下文)
	defer 结束取消()
	if 错误 := 当前引擎.仓储.保存步骤(清理, 步骤); 错误 != nil {
		return 错误
	}
	_ = 当前引擎.事件(本次执行, "step.updated", 步骤.标识, "", 步骤)
	if 调用错误 != nil {
		return nil
	}
	标题 := 字符串包.裁剪空间(结果.文本)
	if 索引 := 字符串包.索引任意(标题, "\r\n"); 索引 >= 0 {
		标题 = 字符串包.裁剪空间(标题[:索引])
	}
	标题 = 字符串包.裁剪(标题, "\"“”")
	if 标题 == "" || len(结果.文本) > 4096 {
		return nil
	}
	标题 = 历史包.标题([]模型包.内容块{{类型: "text", 文本: 标题}})
	会话, 已生效, 错误 := 当前引擎.会话集.应用标题(清理, 本次执行.会话.标识, 本次执行.用户.标识, 历史包.指纹(本次执行.用户.内容), 标题)
	if 错误 != nil {
		return 错误
	}
	if 已生效 {
		return 当前引擎.事件(本次执行, "conversation.updated", "", "", 会话)
	}
	return nil
}
