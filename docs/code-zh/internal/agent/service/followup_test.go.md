package 服务

import (
	上下文包 "context"
	错误判断包 "errors"
	同步包 "sync"
	测试包 "testing"
	时间包 "time"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	执行器包 "github.com/xiaoyu98628/callum/internal/agent/executor"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
)

type 空引导仓储 struct{ 会话包.仓储 }

func (空引导仓储) 待处理引导(上下文包.上下文, string) ([]会话包.后续消息, error) {
	return nil, nil
}

func 引导引擎(测试器 *测试包.测试器) *引擎 {
	测试器.标记测试辅助函数()
	管理器, 错误 := 执行器包.新建(测试器.上下文(), 2, 时间包.秒)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(管理器.关闭)
	return &引擎{管理器: 管理器, 会话集: 空引导仓储{},
		引导: make(map[*执行器包.执行租约]*执行引导), 发出事件: func(智能体包.事件) error { return nil }}
}

func 预留引导(测试器 *测试包.测试器, 当前引擎 *引擎, 执行标识, 会话标识 string) (*执行器包.执行租约, *执行引导) {
	测试器.标记测试辅助函数()
	执行租约, 错误 := 当前引擎.管理器.预留智能体上下文(测试器.上下文(), 执行标识, 会话标识)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(执行租约.中止)
	return 执行租约, 当前引擎.注册引导(执行租约, 执行标识)
}

func Test引导提交不内容块另一个执行(测试器 *测试包.测试器) {
	当前条目 := 引导引擎(测试器)
	首个, 首个状态 := 预留引导(测试器, 当前条目, "first", "conversation-a")
	_, 其他状态 := 预留引导(测试器, 当前条目, "second", "conversation-b")
	采样, 结束 := 首个状态.采样上下文(首个.上下文(), 0)
	defer 结束()

	已进入, 释放 := make(chan struct{}), make(chan struct{})
	解除阻塞 := 同步包.一次函数(func() { close(释放) })
	defer 解除阻塞()
	已提交 := make(chan error, 1)
	go func() {
		已提交 <- 当前条目.接受引导(首个, func() (会话包.后续消息, error) {
			close(已进入)
			<-释放
			return 会话包.后续消息{会话标识: "conversation-a"}, nil
		})
	}()
	执行上下文, 取消函数 := 上下文包.创建超时上下文(测试器.上下文(), 3*时间包.秒)
	defer 取消函数()
	select {
	case <-已进入:
	case <-执行上下文.完成():
		测试器.报告致命错误("引导未进入提交边界")
	}
	进展 := make(chan error, 1)
	go func() {
		_, _, 错误 := 当前条目.快照引导(执行上下文, 其他状态)
		if 错误 == nil {
			错误 = 当前条目.之前工具(执行上下文, 其他状态, func() error { return nil })
		}
		进展 <- 错误
	}()
	select {
	case 错误 := <-进展:
		if 错误 != nil || 采样.获取错误() != nil {
			测试器.格式化报告致命错误("另一会话推进失败或引导在提交前取消采样: %v", 错误)
		}
	case <-执行上下文.完成():
		测试器.报告致命错误("会话 A 提交期间阻塞了会话 B")
	}
	解除阻塞()
	if 错误 := <-已提交; 错误 != nil || !错误判断包.是否属于(上下文包.原因(采样), 会话包.已引导错误) || 首个.上下文().获取错误() != nil {
		测试器.格式化报告致命错误("引导提交未按本次采样边界生效: %v", 错误)
	}
}

func Test引导拒绝执行租约来源前一个执行(测试器 *测试包.测试器) {
	当前条目 := 引导引擎(测试器)
	旧, 状态 := 预留引导(测试器, 当前条目, "same", "conversation")
	if 当前条目.引导执行租约("same") != 旧 {
		测试器.报告致命错误("未找到本次执行的引导租约")
	}
	当前条目.加密封装引导(测试器.上下文(), 状态)
	过时写入 := func() (会话包.后续消息, error) {
		测试器.报告致命错误("旧租约仍可提交引导")
		return 会话包.后续消息{}, nil
	}
	if 错误 := 当前条目.接受引导(旧, 过时写入); !错误判断包.是否属于(错误, 会话包.后续消息变化通知错误) {
		测试器.格式化报告致命错误("封存后仍接受引导: %v", 错误)
	}
	当前条目.移除引导(旧)
	旧.中止()

	当前, 当前状态 := 预留引导(测试器, 当前条目, "same", "conversation")
	if 当前 == 旧 || 当前条目.引导执行租约("same") != 当前 {
		测试器.报告致命错误("原地恢复未建立独立引导租约")
	}
	采样, 结束 := 当前状态.采样上下文(当前.上下文(), 0)
	defer 结束()
	if 错误 := 当前条目.接受引导(旧, 过时写入); !错误判断包.是否属于(错误, 会话包.后续消息变化通知错误) || 采样.获取错误() != nil {
		测试器.格式化报告致命错误("旧租约影响了新的模型采样: %v", 错误)
	}
	if 错误 := 当前条目.接受引导(当前, func() (会话包.后续消息, error) {
		return 会话包.后续消息{会话标识: "conversation"}, nil
	}); 错误 != nil || !错误判断包.是否属于(上下文包.原因(采样), 会话包.已引导错误) {
		测试器.格式化报告致命错误("当前租约的引导未生效: %v", 错误)
	}
}
