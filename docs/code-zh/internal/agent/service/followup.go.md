```go
package 服务

import (
	上下文包 "context"
	切片包 "slices"
	同步包 "sync"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	执行器包 "github.com/xiaoyu98628/callum/internal/agent/executor"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	历史包 "github.com/xiaoyu98628/callum/internal/conversation/history"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	工具包 "github.com/xiaoyu98628/callum/internal/tool"
	策略包 "github.com/xiaoyu98628/callum/internal/tool/policy"
)

type 执行引导 struct {
	执行标识 string
	互斥锁  同步包.互斥锁
	生成代次 uint64
	已加密  bool
	取消函数 上下文包.取消原因函数
}

// 引导执行租约 固定本次请求瞄准的执行。原地恢复时，同一 运行 标识 会对应另一个租约。
func (当前引擎 *引擎) 引导执行租约(执行标识 string) *执行器包.执行租约 {
	执行租约 := 当前引擎.管理器.活动执行租约(执行标识)
	if 当前引擎.引导状态(执行租约) == nil {
		return nil
	}
	return 执行租约
}

// 接受引导 先提交数据、发布用户消息，再唤醒当前模型调用。
func (当前引擎 *引擎) 接受引导(执行租约 *执行器包.执行租约, 写入 func() (会话包.后续消息, error)) error {
	状态 := 当前引擎.引导状态(执行租约)
	if 状态 == nil {
		return 会话包.后续消息变化通知错误
	}
	状态.互斥锁.加锁()
	defer 状态.互斥锁.解锁()
	if 状态.已加密 {
		return 会话包.后续消息变化通知错误
	}
	后续消息, 错误 := 写入()
	if 错误 != nil {
		return 错误
	}
	当前引擎.发布引导("message.created", 后续消息.回执)
	当前引擎.后续后续消息已更新(后续消息.会话标识, 状态.执行标识)
	状态.生成代次++
	if 状态.取消函数 != nil {
		状态.取消函数(会话包.已引导错误)
	}
	return nil
}

func (当前引擎 *引擎) 发布引导(类别 string, 消息 *会话包.消息) {
	if 消息 != nil {
		if 错误 := 当前引擎.发出事件(智能体包.事件{类型: 类别, 会话标识: 消息.会话标识, 执行标识: 消息.执行标识, 消息标识: 消息.标识, 数据: *消息}); 错误 != nil {
			当前引擎.日志记录器.警告("发布引导消息失败", "module", "conversation", "conversation_id", 消息.会话标识)
		}
	}
}

func (当前引擎 *引擎) 注册引导(执行租约 *执行器包.执行租约, 执行标识 string) *执行引导 {
	状态 := &执行引导{执行标识: 执行标识}
	当前引擎.引导互斥锁.加锁()
	当前引擎.引导[执行租约] = 状态
	当前引擎.引导互斥锁.解锁()
	return 状态
}

func (当前引擎 *引擎) 引导状态(执行租约 *执行器包.执行租约) *执行引导 {
	当前引擎.引导互斥锁.加锁()
	defer 当前引擎.引导互斥锁.解锁()
	return 当前引擎.引导[执行租约]
}

// 移除引导 在封存后释放登记，已取得旧状态的请求仍会看到 已加密。
func (当前引擎 *引擎) 移除引导(执行租约 *执行器包.执行租约) {
	当前引擎.引导互斥锁.加锁()
	delete(当前引擎.引导, 执行租约)
	当前引擎.引导互斥锁.解锁()
}

// 快照引导 让引导快照与受理互斥，避免在创建模型步骤期间漏掉新引导。
func (当前引擎 *引擎) 快照引导(执行上下文 上下文包.上下文, 状态 *执行引导) ([]会话包.后续消息, uint64, error) {
	状态.互斥锁.加锁()
	defer 状态.互斥锁.解锁()
	待处理, 错误 := 当前引擎.会话集.待处理引导(执行上下文, 状态.执行标识)
	return 待处理, 状态.生成代次, 错误
}

// 采样上下文 只取消模型调用，保留当前执行、已启动的工具和用户取消语义。
func (状态 *执行引导) 采样上下文(执行上下文 上下文包.上下文, 生成代次 uint64) (上下文包.上下文, func()) {
	请求, 取消函数 := 上下文包.带有取消原因(执行上下文)
	状态.互斥锁.加锁()
	if 状态.生成代次 != 生成代次 {
		取消函数(会话包.已引导错误)
	}
	状态.取消函数 = 取消函数
	状态.互斥锁.解锁()
	return 请求, func() {
		状态.互斥锁.加锁()
		状态.取消函数 = nil
		状态.互斥锁.解锁()
		取消函数(nil)
	}
}

func (当前引擎 *引擎) 加密封装引导(执行上下文 上下文包.上下文, 状态 *执行引导) []会话包.后续消息 {
	状态.互斥锁.加锁()
	状态.已加密 = true
	状态.互斥锁.解锁()
	待处理, 错误 := 当前引擎.会话集.待处理引导(执行上下文, 状态.执行标识)
	if 错误 != nil {
		当前引擎.日志记录器.记录错误("读取待恢复引导失败", "module", "run", "run_id", 状态.执行标识)
	}
	return 待处理
}

func (当前引擎 *引擎) 引导已重新排队(本次执行 会话包.执行, 待处理 []会话包.后续消息) {
	if len(待处理) == 0 {
		return
	}
	标识集 := make([]string, len(待处理))
	for 索引, 后续消息 := range 待处理 {
		标识集[索引] = 后续消息.标识
	}
	_ = 当前引擎.发出事件(智能体包.事件{类型: "conversation.steering_requeued", 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 已移除消息标识集: 标识集})
}

func (当前引擎 *引擎) 后续后续消息已更新(会话标识, 执行标识 string) {
	if 错误 := 当前引擎.发出事件(智能体包.事件{类型: "conversation.follow_ups_updated", 会话标识: 会话标识, 执行标识: 执行标识}); 错误 != nil {
		当前引擎.日志记录器.警告("发布后续消息更新失败", "module", "conversation", "conversation_id", 会话标识)
	}
}

// 之前工具 按同一顺序处理引导与工具的持久化启动边界。
func (当前引擎 *引擎) 之前工具(执行上下文 上下文包.上下文, 状态 *执行引导, 启动 func() error) error {
	状态.互斥锁.加锁()
	defer 状态.互斥锁.解锁()
	待处理, 错误 := 当前引擎.会话集.待处理引导(执行上下文, 状态.执行标识)
	if 错误 != nil {
		return 操作存储错误(执行上下文, 错误)
	}
	if len(待处理) > 0 {
		return 会话包.已引导错误
	}
	return 启动()
}

func (当前引擎 *引擎) 加密封装回答(执行上下文 上下文包.上下文, 状态 *执行引导) (bool, error) {
	状态.互斥锁.加锁()
	defer 状态.互斥锁.解锁()
	待处理, 错误 := 当前引擎.会话集.待处理引导(执行上下文, 状态.执行标识)
	if 错误 != nil {
		return false, 操作存储错误(执行上下文, 错误)
	}
	if len(待处理) != 0 {
		return false, nil
	}
	状态.已加密 = true
	return true, nil
}

func 引导消息集(项集 []会话包.后续消息) []模型包.消息 {
	输出 := make([]模型包.消息, 0, len(项集))
	for _, 项 := range 项集 {
		内容块集 := 切片包.复制(历史包.模型内容(项.内容))
		标记 := "以下是用户在当前任务执行期间发出的引导。以较新的明确要求为准，调整与它冲突的计划、范围和交付内容。继续完成原任务中仍适用的部分；如果用户要求停止或切换任务，按其意图执行。"
		if len(内容块集) > 0 && 内容块集[0].类型 == "text" {
			内容块集[0].文本 = 标记 + "\n\n" + 内容块集[0].文本
		} else {
			内容块集 = append([]模型包.内容块{{类型: "text", 文本: 标记}}, 内容块集...)
		}
		输出 = append(输出, 模型包.消息{角色: 模型包.角色用户, 内容块集: 内容块集})
	}
	return 输出
}

func (当前引擎 *引擎) 跳过已引导调用(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 父级 智能体包.步骤, 调用 模型包.工具调用) (模型包.消息, error) {
	步骤, 错误 := 当前引擎.创建工具步骤(执行上下文, 本次执行, 父级, 调用, nil, 策略包.决定{动作: 策略包.拒绝, 原因: "steered"}, nil)
	if 错误 != nil {
		return 模型包.消息{}, 错误
	}
	if 错误 := 当前引擎.事件(本次执行, "step.created", 步骤.标识, "", 步骤); 错误 != nil {
		return 模型包.消息{}, 错误
	}
	return 当前引擎.结束工具(执行上下文, 本次执行, 步骤, 调用, 工具包.结果{操作结果: 工具包.操作结果不已生效}, 会话包.已引导错误)
}

// 引导附在所属模型步骤上，供后续对话按实际使用的位置恢复。
func (当前引擎 *引擎) 应用引导(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 步骤标识 string, 项集 []会话包.后续消息) error {
	if len(项集) == 0 {
		return nil
	}
	标识集 := make([]string, len(项集))
	for 索引, 项 := range 项集 {
		标识集[索引] = 项.标识
	}
	清理, 取消函数 := 清理上下文(执行上下文)
	defer 取消函数()
	消息集, 错误 := 当前引擎.会话集.应用引导(清理, 本次执行.执行.标识, 步骤标识, 标识集)
	if 错误 != nil {
		return 操作存储错误(执行上下文, 错误)
	}
	for 索引 := range 消息集 {
		当前引擎.发布引导("message.updated", &消息集[索引])
	}
	return nil
}
```
