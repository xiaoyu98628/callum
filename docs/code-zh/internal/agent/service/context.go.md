package 服务

import (
	上下文包 "context"
	错误判断包 "errors"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	压缩包 "github.com/xiaoyu98628/callum/internal/conversation/compaction"
	历史包 "github.com/xiaoyu98628/callum/internal/conversation/history"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	词元包 "github.com/xiaoyu98628/callum/internal/model/tokens"
)

// 执行历史上下文 将当前用户要求和未完成批次独立保留，摘要只替换已验证的历史前缀。
type 执行历史上下文 struct {
	轨迹集 map[string][]模型包.消息
	过去  []模型包.消息
	当前  []模型包.消息
	引导  []模型包.消息
}

// 恢复 按仓储契约核对首位检查点，把真实用户引导与模型及工具轨迹分开。
// 检查点虽使用 用户 角色，仍须留在 当前 首位，供后续压缩替换；只有
// 引导 可以进入审批意图，不能仅凭模型角色或文本前缀推断摘要来源。
func (当前执行历史上下文 *执行历史上下文) 恢复(已保存 []模型包.消息, 检查点 *智能体包.上下文检查点) error {
	var 当前, 引导 []模型包.消息
	if 检查点 != nil {
		if len(已保存) == 0 {
			return 模型包.无效响应错误
		}
		首个, 预期 := 已保存[0], 检查点.消息()
		if 首个.角色 != 预期.角色 || 首个.内容 != 预期.内容 || len(首个.内容块集) != 0 ||
			len(首个.工具调用集) != 0 || 首个.工具调用标识 != "" || 首个.推理 != nil {
			return 模型包.无效响应错误
		}
		当前 = append(当前, 首个)
		已保存 = 已保存[1:]
	}
	for _, 消息 := range 已保存 {
		if 消息.角色 == 模型包.角色用户 {
			引导 = append(引导, 消息)
		} else {
			当前 = append(当前, 消息)
		}
	}
	当前执行历史上下文.当前, 当前执行历史上下文.引导 = 当前, 引导
	return nil
}

func 历史边界(本次执行 会话包.执行) (int64, int64) {
	下限 := int64(0)
	if 本次执行.上下文 != nil {
		下限 = 本次执行.上下文.通过序号
	}
	上界 := 本次执行.用户.序号
	if 本次执行.历史更多 {
		上界 = 下限 + 1
		for _, 行 := range 本次执行.历史 {
			if 行.标识 != 本次执行.用户.标识 {
				上界 = max(上界, 行.序号+1)
			}
		}
	}
	return 下限, 上界
}

func (当前引擎 *引擎) 重新加载历史(执行上下文 上下文包.上下文, 本次执行 *会话包.执行) error {
	行集, 版本, 更多, 错误 := 当前引擎.会话集.加载上下文历史(执行上下文, 本次执行.会话.标识, 本次执行.用户.序号)
	if 错误 != nil {
		return 操作存储错误(执行上下文, 错误)
	}
	if 本次执行.用户.标识 != "" {
		行集 = append(行集, 本次执行.用户)
	}
	本次执行.历史, 本次执行.上下文, 本次执行.历史更多 = 行集, 版本, 更多
	return nil
}

// 压缩历史 可在旧执行已超出整段恢复上限时，按有界完整批次修复其恢复上下文。
func (当前引擎 *引擎) 压缩历史(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 已解析 智能体包.已解析) (bool, error) {
	已见 := map[string]bool{}
	var 容量错误 error
	for _, 行 := range 本次执行.历史 {
		if 行.角色 != "user" || 行.标识 == 本次执行.用户.标识 || 已见[行.执行标识] {
			continue
		}
		已见[行.执行标识] = true
		值, _, 错误 := 当前引擎.压缩执行(执行上下文, 本次执行, 已解析, 行.执行标识, 0, false)
		if 错误 != nil {
			if 错误判断包.是否属于(错误, 压缩包.无减少错误) {
				continue
			}
			if 容量错误(错误) {
				容量错误 = 错误
				continue
			}
			return false, 错误
		}
		if 值 != nil {
			return true, nil
		}
	}
	return false, 容量错误
}

// 缩短历史 缩小准备窗口，后续页面仍从已成功激活的摘要边界重新读取，不丢弃历史。
func 缩短历史(本次执行 *会话包.执行) bool {
	var 开始 []int
	for 索引, 行 := range 本次执行.历史 {
		if 行.角色 == "user" && 行.标识 != 本次执行.用户.标识 {
			开始 = append(开始, 索引)
		}
	}
	if len(开始) <= 本次执行.执行.设置快照.压缩.保留轮次集+1 {
		return false
	}
	边界 := 开始[max(len(开始)/2, 本次执行.执行.设置快照.压缩.保留轮次集+1)]
	行集 := append([]会话包.消息(nil), 本次执行.历史[:边界]...)
	if 本次执行.用户.标识 != "" {
		行集 = append(行集, 本次执行.用户)
	}
	本次执行.历史, 本次执行.历史更多 = 行集, true
	return true
}

func (当前引擎 *引擎) 准备历史(执行上下文 上下文包.上下文, 本次执行 *会话包.执行, 已解析 智能体包.已解析, 手动 bool) (map[string][]模型包.消息, []模型包.消息, error) {
	for {
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			return nil, nil, 错误
		}
		下限, 上界 := 历史边界(*本次执行)
		轨迹集, 错误 := 当前引擎.仓储.上下文执行轨迹(执行上下文, 本次执行.会话.标识, 下限, 上界)
		var 消息集 []模型包.消息
		if 错误 == nil {
			消息集, 错误 = 历史包.组装(本次执行.历史, 轨迹集)
		}
		if 错误 != nil {
			if !容量错误(错误) || (!手动 && !自动压缩(*本次执行)) {
				return nil, nil, 错误
			}
			变化通知, 压缩错误 := 当前引擎.压缩历史(执行上下文, *本次执行, 已解析)
			if 压缩错误 != nil && !容量错误(压缩错误) {
				return nil, nil, 压缩错误
			}
			if 变化通知 {
				continue
			}
			if 缩短历史(本次执行) {
				continue
			}
			if 压缩错误 != nil {
				return nil, nil, 压缩错误
			}
			return nil, nil, 错误
		}
		if !本次执行.历史更多 {
			return 轨迹集, 消息集, nil
		}
		if !手动 && !自动压缩(*本次执行) {
			return nil, nil, 模型包.输入上限错误
		}
		保留 := 本次执行.执行.设置快照.压缩.保留轮次集
		if 本次执行.用户.标识 != "" {
			保留++
		}
		变化通知, 错误 := 当前引擎.压缩旧(执行上下文, 本次执行, 已解析, 轨迹集, 保留)
		if 错误 != nil && !容量错误(错误) {
			return nil, nil, 错误
		}
		if !变化通知 {
			变化通知, 压缩错误 := 当前引擎.压缩历史(执行上下文, *本次执行, 已解析)
			if 压缩错误 != nil {
				return nil, nil, 压缩错误
			}
			if 变化通知 {
				continue
			}
			if 错误 != nil {
				return nil, nil, 错误
			}
			return nil, nil, 压缩包.无减少错误
		}
		if 错误 := 当前引擎.重新加载历史(执行上下文, 本次执行); 错误 != nil {
			return nil, nil, 错误
		}
	}
}

func (当前执行历史上下文 *执行历史上下文) 消息集() []模型包.消息 {
	消息集 := make([]模型包.消息, 0, len(当前执行历史上下文.过去)+len(当前执行历史上下文.当前))
	消息集 = append(消息集, 当前执行历史上下文.过去...)
	return append(消息集, 当前执行历史上下文.当前...)
}

// 适配 同时用于模型调用前和工具结果追加后；剩余 仅构造容量占位，不执行或持久化。
func (当前执行历史上下文 *执行历史上下文) 请求(固定 []模型包.消息, 本次执行 会话包.执行) []模型包.消息 {
	return append(append(固定, 上下文消息集(本次执行.上下文, 当前执行历史上下文.消息集())...), 当前执行历史上下文.引导...)
}

func (当前执行历史上下文 *执行历史上下文) 适配(执行上下文 上下文包.上下文, 当前引擎 *引擎, 本次执行 *会话包.执行, 已解析 智能体包.已解析, 固定 []模型包.消息, 定义集 []模型包.工具定义, 计量器 *词元包.计量器, 之前 int64, 剩余 []模型包.工具调用) error {
	for {
		预览 := append(固定, 上下文消息集(本次执行.上下文, 当前执行历史上下文.消息集())...)
		for _, 调用 := range 剩余 {
			预览 = append(预览, 模型包.消息{角色: 模型包.角色工具, 工具调用标识: 调用.标识, 内容: "{}"})
			if len(预览) > 256 {
				break
			}
		}
		预览 = append(预览, 当前执行历史上下文.引导...)
		上限错误 := 输入容量(执行上下文, *本次执行, 已解析, 预览, 定义集, 计量器)
		if 上限错误 == nil {
			return nil
		}
		if !容量错误(上限错误) || !自动压缩(*本次执行) {
			return 上限错误
		}
		变化通知, 错误 := 当前引擎.压缩旧(执行上下文, 本次执行, 已解析, 当前执行历史上下文.轨迹集, 本次执行.执行.设置快照.压缩.保留轮次集+1)
		if 错误 != nil && !容量错误(错误) {
			return 错误
		}
		if 变化通知 {
			当前执行历史上下文.过去, 错误 = 历史包.组装(本次执行.历史, 当前执行历史上下文.轨迹集)
			if 错误 != nil {
				return 错误
			}
			continue
		}
		检查点, 已消费, 错误 := 当前引擎.压缩执行(执行上下文, *本次执行, 已解析, 本次执行.执行.标识, 之前, 之前 == 0)
		if 错误 != nil && !容量错误(错误) {
			return 错误
		}
		if 检查点 != nil {
			偏移 := 已消费
			if 检查点.基准标识 != "" {
				偏移++
			}
			if 偏移 > len(当前执行历史上下文.当前) {
				return 模型包.无效响应错误
			}
			当前执行历史上下文.当前 = append([]模型包.消息{检查点.消息()}, 当前执行历史上下文.当前[偏移:]...)
			continue
		}
		变化通知, 历史错误 := 当前引擎.压缩历史(执行上下文, *本次执行, 已解析)
		if 历史错误 != nil {
			return 历史错误
		}
		if !变化通知 {
			if 错误 != nil {
				return 错误
			}
			return 上限错误
		}
		当前执行历史上下文.轨迹集, 当前执行历史上下文.过去, 错误 = 当前引擎.准备历史(执行上下文, 本次执行, 已解析, false)
		if 错误 != nil {
			return 错误
		}
	}
}
