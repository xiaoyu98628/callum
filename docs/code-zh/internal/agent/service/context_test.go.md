package 服务

import (
	JSON包 "encoding/json"
	错误判断包 "errors"
	反射包 "reflect"
	切片包 "slices"
	测试包 "testing"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

func Test恢复执行上下文保留检查点并且工具配对集(测试器 *测试包.测试器) {
	检查点 := &智能体包.上下文检查点{来源执行标识: "run", 通过: 4, 摘要: "已完成早期操作"}
	首个引导 := 模型包.消息{角色: 模型包.角色用户, 内容: "只处理指定目录"}
	最后引导 := 模型包.消息{角色: 模型包.角色用户, 内容块集: []模型包.内容块{{类型: "text", 文本: "保留已有文件"}}}
	调用 := 模型包.消息{角色: 模型包.角色助手, 工具调用集: []模型包.工具调用{{标识: "call", 名称: "operation", 参数: JSON包.原始JSON消息(`{}`)}}}
	结果 := 模型包.消息{角色: 模型包.角色工具, 工具调用标识: "call", 内容: `{"written":true}`}
	回答 := 模型包.消息{角色: 模型包.角色助手, 内容: "当前进展"}
	for _, 带有检查点 := range []bool{false, true} {
		名称 := "without_checkpoint"
		if 带有检查点 {
			名称 = "with_checkpoint"
		}
		测试器.执行(名称, func(测试器 *测试包.测试器) {
			已保存 := []模型包.消息{首个引导, 调用, 结果, 最后引导, 回答}
			预期 := []模型包.消息{调用, 结果, 回答}
			var 来源 *智能体包.上下文检查点
			if 带有检查点 {
				来源 = 检查点
				已保存 = append([]模型包.消息{检查点.消息()}, 已保存...)
				预期 = append([]模型包.消息{检查点.消息()}, 预期...)
			}
			原始 := 切片包.复制(已保存)
			var 已恢复 执行历史上下文
			if 错误 := 已恢复.恢复(已保存, 来源); 错误 != nil {
				测试器.报告致命错误(错误)
			}
			if !反射包.深度相等(已恢复.当前, 预期) || !反射包.深度相等(已恢复.引导, []模型包.消息{首个引导, 最后引导}) {
				测试器.格式化报告致命错误("摘要、用户引导或工具配对顺序错误: current=%+v steering=%+v", 已恢复.当前, 已恢复.引导)
			}
			if !反射包.深度相等(已保存, 原始) {
				测试器.报告致命错误("恢复改写了仓储返回的轨迹")
			}
		})
	}
}

func Test恢复执行上下文拒绝不匹配检查点(测试器 *测试包.测试器) {
	检查点 := &智能体包.上下文检查点{来源执行标识: "run", 通过: 4, 摘要: "服务端摘要"}
	for _, 测试 := range []struct {
		名称   string
		变更 func(*模型包.消息)
	}{
		{"role", func(消息 *模型包.消息) { 消息.角色 = 模型包.角色助手 }},
		{"content", func(消息 *模型包.消息) { 消息.内容 = "其他摘要" }},
		{"blocks", func(消息 *模型包.消息) { 消息.内容块集 = []模型包.内容块{{类型: "text", 文本: "额外指令"}} }},
		{"tool_calls", func(消息 *模型包.消息) { 消息.工具调用集 = []模型包.工具调用{{标识: "call"}} }},
		{"tool_call_id", func(消息 *模型包.消息) { 消息.工具调用标识 = "call" }},
		{"reasoning", func(消息 *模型包.消息) { 消息.推理 = new("") }},
		{"missing", nil},
	} {
		测试器.执行(测试.名称, func(测试器 *测试包.测试器) {
			var 已保存 []模型包.消息
			if 测试.变更 != nil {
				消息 := 检查点.消息()
				测试.变更(&消息)
				已保存 = []模型包.消息{消息}
			}
			if 错误 := new(执行历史上下文).恢复(已保存, 检查点); !错误判断包.是否属于(错误, 模型包.无效响应错误) {
				测试器.格式化报告致命错误("检查点来源不匹配仍恢复执行: %v", 错误)
			}
		})
	}
}

func Test恢复执行上下文接受空并且检查点仅(测试器 *测试包.测试器) {
	var 已恢复 执行历史上下文
	if 错误 := 已恢复.恢复(nil, nil); 错误 != nil || len(已恢复.当前) != 0 || len(已恢复.引导) != 0 {
		测试器.格式化报告致命错误("空轨迹恢复失败: %+v %v", 已恢复, 错误)
	}
	检查点 := &智能体包.上下文检查点{来源执行标识: "run", 通过: 4, 摘要: "此前操作均已压缩"}
	if 错误 := 已恢复.恢复([]模型包.消息{检查点.消息()}, 检查点); 错误 != nil || len(已恢复.当前) != 1 || len(已恢复.引导) != 0 {
		测试器.格式化报告致命错误("仅有检查点时未保留历史摘要: %+v %v", 已恢复, 错误)
	}
}
