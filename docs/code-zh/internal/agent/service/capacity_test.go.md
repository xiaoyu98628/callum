package 服务

import (
	错误判断包 "errors"
	字符串包 "strings"
	测试包 "testing"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	设置包 "github.com/xiaoyu98628/callum/internal/settings"
)

func Test手动摘要需要预算当自动压缩已停用(测试器 *测试包.测试器) {
	启动配置 := 设置包.默认压缩()
	启动配置.已启用, 启动配置.词元预算 = false, 100
	本次执行 := 会话包.执行{执行: 智能体包.执行{设置快照: &设置包.快照{压缩: &设置包.压缩快照{压缩: 启动配置}}}}
	消息集 := []模型包.消息{{角色: 模型包.角色用户, 内容: 字符串包.重复("必须保留约束", 100)}}
	if 错误 := 输入容量(测试器.上下文(), 本次执行, 智能体包.已解析{}, 消息集, nil, nil); 错误 != nil {
		测试器.格式化报告致命错误("关闭自动压缩仍强制普通请求 Token 预算: %v", 错误)
	}
	if 错误 := 摘要容量(测试器.上下文(), 本次执行, 智能体包.已解析{}, 消息集); !错误判断包.是否属于(错误, 模型包.词元上限错误) {
		测试器.格式化报告致命错误("手动步骤摘要绕过 Token 预算: %v", 错误)
	}
	本次执行.执行.设置快照.压缩.词元预算 = 0
	if 错误 := 摘要容量(测试器.上下文(), 本次执行, 智能体包.已解析{}, 消息集); !错误判断包.是否属于(错误, 模型包.词元预算错误) {
		测试器.格式化报告致命错误("手动步骤摘要猜测未知容量: %v", 错误)
	}
}
