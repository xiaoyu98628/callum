package 服务

// 执行压缩 不创建或更新回答正文；连接取消和应用退出仍经统一调度器收敛。
func (当前引擎 *引擎) 执行压缩(执行上下文 上下文包.上下文, 执行租约 *执行器包.执行租约, 本次执行 会话包.执行, 已解析 智能体包.已解析) {
	错误 := 当前引擎.事件(本次执行, "run.started", "", "", 本次执行.执行)
	当前引擎.开始上下文用量(本次执行, 已解析)
	defer 当前引擎.结束上下文用量(本次执行)
	if 错误 == nil {
		错误 = 当前引擎.事件(本次执行, "step.created", 本次执行.步骤.标识, "", 本次执行.步骤)
	}
	if 错误 == nil && (本次执行.执行.设置快照 == nil || !本次执行.执行.设置快照.有效() || 本次执行.执行.设置快照.压缩 == nil) {
		错误 = 智能体包.无效错误
	}
	if 错误 == nil {
		轨迹集, _, 读取错误 := 当前引擎.准备历史(执行上下文, &本次执行, 已解析, true)
		错误 = 读取错误
		if 错误 == nil {
			错误 = 当前引擎.手动压缩(执行上下文, &本次执行, 已解析, 轨迹集)
		}
	}
	if 最终 := 执行租约.是否正在结束(); 最终 != nil {
		错误 = 错误判断包.合并(最终, 错误)
	}
	本次执行.执行.状态 = 智能体包.已完成
	if 错误 != nil {
		状态, 类别, 消息 := 失败(错误)
		本次执行.执行.状态, 本次执行.执行.错误类别, 本次执行.执行.错误消息 = 状态, &类别, &消息
	}
	当前时间 := 时间包.当前时间().Unix毫秒()
	本次执行.执行.更新时间, 本次执行.执行.结束时间 = 当前时间, &当前时间
	清理, 取消函数 := 清理上下文(执行上下文)
	defer 取消函数()
	if 结束错误 := 当前引擎.仓储.结束(清理, 本次执行.执行); 结束错误 != nil {
		当前引擎.日志记录器.记录错误("保存压缩执行终态失败", "module", "compaction", "run_id", 本次执行.执行.标识)
		本次执行.执行.状态, 本次执行.执行.错误类别, 本次执行.执行.错误消息 = 智能体包.失败, new("internal"), new("保存执行状态失败")
		当前引擎.结束上下文用量(本次执行)
		_ = 当前引擎.发出事件(智能体包.事件{类型: "run.failed", 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 数据: 本次执行.执行, 状态已持久化: new(false)})
		return
	}
	// 用量刷新先于终态通知，连接可在终态后排空并结束。
	当前引擎.结束上下文用量(本次执行)
	_ = 当前引擎.事件(本次执行, "run."+本次执行.执行.状态, "", "", 本次执行.执行)
}

// 手动压缩 按完整轮次持续推进，每批生成可追溯版本；资源有界，不限制调用批次。
func (当前引擎 *引擎) 手动压缩(执行上下文 上下文包.上下文, 本次执行 *会话包.执行, 已解析 智能体包.已解析, 轨迹集 map[string][]模型包.消息) error {
	步骤 := 本次执行.步骤
	首个 := true
	for {
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			return 错误
		}
		计划, 错误 := 当前引擎.压缩计划(执行上下文, *本次执行, 已解析, 轨迹集, 本次执行.执行.设置快照.压缩.保留轮次集)
		if 错误 != nil {
			return 错误
		}
		if len(计划.来源集) == 0 && !首个 {
			return nil
		}
		if !首个 {
			标识, 错误 := UUID包.新建UUIDv7()
			if 错误 != nil {
				return 错误
			}
			当前时间 := 时间包.当前时间().Unix毫秒()
			步骤, 错误 = 当前引擎.仓储.创建步骤(执行上下文, 智能体包.步骤{标识: 标识.字符串(), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 类型: "context_compression", 名称: "继续压缩历史上下文", 状态: 智能体包.待处理, 输入: 本次执行.步骤.输入, 创建时间: 当前时间, 更新时间: 当前时间})
			if 错误 != nil {
				return 操作存储错误(执行上下文, 错误)
			}
			if 错误 := 当前引擎.事件(*本次执行, "step.created", 步骤.标识, "", 步骤); 错误 != nil {
				return 错误
			}
		}
		版本, 错误 := 当前引擎.压缩(执行上下文, *本次执行, 已解析, 步骤, 计划)
		if 错误 != nil {
			return 错误
		}
		if len(计划.来源集) == 0 {
			return nil
		}
		首个 = false
		本次执行.上下文 = 版本
		已保留 := make([]会话包.消息, 0, len(本次执行.历史))
		for _, 行 := range 本次执行.历史 {
			if 行.序号 > 计划.通过 {
				已保留 = append(已保留, 行)
			}
		}
		本次执行.历史 = 已保留
	}
}

func 上下文消息集(版本 *会话包.上下文版本, 消息集 []模型包.消息) []模型包.消息 {
	if 版本 == nil {
		return 消息集
	}
	return append([]模型包.消息{压缩包.摘要消息(*版本)}, 消息集...)
}

// 压缩旧 压缩较早完整轮次；没有可用轮次时由调用方继续尝试步骤级压缩。
func (当前引擎 *引擎) 压缩旧(执行上下文 上下文包.上下文, 本次执行 *会话包.执行, 已解析 智能体包.已解析, 轨迹集 map[string][]模型包.消息, 保留 int) (bool, error) {
	计划, 错误 := 当前引擎.压缩计划(执行上下文, *本次执行, 已解析, 轨迹集, 保留)
	if 错误 != nil {
		return false, 错误
	}
	if len(计划.来源集) == 0 {
		return false, nil
	}
	标识, 错误 := UUID包.新建UUIDv7()
	if 错误 != nil {
		return false, 错误
	}
	当前时间 := 时间包.当前时间().Unix毫秒()
	输入, 错误 := JSON包.序列化(map[string]any{"purpose": "context_compaction", "trigger": map[string]string{"type": "automatic", "reason": "context_capacity"}})
	if 错误 != nil {
		return false, 错误
	}
	步骤, 错误 := 当前引擎.仓储.创建步骤(执行上下文, 智能体包.步骤{标识: 标识.字符串(), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 类型: "context_compression", 名称: "压缩历史上下文", 状态: 智能体包.待处理, 输入: 输入, 创建时间: 当前时间, 更新时间: 当前时间})
	if 错误 != nil {
		return false, 操作存储错误(执行上下文, 错误)
	}
	if 错误 := 当前引擎.事件(*本次执行, "step.created", 步骤.标识, "", 步骤); 错误 != nil {
		return false, 错误
	}
	版本, 错误 := 当前引擎.压缩(执行上下文, *本次执行, 已解析, 步骤, 计划)
	if 错误 != nil {
		return false, 错误
	}
	本次执行.上下文 = 版本
	已保留 := make([]会话包.消息, 0, len(本次执行.历史))
	for _, 行 := range 本次执行.历史 {
		if 行.序号 > 计划.通过 {
			已保留 = append(已保留, 行)
		}
	}
	本次执行.历史 = 已保留
	return true, nil
}

func 摘要输入(版本 *会话包.上下文版本, 消息集 []模型包.消息) []模型包.消息 {
	输入 := append([]模型包.消息{{角色: 模型包.角色系统, 内容: 压缩包.提示词}}, 上下文消息集(版本, 消息集)...)
	return append(输入, 模型包.消息{角色: 模型包.角色用户, 内容: 压缩包.请求})
}

func (当前引擎 *引擎) 压缩计划(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 已解析 智能体包.已解析, 轨迹集 map[string][]模型包.消息, 保留 int) (压缩包.计划, error) {
	启动配置 := 本次执行.执行.设置快照.压缩
	预算, 预算错误 := 启动配置.输入预算(已解析.上下文窗口, 已解析.最大输出词元)
	var 适配错误 error
	计划, 错误 := 压缩包.构建范围内(本次执行.历史, 轨迹集, 保留, func(消息集 []模型包.消息) bool {
		if 预算错误 != nil {
			适配错误 = 模型包.词元预算错误
			return false
		}
		if 词元包.估算(摘要输入(本次执行.上下文, 消息集), nil, 启动配置.图片词元估算) > 预算 {
			适配错误 = 模型包.词元上限错误
			return false
		}
		适配错误 = 输入容量(执行上下文, 本次执行, 已解析, 摘要输入(本次执行.上下文, 消息集), nil, nil)
		return 适配错误 == nil
	})
	if 错误 != nil && 适配错误 != nil {
		return 压缩包.计划{}, 适配错误
	}
	if 预算错误 != nil && 错误 != nil {
		return 压缩包.计划{}, 模型包.词元预算错误
	}
	// 每次成功必须消费新的源范围；无进展直接失败，不通过固定次数掩盖循环问题。
	if 错误 == nil && len(计划.来源集) > 0 {
		下限 := int64(0)
		if 本次执行.上下文 != nil {
			下限 = 本次执行.上下文.通过序号
		}
		if len(计划.消息集) == 0 || 计划.通过 <= 下限 || len(本次执行.历史) == 0 || 计划.通过 < 本次执行.历史[0].序号 {
			return 压缩包.计划{}, 压缩包.无减少错误
		}
	}
	return 计划, 错误
}

func (当前引擎 *引擎) 压缩(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 已解析 智能体包.已解析, 步骤 智能体包.步骤, 计划 压缩包.计划) (版本 *会话包.上下文版本, 返回错误 error) {
	var 元数据 map[string]any
	if 错误 := JSON包.反序列化(步骤.输入, &元数据); 错误 != nil {
		return nil, 错误
	}
	元数据["through_sequence"], 元数据["source_message_count"] = 计划.通过, len(计划.来源集)
	if 本次执行.上下文 != nil {
		元数据["base_context_version_id"] = 本次执行.上下文.标识
	}
	if 快照 := 本次执行.执行.设置快照; 快照 != nil && 快照.压缩 != nil {
		元数据["compaction_settings_revision"] = 快照.压缩.修订号
	}
	输入元数据, 错误 := JSON包.序列化(元数据)
	if 错误 != nil {
		return nil, 错误
	}
	步骤.输入 = 输入元数据
	当前时间 := 时间包.当前时间().Unix毫秒()
	步骤.状态, 步骤.开始时间, 步骤.更新时间 = 智能体包.运行中, &当前时间, 当前时间
	if 错误 := 当前引擎.仓储.保存步骤(执行上下文, 步骤); 错误 != nil {
		return nil, 操作存储错误(执行上下文, 错误)
	}
	if 错误 := 当前引擎.事件(本次执行, "step.updated", 步骤.标识, "", 步骤); 错误 != nil {
		return nil, 错误
	}
	已提交 := false
	defer func() {
		if 已提交 {
			return
		}
		清理, 取消函数 := 清理上下文(执行上下文)
		defer 取消函数()
		if 版本 != nil {
			返回错误 = 错误判断包.合并(返回错误, 当前引擎.会话集.失败上下文(清理, 版本.标识))
		}
		步骤.状态 = 智能体包.已完成
		if 返回错误 != nil {
			状态, 类别, 消息 := 失败(返回错误)
			步骤.状态, 步骤.错误类别, 步骤.错误消息 = 状态, &类别, &消息
		}
		当前时间 := 时间包.当前时间().Unix毫秒()
		步骤.更新时间, 步骤.结束时间 = 当前时间, &当前时间
		if 错误 := 当前引擎.仓储.保存步骤(清理, 步骤); 错误 != nil {
			返回错误 = 错误判断包.合并(返回错误, 智能体包.存储错误, 错误)
		} else {
			_ = 当前引擎.事件(本次执行, "step.updated", 步骤.标识, "", 步骤)
		}
	}()
	if len(计划.来源集) == 0 {
		步骤.输出 = JSON包.原始JSON消息(`{"compressed":false,"reason":"no_earlier_complete_turns"}`)
		return 本次执行.上下文, nil
	}
	来源 := 上下文消息集(本次执行.上下文, 计划.消息集)
	输入 := 摘要输入(本次执行.上下文, 计划.消息集)
	启动配置 := 本次执行.执行.设置快照.压缩
	预算, 错误 := 启动配置.输入预算(已解析.上下文窗口, 已解析.最大输出词元)
	if 错误 != nil {
		return nil, 模型包.词元预算错误
	}
	if 词元包.估算(输入, nil, 启动配置.图片词元估算) > 预算 {
		return nil, 模型包.词元上限错误
	}
	if 错误 := 输入容量(执行上下文, 本次执行, 已解析, 输入, nil, nil); 错误 != nil {
		return nil, 错误
	}
	标识, 错误 := UUID包.新建UUIDv7()
	if 错误 != nil {
		return nil, 错误
	}
	值 := 会话包.上下文版本{标识: 标识.字符串(), 会话标识: 本次执行.会话.标识, 已生成依据步骤标识: 步骤.标识, 来源集: 计划.来源集, 通过序号: 计划.通过}
	if 本次执行.上下文 != nil {
		值.基准版本标识 = &本次执行.上下文.标识
	}
	值, 错误 = 当前引擎.会话集.创建上下文(执行上下文, 值)
	if 错误 != nil {
		return nil, 操作存储错误(执行上下文, 错误)
	}
	版本 = &值
	结果, 错误 := 当前引擎.摘要模型(执行上下文, 本次执行, 已解析, 步骤, 输入, 值.标识)
	if 错误 != nil {
		return 版本, 错误
	}
	if len(结果.工具调用集) != 0 {
		return 版本, 模型包.无效响应错误
	}
	摘要, 错误 := 压缩包.校验(结果.文本)
	if 错误 != nil {
		return 版本, 错误
	}
	值.摘要 = 摘要
	之前, 错误 := JSON包.序列化(来源)
	if 错误 != nil {
		return 版本, 错误
	}
	之后, 错误 := JSON包.序列化([]模型包.消息{压缩包.摘要消息(值)})
	if 错误 != nil {
		return 版本, 错误
	}
	之前词元 := 词元包.估算(来源, nil, 启动配置.图片词元估算)
	之后词元 := 词元包.估算([]模型包.消息{压缩包.摘要消息(值)}, nil, 启动配置.图片词元估算)
	if 之后词元 >= 之前词元 {
		return 版本, 压缩包.无减少错误
	}
	值.统计, 错误 = JSON包.序列化(map[string]any{"before_messages": len(来源), "after_messages": 1, "before_bytes": len(之前), "after_bytes": len(之后), "estimated_before_tokens": 之前词元, "estimated_after_tokens": 之后词元, "estimation_method": 词元包.方法})
	if 错误 != nil {
		return 版本, 错误
	}
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 版本, 错误
	}
	if 错误 := 当前引擎.会话集.激活上下文(执行上下文, 值); 错误 != nil {
		return 版本, 操作存储错误(执行上下文, 错误)
	}
	已提交 = true
	当前时间 = 时间包.当前时间().Unix毫秒()
	值.状态, 值.已激活位于, 值.更新时间 = "active", &当前时间, 当前时间
	版本 = &值
	步骤.状态, 步骤.结束时间, 步骤.更新时间 = 智能体包.已完成, &当前时间, 当前时间
	步骤.输出, 错误 = JSON包.序列化(map[string]any{"compressed": true, "context_version_id": 值.标识, "size_stats": 值.统计})
	if 错误 != nil {
		return 版本, 错误
	}
	return 版本, 当前引擎.事件(本次执行, "step.updated", 步骤.标识, "", 步骤)
}
