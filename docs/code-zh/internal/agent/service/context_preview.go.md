package 服务

import (
	上下文包 "context"
	时间包 "time"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	历史包 "github.com/xiaoyu98628/callum/internal/conversation/history"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	词元包 "github.com/xiaoyu98628/callum/internal/model/tokens"
)

// 预览上下文 分批恢复同一消息边界的有效历史，只估算、不压缩、不调用模型。
func (当前引擎 *引擎) 预览上下文(执行上下文 上下文包.上下文, 输入 会话包.上下文用量请求, 状态 会话包.上下文状态, 已解析 智能体包.已解析) (会话包.上下文用量, error) {
	值 := 用量基准(输入.会话标识, "", 已解析, 状态.设置.压缩.压缩)
	值.修订号, 值.更新时间 = 当前引擎.上下文修订号(), 时间包.当前时间().Unix毫秒()
	if 状态.活动执行 != nil {
		值.执行标识 = 状态.活动执行.标识
		值.输入上限, 值.上下文窗口, 值.原因 = nil, nil, "context_preparing"
		return 值, nil
	}
	会话 := 状态.会话
	if 输入.会话标识 == "" {
		会话.工作区标识 = 输入.工作区标识
	}
	var 事实集 []智能体包.操作事实
	var 错误 error
	if 输入.会话标识 != "" {
		事实集, 错误 = 当前引擎.仓储.操作事实集(执行上下文, 输入.会话标识)
		if 错误 != nil {
			return 值, 操作存储错误(执行上下文, 错误)
		}
	}
	固定, 定义集, 错误 := 当前引擎.请求部分集(执行上下文, 会话, &状态.设置, 事实集, 当前引擎.额外工具集(执行上下文, false, 状态.设置.MCP修订号)...)
	if 错误 != nil {
		return 值, 错误
	}
	if 状态.上下文 != nil {
		固定 = append(固定, 上下文消息集(状态.上下文, nil)...)
	}
	总计 := 词元包.估算(固定, 定义集, 状态.设置.压缩.图片词元估算)
	之后 := int64(0)
	if 状态.上下文 != nil {
		之后 = 状态.上下文.通过序号
	}
	if 输入.会话标识 != "" {
		for {
			if 错误 := 执行上下文.获取错误(); 错误 != nil {
				return 值, 错误
			}
			行集, 更多, 错误 := 当前引擎.会话集.上下文页(执行上下文, 输入.会话标识, 之后, 状态.通过序号+1)
			if 错误 != nil {
				return 预览失败(值, 错误)
			}
			if len(行集) == 0 {
				if 更多 {
					return 值, 模型包.无效响应错误
				}
				break
			}
			最后 := 行集[len(行集)-1].序号
			if 最后 <= 之后 {
				return 值, 模型包.无效响应错误
			}
			轨迹集, 错误 := 当前引擎.仓储.上下文执行轨迹(执行上下文, 输入.会话标识, 之后, 最后+1)
			if 错误 != nil {
				return 预览失败(值, 错误)
			}
			消息集, 错误 := 历史包.组装(行集, 轨迹集)
			if 错误 != nil {
				return 预览失败(值, 错误)
			}
			总计 += 词元包.估算(消息集, nil, 状态.设置.压缩.图片词元估算) - 16
			之后 = 最后
			if !更多 {
				break
			}
		}
		// 读取期间可能受理新执行、编辑或切换摘要，不能返回混合版本的占用。
		最新, 错误 := 当前引擎.会话集.上下文状态(执行上下文, 输入.会话标识)
		if 错误 != nil {
			return 值, 错误
		}
		if 实时, 成功, 错误 := 当前引擎.实时上下文用量(输入); 成功 || 错误 != nil {
			return 实时, 错误
		}
		if !相同上下文状态(状态, 最新) {
			值.状态, 值.原因 = "refresh_required", "context_changed"
			return 值, nil
		}
	}
	值.状态, 值.估算词元 = "ready", new(总计)
	if 状态.实际模型标识 == 已解析.快照.标识 {
		值.实际输入词元 = 复制整数(状态.实际输入词元)
	}
	return 带有草稿(值, 输入.草稿, 输入.草稿图片集, 状态.设置.压缩.图片词元估算), nil
}

func 预览失败(值 会话包.上下文用量, 错误 error) (会话包.上下文用量, error) {
	if 容量错误(错误) {
		值.状态, 值.原因, 值.估算词元 = "unavailable", "history_limit", nil
		return 值, nil
	}
	return 值, 错误
}

func 相同上下文状态(当前上下文状态, 当前上下文状态 会话包.上下文状态) bool {
	if 当前上下文状态.活动执行 != nil || 当前上下文状态.通过序号 != 当前上下文状态.通过序号 || 当前上下文状态.最后用户执行标识 != 当前上下文状态.最后用户执行标识 || 当前上下文状态.会话.更新时间 != 当前上下文状态.会话.更新时间 || 当前上下文状态.设置.修订号 != 当前上下文状态.设置.修订号 || 当前上下文状态.设置.MCP修订号 != 当前上下文状态.设置.MCP修订号 || 当前上下文状态.设置.压缩.修订号 != 当前上下文状态.设置.压缩.修订号 {
		return false
	}
	if (当前上下文状态.上下文 == nil) != (当前上下文状态.上下文 == nil) {
		return false
	}
	return 当前上下文状态.上下文 == nil || 当前上下文状态.上下文.标识 == 当前上下文状态.上下文.标识
}
