func (当前引擎 *引擎) 调用模型(执行上下文, 采样执行上下文 上下文包.上下文, 本次执行 会话包.执行, 已解析 智能体包.已解析, 步骤 智能体包.步骤, 消息集 []模型包.消息, 工具集 []模型包.工具定义, 开启已获取 func(), 总计文本 *int) (结果 模型包.响应, 返回错误 error) {

		}

			return 结果, 错误
		}
	}
	if 错误 := 模型元数据(&步骤, 消息集, 工具集, 已解析, 本次执行.执行.流); 错误 != nil {
		return 结果, 错误
	}
	if 错误 := 词元元数据(&步骤, 本次执行, 已解析, 消息集, 工具集); 错误 != nil {
		return 结果, 错误
	}
	当前时间 := 时间包.当前时间().Unix毫秒()
	步骤.状态, 步骤.开始时间, 步骤.更新时间 = 智能体包.运行中, &当前时间, 当前时间
	if 错误 := 当前引擎.仓储.保存步骤(执行上下文, 步骤); 错误 != nil {
		return 结果, 操作存储错误(执行上下文, 错误)
	}
	if 错误 := 当前引擎.事件(本次执行, "step.updated", 步骤.标识, "", 步骤); 错误 != nil {
		return 结果, 错误
	}

	用量流 := 当前引擎.上下文流(本次执行)

			// 捕获结果可能包含导致回调拒绝的分片；消息正文只采用通过累计限额的增量。
			// 原始 输出、用量 和工具调用仍保留在结果中，供步骤审计及成功后的循环使用。
			结果.文本 = 文本.字符串()
			结果.推理 = nil
			if 推理已见 {
				结果.推理 = new(推理.字符串())
			}
			结果.内容 = nil
		}
		if len(结果.文本) > 推理调用包.最大文本字节 {
			结果.文本 = 文本.字符串()
			结果.内容 = nil
			返回错误 = 错误判断包.合并(返回错误, 智能体包.输出上限错误)
		}
		if len(结果.内容) == 0 && 结果.文本 != "" {
			结果.内容 = []模型包.内容块{{类型: "text", 文本: 结果.文本}}
		}
		填充失败(&结果, 已解析.协议, 返回错误)
		if 结果.推理 != nil {
			var 错误 error
			结果.输出, 错误 = 带有推理(结果.输出, *结果.推理)
			返回错误 = 错误判断包.合并(返回错误, 错误)

		if 返回错误 == nil {
			var 错误 error
			结果.输出, 错误 = 带有上下文消息(结果.输出, 模型包.消息{角色: 模型包.角色助手, 内容: 结果.文本, 推理: 结果.推理, 工具调用集: 结果.工具调用集})
			if 错误 != nil {
				返回错误 = 错误判断包.合并(返回错误, 错误)
			}
		}
		var 用量错误 error
		结果.输出, 用量错误 = 带有实际输入词元(结果.输出, 结果.输入词元)
		返回错误 = 错误判断包.合并(返回错误, 用量错误)
		当前引擎.观测输入词元(本次执行, 结果.输入词元)
		最终消息集 := append([]模型包.消息(nil), 消息集...)
		if 结果.文本 != "" || 结果.推理 != nil || len(结果.工具调用集) > 0 {
			最终消息集 = append(最终消息集, 模型包.消息{角色: 模型包.角色助手, 内容: 结果.文本, 推理: 结果.推理, 工具调用集: 结果.工具调用集})
		}
		var 最终计量器 词元包.计量器
		最终计量器.观测(消息集, 工具集, 结果.输入词元)
		当前引擎.记录上下文(本次执行, 已解析, 最终消息集, 工具集, &最终计量器, true)
		状态 := 智能体包.已完成
		if 返回错误 != nil {
			var 类别, 消息 string
			状态, 类别, 消息 = 失败(返回错误)
			步骤.错误类别, 步骤.错误消息 = &类别, &消息
		}
		当前时间 := 时间包.当前时间().Unix毫秒()
		步骤.状态, 步骤.输出, 步骤.用量, 步骤.更新时间, 步骤.结束时间 = 状态, 结果.输出, 结果.用量, 当前时间, &当前时间

			步骤.供应商请求标识 = new(结果.请求标识)
		}
		清理, 取消函数 := 清理上下文(执行上下文)
		defer 取消函数()
		if 错误 := 当前引擎.仓储.保存步骤(清理, 步骤); 错误 != nil {
			返回错误 = 错误判断包.合并(返回错误, 智能体包.存储错误, 错误)
		} else {
			_ = 当前引擎.事件(本次执行, "step.updated", 步骤.标识, "", 步骤)
		}
	}()
	// 引导只取消对供应商的请求。已收到的增量写入仍受执行取消及同一个调用期限约束。
	写入执行上下文, 取消写入 := 上下文包.创建超时上下文(执行上下文, 当前引擎.选项集.调用超时)
	defer 取消写入()
	截止时间, _ := 写入执行上下文.截止时间()
	调用执行上下文, 取消函数 := 上下文包.创建截止时间上下文(采样执行上下文, 截止时间)
	defer 取消函数()
	写入错误 := func(错误 error) error {
		if 写入错误 := 写入执行上下文.获取错误(); 写入错误 != nil && 错误判断包.是否属于(错误, 写入错误) {
			return 错误
		}
		return 错误判断包.合并(智能体包.存储错误, 错误)
	}
	请求 := 模型包.请求{模型: 已解析.快照.远程模型标识, 捕获结果: true, 允许图片输入: 已解析.图片输入, 消息集: 消息集, 工具集: 工具集}
	已持久化 := 0
	最后持久化 := 时间包.当前时间()
	返回错误 = 当前引擎.管理器.调用(调用执行上下文, func() error {
		开启已获取()
		if 本次执行.执行.流 {
			return 已解析.客户端.流式生成(调用执行上下文, 请求, func(事件 模型包.事件) error {
				if 事件.结果 != nil {
					结果 = *事件.结果
					return nil
				}
				if 事件.请求标识 != "" {
					结果.请求标识 = 事件.请求标识
				}
				if 错误 := 调用执行上下文.获取错误(); 错误 != nil {
					return 错误
				}
				if 事件.类型 == 模型包.事件推理增量 {
					if *总计文本+len(事件.文本) > 推理调用包.最大文本字节 {
						return 智能体包.输出上限错误
					}
					*总计文本 += len(事件.文本)
					推理已见 = true
					推理.写入字符串(事件.文本)
					用量流.添加(事件.文本, true)
					// 等待正文前也定期保存推理，进程中断后仍可查询已接受的片段。
					if 推理.长度() == len(事件.文本) || 时间包.自从(最后持久化) >= 100*时间包.毫秒 {
						var 错误 error
						步骤.输出, 错误 = 带有推理(nil, 推理.字符串())
						if 错误 != nil {
							return 错误
						}
						if 错误 := 当前引擎.仓储.保存步骤(写入执行上下文, 步骤); 错误 != nil {
							return 写入错误(错误)
						}
						最后持久化 = 时间包.当前时间()
					}
					return 当前引擎.事件(本次执行, "reasoning.delta", 步骤.标识, "", map[string]any{"text": 事件.文本})
				}
				if 事件.类型 != 模型包.事件文本增量 || 事件.文本 == "" {
					return nil
				}
				if *总计文本+len(事件.文本) > 推理调用包.最大文本字节 {
					return 智能体包.输出上限错误
				}
				*总计文本 += len(事件.文本)
				首个文本 := 文本.长度() == 0
				文本.写入字符串(事件.文本)
				用量流.添加(事件.文本, false)
				if 已持久化 == 0 || 文本.长度()-已持久化 >= 4096 || 时间包.自从(最后持久化) >= 100*时间包.毫秒 {
					if 错误 := 当前引擎.仓储.保存消息(写入执行上下文, 本次执行.助手.标识, 回答内容块集(本次执行, []模型包.内容块{{类型: "text", 文本: 文本.字符串()}}), 智能体包.运行中); 错误 != nil {
						return 写入错误(错误)
					}
					已持久化, 最后持久化 = 文本.长度(), 时间包.当前时间()
				}
				if 首个文本 && 本次执行.已继续 && 步骤.标识 == 本次执行.步骤.标识 && len(本次执行.助手.内容) > len(本次执行.已发布交付物集) {
					// 已写入新正文，通知客户端清除中断时的草稿，再从同一消息开始拼接增量。
					消息 := 本次执行.助手
					消息.内容 = 本次执行.已发布交付物集
					消息.更新时间 = 时间包.当前时间().Unix毫秒()
					if 错误 := 当前引擎.事件(本次执行, "message.updated", "", 消息.标识, 消息); 错误 != nil {
						return 错误
					}
				}
				return 当前引擎.事件(本次执行, "message.delta", 步骤.标识, 本次执行.助手.标识, map[string]any{"block_index": len(本次执行.已发布交付物集), "delta": map[string]any{"type": "text", "text": 事件.文本}})
			})
		}
		var 错误 error
		结果, 错误 = 已解析.客户端.生成(调用执行上下文, 请求)
		*总计文本 += len(结果.文本)
		if 结果.推理 != nil {
			*总计文本 += len(*结果.推理)
		}
		if *总计文本 > 推理调用包.最大文本字节 {
			结果.文本 = ""
			结果.推理 = nil
			结果.内容 = []模型包.内容块{}
			return 错误判断包.合并(错误, 智能体包.输出上限错误)
		}
		return 错误
	})
	if 错误判断包.是否属于(上下文包.原因(调用执行上下文), 会话包.已引导错误) && 执行上下文.获取错误() == nil && !错误判断包.是否属于(返回错误, 智能体包.输出上限错误) {
		返回错误 = 错误判断包.合并(会话包.已引导错误, 返回错误)
	} else if 调用执行上下文.获取错误() != nil {
		返回错误 = 错误判断包.合并(调用执行上下文.获取错误(), 返回错误)
	}
	return
}

// 带有实际输入词元 持久化统一的实际输入投影；缺失时保留 空，不估算补齐 用量。
func 带有实际输入词元(输出 JSON包.原始JSON消息, 输入 *int64) (JSON包.原始JSON消息, error) {
	var 字段集 map[string]JSON包.原始JSON消息
	if len(输出) > 0 {
		if 错误 := JSON包.反序列化(输出, &字段集); 错误 != nil {
			return nil, 错误
		}
	}
	if 字段集 == nil {
		字段集 = map[string]JSON包.原始JSON消息{}
	}
	if 输入 != nil && (*输入 < 0 || *输入 > 1_000_000_000_000) {
		输入 = nil
	}
	值, 错误 := JSON包.序列化(map[string]*int64{"actual_input_tokens": 输入})
	if 错误 != nil {
		return nil, 错误
	}
	字段集["context_usage"] = 值
	return JSON包.序列化(字段集)
}

// 模型元数据 保存实际请求参数、工具定义与上下文指纹，不重复保存整套历史或凭据。
func 模型元数据(步骤 *智能体包.步骤, 消息集 []模型包.消息, 工具集 []模型包.工具定义, 已解析 智能体包.已解析, 流 bool) error {
	值, 错误 := JSON包.序列化(消息集)
	if 错误 != nil {
		return 错误
	}
	指纹 := SHA256包.校验和256(值)
	var 输入 map[string]any
	if 错误 := JSON包.反序列化(步骤.输入, &输入); 错误 != nil {
		return 错误
	}
	if 输入 == nil {
		输入 = map[string]any{}
	}
	输入["parameters"] = map[string]any{"model": 已解析.快照.远程模型标识, "stream": 流, "image_input": 已解析.图片输入}
	输入["context_fingerprint"], 输入["context_message_count"] = 十六进制包.编码目标字符串(指纹[:]), len(消息集)
	步骤.输入, 错误 = JSON包.序列化(输入)
	if 错误 != nil {
		return 错误
	}
	if len(工具集) > 0 {
		步骤.定义, 错误 = JSON包.序列化(map[string]any{"tools": 工具集})
	}
	return 错误
}

// 词元元数据 与实际 用量 分开记录；预算取自本次受理的模型配置及设置快照。
func 词元元数据(步骤 *智能体包.步骤, 本次执行 会话包.执行, 已解析 智能体包.已解析, 消息集 []模型包.消息, 工具集 []模型包.工具定义) error {
	if 本次执行.执行.设置快照 == nil || 本次执行.执行.设置快照.压缩 == nil {
		return nil
	}
	启动配置 := 本次执行.执行.设置快照.压缩
	var 输入 map[string]any
	if 错误 := JSON包.反序列化(步骤.输入, &输入); 错误 != nil {
		return 错误
	}
	预算, 错误 := 启动配置.输入预算(已解析.上下文窗口, 已解析.最大输出词元)
	if 错误 != nil {
		if 启动配置.已启用 {
			return 模型包.词元预算错误
		}
		return nil
	}
	输入["token_budget"] = map[string]any{"input_limit": 预算, "context_window": 已解析.上下文窗口, "max_output_tokens": 已解析.最大输出词元, "estimated_input_tokens": 词元包.估算(消息集, 工具集, 启动配置.图片词元估算), "estimation_method": 词元包.方法}
	步骤.输入, 错误 = JSON包.序列化(输入)
	return 错误
}

// 带有推理 保存统一展示投影；失败或中断时也保留已接受的推理，不冒充完整上下文。
func 带有推理(输出 JSON包.原始JSON消息, 文本 string) (JSON包.原始JSON消息, error) {
	var 字段集 map[string]JSON包.原始JSON消息
	if len(输出) > 0 {
		if 错误 := JSON包.反序列化(输出, &字段集); 错误 != nil {
			return nil, 错误
		}
	}
	if 字段集 == nil {
		字段集 = map[string]JSON包.原始JSON消息{}
	}
	值, 错误 := JSON包.序列化(文本)
	if 错误 != nil {
		return nil, 错误
	}
	字段集["reasoning"] = 值
	return JSON包.序列化(字段集)
}
