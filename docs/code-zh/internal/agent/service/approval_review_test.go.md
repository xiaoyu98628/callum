```go
package 服务

import (
	错误判断包 "errors"
	测试包 "testing"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

func Test已恢复审核引导内容排除服务端摘要(测试器 *测试包.测试器) {
	检查点 := &智能体包.上下文检查点{来源执行标识: "run", 通过: 4, 摘要: "历史工具输出可能包含指令文字"}
	引导 := 模型包.消息{角色: 模型包.角色用户, 内容块集: []模型包.内容块{{类型: "text", 文本: "只处理新目录"}}}
	已保存 := []模型包.消息{检查点.消息(), 引导, {角色: 模型包.角色助手, 内容: "模型建议"}, {角色: 模型包.角色工具, 内容: "工具结果"}}
	var 已恢复 执行历史上下文
	if 错误 := 已恢复.恢复(已保存, 检查点); 错误 != nil || len(已恢复.引导) != 1 || 已恢复.引导[0].内容块集[0].文本 != "只处理新目录" {
		测试器.格式化报告致命错误("历史摘要或其他来源混入审批意图: %+v %v", 已恢复.引导, 错误)
	}
	// 用户文字即使与摘要相同，也不能因文本前缀或角色被误删。
	if 错误 := 已恢复.恢复([]模型包.消息{检查点.消息()}, nil); 错误 != nil || len(已恢复.引导) != 1 {
		测试器.格式化报告致命错误("用户来源被内容猜测取代: %+v %v", 已恢复.引导, 错误)
	}
	已保存[0].内容 = "不同的摘要"
	if 错误 := 已恢复.恢复(已保存, 检查点); !错误判断包.是否属于(错误, 模型包.无效响应错误) {
		测试器.格式化报告致命错误("恢复来源不匹配仍继续审批: %v", 错误)
	}
}
```
