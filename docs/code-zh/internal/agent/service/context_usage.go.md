package 服务

import (
	字符串包 "strings"
	时间包 "time"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	词元包 "github.com/xiaoyu98628/callum/internal/model/tokens"
	设置包 "github.com/xiaoyu98628/callum/internal/settings"
)

type 实时上下文用量 struct {
	值    会话包.上下文用量
	实际   *int64
	图片词元 int64
	最后发布 时间包.时间点
}

func 复制整数(值 *int64) *int64 {
	if 值 == nil {
		return nil
	}
	return new(*值)
}

func 复制用量(值 会话包.上下文用量) 会话包.上下文用量 {
	值.估算词元 = 复制整数(值.估算词元)
	值.输入上限 = 复制整数(值.输入上限)
	值.上下文窗口 = 复制整数(值.上下文窗口)
	值.实际输入词元 = 复制整数(值.实际输入词元)
	return 值
}

func 用量基准(标识, 执行标识 string, 已解析 智能体包.已解析, 启动配置 设置包.压缩) 会话包.上下文用量 {
	值 := 会话包.上下文用量{会话标识: 标识, 模型标识: 已解析.快照.标识, 执行标识: 执行标识, 状态: "preparing", 输出预留: 启动配置.输出预留, 安全余量: 启动配置.安全余量, 估算方法: 词元包.方法}
	if 已解析.上下文窗口 > 0 {
		值.上下文窗口 = new(已解析.上下文窗口)
	}
	if 已解析.最大输出词元 > 0 {
		值.输出预留 = min(值.输出预留, 已解析.最大输出词元)
	}
	if 预算, 错误 := 启动配置.输入预算(已解析.上下文窗口, 已解析.最大输出词元); 错误 == nil {
		值.输入上限 = new(预算)
	} else {
		值.原因 = "capacity_unconfigured"
	}
	return 值
}

// 上下文修订号 用于丢弃晚于运行状态变化才返回的只读预览，不写入业务数据库。
func (当前引擎 *引擎) 上下文修订号() int64 { return 当前引擎.用量修订号.加载() }

func (当前引擎 *引擎) 开始上下文用量(本次执行 会话包.执行, 已解析 智能体包.已解析) {
	if 本次执行.执行.设置快照 == nil || 本次执行.执行.设置快照.压缩 == nil {
		return
	}
	值 := 用量基准(本次执行.会话.标识, 本次执行.执行.标识, 已解析, 本次执行.执行.设置快照.压缩.压缩)
	当前引擎.用量互斥锁.加锁()
	当前引擎.用量[本次执行.会话.标识] = 实时上下文用量{值: 值, 图片词元: 本次执行.执行.设置快照.压缩.图片词元估算}
	当前引擎.用量互斥锁.解锁()
	当前引擎.存储上下文用量(值, true)
}

func (当前引擎 *引擎) 发布上下文用量(值 会话包.上下文用量) {
	if 错误 := 当前引擎.发出事件(智能体包.事件{类型: "conversation.context_usage.updated", 会话标识: 值.会话标识, 执行标识: 值.执行标识, 数据: 值}); 错误 != nil {
		当前引擎.日志记录器.警告("发布上下文占用失败", "module", "context_usage", "conversation_id", 值.会话标识, "run_id", 值.执行标识)
	}
}

// 存储上下文用量 只保存有界活动执行的数字快照；高频生成通知最多每 100ms 发布一次。
// 每次分片仍更新查询快照，既不保留正文副本，也不创建后台定时任务。
func (当前引擎 *引擎) 存储上下文用量(值 会话包.上下文用量, 强制 bool) {
	当前时间 := 时间包.当前时间()
	当前引擎.用量互斥锁.加锁()
	实时, 成功 := 当前引擎.用量[值.会话标识]
	if !成功 || 实时.值.执行标识 != 值.执行标识 {
		当前引擎.用量互斥锁.解锁()
		return
	}
	值.修订号, 值.更新时间 = 当前引擎.用量修订号.添加(1), 当前时间.Unix毫秒()
	值.实际输入词元 = 复制整数(实时.实际)
	实时.值 = 复制用量(值)
	发布 := 强制 || 当前时间.订阅(实时.最后发布) >= 100*时间包.毫秒
	if 发布 {
		实时.最后发布 = 当前时间
	}
	当前引擎.用量[值.会话标识] = 实时
	当前引擎.用量互斥锁.解锁()
	if 发布 {
		当前引擎.发布上下文用量(值)
	}
}

func (当前引擎 *引擎) 结束上下文用量(本次执行 会话包.执行) {
	当前引擎.用量互斥锁.加锁()
	实时, 成功 := 当前引擎.用量[本次执行.会话.标识]
	if 成功 && 实时.值.执行标识 == 本次执行.执行.标识 {
		delete(当前引擎.用量, 本次执行.会话.标识)
	} else {
		成功 = false
	}
	当前引擎.用量互斥锁.解锁()
	if !成功 {
		return
	}
	值 := 复制用量(实时.值)
	值.状态, 值.原因, 值.估算词元 = "refresh_required", "", nil
	值.修订号, 值.更新时间 = 当前引擎.用量修订号.添加(1), 时间包.当前时间().Unix毫秒()
	当前引擎.发布上下文用量(值)
}

func 带有草稿(值 会话包.上下文用量, 草稿 string, 图片数量 int, 图片词元 int64) 会话包.上下文用量 {
	值 = 复制用量(值)
	值.草稿词元 = int64(图片数量) * 图片词元
	if 草稿 = 字符串包.裁剪空间(草稿); 草稿 != "" {
		值.草稿词元 += 词元包.估算([]模型包.消息{{角色: 模型包.角色用户, 内容: 草稿}}, nil, 0) - 16
	}
	if 值.估算词元 != nil {
		*值.估算词元 += 值.草稿词元
	}
	return 值
}

// 实时上下文用量 只读取所属执行的数字快照，返回值与运行缓存相互隔离。
func (当前引擎 *引擎) 实时上下文用量(输入 会话包.上下文用量请求) (会话包.上下文用量, bool, error) {
	当前引擎.用量互斥锁.加读锁()
	实时, 成功 := 当前引擎.用量[输入.会话标识]
	值 := 复制用量(实时.值)
	当前引擎.用量互斥锁.解读锁()
	if !成功 {
		return 值, false, nil
	}
	if 输入.模型标识 != "" && 输入.模型标识 != 值.模型标识 {
		return 值, true, 会话包.繁忙错误
	}
	return 带有草稿(值, 输入.草稿, 输入.草稿图片集, 实时.图片词元), true, nil
}

func (当前引擎 *引擎) 记录上下文(本次执行 会话包.执行, 已解析 智能体包.已解析, 消息集 []模型包.消息, 定义集 []模型包.工具定义, 计量器 *词元包.计量器, 强制 bool) {
	if 本次执行.执行.设置快照 == nil || 本次执行.执行.设置快照.压缩 == nil {
		return
	}
	启动配置 := 本次执行.执行.设置快照.压缩
	值 := 用量基准(本次执行.会话.标识, 本次执行.执行.标识, 已解析, 启动配置.压缩)
	var 估算 词元包.估算结果
	if 计量器 != nil {
		估算 = 计量器.估算(消息集, 定义集, 启动配置.图片词元估算)
	} else {
		估算 = 词元包.估算结果{词元: 词元包.估算(消息集, 定义集, 启动配置.图片词元估算), 方法: 词元包.方法}
	}
	值.状态, 值.估算词元, 值.估算方法 = "ready", new(估算.词元), 估算.方法
	当前引擎.存储上下文用量(值, 强制)
}

type 流上下文用量 struct {
	引擎     *引擎
	基准     会话包.上下文用量
	文本, 推理 词元包.文本计数器
}

func (当前引擎 *引擎) 上下文流(本次执行 会话包.执行) 流上下文用量 {
	值, _, _ := 当前引擎.实时上下文用量(会话包.上下文用量请求{会话标识: 本次执行.会话.标识})
	return 流上下文用量{引擎: 当前引擎, 基准: 值}
}

func (当前流上下文用量 *流上下文用量) 添加(文本 string, 推理 bool) {
	if 推理 {
		当前流上下文用量.推理.添加(文本)
	} else {
		当前流上下文用量.文本.添加(文本)
	}
	if 当前流上下文用量.基准.估算词元 == nil {
		return
	}
	值 := 复制用量(当前流上下文用量.基准)
	*值.估算词元 += 8 + 词元包.文本(string(模型包.角色助手)) + 当前流上下文用量.文本.词元() + 当前流上下文用量.推理.词元()
	当前流上下文用量.引擎.存储上下文用量(值, false)
}

func (当前引擎 *引擎) 观测输入词元(本次执行 会话包.执行, 实际 *int64) {
	当前引擎.用量互斥锁.加锁()
	if 实时, 成功 := 当前引擎.用量[本次执行.会话.标识]; 成功 && 实时.值.执行标识 == 本次执行.执行.标识 {
		实时.实际 = nil
		if 实际 != nil && *实际 >= 0 && *实际 <= 1_000_000_000_000 {
			实时.实际 = 复制整数(实际)
		}
		实时.值.实际输入词元 = 复制整数(实时.实际)
		当前引擎.用量[本次执行.会话.标识] = 实时
	}
	当前引擎.用量互斥锁.解锁()
}
