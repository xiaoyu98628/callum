package 服务

import (
	上下文包 "context"
	JSON包 "encoding/json"
	错误判断包 "errors"
	时间包 "time"

	UUID包 "github.com/google/uuid"
	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

// 获取输入 与 输入集 返回固定问题和已保存回复；不暴露写入步骤的通用能力。
func (当前引擎 *引擎) 获取输入(执行上下文 上下文包.上下文, 标识 string) (智能体包.用户输入, error) {
	if !有效标识(标识) {
		return 智能体包.用户输入{}, 智能体包.无效错误
	}
	步骤, 错误 := 当前引擎.仓储.获取步骤(执行上下文, 标识)
	if 错误 != nil {
		return 智能体包.用户输入{}, 错误
	}
	return 智能体包.输入来源步骤(步骤)
}
func (当前引擎 *引擎) 输入集(执行上下文 上下文包.上下文, 执行标识 string, 待处理 bool, 上限, 偏移 int) ([]智能体包.用户输入, error) {
	if !有效标识(执行标识) || 上限 < 1 || 上限 > 200 || 偏移 < 0 {
		return nil, 智能体包.无效错误
	}
	行集, 错误 := 当前引擎.仓储.用户输入步骤集(执行上下文, 执行标识, 待处理, 上限, 偏移)
	if 错误 != nil {
		return nil, 错误
	}
	输出 := make([]智能体包.用户输入, 0, len(行集))
	for _, 行 := range 行集 {
		值, 错误 := 智能体包.输入来源步骤(行)
		if 错误 != nil {
			return nil, 错误
		}
		输出 = append(输出, 值)
	}
	return 输出, nil
}

func (当前引擎 *引擎) 提交输入(执行上下文 上下文包.上下文, 标识, 执行标识 string, 回答 智能体包.输入回答) (智能体包.用户输入, error) {
	if !有效标识(标识) || !有效标识(执行标识) {
		return 智能体包.用户输入{}, 智能体包.无效错误
	}
	// 落库、发布和唤醒有固定次序，执行器不会先生成下一轮再发布旧的等待状态。
	当前引擎.输入互斥锁.加锁()
	defer 当前引擎.输入互斥锁.解锁()
	步骤, 变化通知, 错误 := 当前引擎.仓储.回答用户输入(执行上下文, 标识, 执行标识, 回答)
	if 错误 != nil {
		return 智能体包.用户输入{}, 错误
	}
	值, 错误 := 智能体包.输入来源步骤(步骤)
	if 错误 != nil {
		return 值, 错误
	}
	if 变化通知 {
		发布错误 := 当前引擎.发布输入(执行上下文, "user_input.updated", 步骤)
		if 通道 := 当前引擎.输入信号集[标识]; 通道 != nil {
			select {
			case 通道 <- struct{}{}:
			default:
			}
		}
		return 值, 发布错误
	}
	return 值, nil
}

func (当前引擎 *引擎) 发布输入(执行上下文 上下文包.上下文, 名称 string, 步骤 智能体包.步骤) error {
	值, 错误 := 智能体包.输入来源步骤(步骤)
	if 错误 != nil {
		return 错误
	}
	if 错误 := 当前引擎.发出事件(智能体包.事件{类型: 名称, 会话标识: 步骤.会话标识, 执行标识: 步骤.执行标识, 步骤标识: 步骤.标识, 数据: 值}); 错误 != nil {
		return 错误
	}
	if 错误 := 当前引擎.发出事件(智能体包.事件{类型: "step.updated", 会话标识: 步骤.会话标识, 执行标识: 步骤.执行标识, 步骤标识: 步骤.标识, 数据: 步骤}); 错误 != nil {
		return 错误
	}
	运行, 错误 := 当前引擎.仓储.获取(执行上下文, 步骤.执行标识)
	if 错误 != nil {
		return 错误
	}
	if 错误 := 当前引擎.发出事件(智能体包.事件{类型: "run.updated", 会话标识: 步骤.会话标识, 执行标识: 步骤.执行标识, 数据: 运行}); 错误 != nil {
		return 错误
	}
	行集, 错误 := 当前引擎.会话集.消息集(执行上下文, 步骤.会话标识, 0, 1)
	if 错误 != nil {
		return 错误
	}
	if len(行集) != 1 || 行集[0].执行标识 != 步骤.执行标识 || 行集[0].角色 != "assistant" {
		return 智能体包.冲突错误
	}
	return 当前引擎.发出事件(智能体包.事件{类型: "message.updated", 会话标识: 步骤.会话标识, 执行标识: 步骤.执行标识, 消息标识: 行集[0].标识, 数据: 行集[0]})
}

func (当前引擎 *引擎) 询问用户(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 引导 *执行引导, 父级 智能体包.步骤, 调用 模型包.工具调用) (模型包.消息, error) {
	问题, 错误 := 智能体包.解析输入问题(调用.参数)
	if 错误 != nil {
		return 当前引擎.无效问题(执行上下文, 本次执行, 父级, 调用)
	}
	标识, 错误 := UUID包.新建UUIDv7()
	if 错误 != nil {
		return 模型包.消息{}, 错误
	}
	原始, 错误 := JSON包.序列化(问题)
	if 错误 != nil {
		return 模型包.消息{}, 错误
	}
	定义, 错误 := JSON包.序列化(智能体包.用户输入定义())
	if 错误 != nil {
		return 模型包.消息{}, 错误
	}
	当前时间 := 时间包.当前时间().Unix毫秒()
	var 步骤 智能体包.步骤
	错误 = 当前引擎.之前工具(执行上下文, 引导, func() error {
		var 错误 error
		步骤, 错误 = 当前引擎.仓储.创建用户输入(执行上下文, 智能体包.步骤{标识: 标识.字符串(), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 父级步骤标识: new(父级.标识), 工具调用标识: new(调用.标识), 类型: "user_input", 名称: "等待用户回复", 定义: 定义, 输入: 原始, 状态: 智能体包.等待中用户输入, 创建时间: 当前时间, 更新时间: 当前时间, 开始时间: &当前时间})
		return 错误
	})
	if 错误判断包.是否属于(错误, 会话包.已引导错误) {
		return 模型包.消息{}, 错误
	}
	if 错误 != nil {
		return 模型包.消息{}, 操作存储错误(执行上下文, 错误)
	}
	if 错误 := 当前引擎.事件(本次执行, "step.created", 步骤.标识, "", 步骤); 错误 != nil {
		return 模型包.消息{}, 错误
	}
	if 错误 := 当前引擎.发布输入(执行上下文, "user_input.requested", 步骤); 错误 != nil {
		return 模型包.消息{}, 错误
	}
	步骤, 错误 = 当前引擎.等待输入(执行上下文, 步骤.标识)
	if 错误 != nil {
		return 模型包.消息{}, 错误
	}
	var 输出 struct {
		消息 模型包.消息 `json:"context_message"`
	}
	if JSON包.反序列化(步骤.输出, &输出) != nil || 输出.消息.角色 != 模型包.角色工具 || 输出.消息.工具调用标识 != 调用.标识 {
		return 模型包.消息{}, 智能体包.无效错误
	}
	return 输出.消息, nil
}

func (当前引擎 *引擎) 等待输入(执行上下文 上下文包.上下文, 标识 string) (智能体包.步骤, error) {
	通道 := make(chan struct{}, 1)
	当前引擎.输入互斥锁.加锁()
	if 当前引擎.输入信号集[标识] != nil {
		当前引擎.输入互斥锁.解锁()
		return 智能体包.步骤{}, 智能体包.冲突错误
	}
	当前引擎.输入信号集[标识] = 通道
	当前引擎.输入互斥锁.解锁()
	defer func() { 当前引擎.输入互斥锁.加锁(); delete(当前引擎.输入信号集, 标识); 当前引擎.输入互斥锁.解锁() }()
	for {
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			return 当前引擎.取消输入(执行上下文, 标识, 错误)
		}
		步骤, 错误 := 当前引擎.仓储.获取步骤(执行上下文, 标识)
		if 错误 != nil {
			return 步骤, 错误
		}
		if 步骤.状态 == 智能体包.已完成 {
			return 步骤, nil
		}
		if 步骤.状态 != 智能体包.等待中用户输入 {
			return 步骤, 智能体包.冲突错误
		}
		select {
		case <-执行上下文.完成():
			return 当前引擎.取消输入(执行上下文, 标识, 执行上下文.获取错误())
		case <-通道:
		}
	}
}
func (当前引擎 *引擎) 取消输入(执行上下文 上下文包.上下文, 标识 string, 原因 error) (智能体包.步骤, error) {
	清理, 取消函数 := 清理上下文(执行上下文)
	defer 取消函数()
	步骤, 错误 := 当前引擎.仓储.取消用户输入(清理, 标识)
	if 错误 == nil {
		错误 = 当前引擎.发布输入(清理, "user_input.updated", 步骤)
	}
	return 步骤, 错误判断包.合并(原因, 错误)
}

// 无效提问也保存调用结果，避免后续上下文出现没有对应结果的工具调用。
func (当前引擎 *引擎) 无效问题(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 父级 智能体包.步骤, 调用 模型包.工具调用) (模型包.消息, error) {
	消息 := 模型包.消息{角色: 模型包.角色工具, 工具调用标识: 调用.标识, 内容: `{"error":{"kind":"invalid_arguments","message":"问题参数无效，需要 question 和 options，最多 8 个不重复选项"}}`}
	输出, 错误 := JSON包.序列化(map[string]any{"context_message": 消息})
	if 错误 != nil {
		return 消息, 错误
	}
	标识, 错误 := UUID包.新建UUIDv7()
	if 错误 != nil {
		return 消息, 错误
	}
	当前时间 := 时间包.当前时间().Unix毫秒()
	输入 := 调用.参数
	if !JSON包.有效(输入) {
		输入 = JSON包.原始JSON消息(`{}`)
	}
	步骤, 错误 := 当前引擎.仓储.创建步骤(执行上下文, 智能体包.步骤{标识: 标识.字符串(), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 父级步骤标识: new(父级.标识), 工具调用标识: new(调用.标识), 类型: "tool_call", 名称: 智能体包.用户输入工具, 输入: 输入, 输出: 输出, 状态: 智能体包.失败, 创建时间: 当前时间, 更新时间: 当前时间, 结束时间: &当前时间})
	if 错误 != nil {
		return 消息, 操作存储错误(执行上下文, 错误)
	}
	return 消息, 当前引擎.事件(本次执行, "step.created", 步骤.标识, "", 步骤)
}
