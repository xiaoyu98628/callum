```go
package 服务

import (
	上下文包 "context"
	错误判断包 "errors"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	会话包 "github.com/xiaoyu98628/callum/internal/conversation"
	历史包 "github.com/xiaoyu98628/callum/internal/conversation/history"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	词元包 "github.com/xiaoyu98628/callum/internal/model/tokens"
)

func 容量错误(错误 error) bool {
	return 错误判断包.是否属于(错误, 模型包.输入上限错误) || 错误判断包.是否属于(错误, 模型包.词元上限错误)
}

// 输入容量 统一检查 词元、消息数量和适配器实际编码大小，使用同一执行设置快照。
func 输入容量(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 已解析 智能体包.已解析, 消息集 []模型包.消息, 定义集 []模型包.工具定义, 计量器 *词元包.计量器) error {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	if 错误 := 历史包.检查(消息集); 错误 != nil {
		return 错误
	}
	if 快照 := 本次执行.执行.设置快照; 快照 != nil && 快照.压缩 != nil && 快照.压缩.已启用 {
		启动配置 := 快照.压缩
		预算, 错误 := 启动配置.输入预算(已解析.上下文窗口, 已解析.最大输出词元)
		if 错误 != nil {
			return 模型包.词元预算错误
		}
		var 估算 int64
		if 计量器 == nil {
			估算 = 词元包.估算(消息集, 定义集, 启动配置.图片词元估算)
		} else {
			估算 = 计量器.估算(消息集, 定义集, 启动配置.图片词元估算).词元
		}
		if 估算 > 预算 {
			return 模型包.词元上限错误
		}
	}
	if 校验器, 成功 := 已解析.客户端.(模型包.输入校验器); 成功 {
		return 校验器.校验输入(执行上下文, 模型包.请求{模型: 已解析.快照.远程模型标识, 允许图片输入: 已解析.图片输入, 消息集: 消息集, 工具集: 定义集})
	}
	return nil
}

func 自动压缩(本次执行 会话包.执行) bool {
	return 本次执行.执行.设置快照 != nil && 本次执行.执行.设置快照.压缩 != nil && 本次执行.执行.设置快照.压缩.已启用
}

// 摘要容量 手动压缩也必须具备有效预算，不随自动压缩开关关闭该约束。
func 摘要容量(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 已解析 智能体包.已解析, 消息集 []模型包.消息) error {
	if 本次执行.执行.设置快照 == nil || 本次执行.执行.设置快照.压缩 == nil {
		return 模型包.词元预算错误
	}
	启动配置 := 本次执行.执行.设置快照.压缩
	预算, 错误 := 启动配置.输入预算(已解析.上下文窗口, 已解析.最大输出词元)
	if 错误 != nil {
		return 模型包.词元预算错误
	}
	if 词元包.估算(消息集, nil, 启动配置.图片词元估算) > 预算 {
		return 模型包.词元上限错误
	}
	return 输入容量(执行上下文, 本次执行, 已解析, 消息集, nil, nil)
}
```
