func (当前引擎 *引擎) 循环(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 已解析 智能体包.已解析, 引导 *执行引导, 开始标题 func()) error {

	if 错误 != nil {
		return 错误
	}
	当前 := 执行历史上下文{轨迹集: 轨迹集, 过去: 消息集}

	已拒绝 := map[string]bool{}
	if 本次执行.已继续 {
		已保存, 错误 := 当前引擎.仓储.继续执行轨迹(执行上下文, 本次执行.会话.标识, 本次执行.执行.标识)
		if 错误 != nil {
			return 操作存储错误(执行上下文, 错误)
		}
		检查点, 错误 := 当前引擎.仓储.最新检查点(执行上下文, 本次执行.会话.标识, 本次执行.执行.标识)
		if 错误 != nil {
			return 操作存储错误(执行上下文, 错误)
		}
		if 错误 := 当前.恢复(已保存, 检查点); 错误 != nil {
			return 错误
		}
		审核引导内容 = 切片包.复制(当前.引导)
		键集, 错误 := 当前引擎.仓储.已拒绝操作键集(执行上下文, 本次执行.执行.标识)
		if 错误 != nil {
			return 操作存储错误(执行上下文, 错误)
		}
		for _, 键 := range 键集 {

		}
	}
	事实集, 错误 := 当前引擎.仓储.操作事实集(执行上下文, 本次执行.会话.标识)
	if 错误 != nil {
		return 操作存储错误(执行上下文, 错误)
	}
	待核实 := map[string]bool{}
	for _, 事实 := range 事实集 {

	}
	额外 := 当前引擎.额外工具集(执行上下文, true, 本次执行.执行.设置快照.MCP修订号)
	额外依据名称 := make(map[string]工具包.工具, len(额外))
	for _, 当前项 := range 额外 {
		额外依据名称[当前项.获取定义().名称] = 当前项
	}
	固定, 定义集, 错误 := 当前引擎.请求部分集(执行上下文, 本次执行.会话, 本次执行.执行.设置快照, 事实集, 额外...)
	if 错误 != nil {
		return 错误
	}
	if 本次执行.已继续 {
		固定[0].内容 += "\n\n用户通过「继续任务」恢复上次停止或中断的任务。结合原需求、已收到的引导及工具结果，完成剩余工作。把确认已经生效的操作作为进度使用。结果未知或部分生效时先用只读方式核实，任何进一步操作仍须经过现有权限检查。"

	总计文本 := 0
	步骤 := 本次执行.步骤

	前一个拒绝 := ""
	已生效引导 := len(当前.引导)
	for 轮次 := int64(0); ; 轮次++ {
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			return 错误
		}
		// 参数反馈是本次请求的有界固定上下文，计入预算，不写入用户消息或历史摘要。

		if 纠正 != nil {
			请求固定 = append(请求固定, 纠正.消息())
		}
		待处理, 生成代次, 错误 := 当前引擎.快照引导(执行上下文, 引导)
		if 错误 != nil {
			return 操作存储错误(执行上下文, 错误)
		}
		待处理消息集 := 引导消息集(待处理)
		当前.引导 = append(当前.引导[:已生效引导], 待处理消息集...)
		// 审核只保留真实用户来源；恢复轨迹中的历史摘要即使使用 用户 角色，也不能成为授权依据。
		轮次引导内容 := append(append([]模型包.消息(nil), 审核引导内容...), 待处理消息集...)
		if 错误 := 当前.适配(执行上下文, 当前引擎, &本次执行, 已解析, 请求固定, 定义集, &计量器, 0, nil); 错误 != nil {
			return 错误
		}
		生效 := 当前.请求(请求固定, 本次执行)
		当前引擎.记录上下文(本次执行, 已解析, 生效, 定义集, &计量器, true)
		引导标识集 := make([]string, 0, len(待处理))
		for _, 项 := range 待处理 {
			引导标识集 = append(引导标识集, 项.标识)
		}
		if 轮次 > 0 {
			标识, 错误 := UUID包.新建UUIDv7()
			if 错误 != nil {
				return 错误
			}
			当前时间 := 时间包.当前时间().Unix毫秒()
			字段集 := map[string]any{"purpose": "answer", "protocol": 已解析.协议, "round": 轮次 + 1, "previous_step_id": 步骤.标识}
			if len(引导标识集) > 0 {
				字段集["steering_ids"] = 引导标识集
			}
			if 纠正 != nil {
				字段集["tool_arguments_correction"] = 纠正
			}
			输入, 错误 := JSON包.序列化(字段集)
			if 错误 != nil {
				return 错误
			}
			步骤, 错误 = 当前引擎.仓储.创建步骤(执行上下文, 智能体包.步骤{标识: 标识.字符串(), 会话标识: 本次执行.会话.标识, 执行标识: 本次执行.执行.标识, 类型: "model_call", 名称: "继续生成回答", 状态: 智能体包.待处理, 输入: 输入, 模型: &已解析.快照, 创建时间: 当前时间, 更新时间: 当前时间})
			if 错误 != nil {
				return 操作存储错误(执行上下文, 错误)
			}
			if 错误 := 当前引擎.事件(本次执行, "step.created", 步骤.标识, "", 步骤); 错误 != nil {
				return 错误
			}
		}
		if 轮次 == 0 && len(引导标识集) > 0 {
			var 字段集 map[string]any
			if 错误 := JSON包.反序列化(步骤.输入, &字段集); 错误 != nil {
				return 错误
			}
			字段集["steering_ids"] = 引导标识集
			步骤.输入, 错误 = JSON包.序列化(字段集)
			if 错误 != nil {
				return 错误
			}
		}
		回调 := func() {}
		if 轮次 == 0 {
			回调 = 开始标题
		}
		采样执行上下文, 结束采样 := 引导.采样上下文(执行上下文, 生成代次)
		结果, 调用错误 := 当前引擎.调用模型(执行上下文, 采样执行上下文, 本次执行, 已解析, 步骤, 生效, 定义集, 回调, &总计文本)
		结束采样()
		if 调用错误 == nil {
			if 错误 := 当前引擎.应用引导(执行上下文, 本次执行, 步骤.标识, 待处理); 错误 != nil {
				return 错误
			}
			已生效引导 = len(当前.引导)
			审核引导内容 = 轮次引导内容
		}
		if 错误判断包.是否属于(调用错误, 会话包.已引导错误) && !错误判断包.是否属于(调用错误, 智能体包.存储错误) && 执行上下文.获取错误() == nil {
			if 错误 := 当前引擎.保存回答(执行上下文, 本次执行, []模型包.内容块{}, ""); 错误 != nil {
				return 错误
			}
			continue
		}
		if 调用错误 == nil || !本次执行.已继续 || 轮次 > 0 || len(结果.内容) > 0 || 结果.文本 != "" {
			if 错误 := 当前引擎.保存回答(执行上下文, 本次执行, 结果.内容, 结果.文本); 错误 != nil {
				return 错误
			}
		}
		if 调用错误 != nil {
			if 执行上下文.获取错误() != nil {
				return 错误判断包.合并(执行上下文.获取错误(), 调用错误)
			}
			无效, 可修复 := 唯一参数错误(调用错误)
			if !可修复 || (无效.上限字节 == 0 && 结果.结束原因 != "tool_calls") || len(结果.输出) == 0 || len(结果.工具调用集) != 0 {
				return 调用错误
			}
			// 尚未进入工具循环，整批均未执行；保留失败模型步骤及实际 用量，重新生成参数。
			if 纠正 != nil && 纠正.指纹 == 无效.指纹 {
				return &模型包.工具纠正停滞错误{}
			}
			纠正 = &参数纠正{来源步骤标识: 步骤.标识, 调用索引: 无效.索引, 工具: 无效.工具, 实际字节: 无效.实际字节, 上限字节: 无效.上限字节, 指纹: 无效.指纹}
			计量器.观测(生效, 定义集, 结果.输入词元)
			if 错误 := 当前引擎.保存回答(执行上下文, 本次执行, []模型包.内容块{}, ""); 错误 != nil {
				return 错误
			}
			continue
		}
		纠正 = nil
		计量器.观测(生效, 定义集, 结果.输入词元)
		if len(结果.工具调用集) == 0 {
			已加密, 错误 := 当前引擎.加密封装回答(执行上下文, 引导)
			if 错误 != nil || 已加密 {
				return 错误
			}
			当前.当前 = append(当前.当前, 模型包.消息{角色: 模型包.角色助手, 内容: 结果.文本, 推理: 结果.推理})
			if 错误 := 当前引擎.保存回答(执行上下文, 本次执行, []模型包.内容块{}, ""); 错误 != nil {
				return 错误
			}
			continue
		}
		// 工具调用轮的文本保留在模型步骤中；回答占位恢复为空，再流式展示下一轮正文。
		if 错误 := 当前引擎.保存回答(执行上下文, 本次执行, []模型包.内容块{}, ""); 错误 != nil {
			return 错误
		}
		当前.当前 = append(当前.当前, 模型包.消息{角色: 模型包.角色助手, 内容: 结果.文本, 推理: 结果.推理, 工具调用集: 结果.工具调用集})
		// 调用次数不设配额，但每次追加都检查实际上下文，避免大批调用或结果累积失控。
		之前适配 := len(当前.过去) + len(当前.当前)
		if 错误 := 当前.适配(执行上下文, 当前引擎, &本次执行, 已解析, 固定, 定义集, nil, 步骤.序号, 结果.工具调用集); 错误 != nil {
			return 错误
		}
		if len(当前.过去)+len(当前.当前) != 之前适配 {
			当前引擎.记录上下文(本次执行, 已解析, 当前.请求(固定, 本次执行), 定义集, nil, true)
		}
		轮次结果集 := make([]模型包.消息, 0, len(结果.工具调用集))
		for 索引, 调用 := range 结果.工具调用集 {
			if 错误 := 执行上下文.获取错误(); 错误 != nil {
				return 错误
			}
			var 值 模型包.消息
			var 错误 error
			if 错误 = 当前引擎.之前工具(执行上下文, 引导, func() error { return nil }); 错误 == nil {
				if 调用.名称 == 智能体包.用户输入工具 {
					值, 错误 = 当前引擎.询问用户(执行上下文, 本次执行, 引导, 步骤, 调用)
				} else {
					值, 错误 = 当前引擎.调用工具(执行上下文, 本次执行, 已解析, 引导, 审核引导内容, 步骤, 调用, 已拒绝, 待核实, 额外依据名称[调用.名称])
				}
			}
			if 错误判断包.是否属于(错误, 会话包.已引导错误) {
				值, 错误 = 当前引擎.跳过已引导调用(执行上下文, 本次执行, 步骤, 调用)
			}
			if 错误 != nil {
				return 错误
			}
			轮次结果集 = append(轮次结果集, 值)
			当前.当前 = append(当前.当前, 值)
			当前引擎.记录上下文(本次执行, 已解析, 当前.请求(固定, 本次执行), 定义集, nil, true)
			之前适配 := len(当前.过去) + len(当前.当前)
			if 错误 := 当前.适配(执行上下文, 当前引擎, &本次执行, 已解析, 固定, 定义集, nil, 步骤.序号, 结果.工具调用集[索引+1:]); 错误 != nil {
				return 错误
			}
			// 正常工具追加已统计；只有压缩改写上下文后才再次扫描并发布缩减值。
			if len(当前.过去)+len(当前.当前) != 之前适配 {
				当前引擎.记录上下文(本次执行, 已解析, 当前.请求(固定, 本次执行), 定义集, nil, true)
			}
		}
		已拒绝 := 已拒绝参数轮次(结果.工具调用集, 轮次结果集)
		if 已拒绝 != "" && 已拒绝 == 前一个拒绝 {
			return &模型包.工具纠正停滞错误{}
		}
		前一个拒绝 = 已拒绝
	}
}

func (当前引擎 *引擎) 保存回答(执行上下文 上下文包.上下文, 本次执行 会话包.执行, 内容块集 []模型包.内容块, 文本 string) error {
	if len(内容块集) == 0 && 文本 != "" {
		内容块集 = []模型包.内容块{{类型: "text", 文本: 文本}}
	}
	if 内容块集 == nil {
		内容块集 = []模型包.内容块{}
	}
	内容块集 = 回答内容块集(本次执行, 内容块集)
	清理, 取消函数 := 清理上下文(执行上下文)
	defer 取消函数()
	if 错误 := 当前引擎.仓储.保存消息(清理, 本次执行.助手.标识, 内容块集, 智能体包.运行中); 错误 != nil {
		return 错误判断包.合并(智能体包.存储错误, 错误)
	}
	值 := 本次执行.助手
	值.内容 = 内容块集
	值.更新时间 = 时间包.当前时间().Unix毫秒()
	return 当前引擎.事件(本次执行, "message.updated", "", 值.标识, 值)
}

func 回答内容块集(本次执行 会话包.执行, 内容块集 []模型包.内容块) []模型包.内容块 {
	if len(本次执行.已发布交付物集) == 0 {
		return 内容块集
	}
	// 旧交付物留在正文前，流式增量可以用固定索引创建和更新新的正文块。
	内容 := append([]模型包.内容块(nil), 本次执行.已发布交付物集...)
	已见 := make(map[string]bool, len(内容))
	for _, 内容块 := range 内容 {
		已见[内容块.附件引用] = true
	}
	for _, 内容块 := range 内容块集 {
		if 内容块.附件引用 == "" || !已见[内容块.附件引用] {
			内容 = append(内容, 内容块)
			已见[内容块.附件引用] = true
		}
	}
	return 内容
}

func 工具失败(错误 error) JSON包.原始JSON消息 {
	类别, 消息 := "tool_failed", "工具执行失败"
	switch {
	case 错误判断包.是否属于(错误, 会话包.已引导错误):
		类别, 消息 = "steered", 会话包.已引导错误.错误文本()
	case 错误判断包.是否属于(错误, 工具包.冲突错误):
		类别, 消息 = "file_conflict", 工具包.冲突错误.错误文本()
	case 错误判断包.是否属于(错误, 工具包.未知操作结果错误):
		类别, 消息 = "outcome_unknown", 工具包.未知操作结果错误.错误文本()
	case 错误判断包.是否属于(错误, 工具包.部分操作结果错误):
		类别, 消息 = "partially_applied", 工具包.部分操作结果错误.错误文本()
	case 错误判断包.是否属于(错误, 工具包.路径基准错误):
		类别, 消息 = "path_base_required", 工具包.路径基准错误.错误文本()
	case 错误判断包.是否属于(错误, 工具包.已拒绝错误):
		类别, 消息 = "permission_denied", "操作超出权限范围或已被拒绝"
	case 错误判断包.是否属于(错误, 工具包.无效错误):
		类别, 消息 = "invalid_arguments", "工具参数无效"
		if 详情, 成功 := 错误判断包.按类型匹配[*工具包.参数错误](错误); 成功 {
			消息 = 详情.错误文本()
		}
	case 错误判断包.是否属于(错误, 工具包.不可用错误):
		类别, 消息 = "unavailable", "工具或文件不可用"
	case 错误判断包.是否属于(错误, 工具包.超出上限错误):
		类别, 消息 = "output_limit", "工具输入或结果超过限制"
	case 错误判断包.是否属于(错误, 上下文包.已取消错误):
		类别, 消息 = "cancelled", "操作已取消"
	case 错误判断包.是否属于(错误, 上下文包.超出截止时间错误):
		类别, 消息 = "timeout", "工具执行超时"
	case 错误判断包.是否属于(错误, 模型包.附件错误):
		类别, 消息 = "attachment_unavailable", 模型包.附件错误.错误文本()
	}
	值, _ := JSON包.序列化(map[string]any{"error": map[string]string{"kind": 类别, "message": 消息}})
	return 值
}

// 带有上下文消息 为跨供应商上下文恢复保存统一消息，不解析供应商协议。
func 带有上下文消息(输出 JSON包.原始JSON消息, 消息 模型包.消息) (JSON包.原始JSON消息, error) {
	var 字段集 map[string]JSON包.原始JSON消息
	if len(输出) > 0 {
		if 错误 := JSON包.反序列化(输出, &字段集); 错误 != nil {
			return nil, 错误
		}
	}
	if 字段集 == nil {
		字段集 = map[string]JSON包.原始JSON消息{}
	}
	原始, 错误 := JSON包.序列化(消息)
	if 错误 != nil {
		return nil, 错误
	}
	字段集["context_message"] = 原始
	return JSON包.序列化(字段集)
}
