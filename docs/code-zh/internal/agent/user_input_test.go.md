```go
package 智能体

import (
	JSON包 "encoding/json"
	测试包 "testing"
)

func Test问题并且回答边界集(测试器 *测试包.测试器) {
	for _, 无效 := range []string{
		`{}`, `null`, `{"question":"问题","Question":"重复","options":[]}`, `{"question":"问题","question":"重复","options":[]}`,
		`{"question":"问题","options":null}`, `{"question":"问题","options":["相同"," 相同 "]}`, `{"question":"问题","options":[],"approved":true}`,
		`{"question":"问题","options":["1","2","3","4","5","6","7","8","9"]}`,
	} {
		if _, 错误 := 解析输入问题([]byte(无效)); 错误 == nil {
			测试器.格式化报告致命错误("接受无效问题 %s", 无效)
		}
	}
	问题, 错误 := 解析输入问题([]byte(`{"question":" 请确认 ","options":["已完成","进行中"]}`))
	if 错误 != nil || 问题.问题 != "请确认" {
		测试器.报告致命错误(问题, 错误)
	}
	for _, 无效 := range []输入回答{{}, {选项索引: new(-1)}, {选项索引: new(2)}, {文本: " \n"}, {已跳过: true, 文本: "文字"}, {已跳过: true, 选项索引: new(0)}} {
		if 校验输入回答(问题, 无效) == nil {
			测试器.格式化报告致命错误("接受无效回复 %+v", 无效)
		}
	}
	for _, 回答 := range []输入回答{{选项索引: new(0)}, {文本: "自己的回复"}, {已跳过: true}, {选项索引: new(1), 文本: "补充说明"}} {
		if 错误 := 校验输入回答(问题, 回答); 错误 != nil {
			测试器.报告致命错误(错误)
		}
		输入, _ := JSON包.序列化(问题)
		步骤 := 步骤{标识: "question", 执行标识: "run", 类型: "user_input", 状态: 已完成, 工具调用标识: new("call"), 输入: 输入}
		步骤.输出, 错误 = 输入回答输出(步骤, 回答)
		if 错误 != nil {
			测试器.报告致命错误(错误)
		}
		已解析, 错误 := 输入来源步骤(步骤)
		if 错误 != nil || 已解析.回答 == nil || 已解析.回答.文本 != 回答.文本 {
			测试器.报告致命错误(已解析, 错误)
		}
	}
}
```
