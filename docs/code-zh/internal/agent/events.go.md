```go
package 智能体

// 工具输出 是即时工具输出，不逐片持久化；数据 为 基准64，避免拆分 UTF-8 字符或破坏 JSONL。
// 最终有界结果由工具步骤保存，序号 仅在对应步骤内递增。
type 工具输出 struct {
	序号 int64  `json:"sequence"`
	流  string `json:"stream"`
	编码 string `json:"encoding"`
	数据 string `json:"data"`
}

// 步骤摘要 是通知和列表使用的步骤元数据，大型原始结果通过查询读取。
type 步骤摘要 struct {
	步骤
	输出字节 int `json:"output_bytes"`
	用量字节 int `json:"usage_bytes"`
	定义字节 int `json:"definition_bytes"`
}

// 生成摘要步骤 在事件进入订阅队列之前去掉原始结果，避免队列持有大型响应。
func 生成摘要步骤(步骤 步骤) 步骤摘要 {
	输出 := 步骤摘要{步骤: 步骤, 输出字节: len(步骤.输出), 用量字节: len(步骤.用量), 定义字节: len(步骤.定义)}
	输出.输出, 输出.用量, 输出.定义 = nil, nil, nil
	// 事件可能晚于步骤更新才被消费，不能继续引用执行过程中的可变指针。
	输出.父级步骤标识 = 快照指针(步骤.父级步骤标识)
	输出.模型 = 快照指针(步骤.模型)
	输出.供应商请求标识 = 快照指针(步骤.供应商请求标识)
	输出.工具调用标识 = 快照指针(步骤.工具调用标识)
	输出.错误类别 = 快照指针(步骤.错误类别)
	输出.错误消息 = 快照指针(步骤.错误消息)
	输出.开始时间 = 快照指针(步骤.开始时间)
	输出.结束时间 = 快照指针(步骤.结束时间)
	return 输出
}

func 快照指针[元素类型 any](值 *元素类型) *元素类型 {
	if 值 == nil {
		return nil
	}
	return new(*值)
}
```
