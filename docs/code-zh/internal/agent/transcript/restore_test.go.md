```go
package 执行轨迹测试

import (
	JSON包 "encoding/json"
	错误判断包 "errors"
	切片包 "slices"
	测试包 "testing"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	执行轨迹包 "github.com/xiaoyu98628/callum/internal/agent/transcript"
	模型包 "github.com/xiaoyu98628/callum/internal/model"
	工具包 "github.com/xiaoyu98628/callum/internal/tool"
)

func 编码(测试器 *测试包.测试器, 值 any) JSON包.原始JSON消息 {
	测试器.标记测试辅助函数()
	原始, 错误 := JSON包.序列化(值)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	return 原始
}

func 部分轨迹(测试器 *测试包.测试器) []执行轨迹包.条目 {
	测试器.标记测试辅助函数()
	return []执行轨迹包.条目{
		{步骤标识: "model", 消息: 编码(测试器, 模型包.消息{角色: 模型包.角色助手, 内容: "未完成正文", 工具调用集: []模型包.工具调用{{标识: "a", 名称: "write"}, {标识: "b", 名称: "write"}}})},
		{步骤标识: "tool", 父级步骤标识: new("model"), 工具调用标识: new("a"), 消息: 编码(测试器, 模型包.消息{角色: 模型包.角色工具, 工具调用标识: "a", 内容: "已记录结果"}), 操作: 编码(测试器, 工具包.操作结果记录{版本: 1, 操作结果: 工具包.操作结果未知})},
	}
}

func Test恢复保留事实集并且完成已中断工具配对集(测试器 *测试包.测试器) {
	for _, 状态 := range []string{智能体包.失败, 智能体包.已取消, 智能体包.已中断} {
		测试器.执行(状态, func(测试器 *测试包.测试器) {
			条目集 := 部分轨迹(测试器)
			之前 := string(条目集[0].消息)
			消息集, 错误 := 执行轨迹包.恢复(状态, 条目集)
			if 错误 != nil || len(消息集) != 3 || 消息集[0].内容 != "" || len(消息集[0].工具调用集) != 2 || 消息集[1].内容 != "已记录结果" || 消息集[1].工具调用标识 != "a" || 消息集[2].工具调用标识 != "b" {
				测试器.格式化报告致命错误("恢复丢失事实或调用配对: %+v %v", 消息集, 错误)
			}
			var 缺失 struct {
				操作 工具包.操作结果记录          `json:"operation"`
				错误 struct{ 类别 string } `json:"error"`
			}
			if 错误 := JSON包.反序列化([]byte(消息集[2].内容), &缺失); 错误 != nil || 缺失.操作.操作结果 != 工具包.操作结果不已生效 || !缺失.操作.已恢复 || 缺失.错误.类别 != "not_executed" {
				测试器.格式化报告致命错误("尚未执行的调用被推测为已执行: %+v %v", 缺失, 错误)
			}
			if string(条目集[0].消息) != 之前 {
				测试器.报告致命错误("恢复改写了来源记录")
			}
		})
	}
}

func Test恢复拒绝损坏或者未配对事实集(测试器 *测试包.测试器) {
	for _, 名称 := range []string{"duplicate", "missing_parent", "wrong_call", "invalid_message", "invalid_operation", "orphan_effect"} {
		测试器.执行(名称, func(测试器 *测试包.测试器) {
			条目集 := 部分轨迹(测试器)
			switch 名称 {
			case "duplicate":
				条目集 = append(条目集, 条目集[1])
			case "missing_parent":
				条目集[1].父级步骤标识 = nil
			case "wrong_call":
				条目集[1].工具调用标识 = new("other")
			case "invalid_message":
				条目集[0].消息 = JSON包.原始JSON消息(`{`)
			case "invalid_operation":
				条目集[1].操作 = JSON包.原始JSON消息(`{"version":1,"outcome":"invalid"}`)
			case "orphan_effect":
				条目集[1].父级步骤标识 = new("unrelated")
			}
			if _, 错误 := 执行轨迹包.恢复(智能体包.已中断, 条目集); !错误判断包.是否属于(错误, 模型包.无效响应错误) {
				测试器.格式化报告致命错误("损坏轨迹未被拒绝: %v", 错误)
			}
		})
	}
	条目集 := 部分轨迹(测试器)
	if _, 错误 := 执行轨迹包.恢复(智能体包.已完成, 条目集); !错误判断包.是否属于(错误, 模型包.无效响应错误) {
		测试器.格式化报告致命错误("成功轨迹接受不完整配对: %v", 错误)
	}
	条目集 = append(切片包.复制(条目集), 执行轨迹包.条目{步骤标识: "tool-b", 父级步骤标识: new("model"), 工具调用标识: new("b"), 消息: 编码(测试器, 模型包.消息{角色: 模型包.角色工具, 工具调用标识: "b", 内容: "第二个结果"})})
	条目集 = append(条目集, 执行轨迹包.条目{步骤标识: "final", 消息: 编码(测试器, 模型包.消息{角色: 模型包.角色助手, 内容: "完整回答"})})
	消息集, 错误 := 执行轨迹包.恢复(智能体包.已完成, 条目集)
	if 错误 != nil || len(消息集) != 4 || 消息集[0].内容 != "未完成正文" || 消息集[3].内容 != "完整回答" {
		测试器.格式化报告致命错误("完整轨迹被错误过滤: %+v %v", 消息集, 错误)
	}
}
```
