func Test订阅隔离并且边界(测试器 *测试包.测试器) {
	事件中心, 错误 := 新建(1, 2)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 事件中心.关闭()
	快速, _ := 事件中心.订阅(筛选{会话标识: "a", 类型集: []string{"message.delta"}})
	缓慢, _ := 事件中心.订阅(筛选{})
	if _, 错误 := 事件中心.订阅(筛选{}); !错误判断包.是否属于(错误, 订阅者上限错误) {
		测试器.格式化报告致命错误("订阅上限: %v", 错误)
	}
	// 过滤掉的事件不能占用 快速 容量；缓慢 第二次发布时溢出。
	if 错误 := 事件中心.发布(智能体包.事件{类型: "run.started", 会话标识: "b"}); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	for 索引 := range 3 {
		事件 := 智能体包.事件{类型: "message.delta", 会话标识: "a", 数据: 索引}
		if 错误 := 事件中心.发布(事件); 错误 != nil {
			测试器.报告致命错误(错误)
		}
		实际值, 错误 := 快速.下一个(测试器.上下文())
		if 错误 != nil || 实际值.数据 != 索引 {
			测试器.格式化报告致命错误("快速订阅受到影响: %+v %v", 实际值, 错误)
		}
	}
	if _, 错误 := 缓慢.下一个(测试器.上下文()); !错误判断包.是否属于(错误, 缓慢消费者错误) {
		测试器.格式化报告致命错误("未报告溢出: %v", 错误)
	}
	替换, 错误 := 事件中心.订阅(筛选{})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	替换.关闭()
	快速.关闭()
	快速.关闭()
	if _, 错误 := 快速.下一个(测试器.上下文()); !错误判断包.是否属于(错误, 输入输出包.文件结束) {
		测试器.报告致命错误(错误)
	}
	// 无订阅者时不阻塞，也不使执行失败。
	if 错误 := 事件中心.发布(智能体包.事件{}); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	事件中心.关闭()
	if 错误 := 事件中心.发布(智能体包.事件{}); !错误判断包.是否属于(错误, 已关闭错误) {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 事件中心.订阅(筛选{}); !错误判断包.是否属于(错误, 已关闭错误) {
		测试器.报告致命错误(错误)
	}
}

func Test执行设置不共享执行指针(测试器 *测试包.测试器) {
	事件中心, 错误 := 新建(1, 1)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 事件中心.关闭()
	订阅, 错误 := 事件中心.订阅(筛选{})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	defer 订阅.关闭()
	快照 := &设置包.快照{修订号: 3, 执行: 设置包.默认执行()}
	if 错误 := 事件中心.发布(智能体包.事件{类型: "run.started", 数据: 智能体包.执行{设置快照: 快照}}); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	快照.修订号 = 4
	事件, 错误 := 订阅.下一个(测试器.上下文())
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	运行 := 事件.数据.(智能体包.执行)
	if 运行.设置快照.修订号 != 3 {
		测试器.报告致命错误("队列借用了执行内部的快照")
	}
	运行.设置快照.审批策略 = "never"
	if 快照.审批策略 != "on-request" {
		测试器.报告致命错误("事件消费者改变了执行内部权限")
	}
}

func Test摘要并且排空(测试器 *测试包.测试器) {
	事件中心, _ := 新建(2, 1)
	订阅, _ := 事件中心.订阅(筛选{执行标识: "r"})
	步骤 := 智能体包.步骤{标识: "s", 输出: []byte(`{"large":true}`), 用量: []byte(`{"total_tokens":3}`), 定义: []byte(`{}`)}
	当前时间 := int64(1)
	步骤.开始时间 = &当前时间
	if 错误 := 事件中心.发布(智能体包.事件{类型: "step.updated", 执行标识: "r", 数据: 步骤}); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	当前时间 = 2
	事件中心.关闭()
	事件, 错误 := 订阅.下一个(测试器.上下文())
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	数据, 成功 := 事件.数据.(智能体包.步骤摘要)
	if !成功 || 数据.输出 != nil || 数据.用量 != nil || 数据.定义 != nil || 数据.输出字节 != len(步骤.输出) || 数据.用量字节 != len(步骤.用量) {
		测试器.格式化报告致命错误("队列未投影步骤: %+v", 数据)
	}
	if 数据.开始时间 == nil || *数据.开始时间 != 1 {
		测试器.报告致命错误("已发布快照被后续更新改写")
	}
	if _, 错误 := 订阅.下一个(测试器.上下文()); !错误判断包.是否属于(错误, 输入输出包.文件结束) {
		测试器.报告致命错误(错误)
	}
}

func Test并发发布并且关闭(测试器 *测试包.测试器) {
	事件中心, _ := 新建(4, 8)
	订阅, _ := 事件中心.订阅(筛选{})
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	取消函数()
	if _, 错误 := 订阅.下一个(执行上下文); !错误判断包.是否属于(错误, 上下文包.已取消错误) {
		测试器.报告致命错误(错误)
	}
	var 等待组 同步包.等待组
	for range 8 {
		等待组.Go(func() {
			for range 100 {
				_ = 事件中心.发布(智能体包.事件{类型: "message.delta"})
				下一个, 错误 := 事件中心.订阅(筛选{})
				if 错误 == nil {
					下一个.关闭()
				}
			}
		})
	}
	等待组.Go(func() { 订阅.关闭(); 事件中心.关闭() })
	等待组.等待()
}
