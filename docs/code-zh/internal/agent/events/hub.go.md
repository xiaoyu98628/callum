```go
// 包 事件集 提供运行时内的有界事件订阅，不负责持久化或协议编码。
package 事件集

import (
	上下文包 "context"
	错误判断包 "errors"
	输入输出包 "io"
	切片包 "slices"
	同步包 "sync"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
)

var (
	已关闭错误   = 错误判断包.新建("事件发布器已关闭")
	缓慢消费者错误 = 错误判断包.新建("事件订阅队列已满")
	订阅者上限错误 = 错误判断包.新建("事件订阅数量已达到上限")
)

// 筛选 的非空条件同时生效；空过滤器订阅所有事件。
type 筛选 struct {
	会话标识 string
	执行标识 string
	类型集  []string
}

// 事件中心 由应用运行时持有并关闭，不启动后台任务。
type 事件中心 struct {
	互斥锁    同步包.互斥锁
	已关闭    bool
	容量, 上限 int
	订阅集    map[*订阅]struct{}
}

// 订阅 只允许一个 下一个 调用方；持有者负责 关闭，事件内容只读。
type 订阅 struct {
	事件中心 *事件中心
	筛选   筛选
	队列   chan 智能体包.事件
	已关闭  bool
	原因   error
}

// 新建 的容量和订阅数量必须为正数，限制由装配层确定。
func 新建(容量, 上限 int) (*事件中心, error) {
	if 容量 < 1 || 上限 < 1 {
		return nil, 智能体包.无效错误
	}
	return &事件中心{容量: 容量, 上限: 上限, 订阅集: make(map[*订阅]struct{})}, nil
}

func (当前事件中心 *事件中心) 订阅(筛选 筛选) (*订阅, error) {
	当前事件中心.互斥锁.加锁()
	defer 当前事件中心.互斥锁.解锁()
	if 当前事件中心.已关闭 {
		return nil, 已关闭错误
	}
	if len(当前事件中心.订阅集) >= 当前事件中心.上限 {
		return nil, 订阅者上限错误
	}
	筛选.类型集 = 切片包.复制(筛选.类型集)
	当前状态 := &订阅{事件中心: 当前事件中心, 筛选: 筛选, 队列: make(chan 智能体包.事件, 当前事件中心.容量)}
	当前事件中心.订阅集[当前状态] = struct{}{}
	return 当前状态, nil
}

// 发布 不等待消费者；队列溢出只断开对应订阅，不使业务执行失败。
// 同次发布的内容由订阅者共享，发布方与订阅者均不得修改事件引用的数据。
func (当前事件中心 *事件中心) 发布(事件 智能体包.事件) error {
	if 步骤, 成功 := 事件.数据.(智能体包.步骤); 成功 {
		事件.数据 = 智能体包.生成摘要步骤(步骤)
	}
	if 运行, 成功 := 事件.数据.(智能体包.执行); 成功 && 运行.设置快照 != nil {
		// 消费者遵守只读约定，发布器同时避免队列引用执行器内部的权限指针。
		快照 := *运行.设置快照
		运行.设置快照 = &快照
		事件.数据 = 运行
	}
	事件.已移除消息标识集 = 切片包.复制(事件.已移除消息标识集)
	当前事件中心.互斥锁.加锁()
	defer 当前事件中心.互斥锁.解锁()
	if 当前事件中心.已关闭 {
		return 已关闭错误
	}
	for 当前状态 := range 当前事件中心.订阅集 {
		文件 := 当前状态.筛选
		if (文件.会话标识 != "" && 文件.会话标识 != 事件.会话标识) || (文件.执行标识 != "" && 文件.执行标识 != 事件.执行标识) || (len(文件.类型集) > 0 && !切片包.包含(文件.类型集, 事件.类型)) {
			continue
		}
		select {
		case 当前状态.队列 <- 事件:
		default:
			当前状态.关闭已加锁(缓慢消费者错误)
		}
	}
	return nil
}

// 下一个 返回排队事件；正常关闭后排空队列再返回 文件结束，溢出立即返回错误。
func (当前订阅 *订阅) 下一个(执行上下文 上下文包.上下文) (智能体包.事件, error) {
	select {
	case <-执行上下文.完成():
		return 智能体包.事件{}, 执行上下文.获取错误()
	case 事件, 成功 := <-当前订阅.队列:
		当前订阅.事件中心.互斥锁.加锁()
		原因 := 当前订阅.原因
		当前订阅.事件中心.互斥锁.解锁()
		if 原因 != nil {
			return 智能体包.事件{}, 原因
		}
		if !成功 {
			return 智能体包.事件{}, 输入输出包.文件结束
		}
		return 事件, nil
	}
}

func (当前订阅 *订阅) 关闭已加锁(原因 error) {
	if 当前订阅.已关闭 {
		return
	}
	当前订阅.已关闭, 当前订阅.原因 = true, 原因
	delete(当前订阅.事件中心.订阅集, 当前订阅)
	close(当前订阅.队列)
	if 原因 != nil {
		for range 当前订阅.队列 {
		}
	}
}

// 关闭 解除订阅，可重复调用；已排队的事件仍可读取。
func (当前订阅 *订阅) 关闭() {
	当前订阅.事件中心.互斥锁.加锁()
	defer 当前订阅.事件中心.互斥锁.解锁()
	当前订阅.关闭已加锁(nil)
}

// 关闭 停止发布并关闭全部订阅，应在业务生产者停止后调用。
func (当前事件中心 *事件中心) 关闭() {
	当前事件中心.互斥锁.加锁()
	defer 当前事件中心.互斥锁.解锁()
	当前事件中心.已关闭 = true
	for 当前状态 := range 当前事件中心.订阅集 {
		当前状态.关闭已加锁(nil)
	}
}
```
