```go
// 包 审批 管理确定操作的审批，等待过程不占用事务或模型请求槽位。
package 审批

import (
	上下文包 "context"
	错误判断包 "errors"
	同步包 "sync"
	时间包 "time"
	UTF8包 "unicode/utf8"

	UUID包 "github.com/google/uuid"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
	标识符包 "github.com/xiaoyu98628/callum/internal/id"
)

// 服务 由应用持有，等待与所属执行共享取消信号，不启动独立后台任务。
type 服务 struct {
	仓储 智能体包.审批仓储
	查找     func(上下文包.上下文, string) (智能体包.执行, error)
	发出事件       func(智能体包.事件) error
	互斥锁         同步包.互斥锁
	信号集    map[string]chan struct{}
}

func 新建(仓储 智能体包.审批仓储, 查找 func(上下文包.上下文, string) (智能体包.执行, error), 发出事件 func(智能体包.事件) error) (*服务, error) {
	if 仓储 == nil || 查找 == nil || 发出事件 == nil {
		return nil, 智能体包.审批无效错误
	}
	return &服务{仓储: 仓储, 查找: 查找, 发出事件: 发出事件, 信号集: map[string]chan struct{}{}}, nil
}
func 有效标识(标识 string) bool { return 标识符包.有效版本7(标识) }

func (当前服务 *服务) 获取(执行上下文 上下文包.上下文, 标识 string) (智能体包.审批, error) {
	if !有效标识(标识) {
		return 智能体包.审批{}, 智能体包.审批无效错误
	}
	return 当前服务.仓储.获取审批(执行上下文, 标识)
}
func (当前服务 *服务) 列表(执行上下文 上下文包.上下文, 当前审批筛选 智能体包.审批筛选) ([]智能体包.审批, error) {
	if 当前审批筛选.上限 == 0 {
		当前审批筛选.上限 = 100
	}
	if 当前审批筛选.上限 < 1 || 当前审批筛选.上限 > 200 || 当前审批筛选.偏移 < 0 || (当前审批筛选.执行标识 != "" && !有效标识(当前审批筛选.执行标识)) {
		return nil, 智能体包.审批无效错误
	}
	switch 当前审批筛选.状态 {
	case "", 智能体包.审批待处理, 智能体包.审批已批准, 智能体包.审批已拒绝, 智能体包.审批已取消:
	default:
		return nil, 智能体包.审批无效错误
	}
	return 当前服务.仓储.列表审批集(执行上下文, 当前审批筛选)
}

// 决定 的用户标识与目录范围由服务端确定；客户端只能选择本次或快照提供的会话目录授权。
func (当前服务 *服务) 决定(执行上下文 上下文包.上下文, 标识, 决定 string, 说明 *string, 范围集 ...string) (智能体包.审批, error) {
	if len(范围集) > 1 || (len(范围集) == 1 && 范围集[0] != 智能体包.审批一次 && 范围集[0] != 智能体包.审批会话目录) || !有效标识(标识) || (决定 != 智能体包.审批已批准 && 决定 != 智能体包.审批已拒绝) || (说明 != nil && (len(*说明) > 4096 || !UTF8包.有效字符串(*说明))) {
		return 智能体包.审批{}, 智能体包.审批无效错误
	}
	值, 错误 := 当前服务.仓储.决定审批(执行上下文, 标识, 决定, new("local_user"), 说明, 范围集...)
	if 值.标识 != "" && 错误 == nil {
		当前服务.唤醒(值.标识)
		if 发布错误 := 当前服务.发布(执行上下文, "approval.updated", 值); 发布错误 != nil {
			return 值, 错误判断包.合并(错误, 发布错误)
		}
		if 值.目录授权 != nil {
			if 错误 := 当前服务.发出事件(智能体包.事件{类型: "directory_grant.updated", 会话标识: 值.目录授权.会话标识, 数据: *值.目录授权}); 错误 != nil {
				return 值, 错误
			}
		}
	}
	return 值, 错误
}

func (当前服务 *服务) 唤醒(标识 string) {
	当前服务.互斥锁.加锁()
	defer 当前服务.互斥锁.解锁()
	if 通道 := 当前服务.信号集[标识]; 通道 != nil {
		select {
		case 通道 <- struct{}{}:
		default:
		}
	}
}
func (当前服务 *服务) 发布(执行上下文 上下文包.上下文, 名称 string, 当前审批 智能体包.审批) error {
	执行, 错误 := 当前服务.查找(执行上下文, 当前审批.执行标识)
	if 错误 != nil {
		return 错误
	}
	return 当前服务.发出事件(智能体包.事件{类型: 名称, 会话标识: 执行.会话标识, 执行标识: 当前审批.执行标识, 步骤标识: 当前审批.步骤标识, 数据: 当前审批})
}

// 请求 在落库后才发布审批通知。已完成的独立审核与审批记录一并提交。
func (当前服务 *服务) 请求(执行上下文 上下文包.上下文, 执行标识, 步骤标识 string, 请求 []byte, 哈希 string, 审核 *智能体包.审批审核) (智能体包.审批, error) {
	标识, 错误 := UUID包.新建UUIDv7()
	if 错误 != nil {
		return 智能体包.审批{}, 错误
	}
	当前时间 := 时间包.当前时间()
	输入 := 智能体包.审批{标识: 标识.字符串(), 执行标识: 执行标识, 步骤标识: 步骤标识, 请求: 请求, 请求哈希: 哈希, 状态: 智能体包.审批待处理, 创建时间: 当前时间.Unix毫秒(), 更新时间: 当前时间.Unix毫秒()}
	if 审核 != nil && (审核.决定 == 智能体包.审批已批准 || 审核.决定 == 智能体包.审批已拒绝) {
		输入.状态, 输入.已决定依据, 输入.决定说明, 输入.已决定位于 = 审核.决定, new(智能体包.审批自动审核方), new(审核.原因), new(输入.创建时间)
		输入.审核 = 审核
	}
	值, 错误 := 当前服务.仓储.创建审批(执行上下文, 输入)
	if 错误 != nil {
		return 值, 错误
	}
	if 值.状态 != 智能体包.审批待处理 {
		return 值, 当前服务.发布(执行上下文, "approval.updated", 值)
	}
	return 值, 当前服务.发布(执行上下文, "approval.requested", 值)
}

// 等待 先登记唤醒通道再读取记录，避免决定先于等待导致丢失唤醒。
// 通道数量由活动执行限制，退出时立即删除；取消后以短清理上下文保存状态。
func (当前服务 *服务) 等待(执行上下文 上下文包.上下文, 标识 string) (智能体包.审批, error) {
	通道 := make(chan struct{}, 1)
	当前服务.互斥锁.加锁()
	if 当前服务.信号集[标识] != nil {
		当前服务.互斥锁.解锁()
		return 智能体包.审批{}, 智能体包.冲突错误
	}
	当前服务.信号集[标识] = 通道
	当前服务.互斥锁.解锁()
	defer func() { 当前服务.互斥锁.加锁(); delete(当前服务.信号集, 标识); 当前服务.互斥锁.解锁() }()
	for {
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			return 当前服务.取消待处理(执行上下文, 标识, 错误)
		}
		值, 错误 := 当前服务.仓储.获取审批(执行上下文, 标识)
		if 错误 != nil {
			return 值, 错误
		}
		if 值.状态 != 智能体包.审批待处理 {
			return 值, nil
		}
		// 等待由用户决定或所属执行的取消信号结束，不设置审批期限。
		select {
		case <-执行上下文.完成():
			return 当前服务.取消待处理(执行上下文, 标识, 执行上下文.获取错误())
		case <-通道:
		}
	}
}
func (当前服务 *服务) 取消待处理(执行上下文 上下文包.上下文, 标识 string, 原因 error) (智能体包.审批, error) {
	清理, 取消函数 := 上下文包.创建超时上下文(上下文包.无取消(执行上下文), 5*时间包.秒)
	defer 取消函数()
	值, 错误 := 当前服务.仓储.取消审批(清理, 标识)
	if 错误 == nil {
		错误 = 当前服务.发布(清理, "approval.updated", 值)
	}
	return 值, 错误判断包.合并(原因, 错误)
}
```
