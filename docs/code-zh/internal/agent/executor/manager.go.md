```go
package 执行器

import (
	上下文包 "context"
	错误判断包 "errors"
	同步包 "sync"
	时间包 "time"

	智能体包 "github.com/xiaoyu98628/callum/internal/agent"
)

// 管理器 统一限制普通对话和无历史调用；创建者必须 关闭 并等待资源释放。
type 管理器 struct {
	执行上下文     上下文包.上下文
	取消函数  上下文包.取消函数
	上限   int
	超时 时间包.时长
	互斥锁      同步包.互斥锁
	已关闭  bool
	执行集    map[string]*执行租约
	键集    map[string]string
	调用集   chan struct{}
	变化通知 chan struct{}
	等待组      同步包.等待组
}

// 执行租约 预留执行容量，调用方必须 启动 或 中止，不能遗留未启动的预留。
type 执行租约 struct {
	管理器            *管理器
	标识, 键            string
	执行上下文                上下文包.上下文
	取消函数             上下文包.取消函数
	已开始, 正在结束 bool
	已中止            bool
	最终错误           error
	完成               chan struct{}
	一次               同步包.一次
}

func 新建(执行上下文 上下文包.上下文, 上限 int, 超时 时间包.时长) (*管理器, error) {
	if 执行上下文 == nil || 上限 < 1 || 上限 > 64 || 超时 <= 0 {
		return nil, 智能体包.无效错误
	}
	运行时, 取消函数 := 上下文包.创建可取消上下文(执行上下文)
	return &管理器{执行上下文: 运行时, 取消函数: 取消函数, 上限: 上限, 超时: 超时, 执行集: make(map[string]*执行租约), 键集: make(map[string]string), 调用集: make(chan struct{}, 上限), 变化通知: make(chan struct{})}, nil
}

// 预留上下文 只等待已确定终态的任务退出，防止客户端收到终态后立即发送仍遇到忙碌。
// 正在执行的任务占满容量时仍立即拒绝，不引入后台等待队列。
func (当前管理器 *管理器) 预留上下文(执行上下文 上下文包.上下文, 标识, 键 string) (*执行租约, error) {
	return 当前管理器.内部预留上下文(执行上下文, 标识, 键, false)
}

// 预留智能体上下文 持续到业务完成、取消或错误；不设总调用次数和审批等待超时。
// 每次模型和工具请求分别控制超时，等待审批不消耗模型超时。
func (当前管理器 *管理器) 预留智能体上下文(执行上下文 上下文包.上下文, 标识, 键 string) (*执行租约, error) {
	return 当前管理器.内部预留上下文(执行上下文, 标识, 键, true)
}

func (当前管理器 *管理器) 内部预留上下文(执行上下文 上下文包.上下文, 标识, 键 string, 智能体执行 bool) (*执行租约, error) {
	for {
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			return nil, 错误
		}
		执行租约, 错误 := 当前管理器.内部预留(标识, 键, 智能体执行)
		if !错误判断包.是否属于(错误, 智能体包.繁忙错误) {
			return 执行租约, 错误
		}
		当前管理器.互斥锁.加锁()
		if 当前管理器.已关闭 {
			当前管理器.互斥锁.解锁()
			return nil, 智能体包.已关闭错误
		}
		变化通知 := 当前管理器.变化通知
		等待, 重试 := false, false
		阻塞 := 当前管理器.执行集[标识]
		if 已有 := 当前管理器.执行集[当前管理器.键集[键]]; 键 != "" && 已有 != nil {
			阻塞 = 已有
		}
		if 阻塞 != nil {
			等待 = 阻塞.正在结束
		} else if len(当前管理器.执行集) < 当前管理器.上限 {
			重试 = true
		} else {
			for _, 已有 := range 当前管理器.执行集 {
				等待 = 等待 || 已有.正在结束
			}
		}
		当前管理器.互斥锁.解锁()
		if 重试 {
			continue
		}
		if !等待 {
			return nil, 错误
		}
		select {
		case <-执行上下文.完成():
			return nil, 执行上下文.获取错误()
		case <-当前管理器.执行上下文.完成():
			return nil, 智能体包.已关闭错误
		case <-变化通知:
		}
	}
}

func (当前管理器 *管理器) 上限集() (int, 时间包.时长) { return 当前管理器.上限, 当前管理器.超时 }

// 预留 同时预留全局执行容量和可选会话互斥键，不持有数据库资源。
func (当前管理器 *管理器) 预留(标识, 键 string) (*执行租约, error) { return 当前管理器.内部预留(标识, 键, false) }

func (当前管理器 *管理器) 内部预留(标识, 键 string, 智能体执行 bool) (*执行租约, error) {
	if 标识 == "" {
		return nil, 智能体包.无效错误
	}
	当前管理器.互斥锁.加锁()
	defer 当前管理器.互斥锁.解锁()
	if 当前管理器.已关闭 || 当前管理器.执行上下文.获取错误() != nil {
		return nil, 智能体包.已关闭错误
	}
	if len(当前管理器.执行集) >= 当前管理器.上限 || 当前管理器.执行集[标识] != nil || (键 != "" && 当前管理器.键集[键] != "") {
		return nil, 智能体包.繁忙错误
	}
	var 执行上下文 上下文包.上下文
	var 取消函数 上下文包.取消函数
	if 智能体执行 {
		执行上下文, 取消函数 = 上下文包.创建可取消上下文(当前管理器.执行上下文)
	} else {
		执行上下文, 取消函数 = 上下文包.创建超时上下文(当前管理器.执行上下文, 当前管理器.超时)
	}
	执行租约 := &执行租约{管理器: 当前管理器, 标识: 标识, 键: 键, 执行上下文: 执行上下文, 取消函数: 取消函数, 完成: make(chan struct{})}
	当前管理器.执行集[标识] = 执行租约
	if 键 != "" {
		当前管理器.键集[键] = 标识
	}
	当前管理器.等待组.添加(1)
	return 执行租约, nil
}

func (当前执行租约 *执行租约) 上下文() 上下文包.上下文 { return 当前执行租约.执行上下文 }

// 启动 在受理响应成功排队后调用；即使取消已发生，也启动任务完成持久化清理。
func (当前执行租约 *执行租约) 启动(任务 func(上下文包.上下文)) {
	当前执行租约.管理器.互斥锁.加锁()
	if 当前执行租约.已开始 || 当前执行租约.已中止 {
		当前执行租约.管理器.互斥锁.解锁()
		return
	}
	当前执行租约.已开始 = true
	当前执行租约.管理器.互斥锁.解锁()
	go func() { defer 当前执行租约.释放(); 任务(当前执行租约.执行上下文) }()
}

func (当前执行租约 *执行租约) 中止() {
	当前执行租约.管理器.互斥锁.加锁()
	if 当前执行租约.已开始 || 当前执行租约.已中止 {
		当前执行租约.管理器.互斥锁.解锁()
		return
	}
	当前执行租约.已中止 = true
	当前执行租约.管理器.互斥锁.解锁()
	当前执行租约.释放()
}

func (当前执行租约 *执行租约) 释放() {
	当前执行租约.一次.执行(func() {
		当前执行租约.取消函数()
		映射 := 当前执行租约.管理器
		映射.互斥锁.加锁()
		delete(映射.执行集, 当前执行租约.标识)
		if 当前执行租约.键 != "" {
			delete(映射.键集, 当前执行租约.键)
		}
		close(映射.变化通知)
		映射.变化通知 = make(chan struct{})
		close(当前执行租约.完成)
		映射.互斥锁.解锁()
		映射.等待组.完成()
	})
}

// 结束中 与 取消执行 串行确定终态，返回已经发生的取消或超时。
func (当前执行租约 *执行租约) 是否正在结束() error {
	当前执行租约.管理器.互斥锁.加锁()
	defer 当前执行租约.管理器.互斥锁.解锁()
	if !当前执行租约.正在结束 {
		当前执行租约.正在结束 = true
		当前执行租约.最终错误 = 当前执行租约.执行上下文.获取错误()
	}
	return 当前执行租约.最终错误
}

func (当前管理器 *管理器) 取消执行(标识 string) bool {
	if 执行租约 := 当前管理器.活动执行租约(标识); 执行租约 != nil {
		return 执行租约.取消执行()
	}
	return false
}

// 取消执行 只作用于这个租约，即使相同执行 标识 后续又被受理。
func (当前执行租约 *执行租约) 取消执行() bool {
	映射 := 当前执行租约.管理器
	映射.互斥锁.加锁()
	defer 映射.互斥锁.解锁()
	if 映射.执行集[当前执行租约.标识] != 当前执行租约 || 当前执行租约.正在结束 {
		return false
	}
	当前执行租约.取消函数()
	return true
}

// 活动执行租约 返回当前受理的租约，供连接在退出时精准取消和等待本次执行。
func (当前管理器 *管理器) 活动执行租约(标识 string) *执行租约 {
	当前管理器.互斥锁.加锁()
	defer 当前管理器.互斥锁.解锁()
	return 当前管理器.执行集[标识]
}

func (当前管理器 *管理器) 活动(标识 string) bool { return 当前管理器.活动执行租约(标识) != nil }

// 等待 等待调用时找到的执行退出，调用者不得在该执行自身调用。
func (当前管理器 *管理器) 等待(标识 string) {
	if 执行租约 := 当前管理器.活动执行租约(标识); 执行租约 != nil {
		执行租约.等待()
	}
}

// 等待 等待本次受理退出，调用者不得在该租约自身调用。
func (当前执行租约 *执行租约) 等待() { <-当前执行租约.完成 }

// 调用 限制实际模型请求数；标题请求也使用同一容量，不能绕过限制。
func (当前管理器 *管理器) 调用(执行上下文 上下文包.上下文, 任务 func() error) error {
	select {
	case <-当前管理器.执行上下文.完成():
		return 智能体包.已关闭错误
	case <-执行上下文.完成():
		return 执行上下文.获取错误()
	case 当前管理器.调用集 <- struct{}{}:
	}
	defer func() { <-当前管理器.调用集 }()
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return 错误
	}
	if 当前管理器.执行上下文.获取错误() != nil {
		return 智能体包.已关闭错误
	}
	return 任务()
}

// 尝试调用 不排队等待辅助调用，主回答优先，返回值表示是否取得容量。
func (当前管理器 *管理器) 尝试调用(执行上下文 上下文包.上下文, 任务 func() error) (bool, error) {
	释放, 成功, 错误 := 当前管理器.尝试获取调用(执行上下文)
	if !成功 || 错误 != nil {
		return 成功, 错误
	}
	defer 释放()
	return true, 任务()
}

// 尝试获取调用 为辅助任务立即预留模型请求容量，取得后必须调用返回的释放函数。
// 预留发生在启动 goroutine 之前，避免主回答完成后辅助任务才争抢到空闲容量。
func (当前管理器 *管理器) 尝试获取调用(执行上下文 上下文包.上下文) (func(), bool, error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return nil, false, 错误
	}
	select {
	case <-当前管理器.执行上下文.完成():
		return nil, false, 智能体包.已关闭错误
	case 当前管理器.调用集 <- struct{}{}:
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			<-当前管理器.调用集
			return nil, false, 错误
		}
		if 当前管理器.执行上下文.获取错误() != nil {
			<-当前管理器.调用集
			return nil, false, 智能体包.已关闭错误
		}
		var 一次 同步包.一次
		return func() { 一次.执行(func() { <-当前管理器.调用集 }) }, true, nil
	default:
		return nil, false, nil
	}
}

func (当前管理器 *管理器) 关闭() {
	当前管理器.互斥锁.加锁()
	当前管理器.已关闭 = true
	当前管理器.取消函数()
	for _, 执行租约 := range 当前管理器.执行集 {
		执行租约.取消函数()
	}
	close(当前管理器.变化通知)
	当前管理器.变化通知 = make(chan struct{})
	当前管理器.互斥锁.解锁()
	当前管理器.等待组.等待()
}
```
