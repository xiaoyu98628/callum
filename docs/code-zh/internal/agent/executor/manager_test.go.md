package 执行器

func Test终态决定并且受理之后清理(测试器 *测试包.测试器) {
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	defer 取消函数()
	管理器, 错误 := 新建(执行上下文, 1, 时间包.秒)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(管理器.关闭)
	执行租约, 错误 := 管理器.预留("first", "conversation")
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	正在结束, 释放 := make(chan struct{}), make(chan struct{})
	执行租约.启动(func(执行上下文 上下文包.上下文) {
		if 错误 := 执行租约.是否正在结束(); 错误 != nil {
			测试器.错误文本(错误)
		}
		close(正在结束)
		select {
		case <-释放:
		case <-执行上下文.完成():
		}
	})
	<-正在结束
	if 管理器.取消执行("first") {
		测试器.错误文本("已确定终态仍允许取消")
	}
	已返回 := make(chan *执行租约, 1)
	失败集 := make(chan error, 1)
	go func() {
		下一个, 错误 := 管理器.预留上下文(测试器.上下文(), "second", "conversation")
		if 错误 != nil {
			失败集 <- 错误
			return
		}
		已返回 <- 下一个
	}()
	close(释放)
	select {
	case 下一个 := <-已返回:
		下一个.中止()
	case 错误 := <-失败集:
		测试器.报告致命错误(错误)
	case <-时间包.之后(时间包.秒):
		测试器.报告致命错误("终态任务清理后未释放容量")
	}
	取消函数()
	if 错误 := 执行租约.是否正在结束(); 错误 != nil {
		测试器.格式化报告致命错误("关闭上下文改写了已确定终态: %v", 错误)
	}
	管理器.关闭()
	管理器.关闭()
	if 错误 := 管理器.调用(测试器.上下文(), func() error { 测试器.错误文本("关闭后调用模型"); return nil }); !错误判断包.是否属于(错误, 智能体包.已关闭错误) {
		测试器.格式化报告致命错误("关闭后未拒绝调用: %v", 错误)
	}
}

func Test模型调用集共享上限并且辅助不队列(测试器 *测试包.测试器) {
	管理器, 错误 := 新建(测试器.上下文(), 1, 时间包.秒)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	测试器.注册清理函数(管理器.关闭)
	已开始, 释放 := make(chan struct{}), make(chan struct{})
	完成 := make(chan error, 1)
	go func() {
		完成 <- 管理器.调用(测试器.上下文(), func() error {
			close(已开始)
			select {
			case <-释放:
			case <-测试器.上下文().完成():
			}
			return nil
		})
	}()
	<-已开始
	if 已获取, 错误 := 管理器.尝试调用(测试器.上下文(), func() error { 测试器.错误文本("辅助调用绕过上限"); return nil }); 已获取 || 错误 != nil {
		测试器.格式化报告致命错误("辅助调用不应排队: %v %v", 已获取, 错误)
	}
	close(释放)
	if 错误 := <-完成; 错误 != nil {
		测试器.报告致命错误(错误)
	}
	首个, 错误 := 管理器.预留("first", "same")
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 管理器.预留("second", "same"); !错误判断包.是否属于(错误, 智能体包.繁忙错误) {
		测试器.报告致命错误("未限制同会话")
	}
	首个.中止()
	秒, 错误 := 管理器.预留("second", "same")
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	秒.中止()
}
