```go
package 模型

import (
	错误判断包 "errors"
	格式化包 "fmt"
)

var (
	无效响应错误  = 错误判断包.新建("模型响应格式无效")
	不完整错误   = 错误判断包.新建("模型生成未完整完成")
	已拒绝错误   = 错误判断包.新建("模型拒绝生成内容")
	不支持输出错误 = 错误判断包.新建("暂不支持此类模型输出")
	图片读取错误  = 错误判断包.新建("图片文件不可读取")
	图片格式错误  = 错误判断包.新建("图片格式不支持")
	附件错误    = 错误判断包.新建("附件不可读取或已超过处理范围")
	图片上限错误  = 错误判断包.新建("图片超过处理上限，请缩小或转换后重试")
	输入上限错误  = 错误判断包.新建("模型输入超过字节上限")
	词元预算错误  = 错误判断包.新建("缺少有效的上下文 Token 预算")
	词元上限错误  = 错误判断包.新建("当前必须保留的上下文超过 Token 预算")
)

// 调用错误 保留调用阶段、HTTP 状态和供应商请求标识。
type 调用错误 struct {
	操作   string
	状态代码 int
	请求标识 string
	原因   error
}

func 新建调用错误(
	操作 string,
	状态代码 int,
	请求标识 string,
	原因 error,
) *调用错误 {
	return &调用错误{
		操作:   操作,
		状态代码: 状态代码,
		请求标识: 请求标识,
		原因:   原因,
	}
}

func (当前调用错误 *调用错误) 错误文本() string {
	return 格式化包.格式化字符串(
		"model %s failed (status=%d)",
		当前调用错误.操作,
		当前调用错误.状态代码,
	)
}

func (当前调用错误 *调用错误) 解包() error {
	return 当前调用错误.原因
}

// 工具参数错误 表示工具身份有效，但 参数 不是 JSON 对象或超过单项字节边界。
// 超限可在流式接收中提前终止；仅在整批尚未执行时可请求模型纠正。
// 指纹 是有界参数的摘要，不包含参数原文。
type 工具参数错误 struct {
	索引   int
	指纹   string
	工具   string
	实际字节 int
	上限字节 int
}

func (当前工具参数错误 *工具参数错误) 错误文本() string {
	if 当前工具参数错误.上限字节 > 0 {
		return 格式化包.格式化字符串("模型第 %d 项工具 %s 的参数超过字节上限（%d > %d），本轮工具均未执行", 当前工具参数错误.索引+1, 当前工具参数错误.工具, 当前工具参数错误.实际字节, 当前工具参数错误.上限字节)
	}
	return 格式化包.格式化字符串("模型第 %d 项工具调用的 arguments 必须是合法 JSON 对象，本轮工具均未执行", 当前工具参数错误.索引+1)
}
func (当前工具参数错误 *工具参数错误) 解包() error {
	if 当前工具参数错误.上限字节 > 0 {
		return 输入上限错误
	}
	return 无效响应错误
}

// 工具纠正停滞错误 表示连续提交相同坏参数且未发生执行进展，不能继续自动纠正。
type 工具纠正停滞错误 struct{}

func (*工具纠正停滞错误) 错误文本() string {
	return "模型重复提交相同无效工具参数，没有执行进展，请检查模型服务的工具调用输出"
}
func (*工具纠正停滞错误) 解包() error { return 无效响应错误 }
```
