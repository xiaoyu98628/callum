const 停止分片 = `{"choices":[{"index":0,"delta":{},"finish_reason":"stop"}]}`

func SSE(事件集 ...string) string {
	return "data: " + 字符串包.合并(事件集, "\n\ndata: ") + "\n\n"
}

func 新建流客户端(测试器 *测试包.测试器, 正文 *已跟踪正文) *客户端 {
	测试器.标记测试辅助函数()
	return 新建测试客户端(测试器, func(请求 *HTTP包.请求) (*HTTP包.响应, error) {
		var 通信 struct {
			流 bool `json:"stream"`
		}
		if 错误 := JSON包.新建解码器(请求.正文).解码(&通信); 错误 != nil {
			return nil, 错误
		}
		if !通信.流 {
			测试器.错误文本("流式请求未设置 stream=true")
		}
		return 测试响应(请求, "text/event-stream", 正文), nil
	})
}

type 片段读取器 struct {
	输入输出包.读取器
	大小 int
}

func (当前片段读取器 片段读取器) 读取(字节数据 []byte) (int, error) {
	return 当前片段读取器.读取器.读取(字节数据[:min(len(字节数据), 当前片段读取器.大小)])
}

type 错误读取器 struct {
	错误 error
}

func (当前错误读取器 错误读取器) 读取([]byte) (int, error) {
	return 0, 当前错误读取器.错误
}

func Test流拒绝音频输出(测试器 *测试包.测试器) {

		for _, 内容 := range []string{"", "正文"} {

func Test流取消释放响应(测试器 *测试包.测试器) {
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	defer 取消函数()
	正文 := &已跟踪正文{读取器: 字符串包.新建读取器(SSE(文本分片, 停止分片, "[DONE]"))}
	客户端 := 新建流客户端(测试器, 正文)
	已完成 := false
	错误 := 客户端.流式生成(执行上下文, 测试请求(), func(事件 模型包.事件) error {
		if 事件.类型 == 模型包.事件文本增量 {
			取消函数()
		}
		已完成 = 已完成 || 事件.类型 == 模型包.事件已完成
		return nil
	})
	if !错误判断包.是否属于(错误, 上下文包.已取消错误) {
		测试器.格式化报告致命错误("取消错误链丢失：%v", 错误)
