```go
package 聊天补全

import (
	上下文包 "context"
	错误判断包 "errors"
	HTTP包 "net/http"
	字符串包 "strings"
	测试包 "testing"

	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

func Test输入执行前检查使用已编码请求无网络(测试器 *测试包.测试器) {
	调用集 := 0
	客户端 := 新建测试客户端(测试器, func(*HTTP包.请求) (*HTTP包.响应, error) { 调用集++; return nil, 错误判断包.新建("不应访问网络") })
	请求 := 测试请求()
	// 有效文本低于 12 MiB，实际 JSON 转义后超过协议边界。
	请求.消息集 = []模型包.消息{{角色: 模型包.角色用户, 内容: 字符串包.重复("\x01", (最大输入字节/6)+100)}}
	错误 := 客户端.校验输入(测试器.上下文(), 请求)
	上限, 成功 := 错误判断包.按类型匹配[*模型包.输入上限错误详情](错误)
	if !成功 || !错误判断包.是否属于(错误, 模型包.输入上限错误) || 上限.尺寸 != "请求字节" || 上限.实际 <= 上限.上限 || 调用集 != 0 {
		测试器.格式化报告致命错误("实际编码预检错误: %+v calls=%d", 错误, 调用集)
	}
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	取消函数()
	if 错误 := 客户端.校验输入(执行上下文, 测试请求()); !错误判断包.是否属于(错误, 上下文包.已取消错误) {
		测试器.格式化报告致命错误("预检忽略取消: %v", 错误)
	}
}
```
