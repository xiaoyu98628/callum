package 聊天补全

import (
	JSON包 "encoding/json"
	错误判断包 "errors"
	HTTP包 "net/http"
	字符串包 "strings"
	测试包 "testing"

	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

func Test推理响应并且工具轮次往返(测试器 *测试包.测试器) {
	for _, 值 := range []string{`"分析"`, `""`, `null`, `42`} {
		测试器.执行(值, func(测试器 *测试包.测试器) {
			正文 := &已跟踪正文{读取器: 字符串包.新建读取器(`{"choices":[{"index":0,"message":{"role":"assistant","content":"回答","reasoning_content":` + 值 + `},"finish_reason":"stop"}]}`)}
			客户端 := 新建测试客户端(测试器, func(请求 *HTTP包.请求) (*HTTP包.响应, error) {
				return 测试响应(请求, "application/json", 正文), nil
			})
			结果, 错误 := 客户端.生成(测试器.上下文(), 测试请求())
			if 值 == `42` {
				if !错误判断包.是否属于(错误, 模型包.无效响应错误) {
					测试器.格式化报告致命错误("接受了无效推理字段: %v", 错误)
				}
			} else if 错误 != nil {
				测试器.报告致命错误(错误)
			} else if 值 == `null` {
				if 结果.推理 != nil {
					测试器.报告致命错误("空字段被伪造为推理")
				}
			} else if 结果.推理 == nil {
				测试器.报告致命错误("显式推理字段丢失")
			}
			if 正文.关闭次数 != 1 {
				测试器.报告致命错误("响应未释放")
			}
		})
	}
	请求 := 测试请求()
	请求.消息集 = append(请求.消息集, 模型包.消息{角色: 模型包.角色助手, 推理: new("分析"), 工具调用集: []模型包.工具调用{{标识: "c", 名称: "read_file", 参数: JSON包.原始JSON消息(`{}`)}}}, 模型包.消息{角色: 模型包.角色工具, 工具调用标识: "c", 内容: "结果"})
	参数, 错误 := (&客户端{}).构建请求(测试器.上下文(), 请求)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	原始, 错误 := JSON包.序列化(参数)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if !字符串包.包含(string(原始), `"reasoning_content":"分析"`) {
		测试器.格式化报告致命错误("工具轮推理未回传: %s", 原始)
	}
}

func Test推理流回调失败关闭响应(测试器 *测试包.测试器) {
	正文 := &已跟踪正文{读取器: 字符串包.新建读取器(SSE(`{"choices":[{"index":0,"delta":{"reasoning_content":"分析"}}]}`, 停止分片, "[DONE]"))}
	客户端 := 新建流客户端(测试器, 正文)
	已拒绝 := 错误判断包.新建("controlled rejection")
	错误 := 客户端.流式生成(测试器.上下文(), 测试请求(), func(事件 模型包.事件) error {
		if 事件.类型 == 模型包.事件推理增量 {
			return 已拒绝
		}
		if 事件.类型 == 模型包.事件已完成 {
			测试器.错误文本("回调失败被当作正常结束")
		}
		return nil
	})
	if !错误判断包.是否属于(错误, 已拒绝) || 正文.关闭次数 != 1 {
		测试器.格式化报告致命错误("回调失败处理错误: %v closes=%d", 错误, 正文.关闭次数)
	}
}
