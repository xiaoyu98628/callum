```go
package 聊天补全

import (
	错误判断包 "errors"
	格式化包 "fmt"
	HTTP包 "net/http"
	字符串包 "strings"
	测试包 "testing"

	OpenAI包 "github.com/openai/openai-go/v3"
)

func Test包装错误保留原因无暴露详情(测试器 *测试包.测试器) {
	原因 := 错误判断包.新建("测试原始错误")
	密钥 := "test-secret-do-not-log"
	错误 := 包装错误("generate", &HTTP包.响应{
		状态代码: HTTP包.状态无效网关,
		头部:   HTTP包.头部{"X-Request-Id": {测试请求标识}},
	}, 格式化包.格式化创建错误("credential=%s: %w", 密钥, 原因))
	if !错误判断包.是否属于(错误, 原因) {
		测试器.报告致命错误("原始错误链丢失")
	}
	调用错误 := 断言调用错误(测试器, 错误, HTTP包.状态无效网关, 测试请求标识)
	if 调用错误.操作 != "generate" {
		测试器.格式化创建错误("调用阶段为 %q", 调用错误.操作)
	}
	if 字符串包.包含(错误.错误文本(), 密钥) {
		测试器.错误文本("普通错误输出包含原始敏感信息")
	}
}

func Test包装错误回退返回目标SDK响应(测试器 *测试包.测试器) {
	API错误 := &OpenAI包.错误{
		状态代码: HTTP包.状态过于多个请求集,
		响应: &HTTP包.响应{
			状态代码: HTTP包.状态过于多个请求集,
			头部:   HTTP包.头部{"X-Request-Id": {测试请求标识}},
		},
	}
	错误 := 包装错误("generate", nil, 格式化包.格式化创建错误("SDK 调用失败: %w", API错误))
	断言调用错误(测试器, 错误, HTTP包.状态过于多个请求集, 测试请求标识)
	if 已解包, 成功 := 错误判断包.按类型匹配[*OpenAI包.错误](错误); !成功 || 已解包 != API错误 {
		测试器.报告致命错误("供应商错误链丢失")
	}
}
```
