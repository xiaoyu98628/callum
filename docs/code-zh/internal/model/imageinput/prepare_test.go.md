package 图片输入

func Test图片准备遵循补丁并且像素预算(测试器 *测试包.测试器) {
	var 原始 字节包.缓冲区
	if 错误 := PNG包.编码(&原始, 图片包.新建RGBA(图片包.矩形(0, 0, 1800, 1800))); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	已准备, 错误 := 准备(测试器.上下文(), 原始.字节(), "image/png")
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	配置模块, 错误 := PNG包.解码配置(字节包.新建读取器(已准备))
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 配置模块.宽度 != 1600 || 配置模块.高度 != 1600 || len(已准备) > 最大字节 {
		测试器.格式化报告致命错误("图片未按 patch 预算缩放: %dx%d %d bytes", 配置模块.宽度, 配置模块.高度, len(已准备))
	}
}

func Test图片准备接受大型已编码来源范围内像素预算(测试器 *测试包.测试器) {
	var 原始 字节包.缓冲区
	if 错误 := PNG包.编码(&原始, 图片包.新建RGBA(图片包.矩形(0, 0, 12, 9))); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	// 合法的 PNG 文本块使原文件超过发送边界，像素尺寸仍然很小。
	分片 := make([]byte, 12+(5<<20))
	二进制包.大端字节序.放入无符号整数32(分片, uint32(len(分片)-12))
	copy(分片[4:], "tEXt")
	copy(分片[8:], "Description\x00")
	二进制包.大端字节序.放入无符号整数32(分片[len(分片)-4:], CRC32包.校验和IEEE(分片[4:len(分片)-4]))
	数据 := append(append(append([]byte(nil), 原始.字节()[:33]...), 分片...), 原始.字节()[33:]...)
	已准备, 错误 := 准备(测试器.上下文(), 数据, "image/png")
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	启动配置, 错误 := PNG包.解码配置(字节包.新建读取器(已准备))
	if 错误 != nil || 启动配置.宽度 != 12 || 启动配置.高度 != 9 || len(已准备) > 最大字节 {
		测试器.格式化报告致命错误("小尺寸原图重编码不正确: %+v %v", 启动配置, 错误)
	}
}
