//go:build unix

package 图片输入

import (
	上下文包 "context"
	错误判断包 "errors"
	操作系统包 "os"
	文件路径包 "path/filepath"
	系统调用包 "syscall"
	测试包 "testing"
	时间包 "time"

	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

func Test图片文件已替换依据先进先出不内容块(测试器 *测试包.测试器) {
	路径 := 文件路径包.合并(测试器.创建临时目录(), "input.png")
	if 错误 := 操作系统包.写入文件(路径, []byte("image-content"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	预期, 错误 := 操作系统包.状态信息(路径)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 操作系统包.移除(路径); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 系统调用包.创建命名管道(路径, 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	// 确定性地在检查与打开之间替换目标，不依赖高频路径竞争。
	执行上下文, 取消函数 := 上下文包.创建超时上下文(测试器.上下文(), 时间包.秒)
	defer 取消函数()
	完成 := make(chan error, 1)
	go func() {
		_, 错误 := 读取文件(执行上下文, 路径, 预期)
		完成 <- 错误
	}()
	select {
	case 错误 := <-完成:
		if !错误判断包.是否属于(错误, 模型包.图片读取错误) {
			测试器.格式化报告致命错误("未拒绝被替换的图片: %v", 错误)
		}
	case <-执行上下文.完成():
		// 回归时用写端解除阻塞的 打开，确保失败测试也不遗留读取 goroutine。
		写入器, 错误 := 操作系统包.打开文件(路径, 操作系统包.仅写打开标志|系统调用包.非阻塞标志, 0)
		if 错误 != nil {
			测试器.错误文本(错误)
		} else if 错误 := 写入器.关闭(); 错误 != nil {
			测试器.错误文本(错误)
		}
		select {
		case <-完成:
		case <-时间包.之后(时间包.秒):
			测试器.错误文本("解除命名管道阻塞后读取仍未退出")
		}
		测试器.报告致命错误("命名管道替换导致图片打开阻塞，取消无法结束")
	}
}
