package 图片输入

import (
	上下文包 "context"
	错误判断包 "errors"
	操作系统包 "os"
	文件路径包 "path/filepath"
	测试包 "testing"

	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

func Test图片文件读取拒绝替换并且取消(测试器 *测试包.测试器) {
	目录 := 测试器.创建临时目录()
	路径, 替换 := 文件路径包.合并(目录, "input.png"), 文件路径包.合并(目录, "replacement.png")
	for _, 名称 := range []string{路径, 替换} {
		if 错误 := 操作系统包.写入文件(名称, []byte("image-content"), 0600); 错误 != nil {
			测试器.报告致命错误(错误)
		}
	}
	预期, 错误 := 操作系统包.状态信息(路径)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 操作系统包.移除(路径); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 错误 := 操作系统包.重命名(替换, 路径); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 内容, 错误 := 读取文件(测试器.上下文(), 路径, 预期); !错误判断包.是否属于(错误, 模型包.图片读取错误) || 内容 != nil {
		测试器.格式化报告致命错误("替换后的文件仍被读取: bytes=%d err=%v", len(内容), 错误)
	}
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	取消函数()
	if _, 错误 := 读取文件(执行上下文, 路径, 预期); !错误判断包.是否属于(错误, 上下文包.已取消错误) {
		测试器.格式化报告致命错误("打开图片前忽略取消: %v", 错误)
	}
}

func Test图片文件增长保留上限错误(测试器 *测试包.测试器) {
	路径 := 文件路径包.合并(测试器.创建临时目录(), "input.png")
	if 错误 := 操作系统包.写入文件(路径, []byte("image"), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	预期, 错误 := 操作系统包.状态信息(路径)
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	// 同一文件在大小检查后增长，仍由实际读取上限拒绝，不改成普通读取错误。
	if 错误 := 操作系统包.写入文件(路径, make([]byte, 最大来源字节+1), 0600); 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if 内容, 错误 := 读取文件(测试器.上下文(), 路径, 预期); !错误判断包.是否属于(错误, 模型包.图片上限错误) || 错误判断包.是否属于(错误, 模型包.图片读取错误) || 内容 != nil {
		测试器.格式化报告致命错误("文件增长后未保留输入限额错误: bytes=%d err=%v", len(内容), 错误)
	}
}
