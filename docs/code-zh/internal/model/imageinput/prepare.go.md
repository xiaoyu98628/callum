package 图片输入

import (
	字节包 "bytes"
	上下文包 "context"
	二进制包 "encoding/binary"
	错误判断包 "errors"
	图片包 "image"
	颜色包 "image/color"
	_ "image/gif"
	JPEG包 "image/jpeg"
	PNG包 "image/png"

	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

const (
	最大来源像素集 = 16_000_000
	最大来源尺寸  = 16_384
	最大尺寸    = 2048
	补丁大小    = 32
	最大补丁集   = 2500
)

// 准备 在本地约束内保留原编码，或等比缩小 PNG/JPEG 的发送版本。
// GIF/WebP 保留原始帧及编码，超过发送边界时要求用户转换。
func 准备(执行上下文 上下文包.上下文, 数据 []byte, MIME string) ([]byte, error) {
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return nil, 错误
	}
	宽度, 高度, 错误 := 尺寸集(数据, MIME)
	if 错误 != nil {
		return nil, 错误
	}
	if 宽度 <= 0 || 高度 <= 0 || 宽度 > 最大来源尺寸 || 高度 > 最大来源尺寸 || int64(宽度) > 最大来源像素集/int64(高度) {
		return nil, 模型包.图片上限错误
	}
	写入器, 处理器 := 目标大小(宽度, 高度)
	if MIME == "image/webp" {
		if 写入器 != 宽度 || 处理器 != 高度 || len(数据) > 最大字节 {
			return nil, 模型包.图片上限错误
		}
		return 数据, nil
	}
	// 解码 也验证完整的静态图像。GIF 只解码首帧来校验并保留原始编码。
	来源, _, 错误 := 图片包.解码(上下文读取器{执行上下文, 字节包.新建读取器(数据)})
	if 错误 != nil {
		return nil, 错误判断包.合并(模型包.图片格式错误, 错误)
	}
	if 错误 := 执行上下文.获取错误(); 错误 != nil {
		return nil, 错误
	}
	if 写入器 == 宽度 && 处理器 == 高度 && len(数据) <= 最大字节 {
		return 数据, nil
	}
	if MIME == "image/gif" {
		return nil, 模型包.图片上限错误
	}
	for {
		已缩放, 错误 := 缩放(执行上下文, 来源, 写入器, 处理器)
		if 错误 != nil {
			return nil, 错误
		}
		var 缓冲区 字节包.缓冲区
		写入器 := 上下文写入器{执行上下文, &缓冲区}
		if MIME == "image/jpeg" {
			错误 = JPEG包.编码(写入器, 已缩放, &JPEG包.选项集{质量: 90})
		} else {
			错误 = PNG包.编码(写入器, 已缩放)
		}
		if 错误 != nil {
			return nil, 错误
		}
		if 缓冲区.长度() <= 最大字节 {
			return 缓冲区.字节(), 执行上下文.获取错误()
		}
		写入器, 处理器 = max(1, 写入器*3/4), max(1, 处理器*3/4)
	}
}

type 上下文写入器 struct {
	执行上下文 上下文包.上下文
	缓冲区   *字节包.缓冲区
}

func (当前上下文写入器 上下文写入器) 写入(字节数据 []byte) (int, error) {
	if 错误 := 当前上下文写入器.执行上下文.获取错误(); 错误 != nil {
		return 0, 错误
	}
	return 当前上下文写入器.缓冲区.写入(字节数据)
}

func 目标大小(宽度, 高度 int) (int, int) {
	长 := max(宽度, 高度)
	下界, 上界 := 1, min(长, 最大尺寸)
	for 下界 < 上界 {
		消息标识 := (下界 + 上界 + 1) / 2
		写入器, 处理器 := max(1, int(int64(宽度)*int64(消息标识)/int64(长))), max(1, int(int64(高度)*int64(消息标识)/int64(长)))
		if ((写入器+补丁大小-1)/补丁大小)*((处理器+补丁大小-1)/补丁大小) <= 最大补丁集 {
			下界 = 消息标识
		} else {
			上界 = 消息标识 - 1
		}
	}
	return max(1, int(int64(宽度)*int64(下界)/int64(长))), max(1, int(int64(高度)*int64(下界)/int64(长)))
}

// 每个目标像素取对应源区域的平均值，缩小截图时保留细线信息。
func 缩放(执行上下文 上下文包.上下文, 来源 图片包.图片, 宽度, 高度 int) (*图片包.RGBA, error) {
	边界 := 来源.边界()
	目标 := 图片包.新建RGBA(图片包.矩形(0, 0, 宽度, 高度))
	for 纵坐标 := range 高度 {
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			return nil, 错误
		}
		纵坐标0, 纵坐标1 := 纵坐标*边界.纵向偏移()/高度, (纵坐标+1)*边界.纵向偏移()/高度
		纵坐标1 = min(边界.纵向偏移(), max(纵坐标0+1, 纵坐标1))
		for 横坐标 := range 宽度 {
			横坐标0, 横坐标1 := 横坐标*边界.横向偏移()/宽度, (横坐标+1)*边界.横向偏移()/宽度
			横坐标1 = min(边界.横向偏移(), max(横坐标0+1, 横坐标1))
			var 当前读取器, 组, 乙, 甲 uint64
			for 纵向比例 := 纵坐标0; 纵向比例 < 纵坐标1; 纵向比例++ {
				for 横向比例 := 横坐标0; 横向比例 < 横坐标1; 横向比例++ {
					预乘红色, 预乘绿色, 预乘蓝色, 预乘透明度 := 来源.位于(边界.最小.横坐标+横向比例, 边界.最小.纵坐标+纵向比例).RGBA()
					当前读取器, 组, 乙, 甲 = 当前读取器+uint64(预乘红色), 组+uint64(预乘绿色), 乙+uint64(预乘蓝色), 甲+uint64(预乘透明度)
				}
			}
			数量 := uint64((横坐标1-横坐标0)*(纵坐标1-纵坐标0)) * 257
			目标.设置RGBA(横坐标, 纵坐标, 颜色包.RGBA{红色: uint8(当前读取器 / 数量), 绿色: uint8(组 / 数量), 蓝色: uint8(乙 / 数量), 透明度: uint8(甲 / 数量)})
		}
	}
	return 目标, nil
}

func 尺寸集(数据 []byte, MIME string) (int, int, error) {
	if MIME != "image/webp" {
		启动配置, _, 错误 := 图片包.解码配置(字节包.新建读取器(数据))
		if 错误 != nil {
			return 0, 0, 错误判断包.合并(模型包.图片格式错误, 错误)
		}
		return 启动配置.宽度, 启动配置.高度, nil
	}
	// WebP 的首个块记录静态位流尺寸或扩展画布尺寸。
	if len(数据) < 20 || string(数据[:4]) != "RIFF" || string(数据[8:12]) != "WEBP" ||
		uint64(二进制包.小端字节序.无符号整数32(数据[4:8]))+8 != uint64(len(数据)) {
		return 0, 0, 模型包.图片格式错误
	}
	长度 := 二进制包.小端字节序.无符号整数32(数据[16:20])
	if uint64(长度)+20 > uint64(len(数据)) {
		return 0, 0, 模型包.图片格式错误
	}
	乙 := 数据[20 : 20+int(长度)]
	switch string(数据[12:16]) {
	case "VP8X":
		if len(乙) == 10 {
			return 1 + int(乙[4]) + int(乙[5])<<8 + int(乙[6])<<16, 1 + int(乙[7]) + int(乙[8])<<8 + int(乙[9])<<16, nil
		}
	case "VP8 ":
		if len(乙) >= 10 && 字节包.相等(乙[3:6], []byte{0x9d, 0x01, 0x2a}) {
			return int(二进制包.小端字节序.无符号整数16(乙[6:8]) & 0x3fff), int(二进制包.小端字节序.无符号整数16(乙[8:10]) & 0x3fff), nil
		}
	case "VP8L":
		if len(乙) >= 5 && 乙[0] == 0x2f {
			位 := 二进制包.小端字节序.无符号整数32(乙[1:5])
			return 1 + int(位&0x3fff), 1 + int((位>>14)&0x3fff), nil
		}
	}
	return 0, 0, 模型包.图片格式错误
}
