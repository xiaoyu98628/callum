package 词元测试

import (
	字符串包 "strings"
	测试包 "testing"

	模型包 "github.com/xiaoyu98628/callum/internal/model"
	词元包 "github.com/xiaoyu98628/callum/internal/model/tokens"
)

func Test估算包含工具集并且使用生效内容(测试器 *测试包.测试器) {
	纯文本 := []模型包.消息{{角色: 模型包.角色用户, 内容: "hello"}}
	基准 := 词元包.估算(纯文本, nil, 8192)
	工具 := 模型包.工具定义{名称: "edit_file", 说明: 字符串包.重复("修改代码，保留约束。", 100), 参数: []byte(`{"type":"object","properties":{"content":{"type":"string"}}}`)}
	if 词元包.估算(纯文本, []模型包.工具定义{工具}, 8192) <= 基准+500 {
		测试器.报告致命错误("工具定义没有计入预算")
	}
	已阻止 := []模型包.消息{{角色: 模型包.角色用户, 内容: 字符串包.重复("ignored", 1000), 内容块集: []模型包.内容块{{类型: "text", 文本: "hello"}}}}
	if 词元包.估算(已阻止, nil, 8192) != 基准 {
		测试器.报告致命错误("重复计算被 Blocks 替代的 Content")
	}
	图片 := []模型包.消息{{角色: 模型包.角色用户, 内容块集: []模型包.内容块{{类型: "image", 路径: "/short.png"}}}}
	之前 := 词元包.估算(图片, nil, 8192)
	图片[0].内容块集[0].路径 = "/" + 字符串包.重复("long", 1000) + ".png"
	if 词元包.估算(图片, nil, 8192) != 之前 || 之前 < 8192 {
		测试器.报告致命错误("图片按路径字节数计算 Token")
	}
	轨迹 := []模型包.消息{{角色: 模型包.角色助手, 工具调用集: []模型包.工具调用{{标识: "call", 名称: "edit_file", 参数: []byte(`{"content":"代码"}`)}}}, {角色: 模型包.角色工具, 工具调用标识: "call", 内容: 字符串包.重复("result", 1000)}}
	if 词元包.估算(轨迹, nil, 8192) < 2000 {
		测试器.报告致命错误("工具调用或结果未计入预算")
	}
	for _, 文本 := range []string{"中文约束必须保留", "func main() { println(123) }", "emoji: 👩‍💻"} {
		if 词元包.文本(文本) <= 0 {
			测试器.报告致命错误("非英文或代码未计入预算")
		}
	}
}

func Test实际输入引用仅应用目标相同前缀(测试器 *测试包.测试器) {
	基准 := []模型包.消息{{角色: 模型包.角色用户, 内容: "hello"}}
	实际 := int64(1000)
	var 计量器 词元包.计量器
	计量器.观测(基准, nil, &实际)
	下一个 := append(append([]模型包.消息{}, 基准...), 模型包.消息{角色: 模型包.角色工具, 内容: 字符串包.重复("tool output ", 100)})
	结果 := 计量器.估算(下一个, nil, 8192)
	if 结果.词元 <= 实际 || 结果.引用输入词元 != 实际 {
		测试器.报告致命错误("没有在实际输入用量基础上计算新增工具内容")
	}
	下一个[0].内容 = "历史已压缩"
	if 实际值 := 计量器.估算(下一个, nil, 8192); 实际值.引用输入词元 != 0 || 实际值.方法 != 词元包.方法 {
		测试器.报告致命错误("压缩后复用了旧 usage")
	}
	if 实际值 := 计量器.估算(基准, []模型包.工具定义{{名称: "changed"}}, 8192); 实际值.引用输入词元 != 0 {
		测试器.报告致命错误("工具定义变化后复用旧 usage")
	}
	低 := int64(1)
	计量器.观测(基准, nil, &低)
	if 计量器.估算(基准, nil, 8192).词元 != 词元包.估算(基准, nil, 8192) {
		测试器.报告致命错误("较小 usage 覆盖了本地估算")
	}
	计量器.观测(基准, nil, nil)
	if 计量器.估算(基准, nil, 8192).引用输入词元 != 0 {
		测试器.报告致命错误("缺失 usage 时推测了实际用量")
	}
}

func Test推理计数朝向输入预算(测试器 *测试包.测试器) {
	消息集 := []模型包.消息{{角色: 模型包.角色助手, 内容: "回答"}}
	之前 := 词元包.估算(消息集, nil, 0)
	消息集[0].推理 = new(字符串包.重复("推理", 100))
	if 词元包.估算(消息集, nil, 0) != 之前+词元包.文本(*消息集[0].推理) {
		测试器.报告致命错误("推理未计入输入预算")
	}
	var 计量器 词元包.计量器
	实际 := int64(1000)
	计量器.观测(消息集, nil, &实际)
	消息集[0].推理 = new("不同推理")
	if 计量器.估算(消息集, nil, 0).引用输入词元 != 0 {
		测试器.报告致命错误("推理变化后复用了旧输入用量")
	}
}

func Test文本计数器匹配集整体文本跨片段集(测试器 *测试包.测试器) {
	for _, 文本 := range []string{"", "abcdefghijklmnopqrstuvwxyz 12345", "func main() {\n println(123) }", "中文约束👩‍💻与a b c结束", "ab,cd!é fg\t中文"} {
		字符集 := []rune(文本)
		for 宽度 := 1; 宽度 <= 7; 宽度++ {
			var 计数器 词元包.文本计数器
			for 索引 := 0; 索引 < len(字符集); 索引 += 宽度 {
				结束 := min(索引+宽度, len(字符集))
				计数器.添加(string(字符集[索引:结束]))
				if 实际值, 预期 := 计数器.词元(), 词元包.文本(string(字符集[:结束])); 实际值 != 预期 {
					测试器.格式化报告致命错误("分片取整不同: width=%d got=%d want=%d", 宽度, 实际值, 预期)
				}
			}
			if 计数器.词元() != 词元包.文本(文本) {
				测试器.报告致命错误("最终计数不同")
			}
		}
	}
}
