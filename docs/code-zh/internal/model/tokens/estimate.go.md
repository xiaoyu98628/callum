// 包 词元 估算统一请求的输入 词元，用于压缩规划，不提供计费用量或容量保证。
package 词元

import (
	UTF8包 "unicode/utf8"

	模型包 "github.com/xiaoyu98628/callum/internal/model"
)

// 方法 固定估算规则版本，供步骤及摘要收益审计使用。
const 方法 = "unicode-v1"

// 文本 对 ASCII 字母、数字、空白按三字符一 词元 估算，标点单独计数；
// 非 ASCII 按每码点二 词元、四字节码点四 词元 预留。它不是厂商 tokenizer。
func 文本(值 string) int64 {
	var 计数器 文本计数器
	计数器.添加(值)
	return 计数器.词元()
}

// 文本计数器 按完整 UTF-8 文本分片增量计算同一段文本，不因分片边界重复取整。
type 文本计数器 struct{ 总计, 运行 int64 }

func (当前文本计数器 *文本计数器) 添加(值 string) {
	for _, 当前读取器 := range 值 {
		if 当前读取器 < 128 && ((当前读取器 >= 'a' && 当前读取器 <= 'z') || (当前读取器 >= 'A' && 当前读取器 <= 'Z') || (当前读取器 >= '0' && 当前读取器 <= '9') || 当前读取器 == ' ' || 当前读取器 == '\n' || 当前读取器 == '\r' || 当前读取器 == '\t') {
			当前文本计数器.运行++
			continue
		}
		当前文本计数器.总计 += (当前文本计数器.运行 + 2) / 3
		当前文本计数器.运行 = 0
		if 当前读取器 < 128 {
			当前文本计数器.总计++
		} else if UTF8包.字符长度(当前读取器) == 4 {
			当前文本计数器.总计 += 4
		} else {
			当前文本计数器.总计 += 2
		}
	}
}

func (当前文本计数器 *文本计数器) 词元() int64 { return 当前文本计数器.总计 + (当前文本计数器.运行+2)/3 }

// 估算 统计有效消息内容、工具参数及定义，图片使用明确配置的单图估算，忽略路径长度。
func 估算(消息集 []模型包.消息, 工具集 []模型包.工具定义, 图片词元 int64) int64 {
	总计 := int64(16)
	for _, 映射 := range 消息集 {
		if 映射.推理 != nil {
			总计 += 文本(*映射.推理)
		}
		总计 += 8 + 文本(string(映射.角色)) + 文本(映射.工具调用标识)
		if len(映射.内容块集) == 0 {
			总计 += 文本(映射.内容)
		}
		for _, 乙 := range 映射.内容块集 {
			if 乙.类型 == "image" {
				总计 += 图片词元
			} else if 乙.类型 == "file" {
				总计 += 文本(模型包.附件文本(乙))
			} else {
				总计 += 文本(乙.文本)
			}
		}
		for _, 调用 := range 映射.工具调用集 {
			总计 += 12 + 文本(调用.标识) + 文本(调用.名称) + 文本(string(调用.参数))
		}
	}
	for _, 工具 := range 工具集 {
		总计 += 16 + 文本(工具.名称) + 文本(工具.说明) + 文本(string(工具.参数))
	}
	return 总计
}
