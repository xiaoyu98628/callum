package 模型

import (
	上下文包 "context"
	格式化包 "fmt"
)

// 输入上限错误详情 区分资源维度；仍可通过 输入上限错误 识别输入资源边界。
type 输入上限错误详情 struct {
	尺寸     string
	实际, 上限 int64
}

func (当前输入上限错误详情 *输入上限错误详情) 错误文本() string {
	return 格式化包.格式化字符串("模型输入%s超限（%d > %d）", 当前输入上限错误详情.尺寸, 当前输入上限错误详情.实际, 当前输入上限错误详情.上限)
}
func (当前输入上限错误详情 *输入上限错误详情) 解包() error { return 输入上限错误 }

// 输入校验器 在网络请求前检查实际协议封装，资源由实现自行释放，不发起模型调用。
type 输入校验器 interface {
	校验输入(上下文包.上下文, 请求) error
}
