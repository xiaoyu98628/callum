```go
package 服务

import (
	格式化包 "fmt"
	URL包 "net/url"
	字符串包 "strings"

	供应商包 "github.com/xiaoyu98628/callum/internal/model/provider"
)

func 校验供应商(输入 供应商包.供应商输入) error {
	if 字符串包.裁剪空间(输入.名称) == "" {
		return 格式化包.格式化创建错误("%w: 厂商名称不能为空", 供应商包.无效配置错误)
	}
	if 输入.协议 != "chatcompletions" {
		return 格式化包.格式化创建错误("%w: 当前不支持此协议", 供应商包.无效配置错误)
	}
	地址, 错误 := URL包.解析(输入.基础URL)
	if 错误 != nil || (地址.方案 != "http" && 地址.方案 != "https") || 地址.主机 == "" || 地址.用户 != nil || 地址.原始查询 != "" || 地址.片段 != "" {
		return 格式化包.格式化创建错误("%w: API 根地址无效", 供应商包.无效配置错误)
	}
	return nil
}

func 校验模型(输入 供应商包.模型输入) error {
	if !供应商包.有效标识(输入.供应商标识) || 字符串包.裁剪空间(输入.显示名称) == "" || 字符串包.裁剪空间(输入.远程模型标识) == "" {
		return 格式化包.格式化创建错误("%w: 厂商、名称和远端模型标识必须有效", 供应商包.无效配置错误)
	}
	if 输入.上下文窗口 != nil && *输入.上下文窗口 <= 0 || 输入.最大输出词元 != nil && *输入.最大输出词元 <= 0 {
		return 格式化包.格式化创建错误("%w: Token 容量必须大于零", 供应商包.无效配置错误)
	}
	for 功能 := range 输入.能力集 {
		if !已知功能(功能) {
			return 格式化包.格式化创建错误("%w: 未识别的模型功能", 供应商包.无效配置错误)
		}
	}
	已选择 := make(map[string]bool, len(输入.模型类型))
	for _, 类别 := range 输入.模型类型 {
		if !供应商包.有效模型类型(类别) || 已选择[类别] {
			return 格式化包.格式化创建错误("%w: 模型内容类型无效或重复", 供应商包.无效配置错误)
		}
		已选择[类别] = true
	}
	等级集 := make(map[string]bool, len(输入.推理等级集))
	if 受支持, 已知 := 输入.能力集["reasoning"]; 已知 && !受支持 && len(输入.推理等级集) > 0 {
		return 格式化包.格式化创建错误("%w: 不支持推理的模型不能配置推理等级", 供应商包.无效配置错误)
	}
	for _, 级别 := range 输入.推理等级集 {
		if 字符串包.裁剪空间(级别.显示名称) == "" || 字符串包.裁剪空间(级别.远程值) == "" || 等级集[级别.远程值] {
			return 格式化包.格式化创建错误("%w: 推理等级的名称或远端值无效", 供应商包.无效配置错误)
		}
		等级集[级别.远程值] = true
	}
	return nil
}

func 已知功能(功能 string) bool {
	switch 功能 {
	case "reasoning", "image_input", "tools":
		return true
	default:
		return false
	}
}
```
