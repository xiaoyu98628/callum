package 服务

import (
	上下文包 "context"
	错误判断包 "errors"
	测试包 "testing"

	供应商包 "github.com/xiaoyu98628/callum/internal/model/provider"
)

const 测试标识 = "01900000-0000-7000-8000-000000000001"

type 仓储桩 struct {
	供应商包.仓储
	创建供应商回调 func(上下文包.上下文, string, 供应商包.供应商输入) error
	加载运行时回调    func(上下文包.上下文, string) (供应商包.运行时配置, error)
	获取模型回调       func(上下文包.上下文, string) (供应商包.模型, error)
	列出模型回调     func(上下文包.上下文, bool) ([]供应商包.模型, error)
}

func (当前仓储桩 仓储桩) 创建供应商(执行上下文 上下文包.上下文, 标识 string, 输入 供应商包.供应商输入) error {
	return 当前仓储桩.创建供应商回调(执行上下文, 标识, 输入)
}
func (当前仓储桩 仓储桩) 加载运行时(执行上下文 上下文包.上下文, 标识 string) (供应商包.运行时配置, error) {
	return 当前仓储桩.加载运行时回调(执行上下文, 标识)
}
func (当前仓储桩 仓储桩) 获取模型(执行上下文 上下文包.上下文, 标识 string) (供应商包.模型, error) {
	return 当前仓储桩.获取模型回调(执行上下文, 标识)
}
func (当前仓储桩 仓储桩) 列表模型集(执行上下文 上下文包.上下文, 仅 bool, _ ...供应商包.列表选项集) (供应商包.页[供应商包.模型], error) {
	项集, 错误 := 当前仓储桩.列出模型回调(执行上下文, 仅)
	return 供应商包.页[供应商包.模型]{项集: 项集}, 错误
}

func Test服务校验之前存储并且分配标识(测试器 *测试包.测试器) {
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	defer 取消函数()
	已调用 := false
	已存储标识 := ""
	服务, 错误 := 新建(仓储桩{创建供应商回调: func(实际值 上下文包.上下文, 标识 string, 输入 供应商包.供应商输入) error {
		已调用 = true
		已存储标识 = 标识
		if 实际值 != 执行上下文 || !供应商包.有效标识(标识) || 输入.名称 != "厂商" {
			测试器.报告致命错误("上下文、标识或配置未正确传给存储")
		}
		return nil
	}})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 服务.创建供应商(执行上下文, 供应商包.供应商输入{}); !错误判断包.是否属于(错误, 供应商包.无效配置错误) || 已调用 {
		测试器.格式化报告致命错误("无效配置触达存储: %v", 错误)
	}
	标识, 错误 := 服务.创建供应商(执行上下文, 供应商包.供应商输入{名称: "厂商", 协议: "chatcompletions", 基础URL: "https://example.invalid"})
	if 错误 != nil || 标识 != 已存储标识 || !已调用 {
		测试器.格式化报告致命错误("创建失败: %s %v", 标识, 错误)
	}
	if _, 错误 := 新建(nil); 错误 == nil {
		测试器.报告致命错误("接受了空存储")
	}
}

func Test服务强制运行时规则集并且保留上下文(测试器 *测试包.测试器) {
	运行时 := 供应商包.运行时配置{模型: 供应商包.模型{标识: 测试标识, 模型输入: 供应商包.模型输入{供应商标识: 测试标识, 显示名称: "模型", 远程模型标识: "remote"}}, API密钥: "fake-service-key"}
	读取 := 0
	服务, 错误 := 新建(仓储桩{加载运行时回调: func(执行上下文 上下文包.上下文, 标识 string) (供应商包.运行时配置, error) {
		读取++
		if 错误 := 执行上下文.获取错误(); 错误 != nil {
			return 供应商包.运行时配置{}, 错误
		}
		return 运行时, nil
	}})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 服务.解析(测试器.上下文(), "invalid"); !错误判断包.是否属于(错误, 供应商包.无效配置错误) || 读取 != 0 {
		测试器.格式化报告致命错误("无效标识触达存储: %v", 错误)
	}
	if _, 错误 := 服务.解析(测试器.上下文(), 测试标识); !错误判断包.是否属于(错误, 供应商包.已停用错误) {
		测试器.格式化报告致命错误("未阻止禁用模型: %v", 错误)
	}
	运行时.模型.已启用 = true
	实际值, 错误 := 服务.解析(测试器.上下文(), 测试标识)
	if 错误 != nil || 实际值.API密钥 != 运行时.API密钥 {
		测试器.格式化报告致命错误("有效运行配置丢失: %v", 错误)
	}
	执行上下文, 取消函数 := 上下文包.创建可取消上下文(测试器.上下文())
	取消函数()
	if _, 错误 := 服务.解析(执行上下文, 测试标识); !错误判断包.是否属于(错误, 上下文包.已取消错误) {
		测试器.格式化报告致命错误("取消信号丢失: %v", 错误)
	}
	运行时.模型.模型类型 = []string{"tools"}
	if _, 错误 := 服务.解析(测试器.上下文(), 测试标识); !错误判断包.是否属于(错误, 供应商包.无效配置错误) {
		测试器.格式化报告致命错误("无效持久化配置未被检查: %v", 错误)
	}
}

func Test服务校验读取模型集并且保留存储错误(测试器 *测试包.测试器) {
	无效 := 供应商包.模型{标识: 测试标识, 供应商标识: 测试标识, 显示名称: "m", 远程模型标识: "r", 上下文窗口: new(int64(0))}
	存储错误 := 错误判断包.新建("存储失败")
	服务, 错误 := 新建(仓储桩{
		获取模型回调:       func(上下文包.上下文, string) (供应商包.模型, error) { return 无效, nil },
		列出模型回调:     func(上下文包.上下文, bool) ([]供应商包.模型, error) { return []供应商包.模型{无效}, nil },
		创建供应商回调: func(上下文包.上下文, string, 供应商包.供应商输入) error { return 存储错误 },
	})
	if 错误 != nil {
		测试器.报告致命错误(错误)
	}
	if _, 错误 := 服务.获取模型(测试器.上下文(), 测试标识); !错误判断包.是否属于(错误, 供应商包.无效配置错误) {
		测试器.格式化报告致命错误("查询未校验配置: %v", 错误)
	}
	if 行集, 错误 := 服务.列表模型集(测试器.上下文(), false); !错误判断包.是否属于(错误, 供应商包.无效配置错误) || 行集.项集 != nil {
		测试器.格式化报告致命错误("返回了无效列表: %v", 错误)
	}
	if _, 错误 := 服务.创建供应商(测试器.上下文(), 供应商包.供应商输入{名称: "p", 协议: "chatcompletions", 基础URL: "https://example.invalid"}); !错误判断包.是否属于(错误, 存储错误) {
		测试器.格式化报告致命错误("存储错误链丢失: %v", 错误)
	}
}
