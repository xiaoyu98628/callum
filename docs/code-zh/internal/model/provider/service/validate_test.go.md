package 服务

import (
	错误判断包 "errors"
	字符串包 "strings"
	测试包 "testing"

	供应商包 "github.com/xiaoyu98628/callum/internal/model/provider"

func Test无效模型集(测试器 *测试包.测试器) {
	用例集 := []供应商包.模型输入{
		{供应商标识: "01900000-0000-7000-8000-000000000001", 显示名称: "m", 远程模型标识: "r", 能力集: map[string]bool{"reasoning": false}, 推理等级集: []供应商包.推理级别{{显示名称: "中", 远程值: "medium"}}},
		{},
		{供应商标识: "01900000-0000-7000-8000-000000000001", 显示名称: "m", 远程模型标识: "r", 上下文窗口: new(int64(0))},
		{供应商标识: "01900000-0000-7000-8000-000000000001", 显示名称: "m", 远程模型标识: "r", 模型类型: []string{"reasoning"}},
		{供应商标识: "01900000-0000-7000-8000-000000000001", 显示名称: "m", 远程模型标识: "r", 能力集: map[string]bool{"reasoning": true}, 模型类型: []string{"reasoning", "reasoning"}},
		{供应商标识: "01900000-0000-7000-8000-000000000001", 显示名称: "m", 远程模型标识: "r", 模型类型: []string{"text", "text"}},
		{供应商标识: "01900000-0000-7000-8000-000000000001", 显示名称: "m", 远程模型标识: "r", 模型类型: []string{"image_input"}},
		{供应商标识: "01900000-0000-7000-8000-000000000001", 显示名称: "m", 远程模型标识: "r", 模型类型: []string{"tools"}},
		{供应商标识: "01900000-0000-7000-8000-000000000001", 显示名称: "m", 远程模型标识: "r", 模型类型: []string{"unknown"}},
		{供应商标识: "01900000-0000-7000-8000-000000000001", 显示名称: "m", 远程模型标识: "r", 推理等级集: []供应商包.推理级别{{显示名称: "中", 远程值: "x"}, {显示名称: "高", 远程值: "x"}}},
	}
	for _, 输入 := range 用例集 {
		if 错误 := 校验模型(输入); !错误判断包.是否属于(错误, 供应商包.无效配置错误) {
			测试器.格式化报告致命错误("未拒绝无效配置: %+v %v", 输入, 错误)
		}
	}
}

func Test内容类型集为独立所属能力集(测试器 *测试包.测试器) {
	for _, 能力集 := range []map[string]bool{nil, {}, {"tools": false}, {"reasoning": true}} {
		输入 := 供应商包.模型输入{供应商标识: 测试标识, 显示名称: "模型", 远程模型标识: "remote", 能力集: 能力集, 模型类型: []string{"text", "image", "audio", "video"}}
		if 错误 := 校验模型(输入); 错误 != nil {
			测试器.格式化报告致命错误("内容类型被能力开关限制: %v", 错误)
		}
		输入.模型类型 = nil
		if 错误 := 校验模型(输入); 错误 != nil {
			测试器.格式化报告致命错误("未标注类型的配置被拒绝: %v", 错误)
		}
	}
}

func Test供应商校验不暴露密钥集(测试器 *测试包.测试器) {
	当前对象 := 供应商包.供应商输入{名称: "p", 协议: "chatcompletions", 基础URL: "https://secret:password@example.invalid", API密钥: new("private-key")}
	错误 := 校验供应商(当前对象)
	if !错误判断包.是否属于(错误, 供应商包.无效配置错误) || 字符串包.包含(错误.错误文本(), "password") || 字符串包.包含(错误.错误文本(), "private-key") {
		测试器.格式化报告致命错误("错误未脱敏: %v", 错误)
	}
}
