```go
package 供应商

import 上下文包 "context"

// 仓储 定义配置业务所需的持久化操作，由具体存储包实现。
// 创建记录的标识由 服务 提供；更新不能创建缺失记录。
// 查询不得返回明文凭据，只有 加载运行时 可读取用于内部装配的凭据快照。
// 实现必须传递上下文，返回前释放查询占用，不能将事务带入后续模型调用。
type 仓储 interface {
	创建供应商(上下文包.上下文, string, 供应商输入) error
	更新供应商(上下文包.上下文, string, 供应商输入) error
	获取供应商(上下文包.上下文, string) (供应商, error)
	列表供应商集(上下文包.上下文, ...列表选项集) (页[供应商], error)
	删除供应商(上下文包.上下文, string) error
	创建模型(上下文包.上下文, string, 模型输入) error
	更新模型(上下文包.上下文, string, 模型输入) error
	获取模型(上下文包.上下文, string) (模型, error)
	列表模型集(上下文包.上下文, bool, ...列表选项集) (页[模型], error)
	设置模型已启用(上下文包.上下文, string, bool) error
	删除模型(上下文包.上下文, string) error
	加载运行时(上下文包.上下文, string) (运行时配置, error)
}
```
